1//! Classic "v1" anisette provider: a plain HTTPS GET that returns the anisette
2//! headers as a flat JSON object. This is the format used by the original
3//! anisette servers and by Sideloadly's remote endpoint
4//! (`.../anisette/<id>?u=<device-hash>`).
5//!
6//! No token or account is sent — this is the free, rate-limited path. A random
7//! per-install device hash is persisted so the server returns a stable identity
8//! across refreshes.
9
10use std::collections::HashMap;
11use std::path::PathBuf;
12
13use log::debug;
14
15use crate::AnisetteError;
16use crate::anisette_headers_provider::AnisetteHeadersProvider;

18const DEVICE_HASH_FILE: &str = "anisette_v1_device";
19
20/// Persists a validated replacement device identity after a successful refresh.
21pub fn save_device_hash(path: &std::path::Path, hash: &str) -> Result<(), AnisetteError> {
22    if hash.len() != 64 || !hash.bytes().all(|b| b.is_ascii_hexdigit()) {
23        return Err(AnisetteError::InvalidArgument("invalid device hash".into()));
24    }
25    std::fs::create_dir_all(path)?;
26    std::fs::write(path.join(DEVICE_HASH_FILE), hash)?;
27    Ok(())
28}

30/// Default Sideloadly instance. The remote anisette endpoint is just a classic v1
31/// server, so any self-hosted instance running the same software works too; this
32/// is only the fallback used when a caps-token is given with no host of its own.
33pub const DEFAULT_SIDELOADLY_HOST: &str = "sideloadly.io";
34
35/// Builds the v1 anisette URL for a Sideloadly(-compatible) instance from a host
36/// (with or without a scheme) and a caps-token, e.g.
37/// `sideloadly.io` + `Fzji…` → `https://sideloadly.io/anisette/Fzji…`.
38pub fn sideloadly_anisette_url(host: &str, token: &str) -> String {
39    let host = host
40        .trim()
41        .trim_start_matches("https://")
42        .trim_start_matches("http://")
43        .trim_matches('/');
44    format!("https://{host}/anisette/{}", token.trim())
45}
46
47/// Ensures the per-install device hash file exists under `path`, creating a new
48/// random one if the file is missing or malformed. Returns the 64-hex-char hash.
49pub fn ensure_device_hash(path: &std::path::Path) -> Result<String, AnisetteError> {
50    std::fs::create_dir_all(path)?;
51    let hash_path = path.join(DEVICE_HASH_FILE);
52    match std::fs::read_to_string(&hash_path) {
53        Ok(s) if s.trim().len() == 64 && s.trim().chars().all(|c| c.is_ascii_hexdigit()) => {
54            Ok(s.trim().to_string())
55        }
56        _ => {
57            let bytes = rand::random::<[u8; 32]>();
58            let hex: String = bytes.iter().map(|b| format!("{b:02x}")).collect();
59            std::fs::write(&hash_path, &hex)?;
60            Ok(hex)
61        }
62    }
63}

65pub struct RemoteAnisetteProviderV1 {
66    url: String,
67    configuration_path: PathBuf,
68    device_hash: Option<String>,
69}
70
71impl RemoteAnisetteProviderV1 {
72    pub fn new(url: String, configuration_path: PathBuf) -> RemoteAnisetteProviderV1 {
73        RemoteAnisetteProviderV1 {
74            url: url.trim().trim_end_matches('/').to_string(),
75            configuration_path,
76            device_hash: None,
77        }
78    }
79
80    /// Persistent 32-byte (64 hex char) identifier sent as `?u=`. The server keys
81    /// the provisioned Mac identity off this, so it must stay stable per install.
82    fn device_hash(&mut self) -> Result<String, AnisetteError> {
83        if let Some(h) = &self.device_hash {
84            return Ok(h.clone());
85        }
86        std::fs::create_dir_all(&self.configuration_path)?;
87        let path = self.configuration_path.join(DEVICE_HASH_FILE);
88        let hash = match std::fs::read_to_string(&path) {
89            Ok(s) if s.trim().len() == 64 && s.trim().chars().all(|c| c.is_ascii_hexdigit()) => {
90                s.trim().to_string()
91            }
92            _ => {
93                let bytes = rand::random::<[u8; 32]>();
94                let hex: String = bytes.iter().map(|b| format!("{b:02x}")).collect();
95                std::fs::write(&path, &hex)?;
96                hex
97            }
98        };
99        self.device_hash = Some(hash.clone());
100        Ok(hash)
101    }
102
103    fn request_url(&mut self) -> Result<String, AnisetteError> {
104        // A plain v1 server takes no `u`; a Sideloadly-style endpoint expects one.
105        // Respect a `u` the caller already put in the URL instead of adding a second.
106        if self.url.contains("u=") {
107            return Ok(self.url.clone());
108        }
109        let u = self.device_hash()?;
110        let sep = if self.url.contains('?') { '&' } else { '?' };
111        Ok(format!("{}{sep}u={u}", self.url))
112    }
113
114    /// Probes whether `url` answers with v1 anisette JSON, so callers can tell a
115    /// v1 server apart from a v3 one.
116    #[cfg_attr(not(feature = "async"), remove_async_await::remove_async_await)]
117    pub async fn probe(url: &str, configuration_path: PathBuf) -> bool {
118        let mut p = RemoteAnisetteProviderV1::new(url.to_string(), configuration_path);
119        p.get_anisette_headers(false).await.is_ok()
120    }
121}

123#[cfg_attr(feature = "async", async_trait::async_trait)]
124impl AnisetteHeadersProvider for RemoteAnisetteProviderV1 {
125    #[cfg_attr(not(feature = "async"), remove_async_await::remove_async_await)]
126    async fn get_anisette_headers(
127        &mut self,
128        _skip_provisioning: bool,
129    ) -> Result<HashMap<String, String>, AnisetteError> {
130        let url = self.request_url()?;
131        debug!("Fetching v1 anisette from {url}");
132        #[cfg(feature = "async")]
133        let headers = reqwest::Client::new()
134            .get(&url)
135            .header("Accept", "application/json")
136            .send()
137            .await?
138            .error_for_status()?
139            .json::<HashMap<String, String>>()
140            .await?;
141        #[cfg(not(feature = "async"))]
142        let headers = reqwest::blocking::Client::new()
143            .get(&url)
144            .header("Accept", "application/json")
145            .send()?
146            .error_for_status()?
147            .json::<HashMap<String, String>>()?;
148
149        // A real anisette response always carries the one-time password / machine id.
150        if !headers.contains_key("X-Apple-I-MD") && !headers.contains_key("X-Apple-I-MD-M") {
151            return Err(AnisetteError::InvalidArgument(format!(
152                "{} did not return anisette headers",
153                self.url
154            )));
155        }
156        Ok(self.normalize_headers(headers))
157    }
158}

160#[cfg(test)]
161mod tests {
162    use super::*;
163
164    #[test]
165    fn builds_sideloadly_url_for_any_instance() {
166        assert_eq!(
167            sideloadly_anisette_url(DEFAULT_SIDELOADLY_HOST, "Fzji"),
168            "https://sideloadly.io/anisette/Fzji"
169        );
170        // A self-hosted host, with or without a scheme or trailing slash.
171        assert_eq!(
172            sideloadly_anisette_url("https://ani.example.com/", " Fzji "),
173            "https://ani.example.com/anisette/Fzji"
174        );
175    }
