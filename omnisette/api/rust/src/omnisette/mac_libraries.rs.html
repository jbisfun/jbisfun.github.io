1//! Fetch the known Mac CoreADI build directly from Apple's iTunes installer.
2
3use crate::AnisetteError;
4use sha2::{Digest, Sha256};
5use std::{

11pub const LIBRARY_SHA256: &str = "2eafc3c79c7f41a8afbfbeaaf23ee5dab5ad5bda55ec8e54dbc0ddf1d20ce630";
12const PAYLOAD_SHA256: &str = "8a6c068c95036326eb2d2885fa60912ed0f7229d0939ae92934a70a3bb6cf025";
13const URL: &str = "https://secure-appldnld.apple.com/itunes12/041-56365-20210413-79A5AC6D-6B0A-46EC-B399-5A381183E3BE/iTunes12.8.3.dmg";
14const OFFSET: u64 = 184_678_769;
15const LENGTH: usize = 3_471_595;
16const FRAMEWORK: &str = "System/Library/PrivateFrameworks/CoreADI.framework/Versions/A/CoreADI";
17
18pub fn library_path(root: &Path) -> PathBuf {
19    root.join("lib/mac/CoreADI")
20}
21
22#[cfg_attr(not(feature = "async"), remove_async_await::remove_async_await)]
23pub async fn ensure_library(root: &Path) -> Result<(), AnisetteError> {
24    let path = library_path(root);
25    if std::fs::read(&path).is_ok_and(|bytes| hex::encode(Sha256::digest(bytes)) == LIBRARY_SHA256)
26    {
27        return Ok(());
28    }
29    #[cfg(feature = "async")]
30    use reqwest::Client;
31    #[cfg(not(feature = "async"))]
32    use reqwest::blocking::Client;
33
34    log::info!("Downloading Mac CoreADI from Apple's iTunes installer");
35    let response = Client::builder()
36        .timeout(Duration::from_secs(60))
37        .no_gzip()
38        .build()?
39        .get(URL)
40        .header(
41            "Range",
42            format!("bytes={OFFSET}-{}", OFFSET + LENGTH as u64 - 1),
43        )
44        .header("Accept-Encoding", "identity")
45        .send()
46        .await?
47        .error_for_status()?;
48    if response.status() != reqwest::StatusCode::PARTIAL_CONTENT
49        || response.content_length() != Some(LENGTH as u64)
50    {
51        return Err(
52            anyhow::anyhow!("Apple did not return the expected Mac CoreADI byte range").into(),
53        );
54    }
55    let payload = response.bytes().await?;
56    let library = extract_library(&payload)?;
57    std::fs::create_dir_all(path.parent().unwrap())?;
58    let temporary = path.with_extension("partial");
59    std::fs::write(&temporary, library)?;
60    std::fs::rename(temporary, path)?;
61    Ok(())
62}

64fn extract_library(payload: &[u8]) -> Result<Vec<u8>, AnisetteError> {
65    if payload.len() != LENGTH || hex::encode(Sha256::digest(payload)) != PAYLOAD_SHA256 {
66        return Err(AnisetteError::InvalidLibraryFormat);
67    }
68    let mut cpio = Vec::new();
69    flate2::read::GzDecoder::new(payload)
70        .take(64 * 1024 * 1024)
71        .read_to_end(&mut cpio)?;
72    let fat = cpio_file(&cpio, FRAMEWORK)?;
73    let library = x86_64_slice(fat)?;
74    if hex::encode(Sha256::digest(library)) != LIBRARY_SHA256 {
75        return Err(AnisetteError::InvalidLibraryFormat);
76    }
77    Ok(library.to_vec())
78}

80fn cpio_file<'a>(bytes: &'a [u8], wanted: &str) -> Result<&'a [u8], AnisetteError> {
81    let bad = || AnisetteError::InvalidLibraryFormat;
82    let octal = |bytes: &[u8]| {
83        usize::from_str_radix(std::str::from_utf8(bytes).map_err(|_| bad())?, 8).map_err(|_| bad())
84    };
85    let mut offset = 0usize;
86    loop {
87        let header = bytes
88            .get(offset..offset.checked_add(76).ok_or_else(bad)?)
89            .ok_or_else(bad)?;
90        if &header[..6] != b"070707" {
91            return Err(bad());
92        }
93        let name_length = octal(&header[59..65])?;
94        let size = octal(&header[65..76])?;
95        let name_start = offset + 76;
96        let data_start = name_start.checked_add(name_length).ok_or_else(bad)?;
97        let name = bytes
98            .get(name_start..data_start)
99            .and_then(|name| name.strip_suffix(&[0]))
100            .ok_or_else(bad)?;
101        let name = std::str::from_utf8(name).map_err(|_| bad())?;
102        let end = data_start.checked_add(size).ok_or_else(bad)?;
103        let data = bytes.get(data_start..end).ok_or_else(bad)?;
104        if name.strip_prefix("./").unwrap_or(name) == wanted {
105            return Ok(data);
106        }
107        if name == "TRAILER!!!" {
108            return Err(bad());
109        }
110        offset = end;
111    }
112}

114fn x86_64_slice(bytes: &[u8]) -> Result<&[u8], AnisetteError> {
115    let bad = || AnisetteError::InvalidLibraryFormat;
116    let word = |offset: usize| -> Result<u32, AnisetteError> {
117        let value = bytes
118            .get(offset..offset.checked_add(4).ok_or_else(bad)?)
119            .ok_or_else(bad)?;
120        Ok(u32::from_be_bytes(value.try_into().map_err(|_| bad())?))
121    };
122    if word(0)? != 0xcafebabe {
123        return Err(bad());
124    }
125    let count = word(4)? as usize;
126    if count > bytes.len().saturating_sub(8) / 20 {
127        return Err(bad());
128    }
129    for index in 0..count {
130        let entry = 8 + index * 20;
131        if word(entry)? == 0x01000007 {
132            let start = word(entry + 8)? as usize;
133            let size = word(entry + 12)? as usize;
134            return bytes
135                .get(start..start.checked_add(size).ok_or_else(bad)?)
136                .ok_or_else(bad);
137        }
138    }
139    Err(bad())
140}

142#[cfg(test)]
143mod tests {
144    use super::*;
145
146    #[test]
147    fn malformed_installers_and_out_of_bounds_slices_are_rejected() {
148        for bytes in [vec![], vec![0; 100], b"070707".to_vec()] {
149            assert!(extract_library(&bytes).is_err());
150            assert!(cpio_file(&bytes, FRAMEWORK).is_err());
151            assert!(x86_64_slice(&bytes).is_err());
152        }
153        let mut fat = vec![0; 28];
154        fat[..4].copy_from_slice(&0xcafebabeu32.to_be_bytes());
155        fat[4..8].copy_from_slice(&1u32.to_be_bytes());
156        fat[8..12].copy_from_slice(&0x01000007u32.to_be_bytes());
157        fat[16..20].copy_from_slice(&u32::MAX.to_be_bytes());
158        fat[20..24].copy_from_slice(&100u32.to_be_bytes());
159        assert!(x86_64_slice(&fat).is_err());
160    }
