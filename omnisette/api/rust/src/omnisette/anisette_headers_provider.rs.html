5/// Older anisette servers advertise Xcode, which Apple's GrandSlam edge rejects.
6/// Keep the server's hardware/OS identity and only update the AuthKit client.
7pub(crate) fn normalize_client_info(client_info: &str) -> String {
8    let mut client_info = client_info.to_owned();
9    if let Some(start) = client_info.find("com.apple.dt.Xcode") {
10        let end = client_info[start..]
11            .find(|c: char| c == ')' || c == '>' || c.is_whitespace())
12            .map_or(client_info.len(), |end| start + end);
13        client_info.replace_range(start..end, "com.apple.akd/1.0");
14    }
15    client_info
16}

18#[cfg_attr(feature = "async", async_trait::async_trait)]
19pub trait AnisetteHeadersProvider: Send + Sync {
20    #[cfg_attr(not(feature = "async"), remove_async_await::remove_async_await)]
21    async fn get_anisette_headers(
22        &mut self,
23        skip_provisioning: bool,
24    ) -> Result<HashMap<String, String>, AnisetteError>;
25
26    #[cfg_attr(not(feature = "async"), remove_async_await::remove_async_await)]
27    async fn get_authentication_headers(
28        &mut self,
29    ) -> Result<HashMap<String, String>, AnisetteError> {
30        let headers = self.get_anisette_headers(false).await?;
31        Ok(self.normalize_headers(headers))
32    }
33
34    /// Normalizes headers to ensure that all the required headers are given.
35    fn normalize_headers(
36        &mut self,
37        mut headers: HashMap<String, String>,
38    ) -> HashMap<String, String> {
39        if let Some(client_info) = headers.remove("X-MMe-Client-Info") {
40            headers.insert("X-Mme-Client-Info".to_string(), client_info);
41        }
42        for (name, value) in &mut headers {
43            if name.eq_ignore_ascii_case("X-Mme-Client-Info") {
44                *value = normalize_client_info(value);
45            }
46        }
47
48        headers
49    }
50}
