83/// Where anisette data comes from.
84#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash)]
85pub enum AnisetteProvider {
86    /// Computed on this machine with Apple's Mac CoreADI, using a persistent Mac identity.
87    Local,
88    /// Requested from a remote anisette v3 server.
89    #[default]
90    Remote,
91    /// Computed on this machine with Apple's ADI libraries, using a spoofed,
92    /// disposable Mac identity that is not tied to this machine.
93    Fake,
94}
95
96impl AnisetteProvider {
97    pub const ALL: [AnisetteProvider; 3] = [
98        AnisetteProvider::Local,
99        AnisetteProvider::Remote,
100        AnisetteProvider::Fake,
101    ];
102
103    /// Whether this provider needs Apple's ADI libraries on disk.
104    pub fn needs_libraries(&self) -> bool {
105        matches!(self, AnisetteProvider::Local | AnisetteProvider::Fake)
106    }
107}

260/// Remote provider that works with either protocol a server might speak: the
261/// classic v1 JSON endpoint (also what Sideloadly's remote uses) or the v3
262/// provisioning protocol. It tries v3 first and falls back to v1, remembering
263/// whichever answered so later refreshes skip the probe.
264enum RemoteInner {
265    Undecided,
266    V1(remote_anisette_v1::RemoteAnisetteProviderV1),
267    #[cfg(feature = "remote-anisette-v3")]
268    V3(remote_anisette_v3::RemoteAnisetteProviderV3),
269}
270
271pub struct AutoRemoteProvider {
272    url: String,
273    configuration_path: PathBuf,
274    serial: String,
275    inner: RemoteInner,
276}
277
278impl AutoRemoteProvider {
279    pub fn new(url: String, configuration_path: PathBuf, serial: String) -> AutoRemoteProvider {
280        AutoRemoteProvider {
281            url,
282            configuration_path,
283            serial,
284            inner: RemoteInner::Undecided,
285        }
286    }
287}

289#[cfg_attr(feature = "async", async_trait::async_trait)]
290impl AnisetteHeadersProvider for AutoRemoteProvider {
291    #[cfg_attr(not(feature = "async"), remove_async_await::remove_async_await)]
292    async fn get_anisette_headers(
293        &mut self,
294        skip_provisioning: bool,
295    ) -> Result<std::collections::HashMap<String, String>, AnisetteError> {
296        match &mut self.inner {
297            RemoteInner::V1(p) => return p.get_anisette_headers(skip_provisioning).await,
298            #[cfg(feature = "remote-anisette-v3")]
299            RemoteInner::V3(p) => return p.get_anisette_headers(skip_provisioning).await,
300            RemoteInner::Undecided => {}
301        }
302
303        #[cfg(feature = "remote-anisette-v3")]
304        let v3_error = {
305            let mut v3 = remote_anisette_v3::RemoteAnisetteProviderV3::new(
306                self.url.clone(),
307                self.configuration_path.clone(),
308                self.serial.clone(),
309            );
310            match v3.get_anisette_headers(skip_provisioning).await {
311                Ok(headers) => {
312                    self.inner = RemoteInner::V3(v3);
313                    return Ok(headers);
314                }
315                Err(error) => {
316                    log::debug!("Remote anisette v3 failed ({error}); trying v1");
317                    error.to_string()
318                }
319            }
320        };
321        #[cfg(not(feature = "remote-anisette-v3"))]
322        let _ = &self.serial;
323
324        let mut v1 = remote_anisette_v1::RemoteAnisetteProviderV1::new(
325            self.url.clone(),
326            self.configuration_path.clone(),
327        );
328        match v1.get_anisette_headers(skip_provisioning).await {
329            Ok(headers) => {
330                log::debug!("Remote server {} speaks anisette v1", self.url);
331                self.inner = RemoteInner::V1(v1);
332                Ok(headers)
333            }
334            Err(error) => {
335                #[cfg(feature = "remote-anisette-v3")]
336                return Err(AnisetteError::InvalidArgument(format!(
337                    "Anisette failed using both protocols. v3: {v3_error}; v1: {error}. Check the configured server, system proxy and certificate settings."
338                )));
339                #[cfg(not(feature = "remote-anisette-v3"))]
340                Err(error)
341            }
342        }
343    }
344}

346/// Include the transport's source chain so TLS and proxy failures remain visible.
347/// The origin identifies the endpoint without exposing tokens in URL paths or queries.
348pub fn transport_error_detail(error: &reqwest::Error) -> String {
349    use std::error::Error;
350    let origin = error
351        .url()
352        .map(|url| url.origin().ascii_serialization())
353        .unwrap_or_else(|| "unknown endpoint".into());
354    let kind = if error.is_timeout() {
355        "Request timed out"
356    } else if error.is_decode() {
357        "Invalid response"
358    } else if error.is_connect() {
359        "Connection failed"
360    } else {
361        "HTTP request failed"
362    };
363    let mut details = format!("{kind} at {origin}");
364    if let Some(status) = error.status() {
365        details.push_str(&format!(" (HTTP {})", status.as_u16()));
366    }
367    let mut source = error.source();
368    while let Some(cause) = source {
369        details.push_str(": ");
370        details.push_str(&cause.to_string());
371        source = cause.source();
372    }
373    details
374}

376#[cfg(all(test, feature = "async"))]
377mod transport_tests {
378    use super::*;
379    #[tokio::test]
380    async fn reports_transport_cause_without_url_tokens() {
381        let listener = std::net::TcpListener::bind("127.0.0.1:0").unwrap();
382        let address = listener.local_addr().unwrap();
383        drop(listener);
384        let error = reqwest::Client::new()
385            .get(format!("http://{address}/private/token?secret=sensitive"))
386            .send()
387            .await
388            .unwrap_err();
389        let message = transport_error_detail(&error);
390        assert!(message.contains(&address.to_string()));
391        assert!(message.contains("Connection failed"));
392        assert!(!message.contains("sensitive"));
393        assert!(!message.contains("private"));
394    }
