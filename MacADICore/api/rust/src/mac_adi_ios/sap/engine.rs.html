1//! arm64 Unicorn engine + shim-dispatch framework for the modern-ADI guest.
2//!
3//! Ports ipatool's internal/sap/machine {machine.go engine/invoke, shims.go,
4//! shim_memory.go, arm64.go}. Same shape as our x86 `vm.rs`: imported functions
5//! live on stub pages holding `ret`; a code hook over the shim region runs the
6//! Rust handler, which sets X0 and returns. Platform services (dyld/Mach/XPC/
7//! dispatch/FairPlay) are added by `shim_platform.rs`.

166impl Engine {
167    pub fn new() -> Result<Engine, Error> {
168        let mut uc = Unicorn::new_with_data(Arch::ARM64, Mode::ARM, SapState::default())?;
169        for (addr, size) in [
170            (RETURN_ADDRESS, PAGE),
171            (SCRATCH_BASE, SCRATCH_SIZE),
172            (HEAP_BASE, HEAP_SIZE),
173            (STACK_BASE, STACK_SIZE),
174            (SHIM_BASE, SHIM_SIZE),
175        ] {
176            uc.mem_map(addr, size, Prot::ALL)?;
177        }
178        // `ret` at the return trampoline.
179        uc.mem_write(RETURN_ADDRESS, &0xd65f03c0u32.to_le_bytes())?;
180
181        {
182            let st = uc.get_data_mut();
183            st.code_cursor = SHIM_BASE;
184            st.data_cursor = SHIM_BASE + SHIM_CODE_SIZE;
185            st.heap_cursor = 0;
186            st.xpc_next_object = 0xffff_0000_0000_0000;
187        }
188
189        // Dispatch hook over the shim code region.
190        uc.add_code_hook(SHIM_BASE, SHIM_BASE + SHIM_CODE_SIZE - 1, |uc, addr, _| {
191            let handler = match uc.get_data().handlers.get(&addr).copied() {
192                Some(h) => h,
193                None => {
194                    uc.get_data_mut().fault =
195                        Some(format!("guest entered unknown service address {addr:#x}"));
196                    let _ = uc.emu_stop();
197                    return;
198                }
199            };
200            let mut st = std::mem::take(uc.get_data_mut());
201            let result = st.dispatch(uc, handler);
202            *uc.get_data_mut() = st;
203            if let Err(message) = result {
204                let data = uc.get_data_mut();
205                if data.fault.is_none() {
206                    data.fault = Some(message);
207                }
208                let _ = uc.emu_stop();
209            }
210        })?;
211
212        // Fault on unmapped access, like vm.rs.
213        uc.add_mem_hook(
214            HookType::MEM_READ_UNMAPPED | HookType::MEM_WRITE_UNMAPPED | HookType::MEM_FETCH_UNMAPPED,
215            0,
216            u64::MAX,
217            |uc, kind, addr, size, _| {
218                let pc = uc.reg_read(RegisterARM64::PC).unwrap_or(0);
219                let k = match kind {
220                    MemType::READ_UNMAPPED => "read",
221                    MemType::WRITE_UNMAPPED => "write",
222                    _ => "fetch",
223                };
224                uc.get_data_mut().fault =
225                    Some(format!("{k}-unmapped at {addr:#x} (size {size}) pc={pc:#x}"));
226                false
227            },
228        )?;
229
230        Ok(Engine { uc })
231    }

265    /// Resolve a symbol to a stub address, creating an "unsupported import" stub
266    /// if unknown (ports shims.resolve).
267    pub fn resolve(&mut self, name: &str) -> Result<u64, Error> {
268        if let Some(&addr) = self.uc.get_data().symbols.get(name) {
269            return Ok(addr);
270        }
271        self.add_function(name, Handler::Unsupported)
272    }
273
274    pub fn add_function(&mut self, name: &str, handler: Handler) -> Result<u64, Error> {
275        if let Some(&addr) = self.uc.get_data().symbols.get(name) {
276            return Ok(addr);
277        }
278        let cursor = self.uc.get_data().code_cursor;
279        if cursor + SHIM_SLOT > SHIM_BASE + SHIM_CODE_SIZE {
280            return Err(Error::Emulation("guest service code area full".into()));
281        }
282        self.uc.mem_write(cursor, &0xd65f03c0u32.to_le_bytes())?; // ret
283        let st = self.uc.get_data_mut();
284        st.code_cursor += SHIM_SLOT;
285        st.handlers.insert(cursor, handler);
286        st.symbols.insert(name.to_string(), cursor);
287        Ok(cursor)
288    }
289
290    pub fn add_aliases(&mut self, names: &[&str], handler: Handler) -> Result<(), Error> {
291        for name in names {
292            self.add_function(name, handler)?;
293        }
294        Ok(())
295    }
296
297    pub fn add_data(&mut self, name: &str, data: &[u8]) -> Result<u64, Error> {
298        if let Some(&addr) = self.uc.get_data().symbols.get(name) {
299            return Ok(addr);
300        }
301        let mut cursor = align(self.uc.get_data().data_cursor, 8);
302        if cursor + data.len() as u64 > SHIM_BASE + SHIM_SIZE {
303            return Err(Error::Emulation("guest service data area full".into()));
304        }
305        self.uc.mem_write(cursor, data)?;
306        let advance = data.len().max(8) as u64;
307        let st = self.uc.get_data_mut();
308        cursor = cursor; // keep clippy quiet
309        st.data_cursor = align(st.data_cursor, 8) + advance;
310        st.symbols.insert(name.to_string(), cursor);
311        Ok(cursor)
312    }

355    /// Run `function(args...)` to the return trampoline; returns X0.
356    /// Ports machine.invoke including the arm64 invalid-instruction retry.
357    pub fn invoke(&mut self, function: u64, args: &[u64]) -> Result<u64, Error> {
358        if function == 0 {
359            return Err(Error::Emulation("guest entry point unavailable".into()));
360        }
361        for (i, reg) in ARG_REGS.iter().enumerate() {
362            self.uc.reg_write(*reg, args.get(i).copied().unwrap_or(0))?;
363        }
364        let extra = args.len().saturating_sub(ARG_REGS.len());
365        let mut sp = STACK_END - (extra as u64) * 8;
366        sp -= sp % 16;
367        for i in 0..extra {
368            self.uc.mem_write(sp + (i as u64) * 8, &args[ARG_REGS.len() + i].to_le_bytes())?;
369        }
370        self.uc.reg_write(RegisterARM64::SP, sp)?;
371        self.uc.reg_write(RegisterARM64::X30, RETURN_ADDRESS)?; // LR
372        self.uc.get_data_mut().fault = None;
373
374        let mut next = function;
375        loop {
376            let res = self.uc.emu_start(next, RETURN_ADDRESS, GUEST_TIMEOUT_US, 0);
377            if let Some(fault) = self.uc.get_data_mut().fault.take() {
378                return Err(Error::Emulation(fault));
379            }
380            match res {
381                Ok(()) => break,
382                Err(_) => {
383                    if self.emulate_invalid_instruction()? {
384                        next = self.uc.reg_read(RegisterARM64::PC)?;
385                        continue;
386                    }
387                    let pc = self.uc.reg_read(RegisterARM64::PC).unwrap_or(0);
388                    return Err(Error::Emulation(format!("guest fault at pc={pc:#x}: {res:?}")));
389                }
390            }
391        }
392        let pc = self.uc.reg_read(RegisterARM64::PC)?;
393        if pc != RETURN_ADDRESS {
394            return Err(Error::Emulation(format!("guest stopped at {pc:#x}")));
395        }
396        Ok(self.uc.reg_read(RegisterARM64::X0)?)
397    }

399    /// Ports arm64.go: emulate CASAL/SWPAL (LSE atomics) and EOR3 that the TCI
400    /// interpreter rejects. Returns true if it handled and advanced PC.
401    fn emulate_invalid_instruction(&mut self) -> Result<bool, Error> {
402        let pc = self.uc.reg_read(RegisterARM64::PC)?;
403        let data = match self.uc.mem_read_as_vec(pc, 4) {
404            Ok(d) => d,
405            Err(_) => return Ok(false),
406        };
407        let inst = u32::from_le_bytes(data.try_into().unwrap());
408
409        const EOR3_MASK: u32 = 0xffe08000;
410        const EOR3: u32 = 0xce000000;
411        if inst & EOR3_MASK == EOR3 {
412            self.emulate_eor3(inst)?;
413            self.uc.reg_write(RegisterARM64::PC, pc + 4)?;
414            return Ok(true);
415        }
416
417        const ATOMIC_MASK: u32 = 0xffe0fc00;
418        const CASAL32: u32 = 0x88e0fc00;
419        const CASAL64: u32 = 0xc8e0fc00;
420        const SWPAL32: u32 = 0xb8e08000;
421        const SWPAL64: u32 = 0xf8e08000;
422        let opcode = inst & ATOMIC_MASK;
423        let width: u64 = match opcode {
424            CASAL32 | SWPAL32 => 4,
425            CASAL64 | SWPAL64 => 8,
426            _ => return Ok(false),
427        };
428        let source = (inst >> 16) & 31;
429        let base = (inst >> 5) & 31;
430        let destination = inst & 31;
431        let address = self.read_base_register(base)?;
432        let mem = self.uc.mem_read_as_vec(address, width as usize)?;
433        let mut old_bytes = [0u8; 8];
434        old_bytes[..width as usize].copy_from_slice(&mem);
435        let old_value = u64::from_le_bytes(old_bytes);
436        let mut source_value = self.read_value_register(source)?;
437        if width == 4 {
438            source_value = source_value as u32 as u64;
439        }
440        match opcode {
441            CASAL32 | CASAL64 => {
442                let mut new_value = self.read_value_register(destination)?;
443                if width == 4 {
444                    new_value = new_value as u32 as u64;
445                }
446                if old_value == source_value {
447                    self.write_memory_value(address, new_value, width)?;
448                }
449                self.write_value_register(source, old_value)?;
450            }
451            SWPAL32 | SWPAL64 => {
452                self.write_memory_value(address, source_value, width)?;
453                self.write_value_register(destination, old_value)?;
454            }
455            _ => unreachable!(),
456        }
457        self.uc.reg_write(RegisterARM64::PC, pc + 4)?;
458        Ok(true)
459    }

461    fn emulate_eor3(&mut self, inst: u32) -> Result<(), Error> {
462        let d = inst & 31;
463        let a = (inst >> 5) & 31;
464        let m = (inst >> 10) & 31;
465        let n = (inst >> 16) & 31;
466        let va = self.read_vector(a)?;
467        let vn = self.read_vector(n)?;
468        let vm = self.read_vector(m)?;
469        let mut r = [0u8; 16];
470        for i in 0..16 {
471            r[i] = va[i] ^ vn[i] ^ vm[i];
472        }
473        self.write_vector(d, r)
474    }

476    fn read_vector(&self, index: u32) -> Result<[u8; 16], Error> {
477        let reg = vreg(index);
478        let raw = self.uc.reg_read_long(reg)?;
479        let mut out = [0u8; 16];
480        let n = raw.len().min(16);
481        out[..n].copy_from_slice(&raw[..n]);
482        Ok(out)
483    }
484    fn write_vector(&mut self, index: u32, value: [u8; 16]) -> Result<(), Error> {
485        Ok(self.uc.reg_write_long(vreg(index), &value)?)
486    }

488    fn general_register(index: u32) -> Option<RegisterARM64> {
489        Some(match index {
490            0..=28 => GP_REGS[index as usize],
491            29 => RegisterARM64::X29,
492            30 => RegisterARM64::X30,
493            _ => return None,
494        })
495    }
496    fn read_base_register(&self, index: u32) -> Result<u64, Error> {
497        let reg = if index == 31 {
498            RegisterARM64::SP
499        } else {
500            Self::general_register(index).ok_or_else(|| Error::Emulation("bad base reg".into()))?
501        };
502        self.reg_read(reg)
503    }
504    fn read_value_register(&self, index: u32) -> Result<u64, Error> {
505        if index == 31 {
506            return Ok(0);
507        }
508        let reg = Self::general_register(index).ok_or_else(|| Error::Emulation("bad reg".into()))?;
509        self.reg_read(reg)
510    }
511    fn write_value_register(&mut self, index: u32, value: u64) -> Result<(), Error> {
512        if index == 31 {
513            return Ok(());
514        }
515        let reg = Self::general_register(index).ok_or_else(|| Error::Emulation("bad reg".into()))?;
516        self.reg_write(reg, value)
517    }
518    fn write_memory_value(&mut self, address: u64, value: u64, width: u64) -> Result<(), Error> {
519        let bytes = value.to_le_bytes();
520        self.mem_write(address, &bytes[..width as usize])
521    }

552pub(super) fn arg(uc: &Uc<'_>, i: usize) -> u64 {
553    if i < ARG_REGS.len() {
554        return uc.reg_read(ARG_REGS[i]).unwrap_or(0);
555    }
556    let sp = uc.reg_read(RegisterARM64::SP).unwrap_or(0);
557    uc.mem_read_as_vec(sp + ((i - ARG_REGS.len()) as u64) * 8, 8)
558        .map(|b| u64::from_le_bytes(b.try_into().unwrap()))
559        .unwrap_or(0)
560}

566pub(super) fn read_cstring(uc: &Uc<'_>, mut address: u64) -> String {
567    let mut out = Vec::new();
568    while out.len() < 4096 {
569        match uc.mem_read_as_vec(address, 1) {
570            Ok(b) if b[0] != 0 => {
571                out.push(b[0]);
572                address += 1;
573            }
574            _ => break,
575        }
576    }
577    String::from_utf8_lossy(&out).into_owned()
578}

580impl SapState {
581    fn dispatch(&mut self, uc: &mut Uc<'_>, handler: Handler) -> Result<(), String> {
582        use Handler::*;
583        match handler {
584            Malloc => {
585                let size = arg(uc, 0);
586                let a = self.allocate(uc, size)?;
587                set_result(uc, a);
588            }
589            MallocAligned => {
590                let size = arg(uc, 0);
591                let alignment = arg(uc, 1);
592                if alignment == 0 || alignment & (alignment - 1) != 0 {
593                    return Err(format!("invalid allocation alignment {alignment:#x}"));
594                }
595                let a = self.allocate_aligned(uc, size, alignment)?;
596                set_result(uc, a);
597            }
598            MallocTyped => {
599                let view = arg(uc, 0);
600                let size = read_u32(uc, view + 44)? & 0x00ff_ffff;
601                let a = self.allocate(uc, size as u64)?;
602                set_result(uc, a);
603            }
604            MallocTypedVariable => {
605                let size = arg(uc, 1);
606                let a = self.allocate(uc, size)?;
607                set_result(uc, a);
608            }
609            MallocLock => {
610                let a = self.allocate(uc, 8)?;
611                uc.mem_write(a, &[0u8; 8]).map_err(ue)?;
612                set_result(uc, a);
613            }
614            MallocGoodSize => set_result(uc, align(arg(uc, 0).max(1), 16)),
615            MallocSize => {
616                let a = arg(uc, 0);
617                set_result(uc, self.allocations.get(&a).map(|x| x.reserved).unwrap_or(0));
618            }
619            Calloc => {
620                let count = arg(uc, 0);
621                let size = arg(uc, 1);
622                if count != 0 && size > u64::MAX / count {
623                    return Err("allocation size overflows".into());
624                }
625                let total = count * size;
626                let a = self.allocate(uc, total)?;
627                if total != 0 {
628                    uc.mem_write(a, &vec![0u8; total as usize]).map_err(ue)?;
629                }
630                set_result(uc, a);
631            }
632            Realloc => {
633                let old = arg(uc, 0);
634                let new_size = arg(uc, 1);
635                if old == 0 {
636                    let a = self.allocate(uc, new_size)?;
637                    set_result(uc, a);
638                } else {
639                    let oldalloc = *self
640                        .allocations
641                        .get(&old)
642                        .ok_or_else(|| format!("reallocate unknown pointer {old:#x}"))?;
643                    if new_size <= oldalloc.reserved {
644                        let mut a = oldalloc;
645                        a.size = new_size;
646                        self.allocations.insert(old, a);
647                        set_result(uc, old);
648                    } else {
649                        let newaddr = self.allocate(uc, new_size)?;
650                        let data = uc.mem_read_as_vec(old, oldalloc.size as usize).map_err(ue)?;
651                        uc.mem_write(newaddr, &data).map_err(ue)?;
652                        self.release(uc, old)?;
653                        set_result(uc, newaddr);
654                    }
655                }
656            }
657            Free => {
658                let a = arg(uc, 0);
659                if a != 0 {
660                    self.release(uc, a)?;
661                }
662                set_result(uc, 0);
663            }
664            FreeTyped => {
665                let a = arg(uc, 1);
666                if a != 0 {
667                    self.release(uc, a)?;
668                }
669                set_result(uc, 0);
670            }
671            Mmap => {
672                let length = arg(uc, 1);
673                match self.allocate_aligned(uc, length, PAGE) {
674                    Ok(a) => {
675                        uc.mem_write(a, &vec![0u8; length as usize]).map_err(ue)?;
676                        set_result(uc, a);
677                    }
678                    Err(_) => set_result(uc, u64::MAX),
679                }
680            }
681            Munmap => {
682                let a = arg(uc, 0);
683                match self.release(uc, a) {
684                    Ok(()) => set_result(uc, 0),
685                    Err(_) => set_result(uc, u64::MAX),
686                }
687            }
688            VmAllocate => {
689                let field = arg(uc, 1);
690                let size = arg(uc, 2);
691                match self.allocate_aligned(uc, size, PAGE) {
692                    Ok(a) => {
693                        uc.mem_write(a, &vec![0u8; size as usize]).map_err(ue)?;
694                        write_u64(uc, field, a)?;
695                        set_result(uc, 0);
696                    }
697                    Err(_) => set_result(uc, 3), // KERN_NO_SPACE
698                }
699            }
700            VmDeallocate => {
701                let a = arg(uc, 1);
702                match self.release(uc, a) {
703                    Ok(()) => set_result(uc, 0),
704                    Err(_) => set_result(uc, 1), // KERN_INVALID_ADDRESS
705                }
706            }
707            Memmove => {
708                let dst = arg(uc, 0);
709                let src = arg(uc, 1);
710                let len = arg(uc, 2);
711                checked_size(len)?;
712                let data = uc.mem_read_as_vec(src, len as usize).map_err(ue)?;
713                uc.mem_write(dst, &data).map_err(ue)?;
714                set_result(uc, dst);
715            }
716            Memset => {
717                let dst = arg(uc, 0);
718                let value = arg(uc, 1) as u8;
719                let len = checked_size(arg(uc, 2))?;
720                uc.mem_write(dst, &vec![value; len]).map_err(ue)?;
721                set_result(uc, dst);
722            }
723            Bzero => {
724                let dst = arg(uc, 0);
725                let len = checked_size(arg(uc, 1))?;
726                uc.mem_write(dst, &vec![0u8; len]).map_err(ue)?;
727                set_result(uc, dst);
728            }
729            CheckedMemcpy => {
730                let len = arg(uc, 2);
731                let cap = arg(uc, 3);
732                if len > cap {
733                    return Err("checked copy exceeds destination".into());
734                }
735                return self.dispatch(uc, Memmove);
736            }
737            CheckedMemset => {
738                let len = arg(uc, 2);
739                let cap = arg(uc, 3);
740                if len > cap {
741                    return Err("checked fill exceeds destination".into());
742                }
743                return self.dispatch(uc, Memset);
744            }
745            Memcmp => {
746                let left = arg(uc, 0);
747                let right = arg(uc, 1);
748                let len = arg(uc, 2);
749                checked_size(len)?;
750                let a = uc.mem_read_as_vec(left, len as usize).map_err(ue)?;
751                let b = uc.mem_read_as_vec(right, len as usize).map_err(ue)?;
752                set_result(uc, cmp_to_u64(a.cmp(&b)));
753            }
754            Strcmp => {
755                let a = read_cstring(uc, arg(uc, 0));
756                let b = read_cstring(uc, arg(uc, 1));
757                set_result(uc, cmp_to_u64(a.as_bytes().cmp(b.as_bytes())));
758            }
759            Strncmp => {
760                let left = arg(uc, 0);
761                let right = arg(uc, 1);
762                let len = arg(uc, 2);
763                checked_size(len)?;
764                let a = uc.mem_read_as_vec(left, len as usize).map_err(ue)?;
765                let b = uc.mem_read_as_vec(right, len as usize).map_err(ue)?;
766                let mut result = 0u64;
767                for i in 0..len as usize {
768                    if a[i] != b[i] {
769                        result = (a[i] as i64 - b[i] as i64) as u64;
770                        break;
771                    }
772                    if a[i] == 0 {
773                        break;
774                    }
775                }
776                set_result(uc, result);
777            }
778            Strlen => {
779                let v = read_cstring(uc, arg(uc, 0));
780                set_result(uc, v.len() as u64);
781            }
782            Strnlen => {
783                let address = arg(uc, 0);
784                let limit = arg(uc, 1);
785                if limit > MAX_GUEST_TRANSFER {
786                    return Err(format!("strnlen limit {limit} too large"));
787                }
788                let data = uc.mem_read_as_vec(address, limit as usize).map_err(ue)?;
789                let n = data.iter().position(|&b| b == 0).unwrap_or(limit as usize);
790                set_result(uc, n as u64);
791            }
792            ReturnZero => set_result(uc, 0),
793            Unsupported => {
794                let pc = uc.reg_read(RegisterARM64::PC).map_err(ue)?;
795                let name = self.symbols.iter().find(|(_, a)| **a == pc).map(|(n, _)| n.as_str()).unwrap_or("?");
796                return Err(format!("guest called unsupported import {name}"));
797            },
798            Platform(PlatformHandler::Named(name)) => return self.dispatch_platform(uc, name),
799        }
800        Ok(())
801    }
802
803    pub(super) fn allocate(&mut self, uc: &mut Uc<'_>, size: u64) -> Result<u64, String> {
804        if size > MAX_GUEST_TRANSFER {
805            return Err(format!("allocation size {size} exceeds limit"));
806        }
807        let reserved = align(size.max(1), 16);
808        if let Some(index) = self.free_blocks.iter().position(|b| b.size >= reserved) {
809            let block = self.free_blocks[index];
810            let address = block.address;
811            if block.size == reserved {
812                self.free_blocks.remove(index);
813            } else {
814                self.free_blocks[index].address += reserved;
815                self.free_blocks[index].size -= reserved;
816            }
817            self.allocations.insert(address, GuestAllocation { size, reserved });
818            return Ok(address);
819        }
820        if self.heap_cursor > HEAP_SIZE || reserved > HEAP_SIZE - self.heap_cursor {
821            return Err("guest heap exhausted".into());
822        }
823        let address = HEAP_BASE + self.heap_cursor;
824        self.heap_cursor += reserved;
825        self.allocations.insert(address, GuestAllocation { size, reserved });
826        let _ = uc;
827        Ok(address)
828    }
829
830    pub(super) fn allocate_aligned(&mut self, _uc: &mut Uc<'_>, size: u64, alignment: u64) -> Result<u64, String> {
831        if size > MAX_GUEST_TRANSFER {
832            return Err(format!("allocation size {size} exceeds limit"));
833        }
834        let reserved = align(size.max(1), alignment);
835        let cursor = align(self.heap_cursor, alignment);
836        if cursor > HEAP_SIZE || reserved > HEAP_SIZE - cursor {
837            return Err("guest heap exhausted".into());
838        }
839        let address = HEAP_BASE + cursor;
840        self.heap_cursor = cursor + reserved;
841        self.allocations.insert(address, GuestAllocation { size, reserved });
842        Ok(address)
843    }
844
845    fn release(&mut self, uc: &mut Uc<'_>, address: u64) -> Result<(), String> {
846        let allocation = *self
847            .allocations
848            .get(&address)
849            .ok_or_else(|| format!("free unknown pointer {address:#x}"))?;
850        uc.mem_write(address, &vec![0u8; allocation.reserved as usize]).map_err(ue)?;
851        self.allocations.remove(&address);
852        self.free_blocks.push(FreeBlock { address, size: allocation.reserved });
853        self.coalesce_free_blocks();
854        Ok(())
855    }
856
857    fn coalesce_free_blocks(&mut self) {
858        self.free_blocks.sort_by_key(|b| b.address);
859        let mut merged: Vec<FreeBlock> = Vec::new();
860        for block in self.free_blocks.drain(..) {
861            if let Some(last) = merged.last_mut() {
862                if last.address + last.size == block.address {
863                    last.size += block.size;
864                    continue;
865                }
866            }
867            merged.push(block);
868        }
869        self.free_blocks = merged;
870        while let Some(last) = self.free_blocks.last().copied() {
871            if last.address + last.size != HEAP_BASE + self.heap_cursor {
872                break;
873            }
874            self.heap_cursor -= last.size;
875            self.free_blocks.pop();
876        }
877    }
878}

880pub(super) fn ue(e: unicorn_engine::unicorn_const::uc_error) -> String {

884pub(super) fn read_u32(uc: &Uc<'_>, address: u64) -> Result<u32, String> {
885    let b = uc.mem_read_as_vec(address, 4).map_err(ue)?;
886    Ok(u32::from_le_bytes(b.try_into().unwrap()))
887}
888
889pub(super) fn write_u64(uc: &mut Uc<'_>, address: u64, value: u64) -> Result<(), String> {
890    uc.mem_write(address, &value.to_le_bytes()).map_err(ue)
891}
892
893pub(super) fn checked_size(value: u64) -> Result<usize, String> {
894    if value > MAX_GUEST_TRANSFER {
895        return Err(format!("guest transfer size {value} exceeds limit"));
896    }
897    Ok(value as usize)
898}

900fn cmp_to_u64(ordering: std::cmp::Ordering) -> u64 {
901    match ordering {
902        std::cmp::Ordering::Less => (-1i64) as u64,
903        std::cmp::Ordering::Equal => 0,
904        std::cmp::Ordering::Greater => 1,
905    }
