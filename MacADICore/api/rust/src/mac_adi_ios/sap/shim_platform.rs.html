15fn read(uc: &Uc<'_>, a: u64, n: u64) -> Result<Vec<u8>, String> {
16    let n = checked_size(n)?;
17    if n == 0 { return Ok(Vec::new()); }
18    uc.mem_read_as_vec(a, n).map_err(ue)
19}
20fn write(uc: &mut Uc<'_>, a: u64, data: &[u8]) -> Result<(), String> {
21    if data.is_empty() { return Ok(()); }
22    uc.mem_write(a, data).map_err(ue)
23}
24fn u64_at(uc: &Uc<'_>, a: u64) -> Result<u64, String> {
25    Ok(u64::from_le_bytes(read(uc, a, 8)?.try_into().unwrap()))
26}
27fn w32(uc: &mut Uc<'_>, a: u64, v: u32) -> Result<(), String> { write(uc, a, &v.to_le_bytes()) }
28fn random(n: usize) -> Result<Vec<u8>, String> {
29    let mut data = vec![0; n];
30    getrandom::getrandom(&mut data).map_err(|e| e.to_string())?;
31    Ok(data)
32}

35    pub fn register_platform_services(&mut self) -> Result<(), Error> {
36        for (names, handler) in super::platform_table::SERVICES {
37            for name in *names {
38                self.add_function(name, Handler::Platform(PlatformHandler::Named(handler)))?;
39            }
40        }
41        self.state_mut().errno = self.add_data("guest.errno", &[0; 8])?;
42        self.add_data("___stack_chk_guard", &[0xa5, 0x71, 0x3c, 0xd9, 0x86, 0x42, 0xef, 0x10])?;
43        for name in ["_NDR_record", "_kCFAllocatorDefault", "_kCFAllocatorNull", "_kDADiskDescriptionVolumeUUIDKey", "_kIOMasterPortDefault"] {
44            self.add_data(name, &[0; 8])?;
45        }
46        for (name, value) in [("_kCFBooleanFalse", FAKE), ("_kCFBooleanTrue", FAKE), ("__xpc_type_dictionary", 0xffff100000000001), ("__xpc_type_error", 0xffff100000000002), ("__xpc_error_connection_invalid", 0xffff100000000003)] {
47            self.add_data(name, &value.to_le_bytes())?;
48        }
49        self.add_function("callback.complete", Handler::Platform(PlatformHandler::Named("callbackComplete")))?;
50        Ok(())
51    }
52    pub fn allocate(&mut self, n: u64) -> Result<u64, Error> {
53        let mut st = std::mem::take(self.uc.get_data_mut());
54        let result = st.allocate(&mut self.uc, n).map_err(Error::Emulation);
55        *self.uc.get_data_mut() = st;
56        result
57    }

61    pub fn new_xpc_object(&mut self, data: Option<Vec<u8>>) -> u64 {
62        self.xpc_next_object += 1;
63        self.xpc_objects.insert(self.xpc_next_object, XpcObject { has_data: data.is_some(), data: data.unwrap_or_default(), data_address: 0 });

66    fn function(&mut self, uc: &mut Uc<'_>, name: &str, handler: &'static str) -> Result<u64, String> {
67        if let Some(a) = self.symbols.get(name) { return Ok(*a); }
68        let a = self.code_cursor;
69        if a + SHIM_SLOT > SHIM_BASE + SHIM_CODE_SIZE { return Err("service code area full".into()); }
70        write(uc, a, &0xd65f03c0u32.to_le_bytes())?;
71        self.code_cursor += SHIM_SLOT;
72        self.handlers.insert(a, Handler::Platform(PlatformHandler::Named(handler)));
73        self.symbols.insert(name.into(), a);
74        Ok(a)
75    }
76    fn data(&mut self, uc: &mut Uc<'_>, name: &str, data: &[u8]) -> Result<u64, String> {
77        if let Some(a) = self.symbols.get(name) { return Ok(*a); }
78        let a = align(self.data_cursor, 8);
79        if a + data.len() as u64 > SHIM_BASE + SHIM_SIZE { return Err("service data area full".into()); }
80        write(uc, a, data)?;
81        self.data_cursor = a + data.len().max(8) as u64;
82        self.symbols.insert(name.into(), a);
83        Ok(a)
84    }
85    fn callback(&mut self, uc: &mut Uc<'_>, entry: u64) -> Result<(), String> {
86        self.callback_returns.push(uc.reg_read(RegisterARM64::X30).map_err(ue)?);
87        uc.reg_write(RegisterARM64::X30, self.symbols["callback.complete"]).map_err(ue)?;
88        uc.reg_write(RegisterARM64::PC, entry).map_err(ue)
89    }
90    fn descriptor(&mut self, uc: &mut Uc<'_>, buffer: u64, capacity: u64) -> Result<u64, String> {
91        if self.buffer_vtable == 0 {
92            let unhandled = self.function(uc, "$buffer.unhandled", "bufferUnhandled")?;
93            let mut table = unhandled.to_le_bytes().repeat(0x1000 / 8);
94            for (offset, handler) in [(0x28,"returnZero"),(0xc8,"bufferRead"),(0xd0,"bufferWrite"),(0xd8,"bufferPrepare"),(0xe0,"returnZero"),(0x148,"bufferAddress"),(0x218,"bufferPrepare"),(0x220,"returnZero"),(0x308,"bufferAddress")] {
95                let a = self.function(uc, &format!("$buffer.{handler}"), handler)?;
96                table[offset..offset+8].copy_from_slice(&a.to_le_bytes());
97            }
98            self.buffer_vtable = self.data(uc, "kernel.iobuffer.vtable", &table)?;
99        }
100        let a = self.allocate(uc, 24)?;
101        write_u64(uc, a, self.buffer_vtable)?;
102        write_u64(uc, a+8, buffer)?;
103        write_u64(uc, a+16, capacity)?;
104        Ok(a)
105    }

334                    let data=read(uc,source,a[3])?;write(uc,target,&data)?;a[3]
335                }
336            },
337            "ioAESPerform"=>{
338                let n=a[3];
339                if n==0||n>MAX_GUEST_TRANSFER||n%16!=0 {3}else{
340                    let source=u64_at(uc,a[1]+8)?;let source_size=u64_at(uc,a[1]+16)?;
341                    let target=u64_at(uc,a[2]+8)?;let target_size=u64_at(uc,a[2]+16)?;
342                    if a[7]>source_size||n>source_size-a[7]||a[8]>target_size||n>target_size-a[8] {3}else{
343                        let input=read(uc,source+a[7],n)?;let iv=read(uc,a[4],16)?;let key=read(uc,a[6],40)?;
344                        let (out,iv)=emulate_ioaes(&input,&iv,a[5] as u32,&key)?;
345                        write(uc,target+a[8],&out)?;write(uc,a[4],&iv)?;0
346                    }
347                }
348            },
349            _ => return Err(format!("unknown platform service {name}")),
350        };
351        set_result(uc,result);
352        Ok(())
353    }
354}

356pub fn emulate_ioaes(source: &[u8], iv: &[u8], operation: u32, key: &[u8]) -> Result<(Vec<u8>, Vec<u8>), String> {
357    if source.is_empty() || source.len()%16!=0 || iv.len()!=16 || key.len()!=40 {return Err("invalid IOAES parameters".into());}
358    if u32::from_le_bytes(key[36..40].try_into().unwrap())!=0x89c {return Err("unsupported IOAES key handle".into());}
359    if operation>1 {return Err("unsupported IOAES operation".into());}
360    let cipher=Aes128::new(GenericArray::from_slice(&[0;16]));
361    let mut chain=iv.to_vec();let mut output=Vec::with_capacity(source.len());
362    for input in source.chunks_exact(16) {
363        let mut block=GenericArray::clone_from_slice(input);
364        if operation==0 {
365            for i in 0..16 {block[i]^=chain[i];} cipher.encrypt_block(&mut block); chain.copy_from_slice(&block);
366        }else{
367            cipher.decrypt_block(&mut block);for i in 0..16 {block[i]^=chain[i];}chain.copy_from_slice(input);
368        }
369        output.extend_from_slice(&block);
370    }
371    Ok((output,chain))
