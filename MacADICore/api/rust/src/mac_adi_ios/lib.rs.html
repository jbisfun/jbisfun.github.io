15#[derive(Debug)]
16pub enum Error {
17    MachO(String),
18    Emulation(String),
19    Unicorn(unicorn_engine::unicorn_const::uc_error),
20    UnknownBuild(String),
21    /// The library answered with a negative status (`op` is the ADI call).
22    Adi { op: &'static str, ret: i32 },
23    NotProvisioned,
24    AlreadyProvisioned,
25}
26
27impl std::fmt::Display for Error {
28    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
29        match self {
30            Error::MachO(m) => write!(f, "mach-o: {m}"),
31            Error::Emulation(m) => write!(f, "emulation: {m}"),
32            Error::Unicorn(e) => write!(f, "unicorn: {e:?}"),
33            Error::Adi { op, ret } => write!(f, "{op} failed: {ret}"),
34            Error::NotProvisioned => write!(f, "machine is not provisioned"),
35            Error::AlreadyProvisioned => write!(f, "machine is already provisioned"),
36            Error::UnknownBuild(h) => {
37                write!(f, "unknown CoreADI build sha256 {h}; pass its _vdfut768ig offset explicitly")
38            }
39        }
40    }
41}
42
43impl std::error::Error for Error {}
44
45impl From<unicorn_engine::unicorn_const::uc_error> for Error {
46    fn from(e: unicorn_engine::unicorn_const::uc_error) -> Self {
47        Error::Unicorn(e)
48    }
49}

65/// `(name, dispatcher offset)` of a CoreADI image, by the sha256 of its bytes.
66pub fn identify_build(image: &[u8]) -> Result<(&'static str, u64), Error> {
67    let digest: String = Sha256::digest(image).iter().map(|b| format!("{b:02x}")).collect();
68    BUILDS
69        .iter()
70        .find(|(h, _, _)| *h == digest)
71        .map(|(_, n, o)| (*n, *o))
72        .ok_or(Error::UnknownBuild(digest))
73}
