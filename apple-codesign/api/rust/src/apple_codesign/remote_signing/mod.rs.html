62    #[error("bad client state: {0}")]
63    ClientState(&'static str),
64
65    #[error("joining state not wanted for this session type: {0}")]
66    SessionJoinUnwantedState(String),
67
68    #[error("session join string error: {0}")]

71    #[error("base64 decode error: {0}")]
72    Base64(#[from] base64::DecodeError),
73
74    #[error("I/O error: {0}")]
75    Io(#[from] std::io::Error),
76
77    #[error("PEM encoding error: {0}")]
78    Pem(#[from] pem::PemError),
79
80    #[error("JSON serialization error: {0}")]

82
83    #[error("SPAKE error: {0}")]

85
86    #[error("SPKI error: {0}")]

88
89    #[error("websocket error: {0}")]

152/// Websocket message sent from server to client.
153#[derive(Clone, Debug, Deserialize)]
154struct ServerMessage {
155    /// ID of request responsible for this message.
156    request_id: Option<String>,
157    /// The type of message.
158    #[serde(rename = "type")]
159    typ: ServerMessageType,
160    ttl: Option<u64>,
161    payload: Option<serde_json::Value>,
162}
163
164impl ServerMessage {
165    fn into_result(self) -> Result<Self, RemoteSignError> {
166        if self.typ == ServerMessageType::Error {
167            let error = self.as_error()?;
168            Err(RemoteSignError::ServerError(format!(
169                "{}: {}",
170                error.code, error.message
171            )))
172        } else {
173            Ok(self)
174        }
175    }
176
177    fn as_type<T: DeserializeOwned>(
178        &self,
179        message_type: ServerMessageType,
180    ) -> Result<T, RemoteSignError> {
181        if self.typ == message_type {
182            if let Some(value) = &self.payload {
183                Ok(serde_json::from_value(value.clone())?)
184            } else {
185                Err(RemoteSignError::ClientState(
186                    "no payload for requested type",
187                ))
188            }
189        } else {
190            Err(RemoteSignError::ClientState(
191                "requested payload for wrong message type",
192            ))
193        }
194    }
195
196    fn as_error(&self) -> Result<ServerError, RemoteSignError> {
197        self.as_type::<ServerError>(ServerMessageType::Error)
198    }
199
200    fn as_greeting(&self) -> Result<ServerGreeting, RemoteSignError> {
201        self.as_type::<ServerGreeting>(ServerMessageType::Greeting)
202    }
203
204    fn as_session_joined(&self) -> Result<ServerJoined, RemoteSignError> {
205        self.as_type::<ServerJoined>(ServerMessageType::SessionJoined)
206    }
207
208    fn as_peer_message(&self) -> Result<ServerPeerMessage, RemoteSignError> {
209        self.as_type::<ServerPeerMessage>(ServerMessageType::PeerMessage)
210    }
211
212    fn as_session_closed(&self) -> Result<ServerSessionClosed, RemoteSignError> {
213        self.as_type::<ServerSessionClosed>(ServerMessageType::SessionClosed)
214    }
215}

353fn create_websocket(
354    req: impl IntoClientRequest,
355) -> Result<WebSocket<MaybeTlsStream<TcpStream>>, RemoteSignError> {
356    let config = WebSocketConfig {
357        ..Default::default()
358    };
359
360    let req = req.into_client_request()?;
361    warn!("connecting to {}", req.uri());
362
363    let (ws, _) = tungstenite::client::connect_with_config(req, Some(config), 5)?;
364
365    Ok(ws)
366}
367
368fn wait_for_server_response(
369    ws: &mut WebSocket<MaybeTlsStream<TcpStream>>,
370) -> Result<ServerResponse, RemoteSignError> {
371    loop {
372        match ws.read()? {
373            Message::Text(text) => {
374                let message = serde_json::from_str::<ServerMessage>(&text)?;
375                debug!(
376                    "received message; request-id: {}; type: {:?}",
377                    message
378                        .request_id
379                        .as_ref()
380                        .unwrap_or(&"(not set)".to_string()),
381                    message.typ
382                );
383
384                return Ok(ServerResponse::Message(message));
385            }
386            Message::Binary(_) => {
387                return Err(RemoteSignError::ServerUnexpectedMessage(
388                    "binary websocket message".into(),
389                ))
390            }
391            // TODO return error for these?
392            Message::Pong(_) => {}
393            Message::Ping(_) => {}
394            Message::Frame(_) => {}
395            Message::Close(_) => {
396                return Ok(ServerResponse::Closed);
397            }
398        }
399    }
400}
401
402fn wait_for_server_message(
403    ws: &mut WebSocket<MaybeTlsStream<TcpStream>>,
404) -> Result<ServerMessage, RemoteSignError> {
405    match wait_for_server_response(ws)? {
406        ServerResponse::Closed => Err(RemoteSignError::ClientState("server closed connection")),
407        ServerResponse::Message(m) => {
408            debug!(
409                "received server message {:?}; remaining session TTL: {}",
410                m.typ,
411                m.ttl.unwrap_or_default()
412            );
413            Ok(m)
414        }
415    }
416}
417
418fn wait_for_expected_server_message(
419    ws: &mut WebSocket<MaybeTlsStream<TcpStream>>,
420    message_type: ServerMessageType,
421) -> Result<ServerMessage, RemoteSignError> {
422    let res = wait_for_server_message(ws)?.into_result()?;
423
424    if res.typ == message_type {
425        Ok(res)
426    } else {
427        Err(RemoteSignError::ServerUnexpectedMessage(format!(
428            "{:?}",
429            res.typ
430        )))
431    }
432}

434/// A client for the remote signing protocol that has not yet joined a session.
435///
436/// Clients can perform both the initiator and signer roles.

439}

441impl UnjoinedSigningClient {
442    fn new(req: impl IntoClientRequest) -> Result<Self, RemoteSignError> {
443        let ws = create_websocket(req)?;
444
445        let mut slf = Self { ws };
446
447        slf.send_hello()?;
448
449        Ok(slf)
450    }

452    /// Create a new client in the initiator role.
453    pub fn new_initiator(
454        req: impl IntoClientRequest,
455        initiator: Box<dyn SessionInitiatePeer>,
456        session_info_cb: Option<SessionInfoCallback>,
457    ) -> Result<InitiatorClient, RemoteSignError> {
458        let slf = Self::new(req)?;
459        slf.create_session_and_wait_for_signer(initiator, session_info_cb)
460    }
461
462    /// Create a new client in the signer role.
463    pub fn new_signer(
464        joiner: Box<dyn SessionJoinPeerPreJoin>,
465        signing_key: &dyn KeyInfoSigner,
466        signing_cert: CapturedX509Certificate,
467        certificates: Vec<CapturedX509Certificate>,
468        default_server_url: String,
469    ) -> Result<SigningClient<'_>, RemoteSignError> {
470        // An error here could result in the peer hanging indefinitely because the session
471        // is unjoined. Ideally we'd recover from this by attempting to join with an error.
472        // However, we may not even be able to obtain the session ID since sometimes it is
473        // encrypted and the error could be from a decryption failure! So for now, just let
474        // the peer idle.
475        let join_context = joiner.join_context()?;
476
477        let server_url = join_context
478            .server_url
479            .as_ref()
480            .unwrap_or(&default_server_url);
481
482        let slf = Self::new(server_url)?;
483        slf.join_session(join_context, signing_key, signing_cert, certificates)
484    }
485
486    /// Create a new signing session and wait for a signer to arrive.
487    fn create_session_and_wait_for_signer(
488        mut self,
489        initiator: Box<dyn SessionInitiatePeer>,
490        session_info_cb: Option<SessionInfoCallback>,
491    ) -> Result<InitiatorClient, RemoteSignError> {
492        let session_id = initiator.session_id().to_string();
493
494        self.send_request(
495            ApiMethod::CreateSession,
496            Some(ClientPayload::CreateSession {
497                session_id: session_id.clone(),
498                ttl: 600,
499                context: initiator
500                    .session_create_context()
501                    .map(|x| STANDARD_ENGINE.encode(x)),
502            }),
503        )?;
504
505        let sjs_base64 = initiator.session_join_string_base64()?;
506        let sjs_pem = initiator.session_join_string_pem()?;
507
508        wait_for_expected_server_message(&mut self.ws, ServerMessageType::SessionCreated)?;
509        warn!("session successfully created on server");
510
511        if let Some(cb) = session_info_cb {
512            cb(&sjs_base64, &sjs_pem)?;
513        }
514
515        let res = wait_for_expected_server_message(&mut self.ws, ServerMessageType::SessionJoined)?;
516
517        let joined = res.as_session_joined()?;
518        warn!("signer joined session; deriving shared encryption key");
519
520        let context = if let Some(context) = joined.context {
521            Some(STANDARD_ENGINE.decode(context)?)
522        } else {
523            None
524        };
525
526        let keys = initiator.negotiate_session(context)?;
527
528        let mut client = PairedClient {
529            ws: self.ws,
530            session_id,
531            keys,
532        };
533
534        client.send_ping()?;
535
536        let (signing_cert, signing_chain) = client.request_signing_certificate()?;
537
538        if let Some(name) = signing_cert.subject_common_name() {
539            warn!("remote signer will sign with certificate: {}", name);
540        }
541
542        Ok(InitiatorClient {
543            client: RefCell::new(client),
544            signing_cert,
545            signing_chain,
546        })
547    }

549    /// Join a signing session.
550    ///
551    /// This should be called by signers once they have the session ID to join.
552    pub fn join_session(
553        mut self,
554        join_context: SessionJoinContext,
555        signing_key: &dyn KeyInfoSigner,
556        signing_cert: CapturedX509Certificate,
557        certificates: Vec<CapturedX509Certificate>,
558    ) -> Result<SigningClient<'_>, RemoteSignError> {
559        let session_id = join_context.session_id.clone();
560
561        warn!("joining session...");
562        self.send_request(
563            ApiMethod::JoinSession,
564            Some(ClientPayload::JoinSession {
565                session_id: session_id.clone(),
566                context: join_context.peer_context.map(|x| STANDARD_ENGINE.encode(x)),
567            }),
568        )?;
569
570        wait_for_expected_server_message(&mut self.ws, ServerMessageType::SessionJoined)?;
571
572        warn!("successfully joined signing session {}", session_id);
573
574        let keys = join_context.peer_handshake.negotiate_session()?;
575
576        let mut client = PairedClient {
577            ws: self.ws,
578            session_id,
579            keys,
580        };
581
582        warn!("verifying encrypted communications with peer");
583        client.send_ping()?;
584
585        Ok(SigningClient {
586            client: RefCell::new(client),
587            signing_key,
588            signing_cert,
589            certificates,
590        })
591    }
592
593    fn send_request(
594        &mut self,
595        api: ApiMethod,
596        payload: Option<ClientPayload>,
597    ) -> Result<(), RemoteSignError> {
598        let request_id = uuid::Uuid::new_v4().to_string();
599
600        let message = ClientMessage {
601            request_id,
602            api,
603            payload,
604        };
605
606        let body = serde_json::to_string(&message)?;
607        self.ws.send(body.into())?;
608        self.ws.flush()?;
609
610        Ok(())
611    }
612
613    fn send_hello(&mut self) -> Result<(), RemoteSignError> {
614        self.send_request(ApiMethod::Hello, None)?;
615
616        let res = wait_for_expected_server_message(&mut self.ws, ServerMessageType::Greeting)?;
617        let greeting = res.as_greeting()?;
618
619        if let Some(motd) = &greeting.motd {
620            warn!("message from remote server: {}", motd);
621        }
622
623        for required in REQUIRED_ACTIONS {
624            if !greeting.apis.contains(&required.to_string()) {
625                error!("server does not support required action {}", required);
626                return Err(RemoteSignError::ServerIncompatible);
627            }
628        }
629
630        Ok(())
631    }
632}

634/// A remote signing client that has joined a session and is ready to exchange messages.
635pub struct PairedClient {
636    ws: WebSocket<MaybeTlsStream<TcpStream>>,
637    session_id: String,
638    keys: PeerKeys,
639}
640
641impl Drop for PairedClient {
642    fn drop(&mut self) {
643        warn!("disconnecting from relay server");
644    }
645}
646
647impl PairedClient {
648    fn send_request(
649        &mut self,
650        api: ApiMethod,
651        payload: Option<ClientPayload>,
652    ) -> Result<(), RemoteSignError> {
653        let request_id = uuid::Uuid::new_v4().to_string();
654
655        let message = ClientMessage {
656            request_id,
657            api,
658            payload,
659        };
660
661        let body = serde_json::to_string(&message)?;
662        self.ws.send(body.into())?;
663        self.ws.flush()?;
664
665        Ok(())
666    }
667
668    fn decrypt_peer_message(
669        &mut self,
670        message: &ServerPeerMessage,
671    ) -> Result<PeerMessage, RemoteSignError> {
672        let ciphertext = STANDARD_ENGINE.decode(&message.message)?;
673
674        let plaintext = self.keys.open(ciphertext)?;
675
676        Ok(serde_json::from_slice(&plaintext)?)
677    }
678
679    fn send_encrypted_message(
680        &mut self,
681        message_type: PeerMessageType,
682        payload: Option<PeerPayload>,
683    ) -> Result<(), RemoteSignError> {
684        let message = PeerMessage {
685            typ: message_type,
686            payload: if let Some(payload) = payload {
687                Some(serde_json::to_value(payload)?)
688            } else {
689                None
690            },
691        };
692
693        let ciphertext = self.keys.seal(&serde_json::to_vec(&message)?)?;
694
695        self.send_request(
696            ApiMethod::SendMessage,
697            Some(ClientPayload::SendMessage {
698                session_id: self.session_id.clone(),
699                message: STANDARD_ENGINE.encode(ciphertext),
700            }),
701        )?;
702
703        Ok(())
704    }
705
706    fn wait_for_peer_message(&mut self) -> Result<Option<PeerMessage>, RemoteSignError> {
707        let res = wait_for_server_message(&mut self.ws)?.into_result()?;
708
709        if let Ok(closed) = res.as_session_closed() {
710            warn!(
711                "signing session closed; reason: {}",
712                closed
713                    .reason
714                    .as_ref()
715                    .unwrap_or(&"(none given)".to_string())
716            );
717            Ok(None)
718        } else {
719            let message = res.as_peer_message()?;
720
721            Ok(Some(self.decrypt_peer_message(&message)?))
722        }
723    }
724
725    fn wait_for_server_and_peer_response(&mut self) -> Result<PeerMessage, RemoteSignError> {
726        let mut response = None;
727
728        // We should get a server message acknowledging our request plus the response from
729        // the peer. The order they arrive in is random.
730        for _ in 0..2 {
731            let res = wait_for_server_message(&mut self.ws)?.into_result()?;
732
733            match res.typ {
734                ServerMessageType::MessageSent => {}
735                ServerMessageType::PeerMessage => {
736                    let message = res.as_peer_message()?;
737
738                    response = Some(self.decrypt_peer_message(&message)?);
739                }
740                m => return Err(RemoteSignError::ServerUnexpectedMessage(format!("{m:?}"))),
741            }
742        }
743
744        if let Some(response) = response {
745            Ok(response)
746        } else {
747            Err(RemoteSignError::ClientState(
748                "failed to receive response from server or peer",
749            ))
750        }
751    }
752
753    fn send_goodbye(&mut self, reason: Option<String>) -> Result<(), RemoteSignError> {
754        warn!("terminating signing session on relay");
755        self.send_request(
756            ApiMethod::Goodbye,
757            Some(ClientPayload::Goodbye {
758                session_id: self.session_id.clone(),
759                reason,
760            }),
761        )?;
762
763        wait_for_server_message(&mut self.ws)?.into_result()?;
764        warn!("relay server confirmed session termination");
765
766        Ok(())
767    }
768
769    fn send_ping(&mut self) -> Result<(), RemoteSignError> {
770        // We should get a server message acknowledging our request plus a
771        // ping from the peer. The order may not be reliable.
772        self.send_encrypted_message(PeerMessageType::Ping, None)?;
773        let message = self.wait_for_server_and_peer_response()?;
774        if !matches!(message.typ, PeerMessageType::Ping) {
775            return Err(RemoteSignError::ServerUnexpectedMessage(
776                "unexpected response to ping message".into(),
777            ));
778        }
779
780        self.send_encrypted_message(PeerMessageType::Pong, None)?;
781        let message = self.wait_for_server_and_peer_response()?;
782        if !matches!(message.typ, PeerMessageType::Pong) {
783            return Err(RemoteSignError::ServerUnexpectedMessage(
784                "unexpected response to ping message".into(),
785            ));
786        }
787
788        Ok(())
789    }
790
791    /// Request the signing certificate from the peer.
792    pub fn request_signing_certificate(
793        &mut self,
794    ) -> Result<(CapturedX509Certificate, Vec<CapturedX509Certificate>), RemoteSignError> {
795        warn!("requesting signing certificate info from signer");
796        self.send_encrypted_message(PeerMessageType::RequestSigningCertificate, None)?;
797        let res = self
798            .wait_for_server_and_peer_response()?
799            .require_type(PeerMessageType::SigningCertificate)?;
800
801        let cert = res.as_signing_certificate()?;
802
803        if let Some(cert) = cert.certificates.get(0) {
804            let cert_der = STANDARD_ENGINE.decode(&cert.certificate)?;
805            let chain_der = cert
806                .chain
807                .iter()
808                .map(|x| STANDARD_ENGINE.decode(x))
809                .collect::<Result<Vec<_>, base64::DecodeError>>()?;
810
811            let cert = CapturedX509Certificate::from_der(cert_der)?;
812            let chain = chain_der
813                .into_iter()
814                .map(CapturedX509Certificate::from_der)
815                .collect::<Result<Vec<_>, X509CertificateError>>()?;
816
817            return Ok((cert, chain));
818        }
819
820        Err(RemoteSignError::ClientState(
821            "did not receive any signing certificates from peer",
822        ))
823    }
824}

845impl Signer<Signature> for InitiatorClient {
846    fn try_sign(&self, message: &[u8]) -> Result<Signature, signature::Error> {
847        let mut client = self.client.borrow_mut();
848
849        warn!("sending signing request to remote signer");
850
851        client
852            .send_encrypted_message(
853                PeerMessageType::SignRequest,
854                Some(PeerPayload::SignRequest(PeerSignRequest {
855                    message: STANDARD_ENGINE.encode(message),
856                })),
857            )
858            .map_err(signature::Error::from_source)?;
859
860        let response = client
861            .wait_for_server_and_peer_response()
862            .map_err(signature::Error::from_source)?
863            .require_type(PeerMessageType::Signature)
864            .map_err(signature::Error::from_source)?;
865
866        let peer_signature = response
867            .as_signature()
868            .map_err(signature::Error::from_source)?;
869
870        warn!("received signature from remote signer");
871
872        let signature = STANDARD_ENGINE
873            .decode(&peer_signature.signature)
874            .map_err(signature::Error::from_source)?;
875        let oid_der = STANDARD_ENGINE
876            .decode(&peer_signature.algorithm_oid)
877            .map_err(signature::Error::from_source)?;
878
879        bcder::decode::Constructed::decode(oid_der.as_ref(), Mode::Der, |cons| {
880            Oid::take_from(cons)
881        })
882        .map_err(|_| {
883            signature::Error::from_source(RemoteSignError::Crypto(
884                "error parsing signature OID".into(),
885            ))
886        })?;
887
888        // The peer could be acting maliciously (or just be buggy) and sign with a
889        // certificate from the initial one presented. So verify the signature we
890        // received is valid for the message we sent.
891        if let Err(e) = self.signing_cert.verify_signed_data(message, &signature) {
892            error!("Peer issued signature did not verify against the certificate they provided");
893            error!("The peer could be acting maliciously. Or it could just be buggy.");
894            error!("Either way, it didn't issue a valid signature, so we're giving up.");
895
896            return Err(signature::Error::from_source(e));
897        }
898
899        Ok(signature.into())
900    }
901}

975pub struct SigningClient<'key> {
976    client: RefCell<PairedClient>,
977    signing_key: &'key dyn KeyInfoSigner,
978    signing_cert: CapturedX509Certificate,
979    certificates: Vec<CapturedX509Certificate>,
980}
981
982impl<'key> SigningClient<'key> {
983    fn send_signing_certificate(
984        &self,
985        mut client: RefMut<PairedClient>,
986    ) -> Result<(), RemoteSignError> {
987        client.send_encrypted_message(
988            PeerMessageType::SigningCertificate,
989            Some(PeerPayload::SigningCertificate(PeerSigningCertificate {
990                certificates: vec![PeerCertificate {
991                    certificate: STANDARD_ENGINE.encode(self.signing_cert.encode_der()?),
992                    chain: self
993                        .certificates
994                        .iter()
995                        .map(|cert| {
996                            let der = cert.encode_der()?;
997
998                            Ok(STANDARD_ENGINE.encode(der))
999                        })
1000                        .collect::<Result<Vec<_>, RemoteSignError>>()?,
1001                }],
1002            })),
1003        )?;
1004
1005        wait_for_expected_server_message(&mut client.ws, ServerMessageType::MessageSent)?;
1006
1007        Ok(())
1008    }
1009
1010    fn handle_sign_request(
1011        &self,
1012        mut client: RefMut<PairedClient>,
1013        request: PeerSignRequest,
1014    ) -> Result<(), RemoteSignError> {
1015        let message = STANDARD_ENGINE.decode(&request.message)?;
1016
1017        warn!(
1018            "creating signature for remote message: {}",
1019            &request.message
1020        );
1021        let signature = self
1022            .signing_key
1023            .try_sign(&message)
1024            .map_err(|e| RemoteSignError::Crypto(format!("when creating signature: {e}")))?;
1025        let algorithm = self.signing_key.signature_algorithm()?;
1026
1027        let oid = Oid::from(algorithm);
1028        let mut oid_der = vec![];
1029        oid.encode().write_encoded(Mode::Der, &mut oid_der)?;
1030
1031        warn!("sending signature to peer");
1032        client.send_encrypted_message(
1033            PeerMessageType::Signature,
1034            Some(PeerPayload::Signature(PeerSignature {
1035                message: STANDARD_ENGINE.encode(message),
1036                signature: STANDARD_ENGINE.encode(signature),
1037                algorithm_oid: STANDARD_ENGINE.encode(oid_der),
1038            })),
1039        )?;
1040
1041        wait_for_expected_server_message(&mut client.ws, ServerMessageType::MessageSent)?;
1042        warn!("relay acknowledged signature message received");
1043
1044        Ok(())
1045    }
1046
1047    fn process_next_message(&self) -> Result<bool, RemoteSignError> {
1048        let mut client = self.client.borrow_mut();
1049
1050        warn!("waiting for server to send us a message...");
1051        let res = if let Some(res) = client.wait_for_peer_message()? {
1052            res
1053        } else {
1054            return Ok(false);
1055        };
1056
1057        match res.typ {
1058            PeerMessageType::RequestSigningCertificate => {
1059                self.send_signing_certificate(client)?;
1060            }
1061            PeerMessageType::Ping => {
1062                client.send_encrypted_message(PeerMessageType::Pong, None)?;
1063                wait_for_expected_server_message(&mut client.ws, ServerMessageType::MessageSent)?;
1064            }
1065            PeerMessageType::Pong => {}
1066            PeerMessageType::SignRequest => {
1067                self.handle_sign_request(client, res.as_sign_request()?)?;
1068            }
1069            typ => {
1070                warn!("unprocessed message: {:?}", typ);
1071            }
1072        }
1073
1074        Ok(true)
1075    }
1076
1077    pub fn run(self) -> Result<(), RemoteSignError> {
1078        while self.process_next_message()? {}
1079
1080        Ok(())
1081    }
