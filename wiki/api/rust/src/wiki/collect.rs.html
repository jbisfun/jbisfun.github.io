12/// One repo's table in wiki.toml.
13#[derive(Default, Clone, Serialize, Deserialize)]
14#[serde(default)]
15pub struct RepoConfig {
16    /// Leave the repo out of the wiki.
17    pub skip: bool,
18    /// Group label for the wiki header's dropdowns; repos without one land under "Other".
19    pub category: Option<String>,
20    /// One line for the home page; defaults to the README's first sentence.
21    pub description: Option<String>,
22    pub rust: Tool,
23    pub lua: Tool,
24    pub js: Tool,
25}

27/// `tool = false` turns a doc tool off; a table configures it.
28#[derive(Clone, Serialize, Deserialize)]
29#[serde(untagged)]
30pub enum Tool {
31    Enabled(bool),
32    Settings(ToolSettings),
33}
34
35impl Default for Tool {
36    fn default() -> Self {
37        Tool::Enabled(true)
38    }
39}
40
41impl Tool {
42    /// The settings, or None when the tool is turned off.
43    pub fn settings(&self) -> Option<ToolSettings> {
44        match self {
45            Tool::Enabled(false) => None,
46            Tool::Enabled(true) => Some(ToolSettings::default()),
47            Tool::Settings(settings) => Some(settings.clone()),
48        }
49    }
50}

52#[derive(Default, Clone, Serialize, Deserialize)]
53#[serde(default)]
54pub struct ToolSettings {
55    /// rust: cargo packages (`-p`); default: the workspace's default members.
56    pub packages: Vec<String>,
57    /// lua/js: pathspecs of tracked files; default: every tracked file of the language.
58    pub files: Vec<String>,

93/// Every cloned repo that isn't skipped (wiki.toml `skip` or `$WIKI_SKIP`), sorted by name.
94pub fn collect(paths: &Paths) -> Result<Vec<Repo>> {
95    let toml_path = paths.wiki.join("wiki.toml");
96    let config: WikiToml = toml::from_str(&fs::read_to_string(&toml_path)?)
97        .with_context(|| format!("parsing {}", toml_path.display()))?;
98    let skipped: Vec<String> = env::var("WIKI_SKIP")
99        .unwrap_or_default()
100        .split(',')
101        .filter(|name| !name.is_empty())
102        .map(str::to_owned)
103        .collect();
104
105    let output = std::process::Command::new("pull")
106        .args(["status", "--json"])
107        .current_dir(&paths.root)
108        .output()
109        .context("running `pull status --json`")?;
110    if !output.status.success() {
111        bail!("pull status --json: {}", String::from_utf8_lossy(&output.stderr));
112    }
113    let status: Status = serde_json::from_slice(&output.stdout).context("parsing `pull status --json`")?;
114
115    let github = Regex::new(r"github\.com[:/]([^/]+/[^/]+?)(\.git)?/?$")?;
116    let mut repos = Vec::new();
117    for repo in status.repos {
118        let config = config.repos.get(&repo.name).cloned().unwrap_or_default();
119        if !repo.cloned || config.skip || skipped.contains(&repo.name) {
120            continue;
121        }
122        let readme = fs::read_dir(&repo.path)?
123            .filter_map(Result::ok)
124            .filter(|entry| entry.file_type().is_ok_and(|kind| kind.is_file()))
125            .map(|entry| entry.file_name().to_string_lossy().into_owned())
126            .find(|name| name.eq_ignore_ascii_case("readme.md"));
127        let web = git2::Repository::open(&repo.path)
128            .ok()
129            .and_then(|git| git.find_remote("origin").ok().and_then(|remote| remote.url().ok().map(str::to_owned)))
130            .and_then(|url| github.captures(&url).map(|found| format!("https://github.com/{}", &found[1])));
131        repos.push(Repo {
132            name: repo.name,
133            path: repo.path,
134            branch: repo.branch.unwrap_or_else(|| "main".into()),
135            head: repo.head.unwrap_or_default(),
136            web,
137            readme,
138            config,
139        });
140    }
141    repos.sort_by_key(|repo| repo.name.to_lowercase());
142    Ok(repos)
143}
