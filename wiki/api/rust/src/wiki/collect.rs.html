12/// One repo's table in wiki.toml.
13#[derive(Default, Clone, Serialize, Deserialize)]
14#[serde(default)]
15pub struct RepoConfig {
16    /// Leave the repo out of the wiki.
17    pub skip: bool,
18    /// Display name in the header, sidebar and home page; default: the repo
19    /// name without its pull group (`notes-ios` for `fossify-ios/notes-ios`).
20    pub title: Option<String>,
21    /// Group label for the wiki header's dropdowns; default: the pull group
22    /// (`fossify-ios` for `fossify-ios/notes-ios`), else "Other".
23    pub category: Option<String>,
24    /// One line for the home page; defaults to the README's first sentence.
25    pub description: Option<String>,
26    pub rust: Tool,
27    pub lua: Tool,
28    pub js: Tool,
29    /// Machine-translate the README and docs into the wiki.toml `[i18n]`
30    /// languages; `{ files = [..] }` limits it to matching paths.
31    pub translate: Tool,
32}

34/// `tool = false` turns a doc tool off; a table configures it.
35#[derive(Clone, Serialize, Deserialize)]
36#[serde(untagged)]
37pub enum Tool {
38    Enabled(bool),
39    Settings(ToolSettings),
40}
41
42impl Default for Tool {
43    fn default() -> Self {
44        Tool::Enabled(true)
45    }
46}
47
48impl Tool {
49    /// The settings, or None when the tool is turned off.
50    pub fn settings(&self) -> Option<ToolSettings> {
51        match self {
52            Tool::Enabled(false) => None,
53            Tool::Enabled(true) => Some(ToolSettings::default()),
54            Tool::Settings(settings) => Some(settings.clone()),
55        }
56    }
57}

59#[derive(Default, Clone, Serialize, Deserialize)]
60#[serde(default)]
61pub struct ToolSettings {
62    /// rust: cargo packages (`-p`); default: the workspace's default members.
63    pub packages: Vec<String>,
64    /// lua/js: pathspecs of tracked files; default: every tracked file of the language.
65    /// translate: pathspecs of pages; default: all of them.
66    pub files: Vec<String>,
67}

101/// Every cloned repo that isn't skipped (wiki.toml `skip` or `$WIKI_SKIP`), sorted by name.
102pub fn collect(paths: &Paths) -> Result<Vec<Repo>> {
103    let toml_path = paths.wiki.join("wiki.toml");
104    let config: WikiToml = toml::from_str(&fs::read_to_string(&toml_path)?)
105        .with_context(|| format!("parsing {}", toml_path.display()))?;
106    let skipped: Vec<String> = env::var("WIKI_SKIP")
107        .unwrap_or_default()
108        .split(',')
109        .filter(|name| !name.is_empty())
110        .map(str::to_owned)
111        .collect();
112
113    let output = std::process::Command::new("pull")
114        .args(["status", "--json"])
115        .current_dir(&paths.root)
116        .output()
117        .context("running `pull status --json`")?;
118    if !output.status.success() {
119        bail!("pull status --json: {}", String::from_utf8_lossy(&output.stderr));
120    }
121    let status: Status = serde_json::from_slice(&output.stdout).context("parsing `pull status --json`")?;
122
123    let github = Regex::new(r"github\.com[:/]([^/]+/[^/]+?)(\.git)?/?$")?;
124    let mut repos = Vec::new();
125    for repo in status.repos {
126        let config = config.repos.get(&repo.name).cloned().unwrap_or_default();
127        if !repo.cloned || config.skip || skipped.contains(&repo.name) {
128            continue;
129        }
130        let readme = fs::read_dir(&repo.path)?
131            .filter_map(Result::ok)
132            .filter(|entry| entry.file_type().is_ok_and(|kind| kind.is_file()))
133            .map(|entry| entry.file_name().to_string_lossy().into_owned())
134            .find(|name| name.eq_ignore_ascii_case("readme.md"));
135        let web = git2::Repository::open(&repo.path)
136            .ok()
137            .and_then(|git| git.find_remote("origin").ok().and_then(|remote| remote.url().ok().map(str::to_owned)))
138            .and_then(|url| github.captures(&url).map(|found| format!("https://github.com/{}", &found[1])));
139        repos.push(Repo {
140            name: repo.name,
141            path: repo.path,
142            branch: repo.branch.unwrap_or_else(|| "main".into()),
143            head: repo.head.unwrap_or_default(),
144            web,
145            readme,
146            config,
147        });
148    }
149    repos.sort_by_key(|repo| repo.name.to_lowercase());
150    Ok(repos)
151}
