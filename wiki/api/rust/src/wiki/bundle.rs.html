1//! Client assets into dist/_assets: rolldown (embedded) bundles and minifies
2//! assets/main.js with its npm imports; lightningcss bundles assets/style.css
3//! (Bootstrap via @import) plus syntect's CSS for the highlight classes.
4//! assets/icons/ (the logo, favicons) is copied as is.
5use std::fs;
6use std::path::Path;
7
8use anyhow::{Context, Result, anyhow};
9use lightningcss::bundler::{Bundler as CssBundler, FileProvider};
10use lightningcss::stylesheet::{MinifyOptions, ParserOptions, PrinterOptions};
11use rolldown::{Bundler, BundlerOptions, InputItem, OutputFormat, RawMinifyOptions};
12use syntect::highlighting::ThemeSet;
13use syntect::html::{ClassStyle, css_for_theme_with_class_style};
14
15use crate::Paths;
16use crate::markdown::HIGHLIGHT_PREFIX;
17use crate::state::{self, State};

19pub fn bundle(paths: &Paths, state: &mut State) -> Result<()> {
20    let out = paths.dist.join("_assets");
21    let mut inputs = state::dir_ids(&paths.wiki.join("assets"));
22    inputs.push(state::file_id(&paths.wiki.join("pnpm-lock.yaml")));
23    inputs.push(state::file_id(&std::env::current_exe()?));
24    let key = state::key(inputs);
25    if state.assets == key && out.join("main.js").is_file() && out.join("style.css").is_file() {
26        return Ok(());
27    }
28    if out.exists() {
29        fs::remove_dir_all(&out)?;
30    }
31    fs::create_dir_all(&out)?;
32    javascript(paths, &out)?;
33    fs::write(out.join("style.css"), stylesheet(&paths.wiki.join("assets/style.css"))?)?;
34    let icons = paths.wiki.join("assets/icons");
35    if icons.is_dir() {
36        fs::create_dir_all(out.join("icons"))?;
37        for entry in fs::read_dir(&icons)? {
38            let entry = entry?;
39            fs::copy(entry.path(), out.join("icons").join(entry.file_name()))?;
40        }
41    }
42    state.assets = key;
43    Ok(())
44}

46fn javascript(paths: &Paths, out: &Path) -> Result<()> {
47    let mut bundler = Bundler::new(BundlerOptions {
48        input: Some(vec![InputItem { name: Some("main".into()), import: "./assets/main.js".into() }]),
49        cwd: Some(paths.wiki.clone()),
50        dir: Some(out.to_string_lossy().into_owned()),
51        format: Some(OutputFormat::Esm),
52        minify: Some(RawMinifyOptions::Bool(true)),
53        ..Default::default()
54    })
55    .map_err(|errors| anyhow!("rolldown: {errors:?}"))?;
56    let runtime = tokio::runtime::Runtime::new()?;
57    let output = runtime
58        .block_on(bundler.write())
59        .map_err(|errors| anyhow!("rolldown: {errors:?}"))?;
60    for warning in output.warnings {
61        crate::warn(format_args!("rolldown: {warning:?}"));
62    }
63    Ok(())
64}

66fn stylesheet(entry: &Path) -> Result<String> {
67    let provider = FileProvider::new();
68    let mut bundler = CssBundler::new(&provider, None, ParserOptions::default());
69    let mut sheet = bundler
70        .bundle(entry)
71        .map_err(|err| anyhow!("lightningcss: {err}"))
72        .with_context(|| format!("bundling {}", entry.display()))?;
73    sheet.minify(MinifyOptions::default()).map_err(|err| anyhow!("lightningcss: {err}"))?;
74    let mut css = sheet
75        .to_css(PrinterOptions { minify: true, ..PrinterOptions::default() })
76        .map_err(|err| anyhow!("lightningcss: {err}"))?
77        .code;
78    let themes = ThemeSet::load_defaults();
79    let style = ClassStyle::SpacedPrefixed { prefix: HIGHLIGHT_PREFIX };
80    for (scheme, theme) in [("light", "InspiredGitHub"), ("dark", "base16-ocean.dark")] {
81        let rules = css_for_theme_with_class_style(&themes.themes[theme], style)?;
82        css.push_str(&format!("\n@media (prefers-color-scheme: {scheme}) {{\n{rules}}}\n"));
83    }
84    Ok(css)
85}
