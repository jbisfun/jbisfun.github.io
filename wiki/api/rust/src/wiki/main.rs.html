1//! `wiki`: one static site from every repo in a pull workspace.
2//!
3//! Each repo's README is its entry page, its `docs/` folder becomes pages, and
4//! rustdoc / LDoc / JSDoc produce its code reference. Git object ids (blob,
5//! tree and patch ids) decide what has to be regenerated: see [`state`].
6mod apidoc;
7mod bundle;
8mod collect;
9mod markdown;
10mod render;
11mod state;
12
13use std::path::{Path, PathBuf};
14use std::{env, fs};
15
16use anyhow::{Context, Result, bail};
17use clap::{Parser, Subcommand};
18
19#[derive(Parser)]
20#[command(version, about = "Static wiki from every repo in a pull workspace")]
21struct Cli {
22    #[command(subcommand)]
23    command: Option<Command>,
24}
25
26#[derive(Subcommand)]
27enum Command {
28    /// Build the site into dist/, regenerating only what changed.
29    Build {
30        /// Ignore .build/state.json and rebuild everything.
31        #[arg(long)]
32        force: bool,
33    },
34    /// Build, then mirror dist/ into the Pages repo and stage it.
35    Publish {
36        #[arg(long)]
37        force: bool,
38    },
39    /// Remove .build and dist.
40    Clean,
41}
42
43/// Paths of one run.
44pub struct Paths {
45    /// The wiki repo: wiki.toml, templates/, pages/, assets/.
46    pub wiki: PathBuf,
47    /// The pull workspace root.
48    pub root: PathBuf,
49    /// .build: state.json and the code reference per repo.
50    pub build: PathBuf,
51    pub dist: PathBuf,
52}
53
54impl Paths {
55    fn find() -> Result<Self> {
56        let cwd = env::current_dir()?;
57        let pull_root = env::var_os("PULL_ROOT").map(PathBuf::from);
58        let wiki = match cwd.ancestors().find(|dir| dir.join("wiki.toml").is_file()) {
59            Some(dir) => dir.to_path_buf(),
60            None => match &pull_root {
61                Some(root) if root.join("wiki/wiki.toml").is_file() => root.join("wiki"),
62                _ => bail!("no wiki.toml above {} and none in $PULL_ROOT/wiki", cwd.display()),
63            },
64        };
65        let root = match pull_root {
66            Some(root) => root,
67            None => wiki.parent().context("the wiki repo has no parent dir")?.to_path_buf(),
68        };
69        Ok(Self { build: wiki.join(".build"), dist: wiki.join("dist"), wiki, root })
70    }
71}
72
73pub fn warn(message: impl std::fmt::Display) {
74    eprintln!("wiki: {message}");
75}
76
77fn main() -> Result<()> {
78    let cli = Cli::parse();
79    let paths = Paths::find()?;
80    match cli.command.unwrap_or(Command::Build { force: false }) {
81        Command::Build { force } => build(&paths, force),
82        Command::Publish { force } => {
83            build(&paths, force)?;
84            publish(&paths)
85        }
86        Command::Clean => {
87            for dir in [&paths.build, &paths.dist] {
88                if dir.exists() {
89                    fs::remove_dir_all(dir)?;
90                }
91            }
92            Ok(())
93        }
94    }
95}
96
97fn build(paths: &Paths, force: bool) -> Result<()> {
98    let mut state = if force { state::State::default() } else { state::State::load(&paths.build) };
99    let repos = collect::collect(paths)?;
100    let apis = apidoc::generate(paths, &repos, &mut state)?;
101    bundle::bundle(paths, &mut state)?;
102    render::render(paths, &repos, &apis, &mut state)?;
103    state.save(&paths.build)
104}

106/// Mirror dist/ into the GitHub Pages repo and stage it, so the pull precommit
107/// hook lands the site in the same `pull commit`.
108fn publish(paths: &Paths) -> Result<()> {
109    let target = env::var_os("WIKI_PUBLISH_DIR")
110        .map(PathBuf::from)
111        .unwrap_or_else(|| paths.root.join("jbisfun.github.io"));
112    let status = std::process::Command::new("rsync")
113        .args(["-a", "--delete", "--exclude", ".git", "--exclude", "CNAME", "--exclude", "README.md"])
114        .arg(format!("{}/", paths.dist.display()))
115        .arg(format!("{}/", target.display()))
116        .status()
117        .context("running rsync")?;
118    if !status.success() {
119        bail!("rsync into {} failed", target.display());
120    }
121    fs::write(target.join(".nojekyll"), "")?;
122    let staged = stage_all(&target)?;
123    println!("wiki: published to {} ({staged} files staged)", target.display());
124    Ok(())
125}
126
127/// `git add -A` in `dir`; returns how many files differ from HEAD in the index.
128fn stage_all(dir: &Path) -> Result<usize> {
129    let repo = git2::Repository::open(dir).with_context(|| format!("opening {}", dir.display()))?;
130    let mut index = repo.index()?;
131    index.add_all(["*"], git2::IndexAddOption::DEFAULT, None)?;
132    index.update_all(["*"], None)?;
133    index.write()?;
134    let head = repo.head().ok().and_then(|head| head.peel_to_tree().ok());
135    Ok(repo.diff_tree_to_index(head.as_ref(), Some(&index), None)?.deltas().len())
136}
