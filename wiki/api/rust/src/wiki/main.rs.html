1//! `wiki`: one static site from every repo in a pull workspace.
2//!
3//! Each repo's README is its entry page, its `docs/` folder becomes pages, and
4//! rustdoc / LDoc / JSDoc produce its code reference. Git object ids (blob,
5//! tree and patch ids) decide what has to be regenerated: see [`state`].
6mod apidoc;
7mod bundle;
8mod collect;
9mod i18n;
10mod markdown;
11mod render;
12mod state;
13mod translate;

15use std::path::{Path, PathBuf};
16use std::{env, fs};
17
18use anyhow::{Context, Result, bail};
19use clap::{Parser, Subcommand};
20
21#[derive(Parser)]
22#[command(version, about = "Static wiki from every repo in a pull workspace")]
23struct Cli {
24    #[command(subcommand)]
25    command: Option<Command>,
26}
27
28#[derive(Subcommand)]
29enum Command {
30    /// Build the site into dist/, regenerating only what changed.
31    Build {
32        /// Ignore .build/state.json and rebuild everything.
33        #[arg(long)]
34        force: bool,
35    },
36    /// Build, then mirror dist/ into the Pages repo and stage it.
37    Publish {
38        #[arg(long)]
39        force: bool,
40    },
41    /// Remove .build and dist.
42    Clean,
43}
44
45/// Paths of one run.
46pub struct Paths {
47    /// The wiki repo: wiki.toml, templates/, pages/, assets/.
48    pub wiki: PathBuf,
49    /// The pull workspace root.
50    pub root: PathBuf,
51    /// .build: state.json and the code reference per repo.
52    pub build: PathBuf,
53    pub dist: PathBuf,
54}
55
56impl Paths {
57    fn find() -> Result<Self> {
58        let cwd = env::current_dir()?;
59        let pull_root = env::var_os("PULL_ROOT").map(PathBuf::from);
60        let wiki = match cwd.ancestors().find(|dir| dir.join("wiki.toml").is_file()) {
61            Some(dir) => dir.to_path_buf(),
62            None => match &pull_root {
63                Some(root) if root.join("wiki/wiki.toml").is_file() => root.join("wiki"),
64                _ => bail!("no wiki.toml above {} and none in $PULL_ROOT/wiki", cwd.display()),
65            },
66        };
67        let root = match pull_root {
68            Some(root) => root,
69            None => wiki.parent().context("the wiki repo has no parent dir")?.to_path_buf(),
70        };
71        Ok(Self { build: wiki.join(".build"), dist: wiki.join("dist"), wiki, root })
72    }
73}
74
75pub fn warn(message: impl std::fmt::Display) {
76    eprintln!("wiki: {message}");
77}
78
79fn main() -> Result<()> {
80    let cli = Cli::parse();
81    let paths = Paths::find()?;
82    match cli.command.unwrap_or(Command::Build { force: false }) {
83        Command::Build { force } => build(&paths, force),
84        Command::Publish { force } => {
85            build(&paths, force)?;
86            publish(&paths)
87        }
88        Command::Clean => {
89            for dir in [&paths.build, &paths.dist] {
90                if dir.exists() {
91                    fs::remove_dir_all(dir)?;
92                }
93            }
94            Ok(())
95        }
96    }
97}
98
99fn build(paths: &Paths, force: bool) -> Result<()> {
100    let mut state = if force { state::State::default() } else { state::State::load(&paths.build) };
101    let repos = collect::collect(paths)?;
102    let apis = apidoc::generate(paths, &repos, &mut state)?;
103    bundle::bundle(paths, &mut state)?;
104    render::render(paths, &repos, &apis, &mut state)?;
105    state.save(&paths.build)
106}

108/// Mirror dist/ into the GitHub Pages repo and stage it, so the pull precommit
109/// hook lands the site in the same `pull commit`.
110fn publish(paths: &Paths) -> Result<()> {
111    let target = env::var_os("WIKI_PUBLISH_DIR")
112        .map(PathBuf::from)
113        .unwrap_or_else(|| paths.root.join("jbisfun.github.io"));
114    let status = std::process::Command::new("rsync")
115        .args(["-a", "--delete", "--exclude", ".git", "--exclude", "CNAME", "--exclude", "README.md"])
116        .arg(format!("{}/", paths.dist.display()))
117        .arg(format!("{}/", target.display()))
118        .status()
119        .context("running rsync")?;
120    if !status.success() {
121        bail!("rsync into {} failed", target.display());
122    }
123    fs::write(target.join(".nojekyll"), "")?;
124    let staged = stage_all(&target)?;
125    println!("wiki: published to {} ({staged} files staged)", target.display());
126    Ok(())
127}
128
129/// `git add -A` in `dir`; returns how many files differ from HEAD in the index.
130fn stage_all(dir: &Path) -> Result<usize> {
131    let repo = git2::Repository::open(dir).with_context(|| format!("opening {}", dir.display()))?;
132    let mut index = repo.index()?;
133    index.add_all(["*"], git2::IndexAddOption::DEFAULT, None)?;
134    index.update_all(["*"], None)?;
135    index.write()?;
136    let head = repo.head().ok().and_then(|head| head.peel_to_tree().ok());
137    Ok(repo.diff_tree_to_index(head.as_ref(), Some(&index), None)?.deltas().len())
138}
