1//! Pages: every repo's README (its entry page) and docs/ folder, plus the
2//! wiki's own pages/*.jinja, through minijinja templates (templates/).
3//!
4//! A repo's pages are re-rendered only when its key changed: the blob ids of
5//! all its page sources, its code reference keys, its settings, and the global
6//! key (templates, the repo list in the navbar, the wiki binary). `.jinja`
7//! pages can show anything about the site, so they are rendered every time.
8//!
9//! The site is rendered once per language (see [`crate::i18n`]): the source
10//! language at the root, every other one under its prefix with the repo pages
11//! swapped for their machine translations ([`crate::translate`]).

32static IMAGE: LazyLock<Regex> = LazyLock::new(|| Regex::new(r"(?i)\.(png|jpe?g|gif|svg|webp|avif|ico)$").expect("valid regex"));
33static H1: LazyLock<Regex> = LazyLock::new(|| Regex::new(r"(?m)^#\s+(.+?)\s*(\{#[^}]*\})?\s*$").expect("valid regex"));
34static FRONT_MATTER: LazyLock<Regex> = LazyLock::new(|| Regex::new(r"^---\r?\n([\s\S]*?)\r?\n---\r?\n?").expect("valid regex"));
35static SUMMARY_H1: LazyLock<Regex> = LazyLock::new(|| Regex::new(r"(?i)^#\s+Summary\s*\n").expect("valid regex"));
36static BODY_TAG: LazyLock<Regex> = LazyLock::new(|| Regex::new(r"(?i)<body[^>]*>").expect("valid regex"));
37static HEAD_END: LazyLock<Regex> = LazyLock::new(|| Regex::new(r"(?i)</head>").expect("valid regex"));

41/// Relative URL from one output page to another, both site-relative.
42pub fn url(from: &str, to: &str) -> String {
43    if SCHEME.is_match(to) {
44        return to.to_owned();
45    }
46    let mut to = to.trim_start_matches('/').to_owned();
47    if to.is_empty() || to.ends_with('/') {
48        to.push_str("index.html");
49    }
50    let dir = Path::new(from).parent().unwrap_or(Path::new(""));
51    match pathdiff::diff_paths(&to, dir) {
52        Some(path) if !path.as_os_str().is_empty() => path.to_string_lossy().into_owned(),
53        _ => basename(&to).to_owned(),
54    }
55}

67/// Output path of a source page: README/index.md -> index.html, x.md -> x.html,
68/// x.jinja -> x.html (or x as is when it has its own extension), other files as is.
69fn out_name(rel: &str) -> String {
70    let (dir, base) = rel.rsplit_once('/').unwrap_or(("", rel));
71    let name = if is_index(base) && base.to_lowercase().ends_with(".md") {
72        "index.html".to_owned()
73    } else if let Some(stem) = base.strip_suffix(".md") {
74        format!("{stem}.html")
75    } else if let Some(stem) = base.strip_suffix(".jinja") {
76        if stem.contains('.') { stem.to_owned() } else { format!("{stem}.html") }
77    } else {
78        base.to_owned()
79    };
80    if dir.is_empty() { name } else { format!("{dir}/{name}") }
81}

83fn is_page(rel: &str) -> bool {
84    rel.ends_with(".md") || rel.ends_with(".jinja")

87/// A folder's own page: README or index, markdown or Jinja.
88fn is_index(base: &str) -> bool {
89    ["readme.md", "index.md", "readme.jinja", "index.jinja"].contains(&base.to_lowercase().as_str())

92/// Front matter as a JSON object and the body after it. YAML first; the loose
93/// `key: value` lines mdBook accepts (e.g. unquoted colons) as a fallback.
94fn front_matter(source: &str) -> (Data, String) {
95    let Some(found) = FRONT_MATTER.captures(source) else {
96        return (Data::new(), source.to_owned());
97    };
98    let body = source[found[0].len()..].to_owned();
99    let matter = gray_matter::Matter::<gray_matter::engine::YAML>::new();
100    if let Ok(parsed) = matter.parse::<serde_json::Value>(source)
101        && let Some(serde_json::Value::Object(data)) = parsed.data
102    {
103        return (data, body);
104    }
105    let data = found[1]
106        .lines()
107        .filter_map(|line| line.split_once(':'))
108        .filter(|(key, _)| !key.trim().is_empty())
109        .map(|(key, value)| (key.trim().to_owned(), serde_json::Value::String(value.trim().to_owned())))
110        .collect();
111    (data, body)
112}

114/// Front matter `title`, else the first `# heading`, else the file name.
115fn title_of(data: &Data, body: &str, rel: &str) -> String {
116    match data.get("title") {
117        Some(serde_json::Value::String(title)) => return title.clone(),
118        Some(serde_json::Value::Null) | None => {}
119        Some(other) => return other.to_string(),
120    }
121    if let Some(found) = H1.captures(body) {
122        return found[1].replace(['`', '*', '_'], "");
123    }
124    basename(rel).trim_end_matches(".md").trim_end_matches(".jinja").replace(['-', '_'], " ")
125}

127/// Where links inside one repo's pages go.
128struct Links {
129    /// Repo-relative source path -> site-relative output path.
130    by_rel: HashMap<String, String>,
131    blob: Option<String>,
132    raw: Option<String>,
133}
134
135impl Links {
136    /// .md links become pages (in the language `prefix`); links to other repo
137    /// files go to GitHub (images to their raw URL).
138    fn resolve(&self, link: &str, base: &str, from: &str, prefix: &str) -> Option<String> {
139        if link.is_empty() || SCHEME.is_match(link) {
140            return None;
141        }
142        let cut = link.find(['?', '#']).unwrap_or(link.len());
143        let (path, suffix) = link.split_at(cut);
144        if path.is_empty() {
145            return None;
146        }
147        if path.starts_with('/') {
148            return Some(url(from, path) + suffix);
149        }
150        let decoded = percent_encoding::percent_decode_str(path).decode_utf8_lossy();
151        let joined = path_clean::clean(format!("{base}/{decoded}"));
152        let target = joined.to_string_lossy();
153        let target = target.trim_end_matches('/');
154        for candidate in [target.to_owned(), format!("{target}/README.md"), format!("{target}/readme.md"), format!("{target}/index.md")] {
155            if let Some(out) = self.by_rel.get(&candidate) {
156                let out = if is_page(&candidate) { format!("{prefix}{out}") } else { out.clone() };
157                return Some(url(from, &out) + suffix);
158            }
159        }
160        if target != ".." && !target.starts_with("../") {
161            let prefix = if IMAGE.is_match(target) { &self.raw } else { &self.blob };
162            if let Some(prefix) = prefix {
163                return Some(format!("{prefix}{}{suffix}", if target == "." { "" } else { target }));
164            }
165        }
166        None
167    }
168}

184#[derive(Serialize, Default)]
185struct TreeNode {
186    name: String,
187    page: Option<PageRef>,
188    children: Vec<TreeNode>,
189}
190
191impl TreeNode {
192    /// Nested tree of a repo's docs pages, for docs without SUMMARY.md.
193    fn build(pages: &[(String, PageRef)]) -> Self {
194        let mut root = TreeNode::default();
195        for (doc_rel, page) in pages {
196            let parts: Vec<&str> = doc_rel.split('/').collect();
197            let (leaf, dirs) = parts.split_last().expect("split yields one part");
198            let mut node = &mut root;
199            for dir in dirs {
200                let at = match node.children.iter().position(|child| child.name == *dir) {
201                    Some(at) => at,
202                    None => {
203                        node.children.push(TreeNode { name: (*dir).to_owned(), ..Default::default() });
204                        node.children.len() - 1
205                    }
206                };
207                node = &mut node.children[at];
208            }
209            if is_index(leaf) {
210                node.page = Some(page.clone());
211            } else {
212                node.children.push(TreeNode { name: (*leaf).to_owned(), page: Some(page.clone()), children: Vec::new() });
213            }
214        }
215        root
216    }
217}

317/// A repo with its sources resolved.
318struct SiteRepo<'a> {
319    repo: &'a Repo,
320    links: Arc<Links>,
321    /// (repo-relative source, output path) of every docs file that isn't a page.
322    assets: Vec<(String, String)>,
323    /// Indexes into the page list, entry page first.
324    pages: Vec<usize>,
325    apis: Vec<Api>,
326    docs_rel: Option<String>,
327    book_title: Option<String>,
328    summary: Option<String>,
329    /// Key of the repo's own inputs; `render` folds in the global key.
330    key: String,
331}
332
333impl<'a> SiteRepo<'a> {
334    /// Read a repo's README and docs into `pages`; None when it has neither
335    /// pages nor a code reference. `index` is its position in the site's repos.
336    fn load(repo: &'a Repo, apis: Vec<Api>, index: usize, state: &State, pages: &mut Vec<Page>) -> Result<Option<Self>> {
337        // (repo-relative source, docs-relative path, output path)
338        let mut files: Vec<(String, Option<String>, String)> = Vec::new();
339        if let Some(readme) = &repo.readme {
340            files.push((readme.clone(), None, format!("{}/index.html", repo.name)));
341        }
342        let (mut docs_rel, mut book_title) = (None, None);
343        let docs = repo.path.join("docs");
344        if docs.is_dir() {
345            let (root, title) = docs_root(&docs);
346            let rel = root.strip_prefix(&repo.path).unwrap_or(&root).to_string_lossy().into_owned();
347            for file in walk(&root) {
348                files.push((format!("{rel}/{file}"), Some(file.clone()), format!("{}/docs/{}", repo.name, out_name(&file))));
349            }
350            docs_rel = Some(rel);
351            book_title = title;
352        }
353        if apis.is_empty() && !files.iter().any(|(rel, ..)| is_page(rel)) {
354            return Ok(None);
355        }
356
357        let mut inputs = vec![serde_json::to_string(&repo.config)?, repo.branch.clone(), repo.web.clone().unwrap_or_default()];
358        inputs.extend(apis.iter().map(|api| format!("{}\0{}", api.tool, api.path)));
359        let prefix = format!("{}/", repo.name);
360        inputs.extend(state.apidoc.iter().filter(|(id, _)| id.starts_with(&prefix)).map(|(_, api)| api.key.clone()));
361
362        let links = Links {
363            by_rel: files.iter().map(|(rel, _, out)| (rel.clone(), out.clone())).collect(),
364            blob: repo.web.as_ref().map(|web| format!("{web}/blob/{}/", repo.branch)),
365            raw: repo.web.as_ref().map(|web| format!("{web}/raw/{}/", repo.branch)),
366        };
367        let mut site = SiteRepo {
368            repo,
369            links: Arc::new(links),
370            assets: Vec::new(),
371            pages: Vec::new(),
372            apis,
373            docs_rel,
374            book_title,
375            summary: None,
376            key: String::new(),
377        };
378        for (rel, doc_rel, out) in files {
379            let src = repo.path.join(&rel);
380            inputs.push(format!("{rel}\0{}", state::file_id(&src)));
381            if doc_rel.as_deref() == Some("SUMMARY.md") {
382                site.summary = Some(fs::read_to_string(&src)?);
383            } else if !is_page(&rel) {
384                site.assets.push((rel, out));
385            } else {
386                let source = fs::read_to_string(&src).with_context(|| format!("reading {}", src.display()))?;
387                let (data, body) = front_matter(&source);
388                let title = title_of(&data, &body, &rel);
389                let readme = Some(&rel) == repo.readme.as_ref();
390                site.pages.push(pages.len());
391                pages.push(Page { repo: Some(index), rel, doc_rel, out, data, source, body, title, readme, ..Default::default() });
392            }
393        }
394        if repo.readme.is_none() {
395            site.pages.insert(0, pages.len());
396            pages.push(Page {
397                repo: Some(index),
398                out: format!("{}/index.html", repo.name),
399                title: repo.config.title.clone().unwrap_or_else(|| repo.name.clone()),
400                placeholder: true,
401                ..Default::default()
402            });
403        }
404        site.key = state::key(inputs);
405        Ok(Some(site))
406    }
407
408    /// Whether wiki.toml `translate` covers the repo file `rel`.
409    fn translates(&self, rel: &str) -> bool {
410        let Some(settings) = self.repo.config.translate.settings() else { return false };
411        settings.files.is_empty()
412            || git2::Pathspec::new(settings.files.iter())
413                .is_ok_and(|spec| spec.matches_path(Path::new(rel), git2::PathspecFlags::DEFAULT))
414    }
415
416    /// The repo as templates see it in `locale`; `strings` translates the
417    /// wiki.toml strings (descriptions, categories, book titles).
418    fn view(&self, pages: &[Page], markdown: &Markdown, locale: &Locale, i18n: &I18n, strings: &BTreeMap<String, String>) -> RepoView {
419        let repo = self.repo;
420        let tr = |text: &str| strings.get(text).cloned().unwrap_or_else(|| text.to_owned());
421        let readme = self.pages.iter().map(|at| &pages[*at]).find(|page| page.readme);
422        let docs: Vec<(String, PageRef)> = self
423            .pages
424            .iter()
425            .map(|at| &pages[*at])
426            .filter_map(|page| page.doc_rel.clone().map(|doc_rel| (doc_rel, page.page_ref())))
427            .collect();
428        let apis: Vec<ApiView> = self
429            .apis
430            .iter()
431            .map(|api| ApiView {
432                tool: api.tool.clone(),
433                title: api.title.clone(),
434                label: match api.tool.as_str() {
435                    "rust" => "Rust",
436                    "lua" => "Lua",
437                    "js" => "JavaScript",
438                    other => other,
439                }
440                .to_owned(),
441                out: format!("{}/api/{}", repo.name, api.path),
442            })
443            .collect();
444        let mut api_labels: Vec<String> = apis.iter().map(|api| api.label.clone()).collect();
445        api_labels.dedup();
446        let group = repo.name.rsplit_once('/').map(|(group, _)| group);
447        RepoView {
448            name: repo.name.clone(),
449            title: repo.config.title.clone().unwrap_or_else(|| basename(&repo.name).to_owned()),
450            index: format!("{}{}/index.html", locale.prefix, repo.name),
451            web: repo.web.clone(),
452            blob: self.links.blob.clone(),
453            head: repo.head.clone(),
454            category_key: repo.config.category.as_deref().or(group).unwrap_or("Other").to_owned(),
455            category: match repo.config.category.as_deref().or(group) {
456                Some(category) => tr(category),
457                None => i18n.t(locale, "category-other"),
458            },
459            description: match &repo.config.description {
460                Some(description) => Some(tr(description)),
461                None => readme.and_then(|page| markdown.blurb(&page.body)),
462            },
463            readme_page: readme.map(Page::page_ref),
464            doc_pages: docs.iter().map(|(_, page)| page.clone()).collect(),
465            tree: TreeNode::build(&docs),
466            apis,
467            api_labels,
468            book_title: self.book_title.as_deref().map(tr),
469            has_summary: self.summary.is_some(),
470        }
471    }
472}

474/// mdBook projects keep pages in book.toml's `src` (default "src").
475fn docs_root(docs: &Path) -> (PathBuf, Option<String>) {
476    let Ok(text) = fs::read_to_string(docs.join("book.toml")) else { return (docs.to_path_buf(), None) };
477    let book: toml::Table = toml::from_str(&text).unwrap_or_default();
478    let book = book.get("book").and_then(toml::Value::as_table);
479    let src = book.and_then(|book| book.get("src")).and_then(toml::Value::as_str).unwrap_or("src");
480    let title = book.and_then(|book| book.get("title")).and_then(toml::Value::as_str).map(str::to_owned);
481    (docs.join(src), title)
482}

484/// Files under `dir`, sorted, skipping hidden and git-ignored ones.
485fn walk(dir: &Path) -> Vec<String> {
486    let mut files: Vec<String> = ignore::WalkBuilder::new(dir)
487        .build()
488        .filter_map(Result::ok)
489        .filter(|entry| entry.file_type().is_some_and(|kind| kind.is_file()))
490        .filter_map(|entry| entry.path().strip_prefix(dir).ok().map(|rel| rel.to_string_lossy().into_owned()))
491        .filter(|rel| !rel.split('/').any(|part| part == "node_modules"))
492        .collect();
493    files.sort();
494    files
495}

497fn write(dist: &Path, out: &str, html: &str) -> Result<()> {
498    let dest = dist.join(out);
499    if let Some(dir) = dest.parent() {
500        fs::create_dir_all(dir)?;
501    }
502    fs::write(&dest, html).with_context(|| format!("writing {}", dest.display()))
503}

505/// Remove everything in dist/ except the bundled assets.
506fn clear_pages(dist: &Path) -> Result<()> {
507    for entry in fs::read_dir(dist)? {
508        let entry = entry?;
509        if entry.file_name() != "_assets" {
510            let path = entry.path();
511            if path.is_dir() { fs::remove_dir_all(&path)? } else { fs::remove_file(&path)? }
512        }
513    }
514    Ok(())
515}

517pub fn render(paths: &Paths, repos: &[Repo], apis: &BTreeMap<String, Vec<Api>>, state: &mut State) -> Result<()> {
518    let site_title = std::env::var("WIKI_TITLE").unwrap_or_else(|_| "jbisfun wiki".into());
519    let markdown = Arc::new(Markdown::new());
520    let i18n = I18n::load(paths)?;
521    let dist = &paths.dist;
522
523    let mut pages: Vec<Page> = Vec::new();
524    let mut site_repos: Vec<SiteRepo> = Vec::new();
525    for repo in repos {
526        let repo_apis = apis.get(&repo.name).cloned().unwrap_or_default();
527        if let Some(site) = SiteRepo::load(repo, repo_apis, site_repos.len(), state, &mut pages)? {
528            site_repos.push(site);
529        }
530    }
531
532    let mut global_inputs = state::dir_ids(&paths.wiki.join("templates"));
533    global_inputs.push(state::file_id(&std::env::current_exe()?));
534    global_inputs.push(site_title.clone());
535    global_inputs.extend(i18n.input_ids(paths));
536    global_inputs.extend(site_repos.iter().map(|site| site.repo.name.clone()));
537    let global = state::key(global_inputs);
538    if state.global != global && dist.is_dir() {
539        clear_pages(dist)?;
540    }
541    for site in &mut site_repos {
542        site.key = state::key([global.as_str(), site.key.as_str()]);
543    }
544
545    for rel in walk(&paths.wiki.join("pages")).into_iter().filter(|rel| rel.ends_with(".jinja")) {
546        let source = fs::read_to_string(paths.wiki.join("pages").join(&rel))?;
547        let (data, body) = front_matter(&source);
548        let title = data.get("title").and_then(serde_json::Value::as_str).unwrap_or(&site_title).to_owned();
549        pages.push(Page { out: out_name(&rel), rel, data, source, body, title, ..Default::default() });
550    }
551    for page in &mut pages {
552        page.lang = i18n.source().tag.clone();
553    }
554
555    let mut env = Environment::new();
556    env.set_loader(path_loader(paths.wiki.join("templates")));
557    env.set_auto_escape_callback(|_| AutoEscape::Html);
558    let mut run = Run { paths, state, kept: BTreeMap::new(), changed: Vec::new(), rendered: 0, total: 0 };
559    for locale in &i18n.locales {
560        let sources = &pages;
561        let (pages, strings) = if locale.is_source() {
562            (pages.clone(), BTreeMap::new())
563        } else {
564            localize(paths, &i18n, locale, &site_repos, &pages, &markdown)?
565        };
566        let views: Vec<RepoView> = site_repos.iter().map(|site| site.view(&pages, &markdown, locale, &i18n, &strings)).collect();
567        let all_pages: Vec<PageRef> = pages.iter().map(Page::page_ref).collect();
568        let site = context! {
569            title => site_title.clone(),
570            home => format!("{}index.html", locale.prefix),
571            year => jiff::ZonedDateTime::now().year(),
572            lang => locale.tag.clone(),
573            repos => Value::from(Serde(&views)),
574            pages => Value::from(Serde(&all_pages)),
575        };
576        let renderer = Renderer {
577            env: &env,
578            markdown: &markdown,
579            site: &site,
580            site_repos: &site_repos,
581            views: &views,
582            i18n: &i18n,
583            locale,
584            t: i18n.function(locale),
585        };
586        run.locale(&renderer, &pages, sources)?;
587    }
588
589    let Run { kept, changed, rendered, total, .. } = run;
590    for name in state.repos.keys().filter(|name| !kept.contains_key(*name)) {
591        let dir = dist.join(name);
592        if dir.is_dir() {
593            fs::remove_dir_all(dir)?;
594        }
595    }
596    state.repos = kept;
597    state.global = global;
598
599    let api_count: usize = site_repos.iter().map(|site| site.apis.len()).sum();
600    let changed = if changed.is_empty() { "none".to_owned() } else { changed.join(", ") };
601    println!("wiki: {rendered} of {total} pages rendered, {api_count} code references; regenerated: {changed}");
602    Ok(())
603}

605/// Swap the repo pages for their `locale` translations (translating what's
606/// missing or outdated first) and move every page under the locale's prefix.
607/// Also returns the wiki.toml strings' translations.
608fn localize(
609    paths: &Paths,
610    i18n: &I18n,
611    locale: &Locale,
612    site_repos: &[SiteRepo],
613    pages: &[Page],
614    markdown: &Markdown,
615) -> Result<(Vec<Page>, BTreeMap<String, String>)> {
616    let mut cache = Cache::load(&paths.wiki, &locale.tag);
617    // Page index -> (cache key, English blob id) for every translatable page.
618    let mut keys: HashMap<usize, (String, String)> = HashMap::new();
619    let mut strings = BTreeSet::new();
620    let mut jobs = Vec::new();
621    for site in site_repos {
622        for at in &site.pages {
623            let page = &pages[*at];
624            if page.placeholder || !site.translates(&page.rel) {
625                continue;
626            }
627            let key = format!("{}/{}", site.repo.name, page.rel);
628            let source_id = state::text_id(&page.source);
629            if !cache.page(&key, &source_id).is_some_and(|cached| cached.current) {
630                jobs.push(Job::Page { key: key.clone(), source_id: source_id.clone(), text: page.source.clone() });
631            }
632            keys.insert(*at, (key, source_id));
633        }
634        let config = &site.repo.config;
635        let group = site.repo.name.rsplit_once('/').map(|(group, _)| group);
636        strings.extend(config.description.iter().cloned());
637        strings.extend(config.category.as_deref().or(group).map(str::to_owned));
638        strings.extend(site.book_title.iter().cloned());
639    }
640    jobs.extend(strings.iter().filter(|text| cache.string(text).is_none()).map(|text| Job::String { text: text.clone() }));
641    let repos = site_repos.iter().map(|site| site.repo.name.clone()).collect();
642    cache.prune(&repos, &keys.values().map(|(key, _)| key.clone()).collect(), &strings)?;
643    translate::run(&i18n.config.translator, &i18n.t(locale, "translator-language"), &mut cache, jobs)?;
644
645    let mut localized = Vec::with_capacity(pages.len());
646    for (at, page) in pages.iter().enumerate() {
647        let mut page = page.clone();
648        page.original = Some(page.out.clone());
649        page.out = format!("{}{}", locale.prefix, page.out);
650        page.lang = locale.tag.clone();
651        if let Some((key, source_id)) = keys.get(&at) {
652            match cache.page(key, source_id) {
653                Some(cached) => {
654                    if !cached.current {
655                        warn(format_args!("{key}: {} translation is outdated", locale.tag));
656                    }
657                    if !page.is_template() {
658                        page.heading_ids = Some(Arc::new(markdown::heading_ids(&markdown.render(&page.source))?));
659                    }
660                    let text = markdown::join_cjk_lines(&cached.text);
661                    let (data, body) = front_matter(&text);
662                    page.title = title_of(&data, &body, &page.rel);
663                    (page.data, page.body, page.source) = (data, body, text);
664                    page.translation = Some("machine");
665                }
666                None => page.translation = Some("missing"),
667            }
668        } else if page.repo.is_some() && !page.placeholder {
669            page.translation = Some("missing");
670        }
671        localized.push(page);
672    }
673    let strings = strings.into_iter().filter_map(|text| cache.string(&text).map(|done| (text.clone(), done.to_owned()))).collect();
674    Ok((localized, strings))
675}

677/// Bookkeeping across the languages of one render.
678struct Run<'a> {
679    paths: &'a Paths,
680    state: &'a State,
681    /// "<prefix><repo>" -> key of its rendered pages.
682    kept: BTreeMap<String, String>,
683    changed: Vec<String>,
684    rendered: usize,
685    total: usize,
686}
687
688impl Run<'_> {
689    /// Render `pages` (the site in `renderer.locale`; `sources` is the same
690    /// list in the source language).
691    fn locale(&mut self, renderer: &Renderer, pages: &[Page], sources: &[Page]) -> Result<()> {
692        let dist = &self.paths.dist;
693        let prefix = &renderer.locale.prefix;
694        self.total += pages.len();
695        for (index, site) in renderer.site_repos.iter().enumerate() {
696            let name = format!("{prefix}{}", site.repo.name);
697            let key = if renderer.locale.is_source() {
698                site.key.clone()
699            } else {
700                // Translations and translated strings are inputs too.
701                let mut inputs = vec![site.key.clone(), renderer.locale.tag.clone(), serde_json::to_string(&renderer.views[index])?];
702                inputs.extend(site.pages.iter().map(|at| state::text_id(&pages[*at].source)));
703                state::key(inputs)
704            };
705            let fresh = self.state.repos.get(&name) == Some(&key) && dist.join(&name).is_dir();
706            if !fresh {
707                if renderer.locale.is_source() {
708                    renderer.copy_files(self.paths, index)?;
709                } else if dist.join(&name).exists() {
710                    fs::remove_dir_all(dist.join(&name))?;
711                }
712                self.changed.push(name.clone());
713            }
714            for at in &site.pages {
715                let page = &pages[*at];
716                if fresh && !page.is_template() {
717                    continue;
718                }
719                write(dist, &page.out, &renderer.page_or_original(page, &sources[*at])?)?;
720                self.rendered += 1;
721            }
722            self.kept.insert(name, key);
723        }
724        for page in pages.iter().filter(|page| page.repo.is_none()) {
725            write(dist, &page.out, &renderer.page(page)?)?;
726            self.rendered += 1;
727        }
728        Ok(())
729    }
730}

732struct Renderer<'a> {
733    env: &'a Environment<'a>,
734    markdown: &'a Arc<Markdown>,
735    site: &'a Value,
736    site_repos: &'a [SiteRepo<'a>],
737    views: &'a [RepoView],
738    i18n: &'a I18n,
739    locale: &'a Locale,
740    /// The template function `t(key, **args)` for `locale`.
741    t: Value,
742}
743
744impl Renderer<'_> {
745    /// Start a repo's dist/ folder over with its docs assets and code reference.
746    fn copy_files(&self, paths: &Paths, index: usize) -> Result<()> {
747        let site = &self.site_repos[index];
748        let name = &site.repo.name;
749        let dir = paths.dist.join(name);
750        if dir.exists() {
751            fs::remove_dir_all(&dir)?;
752        }
753        for (rel, out) in &site.assets {
754            let dest = paths.dist.join(out);
755            fs::create_dir_all(dest.parent().context("asset without dir")?)?;
756            fs::copy(site.repo.path.join(rel), &dest)?;
757        }
758        let tools: BTreeSet<&str> = site.apis.iter().map(|api| api.tool.as_str()).collect();
759        for tool in tools {
760            let dest = dir.join("api").join(tool);
761            copy_dir(&paths.build.join("api").join(name).join(tool), &dest)?;
762            // rustdoc/LDoc pages are standalone documents, not rendered through
763            // layout.html: embed the site header so they stay reachable. LDoc
764            // has no dark mode, so its header stays light.
765            match tool {
766                "rust" => self.embed_headers(&dest, &paths.dist, index, None)?,
767                "lua" => self.embed_headers(&dest, &paths.dist, index, Some("light"))?,
768                _ => {}
769            }
770        }
771        Ok(())
772    }
773
774    /// Every `.html` file under `dir` (a freshly copied rustdoc/LDoc output tree) gets
775    /// the site header (partials/apidoc-header.html) spliced in after `<body>`. The
776    /// header is a fixed element whose content lives in a shadow root, so the site's
777    /// stylesheet styles it without touching the doc tool's own CSS; the only thing
778    /// added to the page itself is room at the top for the header.
779    fn embed_headers(&self, dir: &Path, dist: &Path, repo: usize, theme: Option<&str>) -> Result<()> {
780        for entry in WalkDir::new(dir).into_iter().filter_map(Result::ok) {
781            let path = entry.path();
782            if entry.file_type().is_file() && path.extension().is_some_and(|ext| ext == "html") {
783                let from = path.strip_prefix(dist)?.to_string_lossy().replace('\\', "/");
784                self.embed_header(path, &from, repo, theme)?;
785            }
786        }
787        Ok(())
788    }
789
790    fn embed_header(&self, path: &Path, from: &str, repo: usize, theme: Option<&str>) -> Result<()> {
791        let html = fs::read_to_string(path).with_context(|| format!("reading {}", path.display()))?;
792        let Some(body) = BODY_TAG.find(&html).map(|found| found.end()) else {
793            return Ok(());
794        };
795        let header = self.env.get_template("partials/apidoc-header.html")?.render(context! {
796            site => self.site.clone(),
797            repo => Value::from(Serde(&self.views[repo])),
798            url => url_function(from),
799            t => self.t.clone(),
800            theme => theme,
801        })?;
802        // header.js measures the header into --wiki-header-height. rustdoc's sidebar is
803        // sticky at the top of the viewport, so it moves down by the same amount.
804        let head = format!(
805            "  <script type=\"module\" src=\"{}\"></script>\n  <style>\
806             body{{padding-top:var(--wiki-header-height,3.5rem)!important}}\
807             .rustdoc .sidebar{{top:var(--wiki-header-height,3.5rem);height:calc(100vh - var(--wiki-header-height,3.5rem))}}\
808             </style>\n",
809            url(from, "_assets/main.js"),
810        );
811        let mut out = String::with_capacity(html.len() + header.len() + head.len());
812        match HEAD_END.find(&html) {
813            Some(found) => {
814                out.push_str(&html[..found.start()]);
815                out.push_str(&head);
816                out.push_str(&html[found.start()..body]);
817            }
818            None => out.push_str(&html[..body]),
819        }
820        out.push_str(&header);
821        out.push_str(&html[body..]);
822        fs::write(path, out).with_context(|| format!("writing {}", path.display()))
823    }

825    /// A translated page whose translation doesn't render (a Jinja page the
826    /// translator broke) falls back to the source language.
827    fn page_or_original(&self, page: &Page, source: &Page) -> Result<String> {
828        match self.page(page) {
829            Err(err) if page.translation == Some("machine") => {
830                warn(format_args!("{} ({}): {err:#}; showing the original", page.out, self.locale.tag));
831                let fallback = Page {
832                    out: page.out.clone(),
833                    lang: page.lang.clone(),
834                    original: page.original.clone(),
835                    translation: Some("missing"),
836                    ..source.clone()
837                };
838                self.page(&fallback)
839            }
840            result => result,
841        }
842    }

844    /// The same page in every language.
845    fn alternates(&self, page: &Page) -> Vec<Alternate> {
846        let path = page.out.strip_prefix(self.locale.prefix.as_str()).unwrap_or(&page.out);
847        self.i18n
848            .locales
849            .iter()
850            .map(|locale| Alternate {
851                lang: locale.tag.clone(),
852                name: self.i18n.t(locale, "language-name"),
853                out: format!("{}{path}", locale.prefix),
854                current: locale.tag == self.locale.tag,
855            })
856            .collect()
857    }
858
859    fn page(&self, page: &Page) -> Result<String> {
860        let from = page.out.as_str();
861        let prefix = self.locale.prefix.clone();
862        // Markdown with the repo's links resolved from this page.
863        let md = {
864            let markdown = self.markdown.clone();
865            let base = page.rel.rsplit_once('/').map_or(".", |(dir, _)| dir).to_owned();
866            let links = page.repo.map(|at| self.site_repos[at].links.clone());
867            let from = from.to_owned();
868            let heading_ids = page.heading_ids.clone();
869            Arc::new(move |source: &str| -> Result<String> {
870                let mut html = markdown.render(source);
871                if let Some(ids) = &heading_ids {
872                    html = markdown::set_heading_ids(&html, ids)?;
873                }
874                match &links {
875                    Some(links) => markdown::rewrite_links(&html, |link| links.resolve(link, &base, &from, &prefix)),
876                    None => Ok(html),
877                }
878            })
879        };
880        let md_function = {
881            let md = md.clone();
882            Value::from_function(move |source: String| -> Result<Value, minijinja::Error> {
883                md(&source)
884                    .map(Value::from_safe_string)
885                    .map_err(|err| minijinja::Error::new(minijinja::ErrorKind::InvalidOperation, err.to_string()))
886            })
887        };
888        let url_value = url_function(from);
889        let sidebar = match page.repo {
890            Some(at) => self.sidebar(at, from, &url_value)?,
891            None => String::new(),
892        };
893        let data = context! {
894            site => self.site.clone(),
895            page => Value::from(Serde(page.view(self.alternates(page)))),
896            repo => page.repo.map(|at| Value::from(Serde(&self.views[at]))),
897            url => url_value,
898            t => self.t.clone(),
899            md => md_function,
900            sidebar => Value::from_safe_string(sidebar),
901        };
902
903        let content = if page.placeholder {
904            self.env.get_template("repo-index.html")?.render(&data)?
905        } else if page.is_template() {
906            let html = self.env.render_named_str(&page.rel, &page.body, &data)?;
907            if page.rel.ends_with(".md") { md(&html)? } else { html }
908        } else {
909            md(&page.source)?
910        };
911        let layout = match page.data.get("layout") {
912            Some(serde_json::Value::Bool(false)) => return Ok(content),
913            Some(serde_json::Value::String(name)) => name.as_str(),
914            _ => "layout.html",
915        };
916        let data = context! { content => Value::from_safe_string(content), ..data };
917        Ok(self.env.get_template(layout)?.render(data)?)
918    }

920    /// Repo navigation: README, code reference, then the docs (mdBook's
921    /// SUMMARY.md when there is one, else the folder tree).
922    fn sidebar(&self, at: usize, from: &str, url_value: &Value) -> Result<String> {
923        let site = &self.site_repos[at];
924        let summary = match &site.summary {
925            Some(source) => {
926                let html = self.markdown.render(&SUMMARY_H1.replace(source, ""));
927                // Summary links are relative to the docs root.
928                let base = site.docs_rel.as_deref().unwrap_or(".");
929                Some(markdown::rewrite_links(&html, |link| site.links.resolve(link, base, from, &self.locale.prefix))?)
930            }
931            None => None,
932        };
933        let html = self.env.get_template("partials/repo-nav.html")?.render(context! {
934            repo => Value::from(Serde(&self.views[at])),
935            summary => summary.map(Value::from_safe_string),
936            url => url_value.clone(),
937            t => self.t.clone(),
938        })?;
939        markdown::mark_active(&html, basename(from))
940    }
