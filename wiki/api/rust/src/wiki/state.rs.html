1//! What needs regenerating, decided with git object ids.
2//!
3//! Every input is identified the way git identifies it: files by their blob id
4//! (`git hash-object`), a repo's sources by its HEAD tree id plus the patch id
5//! of its uncommitted changes. A unit of work (a repo's pages, a code
6//! reference, the asset bundle) gets a key hashed from the ids of its inputs;
7//! `.build/state.json` keeps the key each unit was last built with, and a
8//! unit whose key is unchanged is left alone.
9use std::collections::BTreeMap;
10use std::fs;
11use std::path::Path;

13use anyhow::Result;
14use git2::{DiffOptions, ObjectType, Oid, Repository};
15use serde::{Deserialize, Serialize};

17use crate::apidoc::Api;

19#[derive(Default, Serialize, Deserialize)]
20pub struct State {
21    /// Key of everything every page depends on (templates, nav, the wiki binary).
22    #[serde(default)]
23    pub global: String,
24    #[serde(default)]
25    pub assets: String,
26    /// Repo name -> key of its rendered pages under dist/<repo>.
27    #[serde(default)]
28    pub repos: BTreeMap<String, String>,
29    /// "<repo>/<tool>" -> code reference built with that key.
30    #[serde(default)]
31    pub apidoc: BTreeMap<String, ApiState>,
32}

34#[derive(Serialize, Deserialize, Clone)]
35pub struct ApiState {
36    pub key: String,
37    pub entries: Vec<Api>,
38}

40impl State {
41    pub fn load(build: &Path) -> Self {
42        fs::read(build.join("state.json"))
43            .ok()
44            .and_then(|bytes| serde_json::from_slice(&bytes).ok())
45            .unwrap_or_default()
46    }
47
48    pub fn save(&self, build: &Path) -> Result<()> {
49        fs::create_dir_all(build)?;
50        fs::write(build.join("state.json"), serde_json::to_vec_pretty(self)?)?;
51        Ok(())
52    }
53}

55/// Blob id of a byte string (`git hash-object --stdin`); joins several parts.
56pub fn key<S: AsRef<str>>(parts: impl IntoIterator<Item = S>) -> String {
57    let mut bytes = Vec::new();
58    for part in parts {
59        bytes.extend_from_slice(part.as_ref().as_bytes());
60        bytes.push(0);
61    }
62    Oid::hash_object(ObjectType::Blob, &bytes).map(|oid| oid.to_string()).unwrap_or_default()
63}
64
65/// Blob id of a file (`git hash-object FILE`); empty when it can't be read.
66pub fn file_id(path: &Path) -> String {
67    Oid::hash_file(ObjectType::Blob, path).map(|oid| oid.to_string()).unwrap_or_default()
68}

70/// Blob ids of every file under `dir`, as "relative path\0id" parts.
71pub fn dir_ids(dir: &Path) -> Vec<String> {
72    let mut parts: Vec<String> = walkdir::WalkDir::new(dir)
73        .into_iter()
74        .filter_map(Result::ok)
75        .filter(|entry| entry.file_type().is_file())
76        .map(|entry| {
77            let rel = entry.path().strip_prefix(dir).unwrap_or(entry.path());
78            format!("{}\0{}", rel.display(), file_id(entry.path()))
79        })
80        .collect();
81    parts.sort();
82    parts
83}

85/// Id of a repo's current sources: HEAD's tree id plus the patch id of
86/// everything uncommitted (staged, unstaged and untracked).
87pub fn source_id(path: &Path) -> Result<String> {
88    let repo = Repository::open(path)?;
89    let tree = repo.head().ok().and_then(|head| head.peel_to_tree().ok());
90    let mut options = DiffOptions::new();
91    options.include_untracked(true).recurse_untracked_dirs(true).show_untracked_content(true);
92    let diff = repo.diff_tree_to_workdir_with_index(tree.as_ref(), Some(&mut options))?;
93    let tree_id = tree.map(|tree| tree.id().to_string()).unwrap_or_default();
94    let patch_id = if diff.deltas().len() == 0 { String::new() } else { diff.patchid(None)?.to_string() };
95    Ok(key([tree_id, patch_id]))
96}
