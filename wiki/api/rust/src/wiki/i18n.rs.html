1//! Site languages. The first locale is the source language (English) at the
2//! site root; every other one is a full copy of the site under its own prefix
3//! (`zh-TW` -> `zh-tw/`). The chrome comes from Fluent files
4//! (`locales/<tag>/*.ftl`, through fluent-templates); repo pages are
5//! machine-translated (see [`crate::translate`]).

21/// wiki.toml `[i18n]`.
22#[derive(Deserialize, Serialize, Default)]
23#[serde(default)]
24pub struct I18nConfig {
25    /// Languages besides English, as BCP 47 tags with a `locales/<tag>/` folder.
26    pub locales: Vec<String>,
27    pub translator: TranslatorConfig,
28}
29
30pub struct Locale {
31    pub tag: String,
32    pub lang: LanguageIdentifier,
33    /// Output path prefix: "" for the source language, else "<tag lowercased>/".
34    pub prefix: String,
35}
36
37impl Locale {
38    fn new(tag: &str, source: bool) -> Result<Self> {
39        let lang = tag.parse().map_err(|err| anyhow!("locale {tag}: {err}"))?;
40        let prefix = if source { String::new() } else { format!("{}/", tag.to_lowercase()) };
41        Ok(Self { tag: tag.to_owned(), lang, prefix })
42    }
43
44    pub fn is_source(&self) -> bool {
45        self.prefix.is_empty()
46    }
47}

49pub struct I18n {
50    /// The source language first.
51    pub locales: Vec<Locale>,
52    pub config: I18nConfig,
53    loader: Arc<ArcLoader>,
54}
55
56#[derive(Deserialize)]
57struct WikiToml {
58    #[serde(default)]
59    i18n: I18nConfig,
60}
61
62pub const SOURCE: &str = "en";
63
64impl I18n {
65    pub fn load(paths: &Paths) -> Result<Self> {
66        let toml_path = paths.wiki.join("wiki.toml");
67        let config = toml::from_str::<WikiToml>(&fs::read_to_string(&toml_path)?)
68            .with_context(|| format!("parsing {}", toml_path.display()))?
69            .i18n;
70        let mut locales = vec![Locale::new(SOURCE, true)?];
71        for tag in &config.locales {
72            locales.push(Locale::new(tag, false)?);
73        }
74        let loader = ArcLoader::builder(&paths.wiki.join("locales"), locales[0].lang.clone())
75            // Fluent wraps placeables in Unicode isolation marks; HTML doesn't need them.
76            .customize(|bundle| bundle.set_use_isolating(false))
77            .build()
78            .map_err(|err| anyhow!("loading locales/: {err}"))?;
79        Ok(Self { locales, config, loader: Arc::new(loader) })
80    }
81
82    pub fn source(&self) -> &Locale {
83        &self.locales[0]
84    }
85
86    pub fn t(&self, locale: &Locale, key: &str) -> String {
87        self.loader.lookup(&locale.lang, key)
88    }
89
90    /// Everything every page of every language depends on: the Fluent files and the locale list.
91    pub fn input_ids(&self, paths: &Paths) -> Vec<String> {
92        let mut ids = state::dir_ids(&paths.wiki.join("locales"));
93        ids.push(self.config.locales.join(","));
94        ids
95    }
96
97    /// The template function `t(key, **args)`.
98    pub fn function(&self, locale: &Locale) -> Value {
99        let loader = self.loader.clone();
100        let lang = locale.lang.clone();
101        Value::from_function(move |key: String, kwargs: Kwargs| -> Result<String, minijinja::Error> {
102            let mut args: HashMap<Cow<'static, str>, FluentValue> = HashMap::new();
103            for name in kwargs.args() {
104                let value: Value = kwargs.get(name)?;
105                let value = match f64::try_from(value.clone()) {
106                    Ok(number) if value.is_number() => FluentValue::from(number),
107                    _ => FluentValue::from(value.to_string()),
108                };
109                args.insert(Cow::Owned(name.to_owned()), value);
110            }
111            Ok(loader.lookup_complete(&lang, &key, Some(&args)))
112        })
113    }
114}
