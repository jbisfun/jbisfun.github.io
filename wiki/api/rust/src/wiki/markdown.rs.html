1//! Markdown for docs from many repos (comrak): GFM tables, autolinks, smart
2//! punctuation, `{#id}` heading attributes, syntect highlighting at build
3//! time (CSS classes, bat's syntaxes, no client JS), and `data-sourcepos` on
4//! every block so Radar can map a selection back to its source line. Links are rewritten
5//! afterwards with lol_html, which covers markdown and raw HTML alike.

16/// Class prefix of highlighted tokens; the bundle generates the matching CSS.

19pub static SCHEME: LazyLock<Regex> = LazyLock::new(|| Regex::new(r"(?i)^([a-z][a-z0-9+.-]*:|//|#)").expect("valid regex"));

26impl Markdown {
27    pub fn new() -> Self {
28        let mut options = Options::default();
29        let extension = &mut options.extension;
30        extension.table = true;
31        extension.strikethrough = true;
32        extension.autolink = true;
33        extension.tasklist = true;
34        extension.footnotes = true;
35        extension.header_attributes = true;
36        extension.header_id_prefix = Some(String::new());
37        extension.front_matter_delimiter = Some("---".into());
38        options.parse.smart = true;
39        options.render.r#unsafe = true;
40        options.render.attributes = true;
41        options.render.sourcepos = true;
42        // bat's syntax set (two-face): syntect's defaults lack TOML, TypeScript, Zig, ...
43        let highlighter = SyntectAdapterBuilder::new()
44            .syntax_set(two_face::syntax::extra_newlines())
45            .css_with_class_prefix(HIGHLIGHT_PREFIX)
46            .build();
47        Self { options, highlighter }
48    }

50    /// HTML of a markdown document (front matter is skipped; source positions
51    /// count from the document's first line).
52    pub fn render(&self, source: &str) -> String {
53        let mut plugins = Plugins::default();
54        plugins.render.codefence_syntax_highlighter = Some(&self.highlighter);
55        comrak::markdown_to_html_with_plugins(&preprocess(source), &self.options, &plugins)
56    }

58    /// Home page blurb: the first sentence of the first paragraph as plain
59    /// text (link and emphasis text kept, markup dropped), cut at its first
60    /// top-level comma, semicolon or colon when it runs past `BLURB` chars.
61    pub fn blurb(&self, source: &str) -> Option<String> {
62        const BLURB: usize = 80;
63        let arena = Arena::new();
64        let root = parse_document(&arena, &preprocess(source), &self.options);
65        let paragraph = root
66            .descendants()
67            .find(|node| matches!(node.data().value, NodeValue::Paragraph))?;
68        let mut text = String::new();
69        for node in paragraph.descendants() {
70            match &node.data().value {
71                NodeValue::Text(t) => text.push_str(t),
72                NodeValue::Code(code) => text.push_str(&code.literal),
73                NodeValue::SoftBreak | NodeValue::LineBreak => text.push(' '),
74                _ => {}
75            }
76        }
77        let text = text.trim();
78        let end = [text.find(". ").map(|at| at + 1), text.find('。').map(|at| at + '。'.len_utf8())]
79            .into_iter()
80            .flatten()
81            .min()
82            .unwrap_or(text.len());
83        let sentence = text[..end].trim();
84        if sentence.is_empty() {
85            return None;
86        }
87        // CJK characters are about two Latin ones wide.
88        let width: usize = sentence.chars().map(|ch| if ch >= '\u{2E80}' { 2 } else { 1 }).sum();
89        if width > BLURB {
90            let mut depth = 0;
91            for (at, ch) in sentence.char_indices() {
92                match ch {
93                    '(' | '[' | '（' | '「' => depth += 1,
94                    ')' | ']' | '）' | '」' => depth -= 1,
95                    ',' | ';' | ':' if depth == 0 => return Some(format!("{}.", &sentence[..at])),
96                    '，' | '；' | '：' if depth == 0 => return Some(format!("{}。", &sentence[..at])),
97                    _ => {}
98                }
99            }
100        }
101        Some(sentence.to_owned())
102    }
103}

105/// Join soft line breaks between CJK characters (outside code blocks): a
106/// browser renders a newline as a space, which CJK text doesn't have.
107pub fn join_cjk_lines(source: &str) -> String {
108    static BREAK: LazyLock<Regex> = LazyLock::new(|| {
109        Regex::new(r"([\p{Han}\p{Hiragana}\p{Katakana}\p{Hangul}，。；：、！？）」』])\n[ \t]*([\p{Han}\p{Hiragana}\p{Katakana}\p{Hangul}（「『])")
110            .expect("valid regex")
111    });
112    let mut out = String::with_capacity(source.len());
113    let mut fenced = false;
114    let mut prose = String::new();
115    for line in source.split_inclusive('\n') {
116        if line.trim_start().starts_with("```") || line.trim_start().starts_with("~~~") {
117            if !fenced {
118                out.push_str(&BREAK.replace_all(&prose, "$1$2"));
119                prose.clear();
120            }
121            fenced = !fenced;
122            out.push_str(line);
123        } else if fenced {
124            out.push_str(line);
125        } else {
126            prose.push_str(line);
127        }
128    }
129    out.push_str(&BREAK.replace_all(&prose, "$1$2"));
130    out
131}

133/// Replace the mdBook preprocessor commands used by the zed docs
134/// (`{#kb action}` / `{#action action}`) with code spans.
135fn preprocess(source: &str) -> String {
136    static COMMAND: LazyLock<Regex> = LazyLock::new(|| Regex::new(r"\{#(?:kb|action)\s+([^}]+)\}").expect("valid regex"));
137    COMMAND.replace_all(source, |found: &regex::Captures| format!("`{}`", found[1].trim())).into_owned()
138}

140/// Rewrite every `a[href]` / `img[src]` through `resolve` (None keeps the
141/// link) and give tables Bootstrap's look.
142pub fn rewrite_links(html: &str, resolve: impl Fn(&str) -> Option<String>) -> Result<String> {
143    let resolve = &resolve;
144    let settings = RewriteStrSettings::new()
145        .append_element_content_handler(element!("a[href]", move |el| {
146            if let Some(link) = el.get_attribute("href").and_then(|href| resolve(&href)) {
147                el.set_attribute("href", &link)?;
148            }
149            Ok(())
150        }))
151        .append_element_content_handler(element!("img[src]", move |el| {
152            if let Some(link) = el.get_attribute("src").and_then(|src| resolve(&src)) {
153                el.set_attribute("src", &link)?;
154            }
155            Ok(())
156        }))
157        .append_element_content_handler(element!("table", |el| {
158            el.set_attribute("class", "table table-sm table-striped")?;
159            Ok(())
160        }));
161    Ok(rewrite_str(html, settings)?)
162}

164const HEADINGS: &str = "h1[id], h2[id], h3[id], h4[id], h5[id], h6[id]";

166/// Ids of the headings in rendered HTML, in document order.
167pub fn heading_ids(html: &str) -> Result<Vec<String>> {
168    let ids = std::cell::RefCell::new(Vec::new());
169    let settings = RewriteStrSettings::new().append_element_content_handler(element!(HEADINGS, |el| {
170        ids.borrow_mut().extend(el.get_attribute("id"));
171        Ok(())
172    }));
173    rewrite_str(html, settings)?;
174    Ok(ids.into_inner())
175}

177/// Give the headings of `html` the ids `ids` (in order) and point their `#`
178/// links there too, so a translated page keeps the English anchors that links
179/// from other pages use. Left alone when the heading counts differ.
180pub fn set_heading_ids(html: &str, ids: &[String]) -> Result<String> {
181    let current = heading_ids(html)?;
182    if current.len() != ids.len() || current == ids {
183        return Ok(html.to_owned());
184    }
185    let map: std::collections::HashMap<&str, &str> = current.iter().map(String::as_str).zip(ids.iter().map(String::as_str)).collect();
186    let map = &map;
187    let settings = RewriteStrSettings::new()
188        .append_element_content_handler(element!(HEADINGS, move |el| {
189            if let Some(id) = el.get_attribute("id").and_then(|id| map.get(id.as_str()).copied()) {
190                el.set_attribute("id", id)?;
191            }
192            Ok(())
193        }))
194        .append_element_content_handler(element!("a[href^='#']", move |el| {
195            if let Some(id) = el.get_attribute("href").and_then(|href| map.get(&href[1..]).copied()) {
196                el.set_attribute("href", &format!("#{id}"))?;
197            }
198            Ok(())
199        }));
200    Ok(rewrite_str(html, settings)?)
201}

203/// Mark the links to `href` (the current page) as active.
204pub fn mark_active(html: &str, href: &str) -> Result<String> {
205    let settings = RewriteStrSettings::new().append_element_content_handler(element!("a[href]", |el| {
206        if el.get_attribute("href").as_deref() == Some(href) {
207            el.set_attribute("class", "active")?;
208            el.set_attribute("aria-current", "page")?;
209        }
210        Ok(())
211    }));
212    Ok(rewrite_str(html, settings)?)
213}
