1//! Markdown for docs from many repos (comrak): GFM tables, autolinks, smart
2//! punctuation, `{#id}` heading attributes, syntect highlighting at build
3//! time (CSS classes, bat's syntaxes, no client JS), and `data-sourcepos` on every block so
4//! Radar can map a selection back to its source line. Links are rewritten
5//! afterwards with lol_html, which covers markdown and raw HTML alike.
6use std::sync::LazyLock;
7
8use anyhow::Result;
9use comrak::nodes::NodeValue;
10use comrak::options::Plugins;
11use comrak::plugins::syntect::{SyntectAdapter, SyntectAdapterBuilder};
12use comrak::{Arena, Options, parse_document};
13use lol_html::{RewriteStrSettings, element, rewrite_str};
14use regex::Regex;
15
16/// Class prefix of highlighted tokens; the bundle generates the matching CSS.
17pub const HIGHLIGHT_PREFIX: &str = "hl-";
18
19pub static SCHEME: LazyLock<Regex> = LazyLock::new(|| Regex::new(r"(?i)^([a-z][a-z0-9+.-]*:|//|#)").expect("valid regex"));
20
21pub struct Markdown {
22    options: Options<'static>,
23    highlighter: SyntectAdapter,
24}
25
26impl Markdown {
27    pub fn new() -> Self {
28        let mut options = Options::default();
29        let extension = &mut options.extension;
30        extension.table = true;
31        extension.strikethrough = true;
32        extension.autolink = true;
33        extension.tasklist = true;
34        extension.footnotes = true;
35        extension.header_attributes = true;
36        extension.header_id_prefix = Some(String::new());
37        extension.front_matter_delimiter = Some("---".into());
38        options.parse.smart = true;
39        options.render.r#unsafe = true;
40        options.render.attributes = true;
41        options.render.sourcepos = true;
42        // bat's syntax set (two-face): syntect's defaults lack TOML, TypeScript, Zig, ...
43        let highlighter = SyntectAdapterBuilder::new()
44            .syntax_set(two_face::syntax::extra_newlines())
45            .css_with_class_prefix(HIGHLIGHT_PREFIX)
46            .build();
47        Self { options, highlighter }
48    }
49
50    /// HTML of a markdown document (front matter is skipped; source positions
51    /// count from the document's first line).
52    pub fn render(&self, source: &str) -> String {
53        let mut plugins = Plugins::default();
54        plugins.render.codefence_syntax_highlighter = Some(&self.highlighter);
55        comrak::markdown_to_html_with_plugins(&preprocess(source), &self.options, &plugins)
56    }
57
58    /// Plain text of the first paragraph with text (not just badges), for the home page cards.
59    /// The first sentence of the first paragraph, as plain text (link and
60    /// emphasis text kept, markup dropped).
61    pub fn first_sentence(&self, source: &str) -> Option<String> {
62        let arena = Arena::new();
63        let root = parse_document(&arena, &preprocess(source), &self.options);
64        let paragraph = root
65            .descendants()
66            .find(|node| matches!(node.data().value, NodeValue::Paragraph))?;
67        let mut text = String::new();
68        for node in paragraph.descendants() {
69            match &node.data().value {
70                NodeValue::Text(t) => text.push_str(t),
71                NodeValue::Code(code) => text.push_str(&code.literal),
72                NodeValue::SoftBreak | NodeValue::LineBreak => text.push(' '),
73                _ => {}
74            }
75        }
76        let text = text.trim();
77        let end = text
78            .match_indices(". ")
79            .map(|(at, _)| at + 1)
80            .next()
81            .unwrap_or(text.len());
82        let sentence = text[..end].trim();
83        (!sentence.is_empty()).then(|| sentence.to_owned())
84    }
85}

87/// Replace the mdBook preprocessor commands used by the zed docs
88/// (`{#kb action}` / `{#action action}`) with code spans.
89pub fn preprocess(source: &str) -> String {
90    static COMMAND: LazyLock<Regex> = LazyLock::new(|| Regex::new(r"\{#(?:kb|action)\s+([^}]+)\}").expect("valid regex"));
91    COMMAND.replace_all(source, |found: &regex::Captures| format!("`{}`", found[1].trim())).into_owned()
92}

94/// Rewrite every `a[href]` / `img[src]` through `resolve` (None keeps the
95/// link) and give tables Bootstrap's look.
96pub fn rewrite_links(html: &str, resolve: impl Fn(&str) -> Option<String>) -> Result<String> {
97    let resolve = &resolve;
98    let settings = RewriteStrSettings::new()
99        .append_element_content_handler(element!("a[href]", move |el| {
100            if let Some(link) = el.get_attribute("href").and_then(|href| resolve(&href)) {
101                el.set_attribute("href", &link)?;
102            }
103            Ok(())
104        }))
105        .append_element_content_handler(element!("img[src]", move |el| {
106            if let Some(link) = el.get_attribute("src").and_then(|src| resolve(&src)) {
107                el.set_attribute("src", &link)?;
108            }
109            Ok(())
110        }))
111        .append_element_content_handler(element!("table", |el| {
112            el.set_attribute("class", "table table-sm table-striped")?;
113            Ok(())
114        }));
115    Ok(rewrite_str(html, settings)?)
116}

118/// Mark the links to `href` (the current page) as active.
119pub fn mark_active(html: &str, href: &str) -> Result<String> {
120    let settings = RewriteStrSettings::new().append_element_content_handler(element!("a[href]", |el| {
121        if el.get_attribute("href").as_deref() == Some(href) {
122            el.set_attribute("class", "active")?;
123            el.set_attribute("aria-current", "page")?;
124        }
125        Ok(())
126    }));
127    Ok(rewrite_str(html, settings)?)
128}
