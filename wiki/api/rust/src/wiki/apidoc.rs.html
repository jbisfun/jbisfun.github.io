1//! Code reference per repo, by each language's own tool, into
2//! `.build/api/<repo>/<tool>/`:
3//!   rust: `cargo doc --no-deps` (rustdoc)   lua: LDoc   js: JSDoc
4//! A tool that is missing or fails only warns: the site still builds. A
5//! reference is rebuilt only when the repo's source id (see [`state`]) or its
6//! wiki.toml settings changed.
7use std::collections::BTreeMap;
8use std::fs;
9use std::path::{Path, PathBuf};
10use std::process::Command;
11
12use anyhow::Result;
13use regex::Regex;
14use serde::{Deserialize, Serialize};

20/// One code reference page set: `path` is relative to `.build/api/<repo>`.
21#[derive(Clone, Serialize, Deserialize)]
22pub struct Api {
23    pub tool: String,
24    pub title: String,
25    pub path: String,
26}
27
28/// Repo name -> its code references.
29pub fn generate(paths: &Paths, repos: &[Repo], state: &mut State) -> Result<BTreeMap<String, Vec<Api>>> {
30    let mut apis = BTreeMap::new();
31    let mut kept = BTreeMap::new();
32    for repo in repos {
33        let source = match state::source_id(&repo.path) {
34            Ok(id) => id,
35            Err(err) => {
36                warn(format_args!("{}: {err}", repo.name));
37                continue;
38            }
39        };
40        let mut entries = Vec::new();
41        let tools = [("rust", &repo.config.rust), ("lua", &repo.config.lua), ("js", &repo.config.js)];
42        for (tool, config) in tools {
43            let Some(settings) = config.settings() else { continue };
44            let id = format!("{}/{tool}", repo.name);
45            let key = state::key([source.as_str(), tool, &serde_json::to_string(&settings)?]);
46            let dest = paths.build.join("api").join(&repo.name).join(tool);
47            let built = match state.apidoc.get(&id) {
48                Some(previous) if previous.key == key && (previous.entries.is_empty() || dest.is_dir()) => {
49                    previous.entries.clone()
50                }
51                _ => {
52                    let job = Job { paths, repo, settings: &settings, dest: &dest };
53                    match tool {
54                        "rust" => job.rust(),
55                        "lua" => job.lua(),
56                        _ => job.js(),
57                    }
58                }
59            };
60            entries.extend(built.iter().cloned());
61            kept.insert(id, ApiState { key, entries: built });
62        }
63        if !entries.is_empty() {
64            apis.insert(repo.name.clone(), entries);
65        }
66    }
67    state.apidoc = kept;
68    Ok(apis)
69}
70
71struct Job<'a> {
72    paths: &'a Paths,
73    repo: &'a Repo,
74    settings: &'a ToolSettings,
75    dest: &'a Path,
76}
77
78impl Job<'_> {
79    fn warn(&self, message: impl std::fmt::Display) {
80        warn(format_args!("{}: {message}", self.repo.name));
81    }
82
83    fn entry(&self, tool: &str, title: &str, path: String) -> Api {
84        Api { tool: tool.into(), title: title.into(), path }
85    }
86
87    /// Run a doc tool; on failure warn with its output and return false.
88    fn run(&self, command: &mut Command, what: &str) -> bool {
89        match command.output() {
90            Ok(output) if output.status.success() => true,
91            Ok(output) => {
92                self.warn(format_args!("{what} failed"));
93                eprint!("{}{}", String::from_utf8_lossy(&output.stdout), String::from_utf8_lossy(&output.stderr));
94                false
95            }
96            Err(err) => {
97                self.warn(format_args!("{what} not available ({err}), skipping"));
98                false
99            }
100        }
101    }
102
103    fn reset_dest(&self) -> Result<()> {
104        if self.dest.exists() {
105            fs::remove_dir_all(self.dest)?;
106        }
107        fs::create_dir_all(self.dest)?;
108        Ok(())
109    }
110
111    /// Tracked files matching the `files` pathspecs, or the defaults.
112    fn tracked_files(&self, defaults: &[&str]) -> Vec<String> {
113        let specs: Vec<&str> = if self.settings.files.is_empty() {
114            defaults.to_vec()
115        } else {
116            self.settings.files.iter().map(String::as_str).collect()
117        };
118        let Ok(repo) = git2::Repository::open(&self.repo.path) else { return Vec::new() };
119        let (Ok(index), Ok(pathspec)) = (repo.index(), git2::Pathspec::new(specs.iter().copied())) else {
120            return Vec::new();
121        };
122        index
123            .iter()
124            .map(|entry| String::from_utf8_lossy(&entry.path).into_owned())
125            .filter(|path| pathspec.matches_path(Path::new(path), git2::PathspecFlags::DEFAULT))
126            .collect()
127    }
128
129    fn any_file_contains(&self, files: &[String], pattern: &Regex) -> bool {
130        files.iter().any(|file| fs::read_to_string(self.repo.path.join(file)).is_ok_and(|text| pattern.is_match(&text)))
131    }
132
133    fn rust(&self) -> Vec<Api> {
134        if !self.repo.path.join("Cargo.toml").is_file() {
135            return Vec::new();
136        }
137        self.try_rust().unwrap_or_else(|err| {
138            self.warn(format_args!("rustdoc: {err}"));
139            Vec::new()
140        })
141    }
142
143    fn try_rust(&self) -> Result<Vec<Api>> {
144        // The repo's own target dir, so dependency metadata is shared with its builds.
145        let metadata = Command::new("cargo")
146            .args(["metadata", "--format-version", "1", "--no-deps"])
147            .current_dir(&self.repo.path)
148            .output()?;
149        let metadata: serde_json::Value = serde_json::from_slice(&metadata.stdout)?;
150        let target = PathBuf::from(metadata["target_directory"].as_str().unwrap_or("target"));
151        let mut cargo = Command::new("cargo");
152        cargo.args(["doc", "--no-deps", "--quiet"]).current_dir(&self.repo.path);
153        for package in &self.settings.packages {
154            cargo.args(["-p", package]);
155        }
156        if !self.run(&mut cargo, "cargo doc") {
157            return Ok(Vec::new());
158        }
159        self.reset_dest()?;
160        copy_dir(&target.join("doc"), self.dest)?;
161        let crates_js = fs::read_to_string(self.dest.join("crates.js"))?;
162        let list = Regex::new(r"(?m)^window\.ALL_CRATES = (.*);$")?
163            .captures(&crates_js)
164            .and_then(|found| serde_json::from_str::<Vec<String>>(&found[1]).ok())
165            .unwrap_or_default();
166        Ok(list.iter().map(|name| self.entry("rust", name, format!("rust/{name}/index.html"))).collect())
167    }
168
169    fn lua(&self) -> Vec<Api> {
170        // *.x.lua are preprocessed ycode hook scripts, not plain Lua modules.
171        let defaulted = self.settings.files.is_empty();
172        let files: Vec<String> = self
173            .tracked_files(&["*.lua"])
174            .into_iter()
175            .filter(|file| !(defaulted && file.ends_with(".x.lua")))
176            .collect();
177        // LDoc only documents `---` comments; without any there is nothing to show.
178        let doc_comment = Regex::new(r"(?m)^\s*---").expect("valid regex");
179        if files.is_empty() || !self.any_file_contains(&files, &doc_comment) {
180            return Vec::new();
181        }
182        self.try_lua(&files).unwrap_or_else(|err| {
183            self.warn(format_args!("ldoc: {err}"));
184            Vec::new()
185        })
186    }
187
188    fn try_lua(&self, files: &[String]) -> Result<Vec<Api>> {
189        self.reset_dest()?;
190        let name = &self.repo.name;
191        let conf = self.dest.with_file_name("config.ld");
192        // JSON strings double as Lua string literals. "backtick": Debian ships
193        // LDoc without its markdown module, so only `code` spans are formatted.
194        let json = |text: &str| serde_json::to_string(text).unwrap_or_default();
195        let mut config = format!(
196            "project = {}\ntitle = {}\ndir = {}\nformat = \"backtick\"\nfile = {{\n",
197            json(name),
198            json(&format!("{name} Lua API")),
199            json(&self.dest.to_string_lossy()),
200        );
201        for file in files {
202            config.push_str(&format!("  {},\n", json(&self.repo.path.join(file).to_string_lossy())));
203        }
204        config.push_str("}\n");
205        fs::write(&conf, config)?;
206        let ok = self.run(Command::new("ldoc").arg("--testing").arg("-c").arg(&conf).arg(".").current_dir(&self.repo.path), "ldoc");
207        fs::remove_file(&conf)?;
208        Ok(if ok { vec![self.entry("lua", name, "lua/index.html".into())] } else { Vec::new() })
209    }
210
211    fn js(&self) -> Vec<Api> {
212        if !self.repo.path.join("package.json").is_file() {
213            return Vec::new();
214        }
215        let files = self.tracked_files(&["*.js", "*.mjs"]);
216        let doc_comment = Regex::new(r"/\*\*").expect("valid regex");
217        if files.is_empty() || !self.any_file_contains(&files, &doc_comment) {
218            return Vec::new();
219        }
220        let jsdoc = self.paths.wiki.join("node_modules/.bin/jsdoc");
221        if !jsdoc.is_file() {
222            self.warn("jsdoc not installed (just install), skipping");
223            return Vec::new();
224        }
225        if let Err(err) = self.reset_dest() {
226            self.warn(err);
227            return Vec::new();
228        }
229        // jsdoc.json drops the generation date, so unchanged code gives unchanged pages.
230        let mut command = Command::new(jsdoc);
231        command
232            .arg("-c")
233            .arg(self.paths.wiki.join("jsdoc.json"))
234            .arg("-d")
235            .arg(self.dest)
236            .args(&files)
237            .current_dir(&self.repo.path);
238        if self.run(&mut command, "jsdoc") {
239            vec![self.entry("js", &self.repo.name, "js/index.html".into())]
240        } else {
241            Vec::new()
242        }
243    }
244}

246/// Copy the files under `from` into `to`, keeping the layout.
247pub fn copy_dir(from: &Path, to: &Path) -> Result<()> {
248    for entry in walkdir::WalkDir::new(from) {
249        let entry = entry?;
250        let dest = to.join(entry.path().strip_prefix(from)?);
251        if entry.file_type().is_dir() {
252            fs::create_dir_all(&dest)?;
253        } else {
254            fs::copy(entry.path(), &dest)?;
255        }
256    }
257    Ok(())
