1//! List elements: `list_item`, `list_header` and `section`.
2
3use super::LuaWindow;
4use super::props::Props;
5use gpui::{AnyElement, Context, IntoElement, ParentElement as _, SharedString, Styled as _, Window, div};
6use mlua::Table;
7use ui::{LabelCommon as _, LabelSize, ListHeader, ListItem, ListItemSpacing, ListSubHeader, Toggleable as _};
8
9impl LuaWindow {
10    /// `list_item{ "label" | children, detail, selected, indent, start, end_,
11    /// end_hover, on_click }`: a `ui::ListItem` (dense, inset, rounded hover).
12    pub(super) fn build_list_item(
13        &mut self,
14        t: &Table,
15        p: &Props,
16        start: usize,
17        window: &mut Window,
18        cx: &mut Context<Self>,
19    ) -> mlua::Result<AnyElement> {
20        let id = self.id_or_auto(p, "list-item");
21        let start_slot = self.slot(t, "start", window, cx)?;
22        let end_slot = self.slot(t, "end_", window, cx)?;
23        let end_hover = self.slot(t, "end_hover", window, cx)?;
24        let selected = p.selected.as_ref().is_some_and(|s| s.is_true());
25        let mut item = ListItem::new(SharedString::from(id))
26            .spacing(ListItemSpacing::Dense)
27            .inset(true)
28            .indent_level(p.indent.unwrap_or(0))
29            .toggle_state(selected)
30            .start_slot::<AnyElement>(start_slot)
31            .end_slot::<AnyElement>(end_slot);
32        if let Some(h) = end_hover {
33            item = item.end_slot_on_hover(h);
34        }
35        if let Some(tt) = p.tooltip.clone() {
36            item = item.tooltip(ui::Tooltip::text(tt));
37        }
38        if let Some(f) = Self::callback(t, "on_click")? {
39            item = item.on_click(self.click(f, cx));
40        }
41        // A string label (plus optional detail) reads like a native row; other
42        // children are laid out as given.
43        if let Some(label) = Self::text_of_strings(t, start)? {
44            let mut row = ui::h_flex()
45                .min_w_0()
46                .gap(ui::DynamicSpacing::Base06.rems(cx))
47                .child(Self::label(label, p, None, cx)?.truncate());
48            if let Some(detail) = p.detail.clone() {
49                row = row.child(
50                    ui::Label::new(detail)
51                        .size(LabelSize::Small)
52                        .color(ui::Color::Muted)
53                        .truncate(),
54                );
55            }
56            item = item.child(row);
57        } else {
58            item = item.children(self.children(t, start, window, cx)?);
59        }
60        Ok(item.into_any_element())
61    }

63    /// The positional children if they are all plain strings/numbers.
64    fn text_of_strings(t: &Table, start: usize) -> mlua::Result<Option<String>> {
65        let len = t.raw_len();
66        if start > len {
67            return Ok(None);
68        }
69        for i in start..=len {
70            if matches!(t.raw_get::<mlua::Value>(i)?, mlua::Value::Table(_)) {
71                return Ok(None);
72            }
73        }
74        Self::text_of(t, start)
75    }

77    /// `list_header{ "label", sub, start, end_, end_hover, on_click }`:
78    /// `ui::ListHeader`, or `ui::ListSubHeader` with `sub = true`.
79    pub(super) fn build_list_header(
80        &mut self,
81        t: &Table,
82        p: &Props,
83        start: usize,
84        window: &mut Window,
85        cx: &mut Context<Self>,
86    ) -> mlua::Result<AnyElement> {
87        let label = Self::text_of(t, start)?.or_else(|| p.label.clone()).unwrap_or_default();
88        let end_slot = self.slot(t, "end_", window, cx)?;
89        let header = if p.sub {
90            let mut header = ListSubHeader::new(label).inset(true);
91            if let Some(end_slot) = end_slot {
92                header = header.end_slot(end_slot);
93            }
94            header.into_any_element()
95        } else {
96            let start_slot = self.slot(t, "start", window, cx)?;
97            let end_hover = self.slot(t, "end_hover", window, cx)?;
98            let mut header = ListHeader::new(label)
99                .inset(true)
100                .start_slot::<AnyElement>(start_slot)
101                .end_slot::<AnyElement>(end_slot)
102                .end_hover_slot::<AnyElement>(end_hover);
103            if let Some(f) = Self::callback(t, "on_click")? {
104                let click = self.click(f, cx);
105                header = header.on_toggle(click);
106            }
107            header.into_any_element()
108        };
109        // Native headers are flex_1; in a Lua column that stretches them.
110        Ok(div().flex_none().w_full().child(header).into_any_element())
111    }

113    /// `section{ "Title", action, empty, children }`: a sub-header and its rows.
114    pub(super) fn build_section(
115        &mut self,
116        t: &Table,
117        p: &Props,
118        start: usize,
119        window: &mut Window,
120        cx: &mut Context<Self>,
121    ) -> mlua::Result<AnyElement> {
122        let mut title: Option<String> = p.title.clone();
123        let mut first = start;
124        if title.is_none() {
125            if let mlua::Value::String(s) = t.raw_get::<mlua::Value>(start)? {
126                title = Some(s.to_string_lossy());
127                first = start + 1;
128            }
129        }
130        let mut col = ui::v_flex().flex_none().w_full();
131        if let Some(title) = title {
132            let mut header = ListSubHeader::new(title).inset(true);
133            if let Some(action) = self.slot(t, "action", window, cx)? {
134                header = header.end_slot(action);
135            }
136            col = col.child(div().flex_none().w_full().child(header));
137        }
138        let rows = self.children(t, first, window, cx)?;
139        if rows.is_empty() {
140            if let Some(empty) = p.empty.clone() {
141                // Wraps (errors can be long); indented like list_item labels.
142                col = col.child(
143                    div()
144                        .w_full()
145                        .pl_2p5()
146                        .pr_2()
147                        .py(ui::DynamicSpacing::Base02.rems(cx))
148                        .child(
149                            ui::Label::new(empty)
150                                .size(LabelSize::Small)
151                                .color(ui::Color::Placeholder),
152                        ),
153                );
154            }
155        }
156        Ok(col.children(rows).pb(ui::DynamicSpacing::Base04.rems(cx)).into_any_element())
157    }
