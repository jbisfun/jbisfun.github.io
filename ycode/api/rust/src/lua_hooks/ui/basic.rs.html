14    /// `text` is a `ui::Label`: native size and color, UI font unless `mono`.
15    pub(super) fn build_text(
16        &mut self,
17        t: &Table,
18        p: &Props,
19        start: usize,
20        cx: &mut Context<Self>,
21    ) -> mlua::Result<AnyElement> {
22        let s = Self::text_of(t, start)?.or_else(|| p.label.clone()).unwrap_or_default();
23        let label = Self::label(s, p, None, cx)?;
24        // Clickable text (file links) gets a pointer and an underline on hover.
25        match Self::callback(t, "on_click")? {
26            None => Ok(label.into_any_element()),
27            Some(f) => {
28                use gpui::{InteractiveElement as _, ParentElement as _, StatefulInteractiveElement as _, Styled as _};
29                let id = self.id_or_auto(p, "text");
30                Ok(gpui::div()
31                    .id(SharedString::from(id))
32                    .cursor_pointer()
33                    .hover(|s| s.underline())
34                    .on_click(self.click(f, cx))
35                    .child(label)
36                    .into_any_element())
37            }
38        }
39    }

41    /// A `ui::Label` styled from `size color mono bold truncate`.
42    pub(super) fn label(
43        s: impl Into<SharedString>,
44        p: &Props,
45        default_size: Option<LabelSize>,
46        cx: &App,
47    ) -> mlua::Result<Label> {
48        let mut label = Label::new(s);
49        match &p.size {
50            Some(size) => label = label.size(props::label_size(size)?),
51            None => {
52                if let Some(size) = default_size {
53                    label = label.size(size);
54                }
55            }
56        }
57        if let Some(c) = &p.color {
58            label = label.color(props::color(c, cx)?);
59        }
60        if p.mono {
61            label = label.buffer_font(cx);
62        }
63        if p.bold {
64            label = label.weight(gpui::FontWeight::SEMIBOLD);
65        }
66        if p.truncate {
67            label = label.truncate();
68        }
69        Ok(label)
70    }

72    /// `button` (`ui::Button`, small label, subtle by default) and
73    /// `icon_button` (`ui::IconButton`, small icon).
74    pub(super) fn build_button(
75        &mut self,
76        tag: &str,
77        t: &Table,
78        p: &Props,
79        start: usize,
80        cx: &mut Context<Self>,
81    ) -> mlua::Result<AnyElement> {
82        let id = self.id_or_auto(p, tag);
83        let click = Self::callback(t, "on_click")?.map(|f| self.click(f, cx));
84        let icon = p.icon.as_deref().map(props::icon_name).transpose()?;
85        let style = match p.style.as_deref() {
86            None | Some("subtle") => ui::ButtonStyle::Subtle,
87            Some("filled") => ui::ButtonStyle::Filled,
88            Some("transparent") => ui::ButtonStyle::Transparent,
89            Some(other) => {
90                return Err(rt_err(format!("bad style {other:?}, expected subtle|filled|transparent")));
91            }
92        };
93
94        if tag == "icon_button" {
95            let icon = icon.ok_or_else(|| rt_err("needs an `icon`"))?;
96            let mut btn = ui::IconButton::new(id, icon)
97                .icon_size(IconSize::Small)
98                .style(style)
99                .disabled(p.disabled);
100            if let Some(color) = &p.color {
101                btn = btn.icon_color(props::color(color, cx)?);
102            }
103            if let Some(tt) = p.tooltip.clone() {
104                btn = btn.tooltip(Tooltip::text(tt));
105            }
106            if let Some(f) = click {
107                btn = btn.on_click(f);
108            }
109            return Ok(btn.into_any_element());
110        }
111
112        let label = Self::text_of(t, start)?.or_else(|| p.label.clone()).unwrap_or_default();
113        let mut btn = ui::Button::new(id, label)
114            .label_size(LabelSize::Small)
115            .style(style)
116            .disabled(p.disabled);
117        if let Some(icon) = icon {
118            btn = btn.start_icon(Icon::new(icon).size(IconSize::Small));
119        }
120        if let Some(color) = &p.color {
121            btn = btn.color(props::color(color, cx)?);
122        }
123        if let Some(tt) = p.tooltip.clone() {
124            btn = btn.tooltip(Tooltip::text(tt));
125        }
126        if let Some(f) = click {
127            btn = btn.on_click(f);
128        }
129        Ok(btn.into_any_element())
130    }

132    /// `icon{ name, size, color }` (`ui::Icon`, small by default).
133    pub(super) fn build_icon(p: &Props, cx: &App) -> mlua::Result<AnyElement> {
134        let name = p.name.as_deref().or(p.icon.as_deref()).ok_or_else(|| rt_err("needs a `name`"))?;
135        let mut icon = Icon::new(props::icon_name(name)?).size(match &p.size {
136            Some(s) => props::icon_size(s)?,
137            None => IconSize::Small,
138        });
139        if let Some(c) = &p.color {
140            icon = icon.color(props::color(c, cx)?);
141        }
142        Ok(icon.into_any_element())
143    }

145    /// `checkbox{ checked = true|false|"mixed", label, size, disabled, on_click }`.
146    pub(super) fn build_checkbox(
147        &mut self,
148        t: &Table,
149        p: &Props,
150        cx: &mut Context<Self>,
151    ) -> mlua::Result<AnyElement> {
152        let checked = match &p.checked {
153            None | Some(Toggle::Bool(false)) => ToggleState::Unselected,
154            Some(Toggle::Bool(true)) => ToggleState::Selected,
155            Some(Toggle::Str(s)) if s == "mixed" => ToggleState::Indeterminate,
156            Some(other) => {
157                return Err(rt_err(format!("bad checked {other:?}, expected true|false|\"mixed\"")));
158            }
159        };
160        let id = self.id_or_auto(p, "checkbox");
161        let mut checkbox = Checkbox::new(SharedString::from(id), checked).disabled(p.disabled);
162        if let Some(label) = p.label.clone() {
163            let size = match &p.size {
164                Some(s) => props::label_size(s)?,
165                None => LabelSize::Small,
166            };
167            checkbox = checkbox.label(label).label_size(size);
168        }
169        if let Some(f) = Self::callback(t, "on_click")? {
170            let entity = cx.entity();
171            checkbox = checkbox.on_click(move |_, window, cx| {
172                entity.update(cx, |this, cx| {
173                    this.event(&f, Value::Nil, window, cx);
174                });
175            });
176        }
177        Ok(checkbox.into_any_element())
178    }

180    /// `toggle_group{ options, selected, on_select = fn(w, option) }`
181    /// (`ui::ToggleButtonGroup`, 1-6 options).
182    pub(super) fn build_toggle_group(
183        &mut self,
184        t: &Table,
185        p: &Props,
186        cx: &mut Context<Self>,
187    ) -> mlua::Result<AnyElement> {
188        let options = p.options.clone().unwrap_or_default();
189        if options.is_empty() {
190            return Err(rt_err("needs an `options` list"));
191        }
192        let selected_index = match &p.selected {
193            Some(Toggle::Str(s)) => options.iter().position(|o| o == s).unwrap_or(0),
194            _ => 0,
195        };
196        let group_name = SharedString::from(self.id_or_auto(p, "toggle-group"));
197        let on_select = Self::callback(t, "on_select")?;
198        let entity = cx.entity();
199        let rt = self.rt.clone();
200        let buttons: Vec<ToggleButtonSimple> = options
201            .into_iter()
202            .map(|opt| {
203                let f = on_select.clone();
204                let entity = entity.clone();
205                let rt = rt.clone();
206                ToggleButtonSimple::new(opt.clone(), move |_, window, cx| {
207                    let Some(f) = f.clone() else { return };
208                    let Ok(value) = rt.lua().create_string(&opt) else { return };
209                    entity.update(cx, |this, cx| {
210                        this.event(&f, Value::String(value), window, cx);
211                    });
212                })
213            })
214            .collect();
215        // ToggleButtonGroup is generic over a const column count.
216        macro_rules! group {
217            ($n:literal) => {{
218                let arr: [ToggleButtonSimple; $n] = match buttons.try_into() {
219                    Ok(arr) => arr,
220                    Err(_) => unreachable!(),
221                };
222                ToggleButtonGroup::single_row(group_name, arr)
223                    .selected_index(selected_index)
224                    .auto_width()
225                    .label_size(LabelSize::Small)
226                    .into_any_element()
227            }};
228        }
229        Ok(match buttons.len() {
230            1 => group!(1),
231            2 => group!(2),
232            3 => group!(3),
233            4 => group!(4),
234            5 => group!(5),
235            6 => group!(6),
236            n => return Err(rt_err(format!("supports 1-6 options, got {n}"))),
237        })
238    }

240    /// `disclosure{ open, on_click }` (`ui::Disclosure` chevron).
241    pub(super) fn build_disclosure(
242        &mut self,
243        t: &Table,
244        p: &Props,
245        cx: &mut Context<Self>,
246    ) -> mlua::Result<AnyElement> {
247        let id = self.id_or_auto(p, "disclosure");
248        let mut disclosure = Disclosure::new(SharedString::from(id), p.open);
249        if let Some(f) = Self::callback(t, "on_click")? {
250            let handler: std::sync::Arc<dyn Fn(&gpui::ClickEvent, &mut Window, &mut App)> =
251                std::sync::Arc::new(self.click(f, cx));
252            disclosure = disclosure.on_toggle_expanded(handler);
253        }
254        Ok(disclosure.into_any_element())
255    }

257    /// `divider{ vertical }` (`ui::Divider`).
258    pub(super) fn build_divider(p: &Props) -> AnyElement {
259        let divider = if p.vertical { Divider::vertical() } else { Divider::horizontal() };
260        divider.color(DividerColor::BorderVariant).into_any_element()
261    }
