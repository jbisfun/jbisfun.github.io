1//! Typed element props: every string key of an element table is deserialized
2//! into [`Props`] with `serde(deny_unknown_fields)`, so a misspelled prop is an
3//! error shown in place of the element instead of being silently ignored.
4//! Callbacks and element-valued slots aren't serde data; they are read
5//! directly from the table (see [`CALLBACKS`] and [`SLOTS`]).
6

15pub(super) fn rt_err(msg: impl Into<String>) -> mlua::Error {
16    mlua::Error::runtime(msg.into())
17}

130impl Props {
131    /// Reads the string-keyed props of an element table.
132    pub fn parse(lua: &Lua, t: &Table) -> mlua::Result<Props> {
133        let data = lua.create_table()?;
134        for pair in t.pairs::<Value, Value>() {
135            let (k, v) = pair?;
136            let Value::String(key) = &k else { continue };
137            let key = key.to_str()?;
138            if CALLBACKS.contains(&&*key) || SLOTS.contains(&&*key) {
139                continue;
140            }
141            if matches!(v, Value::Function(_)) {
142                return Err(rt_err(format!(
143                    "unknown callback `{}` (expected one of {})",
144                    &*key,
145                    CALLBACKS.join(", ")
146                )));
147            }
148            data.raw_set(k, v)?;
149        }
150        lua.from_value(Value::Table(data))
151            .map_err(|e| rt_err(format!("bad props: {e}")))
152    }
153}

155thread_local! {
156    static WARNED: RefCell<HashSet<String>> = RefCell::default();
157}

159/// Logs a deprecation once per message instead of on every frame.
160pub(super) fn deprecated(msg: String) {
161    WARNED.with(|w| {
162        if w.borrow_mut().insert(msg.clone()) {
163            log::warn!(target: "lua_hooks", "{msg}");
164        }
165    });
166}

168pub(super) fn spacing(s: &Space, what: &str, cx: &App) -> mlua::Result<AbsoluteLength> {
169    Ok(match s {
170        Space::Token(t) => {
171            let d = match t.as_str() {
172                "none" => DynamicSpacing::Base00,
173                "xs" => DynamicSpacing::Base02,
174                "sm" => DynamicSpacing::Base04,
175                "md" => DynamicSpacing::Base08,
176                "lg" => DynamicSpacing::Base12,
177                "xl" => DynamicSpacing::Base16,
178                other => {
179                    return Err(rt_err(format!(
180                        "bad {what} {other:?}, expected none|xs|sm|md|lg|xl"
181                    )));
182                }
183            };
184            d.rems(cx).into()
185        }
186        Space::Px(n) => {
187            deprecated(format!(
188                "{what} = {n} (px) is deprecated: use none|xs|sm|md|lg|xl, which follow UI density"
189            ));
190            px(*n).into()
191        }
192    })
193}

195/// Width of a `"sidebar"` (rems, so it follows `ui_font_size`).
196pub(super) const SIDEBAR_REMS: f32 = 16.;
197
198pub(super) fn rems_len(l: &Len, what: &str) -> mlua::Result<Option<AbsoluteLength>> {
199    Ok(match l {
200        Len::Name(n) if n == "full" => None,
201        Len::Name(n) if n == "sidebar" => Some(Rems(SIDEBAR_REMS).into()),
202        Len::Name(n) => return Err(rt_err(format!("bad {what} {n:?}, expected full|sidebar|<px>"))),
203        Len::Px(n) => Some(px(*n).into()),
204    })
205}

207pub(super) fn label_size(s: &Size) -> mlua::Result<LabelSize> {
208    Ok(match s {
209        Size::Name(n) => match n.as_str() {
210            "default" => LabelSize::Default,
211            "small" => LabelSize::Small,
212            "xsmall" => LabelSize::XSmall,
213            "large" => LabelSize::Large,
214            other => {
215                return Err(rt_err(format!(
216                    "bad size {other:?}, expected default|small|xsmall|large"
217                )));
218            }
219        },
220        Size::Px(n) => {
221            deprecated(format!("size = {n} (px) is deprecated: use default|small|xsmall|large"));
222            if *n <= 10. {
223                LabelSize::XSmall
224            } else if *n <= 12. {
225                LabelSize::Small
226            } else if *n <= 14. {
227                LabelSize::Default
228            } else {
229                LabelSize::Large
230            }
231        }
232    })
233}

235pub(super) fn icon_size(s: &Size) -> mlua::Result<IconSize> {
236    Ok(match s {
237        Size::Name(n) => match n.as_str() {
238            "indicator" => IconSize::Indicator,
239            "xsmall" => IconSize::XSmall,
240            "small" => IconSize::Small,
241            "default" | "medium" => IconSize::Medium,
242            "large" | "xlarge" => IconSize::XLarge,
243            other => return Err(rt_err(format!("bad icon size {other:?}"))),
244        },
245        Size::Px(n) => return Err(rt_err(format!("bad icon size {n}: use a name"))),
246    })
247}

249/// Theme color by name (`background surface panel editor element hover text muted
250/// accent border error warning success`) or `#rrggbb[aa]`.
251pub(super) fn hsla(name: &str, cx: &App) -> mlua::Result<Hsla> {
252    let c = cx.theme().colors();
253    let s = cx.theme().status();
254    Ok(match name {
255        "background" => c.background,
256        "surface" => c.surface_background,
257        "panel" => c.panel_background,
258        "editor" => c.editor_background,
259        "element" => c.element_background,
260        "hover" => c.element_hover,
261        "text" | "default" => c.text,
262        "muted" => c.text_muted,
263        "accent" => c.text_accent,
264        "border" => c.border,
265        "border_variant" => c.border_variant,
266        "error" => s.error,
267        "warning" => s.warning,
268        "success" => s.success,
269        hex => {
270            let bad = || rt_err(format!("bad color {name:?}"));
271            let hex = hex.strip_prefix('#').ok_or_else(bad)?;
272            let v = u32::from_str_radix(hex, 16).map_err(|_| bad())?;
273            match hex.len() {
274                6 => rgba((v << 8) | 0xff).into(),
275                8 => rgba(v).into(),
276                _ => return Err(bad()),
277            }
278        }
279    })
280}

282/// Named `ui::Color` (for labels and icons), or a custom theme/hex color.
283pub(super) fn color(name: &str, cx: &App) -> mlua::Result<Color> {
284    Ok(match name {
285        "default" | "text" => Color::Default,
286        "muted" => Color::Muted,
287        "accent" => Color::Accent,
288        "error" => Color::Error,
289        "warning" => Color::Warning,
290        "success" => Color::Success,
291        "hidden" => Color::Hidden,
292        "disabled" => Color::Disabled,
293        "placeholder" => Color::Placeholder,
294        "info" => Color::Info,
295        "hint" => Color::Hint,
296        _ => Color::Custom(hsla(name, cx)?),
297    })
298}

300/// Parses a Lua icon name (`"settings"` or `"settings-off"`) into `ui::IconName`.
301pub(super) fn icon_name(name: &str) -> mlua::Result<ui::IconName> {
302    use std::str::FromStr as _;
303    ui::IconName::from_str(&name.replace('-', "_"))
304        .map_err(|_| rt_err(format!("unknown icon {name:?}")))
305}

307#[cfg(test)]
308mod tests {
309    use super::*;
310
311    fn parse(src: &str) -> mlua::Result<Props> {
312        let lua = Lua::new();
313        let t: Table = lua.load(src).eval()?;
314        Props::parse(&lua, &t)
315    }
316
317    #[test]
318    fn typed_props() {
319        let p = parse(r#"{ "list_item", "name", selected = true, indent = 1, fg = "muted",
320            gap = "sm", w = "sidebar", on_click = function() end, start = { "icon", name = "x" } }"#)
321            .unwrap();
322        assert!(p.selected.unwrap().is_true());
323        assert_eq!(p.indent, Some(1));
324        assert_eq!(p.color.as_deref(), Some("muted"));
325        assert!(matches!(p.gap, Some(Space::Token(ref t)) if t == "sm"));
326        assert!(matches!(parse(r#"{ "row", gap = 4 }"#).unwrap().gap, Some(Space::Px(_))));
327        assert!(matches!(parse(r#"{ "checkbox", checked = "mixed" }"#).unwrap().checked,
328            Some(Toggle::Str(_))));
329    }
330
331    #[test]
332    fn rejects_typos() {
333        let e = parse(r#"{ "row", pading = "sm" }"#).unwrap_err().to_string();
334        assert!(e.contains("pading"), "{e}");
335        let e = parse(r#"{ "button", onclick = function() end }"#).unwrap_err().to_string();
336        assert!(e.contains("onclick"), "{e}");
337    }
