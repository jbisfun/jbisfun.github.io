13/// What a hook can reach in the host application during one invocation.
14pub trait Host {
15    /// Run the original implementation, optionally with replacement arguments.
16    fn call_orig(&mut self, args: Option<Json>) -> Result<Option<Json>>;
17    /// Dispatch an action by name (e.g. `editor::SelectAll`).
18    fn dispatch(&mut self, name: &str, args: Option<Json>) -> Result<()>;
19    /// Every action name the app knows about.
20    fn action_names(&self) -> Vec<String>;
21    /// `gpui.open_window{...}` (see `ui.rs`).
22    fn open_window(&mut self, _opts: Table) -> Result<()> {
23        anyhow::bail!("gpui.open_window is not available here")
24    }
25    /// Persistent key-value store (the app database). `None` deletes.
26    fn kv_get(&mut self, _key: &str) -> Result<Option<String>> {
27        anyhow::bail!("ycode.kv is not available here")
28    }
29    fn kv_set(&mut self, _key: &str, _value: Option<String>) -> Result<()> {
30        anyhow::bail!("ycode.kv is not available here")
31    }
32    /// Run `sh -c cmd` off the main thread; later call `callback(ok, output, code)`.
33    fn spawn(&mut self, _cmd: String, _callback: Function) -> Result<()> {
34        anyhow::bail!("ycode.spawn is not available here")
35    }
36    /// Hide every action in `namespace` from the command palette.
37    fn palette_hide(&mut self, _namespace: &str) -> Result<()> {
38        anyhow::bail!("ycode.palette_hide is not available here")
39    }
40    /// Replace the plugin settings layer with `settings` (all `ycode.settings` calls merged).
41    fn set_settings(&mut self, _settings: Json) -> Result<()> {
42        anyhow::bail!("ycode.settings is not available here")
43    }
44    /// Folders open in workspace windows, the active window's first.
45    fn workspace_folders(&self) -> Vec<String> {
46        Vec::new()
47    }
48    /// Open a file in the workspace (editor tab), at `line` (1-based) if given.
49    fn open_path(&mut self, _path: String, _line: Option<u32>) -> Result<()> {
50        anyhow::bail!("ycode.open is not available here")
51    }
52    /// Call `callback(sel)` soon with the active editor's selection
53    /// (`{ path, line, end_line, text }`), or nil without a file editor.
54    fn selection(&mut self, _callback: Function) -> Result<()> {
55        anyhow::bail!("ycode.selection is not available here")
56    }
57}

59/// Recursively merge `patch` into `base` (objects merge, everything else replaces).
60pub fn merge_json(base: &mut Json, patch: Json) {
61    match (base, patch) {
62        (Json::Object(b), Json::Object(p)) => {
63            for (k, v) in p {
64                merge_json(b.entry(k).or_insert(Json::Null), v);
65            }
66        }
67        (b, p) => *b = p,
68    }
69}

71const PRELUDE: &str = r#"
72__ycode = { hooks = {}, ctors = {} }
73function __ycode.def(group, name, is_new, src)
74  local h = { group = group, name = name, new = is_new, src = src }
75  __ycode.hooks[#__ycode.hooks + 1] = h
76  __ycode.touch()
77  return h
78end
79function __ycode.ctor(src)
80  local c = { src = src }
81  __ycode.ctors[#__ycode.ctors + 1] = c
82  return c
83end
84function ycode.hook(name, fn) __ycode.def(nil, name, false, "ycode.hook(" .. name .. ")").fn = fn end
85function ycode.new_action(name, fn) __ycode.def(nil, name, true, "ycode.new_action(" .. name .. ")").fn = fn end
86function ycode.ctor(fn) __ycode.ctor("ycode.ctor").fn = fn end
87"#;

133/// `editor::Copy` matches `editor::Copy`, `editor::*` and `*` / `*::*`.
134pub fn pattern_matches(pattern: &str, name: &str) -> bool {
135    if pattern == name || pattern == "*" || pattern == "*::*" {
136        return true;
137    }
138    match pattern.strip_suffix('*') {
139        Some(prefix) if prefix.ends_with("::") => name.starts_with(prefix),
140        _ => false,
141    }
142}

144/// `lib/` holds `require`-able modules, and dot-directories (.git, ...) are never extensions.
145fn is_module_or_hidden(p: &Path) -> bool {
146    p.file_name()
147        .and_then(|n| n.to_str())
148        .is_none_or(|n| n == "lib" || n.starts_with('.'))
149}

160    fn new_lua() -> mlua::Result<Self> {
161        // Full standard library and C modules (`require` of .so files). Hook files are
162        // trusted user code, like an editor init.lua; safe mode refuses C modules.
163        // SAFETY: only `debug`/C-module loading become available; nothing else differs.
164        let lua = unsafe { Lua::unsafe_new_with(mlua::StdLib::ALL, mlua::LuaOptions::new()) };
165        let shared = Rc::new(Shared::default());
166        let ycode = lua.create_table()?;
167
168        let s = shared.clone();
169        ycode.set(
170            "log",
171            lua.create_function(move |lua, args: Variadic<Value>| {
172                let tostring: Function = lua.globals().get("tostring")?;
173                let parts = args
174                    .into_iter()
175                    .map(|v| tostring.call::<String>(v))
176                    .collect::<mlua::Result<Vec<_>>>()?;
177                let line = parts.join(" ");
178                log::info!(target: "lua_hooks", "{line}");
179                let mut log = s.log.borrow_mut();
180                if log.len() >= 1000 {
181                    log.drain(..500);
182                }
183                log.push(line);
184                Ok(())
185            })?,
186        )?;
187
188        // JSON via serde_json (null <-> nil, arrays <-> sequences).
189        ycode.set(
190            "json_decode",
191            lua.create_function(|lua, s: String| {
192                let v: Json = serde_json::from_str(&s).map_err(mlua::Error::external)?;
193                lua.to_value(&v)
194            })?,
195        )?;
196        ycode.set(
197            "json_encode",
198            lua.create_function(|lua, v: Value| {
199                let v: Json = lua.from_value(v)?;
200                serde_json::to_string(&v).map_err(mlua::Error::external)
201            })?,
202        )?;
203
204        // `print` goes to the same place as ycode.log (hook log / executor), not stdout.
205        lua.globals().set("print", ycode.get::<Function>("log")?)?;
206
207        // LuaSocket (socket, socket.http, ltn12, mime, ...) built in.
208        mlua_socket::preload(&lua)?;
209
210        let s = shared.clone();
211        ycode.set(
212            "init",
213            lua.create_function(move |_, group: Option<String>| {
214                if let Some(g) = group {
215                    s.groups.borrow_mut().insert(g);
216                    s.dirty.set(true);
217                }
218                Ok(())
219            })?,
220        )?;
221
222        let s = shared.clone();
223        ycode.set(
224            "status_button",
225            lua.create_function(move |lua, spec: Value| {
226                let button: StatusButton = lua.from_value(spec)?;
227                let mut buttons = s.status_buttons.borrow_mut();
228                buttons.retain(|b| b.id != button.id);
229                buttons.push(button);
230                Ok(())
231            })?,
232        )?;
233
234        let s = shared.clone();
235        ycode.set(
236            "take_log",
237            lua.create_function(move |_, ()| Ok(std::mem::take(&mut *s.log.borrow_mut())))?,
238        )?;
239
240        let gpui = lua.create_table()?;
241        let s = shared.clone();
242        gpui.set(
243            "register_panel",
244            lua.create_function(move |_, opts: Table| {
245                let id: String = opts
246                    .get::<Option<String>>("id")?
247                    .ok_or_else(|| mlua::Error::runtime("gpui.register_panel: `id` is required"))?;
248                let mut panels = s.panels.borrow_mut();
249                panels.retain(|(i, _)| *i != id);
250                panels.push((id, opts));
251                Ok(())
252            })?,
253        )?;
254
255        for (table, name) in [
256            (&ycode, "dispatch"),
257            (&ycode, "actions"),
258            (&ycode, "eval"),
259            (&ycode, "settings"),
260            (&ycode, "kv_get"),
261            (&ycode, "kv_set"),
262            (&ycode, "palette_hide"),
263            (&ycode, "spawn"),
264            (&ycode, "selection"),
265            (&gpui, "open_window"),
266        ] {
267            let msg = format!("{name} is only available inside a hook, %ctor or window callback");
268            table.set(
269                name,
270                lua.create_function(move |_, _: MultiValue| -> mlua::Result<()> {
271                    Err(mlua::Error::runtime(msg.clone()))
272                })?,
273            )?;
274        }
275        lua.globals().set("ycode", ycode)?;
276        lua.globals().set("gpui", gpui)?;
277        lua.load(PRELUDE).set_name("=ycode").exec()?;
278
279        let s = shared.clone();
280        let internal: Table = lua.globals().get("__ycode")?;
281        internal.set(
282            "touch",
283            lua.create_function(move |_, ()| {
284                s.dirty.set(true);
285                Ok(())
286            })?,
287        )?;
288        shared.dirty.set(true);
289        Ok(Self {
290            lua,
291            shared,
292            index: RefCell::default(),
293            settings: RefCell::new(Json::Object(Default::default())),
294        })
295    }

297    /// Prepend directories to `package.path` / `package.cpath` so `require` finds modules
298    /// next to hook files (`dir/?.lua`, `dir/?/init.lua`, `dir/?.so`).
299    pub fn add_module_dirs(&self, dirs: &[std::path::PathBuf]) -> Result<()> {
300        let package: Table = self.lua.globals().get("package").map_err(lua_err)?;
301        let mut path: String = package.get("path").map_err(lua_err)?;
302        let mut cpath: String = package.get("cpath").map_err(lua_err)?;
303        for dir in dirs.iter().rev() {
304            let d = dir.display();
305            path = format!("{d}/?.lua;{d}/?/init.lua;{path}");
306            cpath = format!("{d}/?.so;{cpath}");
307        }
308        package.set("path", path).map_err(lua_err)?;
309        package.set("cpath", cpath).map_err(lua_err)?;
310        Ok(())
311    }

342    fn load_level(&self, dir: &Path, with_subdirs: bool) -> Vec<String> {
343        let Ok(entries) = std::fs::read_dir(dir) else {
344            return Vec::new();
345        };
346        let mut files: Vec<_> = entries
347            .filter_map(|e| e.ok().map(|e| e.path()))
348            .filter(|p| p.extension().is_some_and(|e| e == "lua"))
349            .collect();
350        files.sort();
351        let mut errors = Vec::new();
352        for path in files {
353            let logos = path.to_string_lossy().ends_with(".x.lua");
354            let res = std::fs::read_to_string(&path)
355                .with_context(|| format!("reading {}", path.display()))
356                .and_then(|src| self.load_str(&path.to_string_lossy(), &src, logos));
357            if let Err(e) = res {
358                errors.push(format!("{e:#}"));
359            }
360        }
361        if with_subdirs {
362            // Extensions: one level of subdirectories (not `lib/`, which holds modules).
363            for sub in Self::extension_dirs(dir) {
364                errors.extend(self.load_files(&sub));
365            }
366        }
367        errors
368    }

398    /// Run (and consume) pending `%ctor` blocks.
399    pub fn run_ctors<D: Host + ?Sized>(&self, host: &mut D) -> Vec<String> {
400        let mut errors = Vec::new();
401        let ctors: Vec<Table> = (|| -> mlua::Result<Vec<Table>> {
402            let internal: Table = self.lua.globals().get("__ycode")?;
403            let list: Table = internal.get("ctors")?;
404            internal.set("ctors", self.lua.create_table()?)?;
405            list.sequence_values().collect()
406        })()
407        .unwrap_or_default();
408        let host = RefCell::new(host);
409        for ctor in ctors {
410            let src: String = ctor.get("src").unwrap_or_default();
411            let Ok(func) = ctor.get::<Function>("fn") else {
412                continue;
413            };
414            let res = self
415                .lua
416                .scope(|scope| self.with_api(scope, &host, || func.call::<()>(())));
417            if let Err(e) = res {
418                errors.push(format!("{src}: {e}"));
419            }
420        }
421        errors
422    }

424    fn refresh(&self) {
425        if !self.shared.dirty.replace(false) {
426            return;
427        }
428        let groups = self.shared.groups.borrow();
429        let hooks = (|| -> mlua::Result<Vec<Hook>> {
430            let internal: Table = self.lua.globals().get("__ycode")?;
431            let list: Table = internal.get("hooks")?;
432            let mut out = Vec::new();
433            for h in list.sequence_values::<Table>() {
434                let h = h?;
435                let group: Option<String> = h.get("group")?;
436                if group.is_some_and(|g| !groups.contains(&g)) {
437                    continue;
438                }
439                let Ok(func) = h.get::<Function>("fn") else {
440                    continue;
441                };
442                out.push(Hook {
443                    name: h.get("name")?,
444                    src: h.get("src")?,
445                    is_new: h.get("new")?,
446                    func,
447                });
448            }
449            Ok(out)
450        })();
451        match hooks {
452            Ok(h) => *self.index.borrow_mut() = h,
453            Err(e) => log::error!(target: "lua_hooks", "reading hook table: {e}"),
454        }
455    }

499    /// Run a REPL chunk (directives allowed). Expressions are returned like the `lua`
500    /// REPL does; any `%ctor` it defines runs immediately.
501    pub fn eval<D: Host + ?Sized>(&self, code: &str, host: &mut D) -> Result<String> {
502        let code = preprocess::preprocess("repl", code)?;
503        let func = match self
504            .lua
505            .load(format!("return {code}"))
506            .set_name("=repl")
507            .into_function()
508        {
509            Ok(f) => f,
510            Err(_) => self
511                .lua
512                .load(&code)
513                .set_name("=repl")
514                .into_function()
515                .map_err(lua_err)?,
516        };
517        let host = RefCell::new(host);
518        let out = self
519            .lua
520            .scope(|scope| {
521                self.with_api(scope, &host, || {
522                    let tostring: Function = self.lua.globals().get("tostring")?;
523                    func.call::<MultiValue>(())?
524                        .into_iter()
525                        .map(|v| tostring.call::<String>(v))
526                        .collect::<mlua::Result<Vec<_>>>()
527                })
528            })
529            .map_err(lua_err)?;
530        let errors = self.run_ctors(host.into_inner());
531        if !errors.is_empty() {
532            anyhow::bail!("{}", errors.join("\n"));
533        }
534        Ok(out.join("\t"))
535    }

552    ) -> mlua::Result<R>

554        H: std::ops::DerefMut<Target = D>,

561    /// Bind `ycode.dispatch` / `ycode.actions` / `ycode.eval` / `gpui.open_window` to
562    /// `host` while `f` runs.
563    fn with_api<'scope, 'env, H, D, R>(
564        &'scope self,
565        scope: &'scope Scope<'scope, 'env>,
566        host: &'scope RefCell<H>,
567        f: impl FnOnce() -> mlua::Result<R>,
568    ) -> mlua::Result<R>
569    where
570        H: std::ops::DerefMut<Target = D>,
571        D: Host + ?Sized,
572    {
573        let ycode: Table = self.lua.globals().get("ycode")?;
574        let gpui: Table = self.lua.globals().get("gpui")?;
575        let prev_dispatch: Value = ycode.get("dispatch")?;
576        let prev_actions: Value = ycode.get("actions")?;
577        let prev_eval: Value = ycode.get("eval")?;
578        let prev_settings: Value = ycode.get("settings")?;
579        let prev_kv_get: Value = ycode.get("kv_get")?;
580        let prev_kv_set: Value = ycode.get("kv_set")?;
581        let prev_palette_hide: Value = ycode.get("palette_hide")?;
582        let prev_spawn: Value = ycode.get("spawn")?;
583        let prev_folders: Value = ycode.get("workspace_folders")?;
584        let prev_open_path: Value = ycode.get("open")?;
585        let prev_selection: Value = ycode.get("selection")?;
586        ycode.set(
587            "open",
588            scope.create_function(move |_, (path, line): (String, Option<u32>)| {
589                let mut host = host
590                    .try_borrow_mut()
591                    .map_err(|_| mlua::Error::runtime("ycode.open: host busy"))?;
592                host.open_path(path, line).map_err(mlua::Error::external)
593            })?,
594        )?;
595        ycode.set(
596            "selection",
597            scope.create_function(move |_, callback: Function| {
598                let mut host = host
599                    .try_borrow_mut()
600                    .map_err(|_| mlua::Error::runtime("ycode.selection: host busy"))?;
601                host.selection(callback).map_err(mlua::Error::external)
602            })?,
603        )?;
604        ycode.set(
605            "workspace_folders",
606            scope.create_function(move |_, ()| {
607                let host = host
608                    .try_borrow()
609                    .map_err(|_| mlua::Error::runtime("ycode.workspace_folders: host busy"))?;
610                Ok(host.workspace_folders())
611            })?,
612        )?;
613        let busy = |what: &'static str| move |_| mlua::Error::runtime(format!("{what}: host busy"));
614        ycode.set(
615            "kv_get",
616            scope.create_function(move |_, key: String| {
617                let mut host = host.try_borrow_mut().map_err(busy("ycode.kv_get"))?;
618                host.kv_get(&key).map_err(mlua::Error::external)
619            })?,
620        )?;
621        ycode.set(
622            "kv_set",
623            scope.create_function(move |_, (key, value): (String, Option<String>)| {
624                let mut host = host.try_borrow_mut().map_err(busy("ycode.kv_set"))?;
625                host.kv_set(&key, value).map_err(mlua::Error::external)
626            })?,
627        )?;
628        ycode.set(
629            "spawn",
630            scope.create_function(move |_, (cmd, callback): (String, Function)| {
631                let mut host = host.try_borrow_mut().map_err(busy("ycode.spawn"))?;
632                host.spawn(cmd, callback).map_err(mlua::Error::external)
633            })?,
634        )?;
635        ycode.set(
636            "palette_hide",
637            scope.create_function(move |_, namespaces: mlua::Variadic<String>| {
638                let mut host = host.try_borrow_mut().map_err(busy("ycode.palette_hide"))?;
639                for ns in namespaces {
640                    host.palette_hide(&ns).map_err(mlua::Error::external)?;
641                }
642                Ok(())
643            })?,
644        )?;
645        ycode.set(
646            "settings",
647            scope.create_function(move |lua, patch: Value| {
648                let patch: Json = lua.from_value(patch)?;
649                let merged = {
650                    let mut settings = self.settings.borrow_mut();
651                    merge_json(&mut settings, patch);
652                    settings.clone()
653                };
654                let mut host = host
655                    .try_borrow_mut()
656                    .map_err(|_| mlua::Error::runtime("ycode.settings: host busy"))?;
657                host.set_settings(merged).map_err(mlua::Error::external)
658            })?,
659        )?;
660        let prev_open: Value = gpui.get("open_window")?;
661        ycode.set(
662            "eval",
663            scope.create_function(move |_, code: String| {
664                let mut host = host
665                    .try_borrow_mut()
666                    .map_err(|_| mlua::Error::runtime("ycode.eval: host busy"))?;
667                Ok(match self.eval(&code, &mut **host) {
668                    Ok(r) => (true, r),
669                    Err(e) => (false, format!("{e:#}")),
670                })
671            })?,
672        )?;
673        gpui.set(
674            "open_window",
675            scope.create_function(move |_, opts: Table| {
676                let mut host = host
677                    .try_borrow_mut()
678                    .map_err(|_| mlua::Error::runtime("gpui.open_window: host busy"))?;
679                host.open_window(opts).map_err(mlua::Error::external)
680            })?,
681        )?;
682        ycode.set(
683            "dispatch",
684            scope.create_function(move |lua, (name, args): (String, Option<Value>)| {
685                let mut args = args.map(|v| lua.from_value::<Json>(v)).transpose()?;
686                let mut name = name;
687                if self.is_new(&name) {
688                    args = Some(serde_json::json!({ "name": name, "args": args }));
689                    name = "ycode::Run".into();
690                }
691                let mut host = host
692                    .try_borrow_mut()
693                    .map_err(|_| mlua::Error::runtime("ycode.dispatch: host busy"))?;
694                host.dispatch(&name, args).map_err(mlua::Error::external)
695            })?,
696        )?;
697        ycode.set(
698            "actions",
699            scope.create_function(move |_, ()| {
700                let host = host
701                    .try_borrow()
702                    .map_err(|_| mlua::Error::runtime("ycode.actions: host busy"))?;
703                Ok(host.action_names())
704            })?,
705        )?;
706        let result = f();
707        ycode.set("dispatch", prev_dispatch)?;
708        ycode.set("actions", prev_actions)?;
709        ycode.set("eval", prev_eval)?;
710        ycode.set("settings", prev_settings)?;
711        ycode.set("kv_get", prev_kv_get)?;
712        ycode.set("kv_set", prev_kv_set)?;
713        ycode.set("palette_hide", prev_palette_hide)?;
714        ycode.set("spawn", prev_spawn)?;
715        ycode.set("workspace_folders", prev_folders)?;
716        ycode.set("open", prev_open_path)?;
717        ycode.set("selection", prev_selection)?;
718        gpui.set("open_window", prev_open)?;
719        result
720    }

722    /// Run every hook matching `name` as a `%orig` chain around `host.call_orig`.
723    /// The most recently defined hook runs first. If a hook errors before calling
724    /// `%orig`, the original still runs (unless this is a `%new` action).
725    pub fn invoke(
726        &self,
727        name: &str,
728        debug: &str,
729        args: Option<Json>,
730        host: &mut dyn Host,
731    ) -> Result<Option<Json>> {
732        self.refresh();
733        let chain: Vec<Hook> = self
734            .index
735            .borrow()
736            .iter()
737            .filter(|h| pattern_matches(&h.name, name))
738            .cloned()
739            .collect();
740        if chain.is_empty() {
741            return host.call_orig(args);
742        }
743        let is_new = chain.iter().any(|h| h.is_new);
744        let orig_calls = Cell::new(0u32);
745        let host = RefCell::new(host);
746
747        let result = self.lua.scope(|scope| {
748            let lua = &self.lua;
749            let args_val = match &args {
750                Some(a) => lua.to_value(a)?,
751                None => Value::Nil,
752            };
753            let mut orig = scope.create_function(|lua, a: Option<Value>| {
754                orig_calls.set(orig_calls.get() + 1);
755                let a = a.map(|v| lua.from_value::<Json>(v)).transpose()?;
756                let r = host
757                    .try_borrow_mut()
758                    .map_err(|_| mlua::Error::runtime("%orig: host busy"))?
759                    .call_orig(a)
760                    .map_err(mlua::Error::external)?;
761                r.map(|j| lua.to_value(&j)).transpose()
762            })?;
763            for hook in &chain {
764                let me = lua.create_table()?;
765                me.set("name", name)?;
766                me.set("debug", debug)?;
767                me.set("src", hook.src.as_str())?;
768                me.set("args", args_val.clone())?;
769                me.set("orig", orig.clone())?;
770                let func = hook.func.clone();
771                let default_args = args_val.clone();
772                orig = scope.create_function(move |_, a: Option<Value>| {
773                    func.call::<MultiValue>((me.clone(), a.unwrap_or(default_args.clone())))
774                })?;
775            }
776            self.with_api(scope, &host, || orig.call::<Value>(args_val.clone()))
777        });
778
779        let host = host.into_inner();
780        match result {
781            Ok(v) => Ok(self.lua.from_value::<Json>(v).ok().filter(|j| !j.is_null())),
782            Err(e) => {
783                log::error!(target: "lua_hooks", "{name}: {e}");
784                if orig_calls.get() == 0 && !is_new {
785                    host.call_orig(args)?;
786                }
787                Err(lua_err(e))
788            }
789        }
790    }
791}
