1//! Lua hooks for ycode, in the spirit of Logos/MobileSubstrate tweaks.
2//!
3//! Every action dispatched in the app (keyboard, menus, command palette, `dispatch_action`)
4//! goes through [`gpui::App::intercept_actions`]. Hook files can wrap any action by name,
5//! call through to the original with `%orig`, replace its arguments, or swallow it, and can
6//! define new actions with `%new` that are bound to keys through `ycode::Run`.
7//!
8//! Files are loaded from `<config dir>/hooks/` in name order: `*.x.lua` files go through the
9//! Logos-style preprocessor in [`preprocess`]; plain `*.lua` files can use the same runtime
10//! via `ycode.hook(name, fn)`, `ycode.new_action(name, fn)` and `ycode.ctor(fn)`.
11//! Run `ycode: reload hooks` to reload them.
12

84/// Build an action by name and dispatch it once the current dispatch has finished.
85/// Dispatches to the active window, or, when that is a Lua window or nothing is focused,
86/// to an editor (workspace) window, where workspace/agent/editor actions are handled.
87pub(crate) fn dispatch_deferred(cx: &mut App, name: &str, args: Option<Json>) -> Result<()> {
88    let action = cx.build_action(name, args)?;
89    cx.defer(move |cx| {
90        let is_lua = |w: &AnyWindowHandle| w.downcast::<ui::LuaWindow>().is_some();
91        let target = cx
92            .active_window()
93            .filter(|w| !is_lua(w))
94            .or_else(|| cx.windows().into_iter().find(|w| !is_lua(w)));
95        match target {
96            Some(window) => {
97                window
98                    .update(cx, |_, window, cx| {
99                        window.dispatch_action(action.boxed_clone(), cx)
100                    })
101                    .ok();
102            }
103            None => cx.dispatch_action(action.as_ref()),
104        }
105    });
106    Ok(())
107}

109/// The workspace windows, the active one first.
110fn workspace_windows(cx: &App) -> Vec<gpui::WindowHandle<workspace::MultiWorkspace>> {
111    let mut windows = cx.windows();
112    if let Some(active) = cx.active_window() {
113        windows.retain(|w| *w != active);
114        windows.insert(0, active);
115    }
116    windows
117        .into_iter()
118        .filter_map(|w| w.downcast::<workspace::MultiWorkspace>())
119        .collect()
120}

122/// `ycode.open(path, line)`: open a file in the active (or first) workspace window,
123/// with the cursor on `line` (1-based) when given.
124pub(crate) fn open_path_deferred(cx: &mut App, path: String, line: Option<u32>) {
125    cx.defer(move |cx| {
126        let Some(mw) = workspace_windows(cx).into_iter().next() else {
127            log::warn!(target: "lua_hooks", "ycode.open: no workspace window");
128            return;
129        };
130        mw.update(cx, |mw, window, cx| {
131            mw.workspace().update(cx, |ws, cx| {
132                let open = ws.open_abs_path(
133                    PathBuf::from(&path),
134                    workspace::OpenOptions::default(),
135                    window,
136                    cx,
137                );
138                cx.spawn_in(window, async move |_, cx| {
139                    let item = open.await?;
140                    if let (Some(line), Some(editor)) = (line, item.downcast::<editor::Editor>()) {
141                        let point = text::Point::new(line.saturating_sub(1), 0);
142                        editor.update_in(cx, |editor, window, cx| {
143                            editor.go_to_singleton_buffer_point(point, window, cx)
144                        })?;
145                    }
146                    anyhow::Ok(())
147                })
148                .detach_and_log_err(&*cx);
149            });
150        })
151        .ok();
152    });
153}

155/// The active editor's file and selection.
156struct Selection {
157    path: String,
158    line: u32,
159    end_line: u32,
160    text: String,
161}
162
163fn active_selection(cx: &mut App) -> Option<Selection> {
164    let mw = workspace_windows(cx).into_iter().next()?;
165    let ws = mw.read(cx).ok()?.workspace().clone();
166    let editor = ws.read(cx).active_item(cx)?.downcast::<editor::Editor>()?;
167    editor.update(cx, |editor, cx| {
168        let snapshot = editor.display_snapshot(cx);
169        let sel = editor.selections.newest::<text::Point>(&snapshot);
170        let buffer = editor.buffer().read(cx).as_singleton()?;
171        let buffer = buffer.read(cx);
172        let path = language::File::as_local(buffer.file()?.as_ref())?.abs_path(cx);
173        let text: String = buffer.text_for_range(sel.start..sel.end).collect();
174        // A selection ending at column 0 of the next line covers only the lines above.
175        let end_row = if sel.end.column == 0 && sel.end.row > sel.start.row {
176            sel.end.row - 1
177        } else {
178            sel.end.row
179        };
180        Some(Selection {
181            path: path.to_string_lossy().into_owned(),
182            line: sel.start.row + 1,
183            end_line: end_row + 1,
184            text,
185        })
186    })
187}

189/// `ycode.selection(callback)`: read the active editor's selection once nothing is
190/// being updated (windows can't be read while they are), then call back.
191pub(crate) fn selection_deferred(cx: &mut App, callback: mlua::Function) {
192    let Some(rt) = LuaHooks::runtime(cx) else {
193        return;
194    };
195    let weak = Rc::downgrade(&rt);
196    cx.defer(move |cx| {
197        let selection = active_selection(cx);
198        call_back(cx, &weak, &callback, |lua| {
199            let Some(s) = selection else {
200                return Ok(mlua::MultiValue::from_vec(vec![mlua::Value::Nil]));
201            };
202            let t = lua.create_table()?;
203            t.set("path", s.path)?;
204            t.set("line", s.line)?;
205            t.set("end_line", s.end_line)?;
206            t.set("text", s.text)?;
207            Ok(mlua::MultiValue::from_vec(vec![mlua::Value::Table(t)]))
208        });
209    });
210}

212/// Call a Lua callback from outside any hook (spawn/selection results) with a host,
213/// then run any ctors it registered. Dropped if the runtime it came from was
214/// replaced by a reload in the meantime.
215fn call_back(
216    cx: &mut App,
217    from: &std::rc::Weak<Runtime>,
218    callback: &mlua::Function,
219    args: impl FnOnce(&mlua::Lua) -> mlua::Result<mlua::MultiValue>,
220) {
221    let (Some(rt), Some(current)) = (from.upgrade(), LuaHooks::runtime(cx)) else {
222        return;
223    };
224    if !Rc::ptr_eq(&rt, &current) {
225        return;
226    }
227    let mut host = ActionHost {
228        cx,
229        action: None,
230        orig: None,
231    };
232    let cell = RefCell::new(&mut host);
233    if let Err(e) = rt.with_host(&cell, |_| callback.call::<()>(args(rt.lua())?)) {
234        log::error!(target: "lua_hooks", "callback: {e}");
235    }
236    drop(cell);
237    for e in rt.run_ctors(&mut host) {
238        log::error!(target: "lua_hooks", "{e}");
239    }
240}

242/// Folders of every workspace, kept up to date from project events. Reading windows on
243/// demand fails for any window that is being updated or drawn at that moment (it is leased
244/// out of `cx.windows()`), which is exactly where Lua runs from actions and panels.
245#[derive(Default)]
246struct WorkspaceFolders(Vec<(gpui::EntityId, AnyWindowHandle, Vec<String>)>);
247impl Global for WorkspaceFolders {}
248
249fn track_workspace_folders(cx: &mut App) {
250    fn record(
251        ws: &workspace::Workspace,
252        window: AnyWindowHandle,
253        cx: &mut gpui::Context<workspace::Workspace>,
254    ) {
255        let id = cx.entity().entity_id();
256        let folders: Vec<String> = ws
257            .root_paths(cx)
258            .iter()
259            .map(|p| p.to_string_lossy().into_owned())
260            .collect();
261        let all = &mut cx.default_global::<WorkspaceFolders>().0;
262        all.retain(|(i, _, _)| *i != id);
263        all.push((id, window, folders));
264    }
265    cx.observe_new(|ws: &mut workspace::Workspace, window, cx| {
266        let Some(window) = window else { return };
267        let handle = window.window_handle();
268        record(ws, handle, cx);
269        let project = ws.project().clone();
270        cx.subscribe(&project, move |ws, _, event, cx| {
271            if matches!(
272                event,
273                project::Event::WorktreeAdded(_) | project::Event::WorktreeRemoved(_)
274            ) {
275                record(ws, handle, cx);
276            }
277        })
278        .detach();
279        let id = cx.entity().entity_id();
280        cx.on_release(move |_, cx| {
281            cx.default_global::<WorkspaceFolders>()
282                .0
283                .retain(|(i, _, _)| *i != id);
284        })
285        .detach();
286    })
287    .detach();
288}

290/// `ycode.workspace_folders()`: folders open in workspaces, the active window's first,
291/// without duplicates.
292pub(crate) fn workspace_folders(cx: &App) -> Vec<String> {
293    let Some(all) = cx.try_global::<WorkspaceFolders>() else {
294        return Vec::new();
295    };
296    let active = cx.active_window();
297    let mut entries: Vec<_> = all.0.iter().collect();
298    // Stable: the active window's workspaces first, otherwise in creation order.
299    entries.sort_by_key(|(_, w, _)| Some(*w) != active);
300    let mut folders: Vec<String> = Vec::new();
301    for (_, _, paths) in entries {
302        for p in paths {
303            if !folders.contains(p) {
304                folders.push(p.clone());
305            }
306        }
307    }
308    folders
309}

338/// `ycode.kv_set`: write synchronously, so code reading the key right after sees it.
339pub(crate) fn kv_set(cx: &App, key: &str, value: Option<String>) -> Result<()> {
340    let store = db::kvp::KeyValueStore::global(cx);
341    let key = key.to_string();
342    gpui::block_on(async move {
343        match value {
344            Some(v) => store.write_kvp(key, v).await,
345            None => store.delete_kvp(key).await,
346        }
347    })
348}

350/// Namespaces hidden from the command palette by Lua; shown again on reload.
351#[derive(Default)]
352struct HiddenNamespaces(Vec<&'static str>);
353impl Global for HiddenNamespaces {}
354
355pub(crate) fn palette_hide(cx: &mut App, namespace: &str) -> Result<()> {
356    let hidden = cx.default_global::<HiddenNamespaces>();
357    let ns: &'static str = match hidden.0.iter().find(|n| **n == namespace) {
358        Some(n) => n,
359        // Leaked once per distinct namespace; the filter only takes &'static str.
360        None => {
361            let n: &'static str = Box::leak(namespace.to_string().into_boxed_str());
362            hidden.0.push(n);
363            n
364        }
365    };
366    CommandPaletteFilter::update_global(cx, |filter, _| filter.hide_namespace(ns));
367    Ok(())
368}
369
370fn palette_reset(cx: &mut App) {
371    let hidden: Vec<&'static str> = cx.default_global::<HiddenNamespaces>().0.clone();
372    CommandPaletteFilter::update_global(cx, |filter, _| {
373        for ns in hidden {
374            filter.show_namespace(ns);
375        }
376    });
377}

384/// `ycode.spawn`: run a shell command on the background executor, then call the Lua
385/// callback on the main thread, unless the runtime was reloaded in the meantime.
386pub(crate) fn spawn_command(cx: &mut App, cmd: String, callback: mlua::Function) -> Result<()> {
387    let Some(rt) = LuaHooks::runtime(cx) else {
388        anyhow::bail!("no Lua runtime");
389    };
390    let weak = Rc::downgrade(&rt);
391    let job = cx.background_spawn(async move {
392        std::process::Command::new("sh")
393            .arg("-c")
394            .arg(&cmd)
395            .stdin(std::process::Stdio::null())
396            .output()
397    });
398    cx.spawn(async move |cx| {
399        let output = job.await;
400        cx.update(|cx| {
401            let (ok, text, code) = match output {
402                Ok(o) => {
403                    let mut text = String::from_utf8_lossy(&o.stdout).into_owned();
404                    text.push_str(&String::from_utf8_lossy(&o.stderr));
405                    (o.status.success(), text, o.status.code().unwrap_or(-1))
406                }
407                Err(e) => (false, e.to_string(), -1),
408            };
409            call_back(cx, &weak, &callback, |lua| (ok, text, code).into_lua_multi(lua));
410        });
411    })
412    .detach();
413    Ok(())
414}

416/// Open a Lua window after the current update (Lua may be running inside another window).
417pub(crate) fn open_window_deferred(cx: &mut App, opts: mlua::Table) {
418    cx.defer(move |cx| {
419        let Some(rt) = LuaHooks::runtime(cx) else {
420            return;
421        };
422        match ui::open_window(cx, rt, opts) {
423            Ok(handle) => {

427            }
428            Err(e) => log::error!(target: "lua_hooks", "gpui.open_window: {e}"),
429        }
430    });
431}

433/// [`Host`] for an action dispatch (or a ctor / `%new` action when `orig` is `None`).
434struct ActionHost<'a> {
435    cx: &'a mut App,
436    action: Option<&'a dyn Action>,
437    orig: Option<&'a mut dyn FnMut(&dyn Action, &mut App)>,
438}
439
440impl Host for ActionHost<'_> {
441    fn call_orig(&mut self, args: Option<Json>) -> Result<Option<Json>> {
442        let (Some(action), Some(orig)) = (self.action, self.orig.as_mut()) else {
443            return Ok(None);
444        };
445        match args {
446            None => orig(action, self.cx),
447            Some(args) => {
448                let replacement = self.cx.build_action(action.name(), Some(args))?;
449                orig(replacement.as_ref(), self.cx);
450            }
451        }
452        Ok(None)
453    }

456        dispatch_deferred(self.cx, name, args)
457    }

461    }

464        workspace_folders(self.cx)
465    }
466
467    fn open_path(&mut self, path: String, line: Option<u32>) -> Result<()> {

469        Ok(())
470    }
471
472    fn selection(&mut self, callback: mlua::Function) -> Result<()> {

474        Ok(())
475    }
476
477    fn open_window(&mut self, opts: mlua::Table) -> Result<()> {

480    }
481

566/// Expose a named hook point to `%hookf name(args)`. Without a matching hook this is just
567/// `orig(args, cx)`. A hook's return value (if not nil) replaces the result.
568pub fn call_point(
569    cx: &mut App,
570    name: &str,
571    args: Json,
572    mut orig: impl FnMut(Json, &mut App) -> Json,
573) -> Json {
574    let Some(rt) = LuaHooks::runtime(cx).filter(|rt| rt.wants(name)) else {
575        return orig(args, cx);
576    };
577    let mut result = None;
578    let mut orig = |a: Json, cx: &mut App| {
579        let r = orig(a, cx);
580        result = Some(r.clone());
581        r
582    };
583    let mut host = PointHost {
584        cx,
585        args: args.clone(),
586        orig: &mut orig,
587    };
588    match rt.invoke(name, name, Some(args), &mut host) {
589        Ok(Some(v)) => v,
590        Ok(None) => result.unwrap_or(Json::Null),
591        Err(e) => {
592            log::error!(target: "lua_hooks", "{name}: {e:#}");
593            result.unwrap_or(Json::Null)
594        }
595    }
596}

598pub fn init(cx: &mut App) {
599    let hooks = LuaHooks::default();
600    cx.set_global(hooks.clone());
601    // Tests install their own runtime instead of the user's hooks directory.
602    #[cfg(not(test))]
603    load(cx);
604
605    cx.intercept_actions(move |action, orig, cx| {
606        let Some(rt) = hooks.runtime.borrow().clone() else {
607            orig(action, cx);
608            return;
609        };
610
611        if let Some(run) = action.as_any().downcast_ref::<Run>() {
612            if !rt.is_new(&run.name) && !rt.wants(&run.name) {
613                log::warn!(target: "lua_hooks", "ycode::Run: no Lua action named {}", run.name);
614                return;
615            }
616            let debug = format!("{action:?}");
617            let mut host = ActionHost {
618                cx,
619                action: None,
620                orig: None,
621            };
622            if let Err(e) = rt.invoke(&run.name, &debug, run.args.clone(), &mut host) {
623                log::error!(target: "lua_hooks", "{}: {e:#}", run.name);
624            }
625            return;
626        }
627
628        let name = action.name();
629        if !rt.wants(name) {
630            orig(action, cx);
631            return;
632        }
633        let debug = format!("{action:?}");
634        let mut host = ActionHost {
635            cx,
636            action: Some(action),
637            orig: Some(orig),
638        };
639        // Errors are logged by the runtime, which also falls back to the original action.
640        rt.invoke(name, &debug, None, &mut host).ok();
641    });
642
643    cx.on_action(|_: &ReloadHooks, cx| load(cx));
644    status::init(cx);
645    panel::init(cx);
646    track_workspace_folders(cx);
647    // `%new` actions show up in the command palette like registered ones.
648    GlobalCommandPaletteSource::set(cx, |cx| {
649        let Some(rt) = LuaHooks::runtime(cx) else {
650            return Vec::new();
651        };
652        rt.new_actions()
653            .into_iter()
654            .map(|name| {
655                let action: Box<dyn Action> = Box::new(Run {
656                    name: name.clone(),
657                    args: None,
658                });
659                (humanize(&name), action)
660            })
661            .collect()
662    });
663
664    #[cfg(all(unix, not(test)))]
665    executor::start(cx);
666}

668/// Evaluate a REPL request on the main thread. `.reload` reloads hook files.
669#[cfg_attr(any(test, not(unix)), allow(dead_code))]
670fn eval_request(cx: &mut App, code: &str) -> Json {
671    if code.trim() == ".reload" {
672        load(cx);
673    }
674    let Some(rt) = LuaHooks::runtime(cx) else {
675        return serde_json::json!({ "ok": false, "error": "no Lua runtime", "log": [] });
676    };
677    let result = if code.trim().is_empty() || code.trim() == ".reload" {
678        Ok(String::new())
679    } else {
680        rt.eval(
681            code,
682            &mut ActionHost {
683                cx,
684                action: None,
685                orig: None,
686            },
687        )
688    };
689    let log = rt.take_log();
690    match result {
691        Ok(r) => serde_json::json!({ "ok": true, "result": r, "log": log }),
692        Err(e) => serde_json::json!({ "ok": false, "error": format!("{e:#}"), "log": log }),
693    }
694}

696/// Live Lua executor: a Unix socket speaking one JSON object per line.
697/// Request `{"code": "..."}`, response `{"ok", "result" | "error", "log"}`, where `log` is
698/// everything `ycode.log` printed since the previous request (including from hooks).
699#[cfg(all(unix, not(test)))]
700mod executor {
701    use super::*;
702    use std::io::{BufRead, BufReader, Write};
703    use std::os::unix::net::{UnixListener, UnixStream};
704
705    pub fn socket_path() -> PathBuf {
706        std::env::var_os("XDG_RUNTIME_DIR")
707            .map(PathBuf::from)
708            .unwrap_or_else(std::env::temp_dir)
709            .join("ycode-lua.sock")
710    }
711
712    type Request = (String, smol::channel::Sender<Json>);
713
714    pub fn start(cx: &mut App) {
715        let path = socket_path();
716        std::fs::remove_file(&path).ok();
717        let listener = match UnixListener::bind(&path) {
718            Ok(l) => l,
719            Err(e) => {
720                log::error!(target: "lua_hooks", "executor: binding {}: {e}", path.display());
721                return;
722            }
723        };
724        log::info!(target: "lua_hooks", "executor listening on {}", path.display());
725        let (tx, rx) = smol::channel::unbounded::<Request>();
726        std::thread::Builder::new()
727            .name("lua-executor".into())
728            .spawn(move || {
729                for stream in listener.incoming().flatten() {
730                    let tx = tx.clone();
731                    std::thread::spawn(move || serve(stream, tx));
732                }
733            })
734            .ok();
735        cx.spawn(async move |cx| {
736            while let Ok((code, reply)) = rx.recv().await {
737                let response = cx.update(|cx| eval_request(cx, &code));
738                reply.send(response).await.ok();
739            }
740        })
741        .detach();
742    }
743
744    fn serve(stream: UnixStream, tx: smol::channel::Sender<Request>) {
745        let Ok(mut writer) = stream.try_clone() else {
746            return;
747        };
748        for line in BufReader::new(stream).lines() {
749            let Ok(line) = line else { return };
750            let code = serde_json::from_str::<Json>(&line)
751                .ok()
752                .and_then(|j| j.get("code")?.as_str().map(String::from))
753                .unwrap_or(line);
754            let (reply_tx, reply_rx) = smol::channel::bounded(1);
755            if tx.send_blocking((code, reply_tx)).is_err() {
756                return;
757            }
758            let Ok(response) = reply_rx.recv_blocking() else {
759                return;
760            };
761            if writeln!(writer, "{response}").is_err() {
762                return;
763            }
764        }
765    }
766}

768/// (Re)load every hook file and run its `%ctor`s, replacing the previous runtime.
769fn load(cx: &mut App) {
770    let dir = hooks_dir();
771    let rt = match Runtime::new() {
772        Ok(rt) => rt,
773        Err(e) => {
774            log::error!(target: "lua_hooks", "creating Lua runtime: {e:#}");
775            return;
776        }
777    };
778    let mut module_dirs = vec![dir.join("lib"), dir.clone()];
779    for ext in Runtime::extension_dirs(&dir) {
780        module_dirs.push(ext.join("lib"));
781    }
782    if let Some(home) = std::env::var_os("HOME").map(PathBuf::from) {
783        // luarocks --local
784        module_dirs.push(home.join(".luarocks/share/lua/5.4"));
785        module_dirs.push(home.join(".luarocks/lib/lua/5.4"));
786    }
787    // Distro-packaged modules: Debian multiarch, Fedora/openSUSE lib64, Arch and others.
788    let arch = std::env::consts::ARCH;
789    for dir in [
790        format!("/usr/lib/{arch}-linux-gnu/lua/5.4"),
791        "/usr/lib64/lua/5.4".to_string(),
792        "/usr/lib/lua/5.4".to_string(),
793        "/usr/share/lua/5.4".to_string(),
794    ] {
795        let dir = PathBuf::from(dir);
796        if dir.is_dir() {
797            module_dirs.push(dir);
798        }
799    }
800    if let Err(e) = rt.add_module_dirs(&module_dirs) {
801        log::error!(target: "lua_hooks", "setting package paths: {e:#}");
802    }
803    if let Err(e) = rt.set_ycode_field("hooks_dir", &dir.to_string_lossy()) {
804        log::error!(target: "lua_hooks", "setting ycode.hooks_dir: {e:#}");
805    }
806    let errors = rt.load_dir(&dir);
807    let errors = install(cx, rt, errors);
808    log::info!(target: "lua_hooks", "hooks dir {} ({} errors)", dir.display(), errors.len());
809}

811/// Run pending `%ctor`s and make `rt` the active runtime. Returns all errors.
812pub fn install(cx: &mut App, rt: Runtime, mut errors: Vec<String>) -> Vec<String> {
813    // Palette filters and the plugin settings layer belong to the runtime being replaced.
814    palette_reset(cx);
815    if cx.has_global::<SettingsStore>() {
816        cx.update_global(|store: &mut SettingsStore, cx| store.set_plugin_settings("", cx))
817            .ok();
818    }
819    let rt = Rc::new(rt);
820    if let Some(hooks) = cx.try_global::<LuaHooks>().cloned() {
821        // Windows run callbacks in the old Lua state; close them with it.
822        for handle in hooks.windows.take() {
823            handle
824                .update(cx, |_, window, _| window.remove_window())
825                .ok();
826        }
827        // Installed before ctors run, so they can use APIs that need it (ycode.spawn).
828        *hooks.runtime.borrow_mut() = Some(rt.clone());
829    }
830    errors.extend(rt.run_ctors(&mut ActionHost {
831        cx,
832        action: None,
833        orig: None,
834    }));
835    for e in &errors {
836        log::error!(target: "lua_hooks", "{e}");
837    }
838    log::info!(target: "lua_hooks", "{} active hooks", rt.active_hooks().len());
839    // Status bar buttons and dock panels come from the runtime; rebuild them.
840    cx.refresh_windows();
841    panel::sync_all(cx);
842    errors
843}
