Enum Command

pub(crate) enum Command {
    Add {
        repo: String,
        file: String,
        line: u32,
        note: Option<String>,
        quote: Option<String>,
        source: String,
    },
    List {
        all: bool,
        json: bool,
    },
    Close {
        number: u64,
    },
    Reopen {
        number: u64,
    },
    Rm {
        number: u64,
    },
    Export,
    Mcp {
        root: Option<PathBuf>,
    },
}
