1//! `radar mcp`: an MCP server (stdio) over the store, for agents.
2//!
3//! The database is opened for each tool call and closed again, so the CLI (the pull
4//! precommit `radar export`, ycode-radar) can use it while an agent is attached.
5

19type ToolResult = Result<String, String>;

21fn err(e: impl std::fmt::Display) -> String {
22    format!("{e:#}")

25#[derive(Deserialize, schemars::JsonSchema)]
26struct AddArgs {
27    repo: String,
28    /// Path relative to the repo.
29    file: String,
30    line: u32,
31    note: Option<String>,
32    /// The selected text, if narrower than the line.
33    quote: Option<String>,
34    /// Where the report came from (default "mcp").
35    source: Option<String>,
36}

38#[derive(Deserialize, schemars::JsonSchema)]
39struct ListArgs {
40    /// Include closed reports.
41    #[serde(default)]
42    all: bool,
43}
44
45#[derive(Deserialize, schemars::JsonSchema)]
46struct NumberArgs {
47    number: u64,
48}

50#[derive(Clone)]
51pub struct RadarMcp {
52    root: PathBuf,
53    tool_router: ToolRouter<Self>,
54}
55
56impl RadarMcp {
57    fn new(root: PathBuf) -> Self {
58        Self {
59            root,
60            tool_router: Self::tool_router(),
61        }
62    }
63
64    fn store(&self) -> Result<Store, String> {
65        Store::open(&self.root).map_err(err)
66    }
67}
68
69#[tool_router]
70impl RadarMcp {
71    #[tool(description = "File a report; the line text is read from the workspace.")]
72    fn add(&self, Parameters(a): Parameters<AddArgs>) -> ToolResult {
73        let report = self
74            .store()?
75            .add(NewReport {
76                repo: a.repo,
77                file: a.file,
78                line: a.line,
79                line_text: None,
80                quote: a.quote,
81                note: a.note,
82                source: Some(a.source.unwrap_or_else(|| "mcp".into())),
83            })
84            .map_err(err)?;
85        Ok(format!(
86            "#{} {}/{}:{}",
87            report.number, report.repo, report.file, report.line
88        ))
89    }
90
91    #[tool(description = "List reports (all: true also includes closed ones).")]
92    fn list(&self, Parameters(a): Parameters<ListArgs>) -> ToolResult {
93        let reports = self.store()?.list(a.all).map_err(err)?;
94        let mut out = String::new();
95        for report in &reports {
96            let closed = if report.status == Status::Closed {
97                " (closed)"
98            } else {
99                ""
100            };
101            let _ = writeln!(
102                out,
103                "#{} {}/{}:{}{closed}  {}\n    {}",
104                report.number,
105                report.repo,
106                report.file,
107                report.line,
108                store::filed_at(report),
109                report.line_text.trim()
110            );
111            if let Some(note) = &report.note {
112                let _ = writeln!(out, "    {note}");
113            }
114        }
115        if out.is_empty() {
116            out = "no reports".into();
117        }
118        Ok(out)
119    }
120
121    #[tool(description = "Close a report.")]
122    fn close(&self, Parameters(a): Parameters<NumberArgs>) -> ToolResult {
123        self.store()?.set_status(a.number, Status::Closed).map_err(err)?;
124        Ok(format!("#{} closed", a.number))
125    }
126
127    #[tool(description = "Reopen a closed report.")]
128    fn reopen(&self, Parameters(a): Parameters<NumberArgs>) -> ToolResult {
129        self.store()?.set_status(a.number, Status::Open).map_err(err)?;
130        Ok(format!("#{} reopened", a.number))
131    }
132
133    #[tool(description = "Delete a report.")]
134    fn rm(&self, Parameters(a): Parameters<NumberArgs>) -> ToolResult {
135        self.store()?.remove(a.number).map_err(err)?;
136        Ok(format!("#{} removed", a.number))
137    }
138
139    #[tool(
140        description = "Write issues/docs/NNNN.md and issues/docs/index.md for open reports; returns the docs directory."
141    )]
142    fn export(&self) -> ToolResult {
143        let docs = self.store()?.export().map_err(err)?;
144        Ok(docs.display().to_string())
145    }
146}
147
148#[tool_handler(router = self.tool_router)]
149impl ServerHandler for RadarMcp {
150    fn get_info(&self) -> ServerConfig {
151        ServerConfig::new(ServerCapabilities::builder().enable_tools().build()).with_instructions(
152            "radar: reports for agents, a line of code plus a note. `add` to file one, \
153             `list` (optionally `all`) to see open/closed reports, `close`/`reopen`/`rm` by \
154             number, `export` to refresh the published markdown. Only one radar process may \
155             hold the database open at a time, so don't run the `radar` CLI against the same \
156             workspace while this server is attached.",
157        )
158    }
159}

161/// Serve MCP on stdin/stdout until the client disconnects.
162pub fn serve(root: PathBuf) -> anyhow::Result<()> {
163    // Fail early if the database can't be opened at all.
164    drop(Store::open(&root)?);
165    let rt = tokio::runtime::Builder::new_multi_thread()
166        .enable_all()
167        .build()?;
168    rt.block_on(async move {
169        let service = RadarMcp::new(root)
170            .serve(rmcp::transport::stdio())
171            .await?;
172        service.waiting().await?;
173        Ok(())
174    })
175}
