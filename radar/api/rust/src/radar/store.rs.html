1//! The report store: a PoloDB database in `<workspace>/issues/radar.db`, and the
2//! markdown it publishes to `<workspace>/issues/docs/`.

4use std::collections::BTreeSet;

7use anyhow::{Context, bail};

12#[derive(Serialize, Deserialize, Clone, Copy, Debug, PartialEq, Eq)]
13#[serde(rename_all = "lowercase")]
14pub enum Status {
15    Open,
16    Closed,
17}
18
19/// One report: where it points (repo, file, line, the line's text) and what was said.
20#[derive(Serialize, Deserialize, Clone, Debug)]
21pub struct Report {
22    pub number: u64,
23    pub repo: String,
24    /// Path relative to the repo root.
25    pub file: String,
26    /// 1-based.
27    pub line: u32,
28    pub line_text: String,
29    /// The text the user selected, when it isn't just the line.
30    #[serde(default)]
31    pub quote: Option<String>,
32    #[serde(default)]
33    pub note: Option<String>,
34    pub status: Status,
35    /// Where it was filed from: cli, wiki, ycode.
36    pub source: String,
37    pub created: chrono::DateTime<chrono::Utc>,
38}

40/// What a client sends. The line text is read from the workspace when omitted.
41#[derive(Deserialize, Debug)]
42pub struct NewReport {
43    pub repo: String,
44    pub file: String,
45    pub line: u32,

47    pub line_text: Option<String>,

56pub struct Store {
57    root: PathBuf,
58    reports: Collection<Report>,
59    // Keeps the database open for the collection handle.

63pub fn issues_dir(root: &Path) -> PathBuf {
64    root.join("issues")
65}
66
67impl Store {
68    pub fn open(root: &Path) -> anyhow::Result<Self> {
69        let path = issues_dir(root).join("radar.db");
70        // Only one process can have the database open; every user (CLI, `radar mcp`)
71        // holds it for a single operation, so wait a little for the lock.
72        let mut tries = 0;
73        let db = loop {
74            match Database::open_path(&path) {
75                Ok(db) => break db,
76                Err(_) if tries < 50 => {
77                    tries += 1;
78                    std::thread::sleep(std::time::Duration::from_millis(100));
79                }
80                Err(e) => {
81                    return Err(e).with_context(|| {
82                        format!("opening {} (another radar process holds it)", path.display())
83                    });
84                }
85            }
86        };
87        Ok(Self {
88            root: root.to_path_buf(),
89            reports: db.collection("reports"),
90            _db: db,
91        })
92    }
93
94    pub fn list(&self, all: bool) -> anyhow::Result<Vec<Report>> {
95        let filter = if all {
96            doc! {}
97        } else {
98            doc! { "status": "open" }
99        };
100        let mut reports = self
101            .reports
102            .find(filter)
103            .run()?
104            .collect::<Result<Vec<_>, _>>()?;
105        reports.sort_by_key(|report| report.number);
106        Ok(reports)
107    }
108
109    pub fn add(&self, new: NewReport) -> anyhow::Result<Report> {
110        let path = self.source_path(&new.repo, &new.file)?;
111        let line_text = match new.line_text {
112            Some(text) => text,
113            None => read_line(&path, new.line)?,
114        };
115        let number = self.list(true)?.last().map_or(1, |last| last.number + 1);
116        let report = Report {
117            number,
118            repo: new.repo,
119            file: new.file,
120            line: new.line,
121            quote: new
122                .quote
123                .map(|quote| quote.trim().to_string())
124                .filter(|quote| !quote.is_empty() && quote != line_text.trim()),
125            line_text,
126            note: new.note.filter(|note| !note.trim().is_empty()),
127            status: Status::Open,
128            source: new.source.unwrap_or_else(|| "cli".into()),
129            created: chrono::Utc::now(),
130        };
131        self.reports.insert_one(&report)?;
132        Ok(report)
133    }
134
135    pub fn set_status(&self, number: u64, status: Status) -> anyhow::Result<()> {
136        let status = serde_json::to_value(status)?;
137        let result = self.reports.update_one(
138            doc! { "number": number as i64 },
139            doc! { "$set": { "status": status.as_str().unwrap_or_default() } },
140        )?;
141        if result.matched_count == 0 {
142            bail!("no report #{number}");
143        }
144        Ok(())
145    }
146
147    pub fn remove(&self, number: u64) -> anyhow::Result<()> {
148        if self
149            .reports
150            .delete_one(doc! { "number": number as i64 })?
151            .deleted_count
152            == 0
153        {
154            bail!("no report #{number}");
155        }
156        Ok(())
157    }
158
159    /// `<repo>/<file>` inside the workspace; rejects paths that escape the repo.
160    fn source_path(&self, repo: &str, file: &str) -> anyhow::Result<PathBuf> {
161        let repo_dir = self
162            .root
163            .join(repo)
164            .canonicalize()
165            .with_context(|| format!("no repo {repo} in {}", self.root.display()))?;
166        if repo_dir.parent() != Some(self.root.canonicalize()?.as_path()) {
167            bail!("{repo} is not a workspace repo");
168        }
169        let path = repo_dir
170            .join(file)
171            .canonicalize()
172            .with_context(|| format!("no file {file} in {repo}"))?;
173        if !path.starts_with(&repo_dir) {
174            bail!("{file} is outside {repo}");
175        }
176        Ok(path)
177    }
178
179    /// Writes `docs/NNNN.md` per open report and `docs/index.md`, removing pages of
180    /// closed and deleted reports.
181    pub fn export(&self) -> anyhow::Result<PathBuf> {
182        let docs = issues_dir(&self.root).join("docs");
183        std::fs::create_dir_all(&docs)?;
184        let reports = self.list(false)?;
185        let mut written = BTreeSet::new();
186        for report in &reports {
187            let name = format!("{:04}.md", report.number);
188            write_if_changed(&docs.join(&name), &report_page(report)?)?;
189            written.insert(name);
190        }
191        write_if_changed(&docs.join("index.md"), &index_page(&reports))?;
192        for entry in std::fs::read_dir(&docs)? {
193            let name = entry?.file_name().to_string_lossy().into_owned();
194            let generated = name.len() > 3
195                && name.ends_with(".md")
196                && name[..name.len() - 3]
197                    .bytes()
198                    .all(|byte| byte.is_ascii_digit());
199            if generated && !written.contains(&name) {
200                std::fs::remove_file(docs.join(&name))?;
201            }
202        }
203        Ok(docs)
204    }
205}

207fn read_line(path: &Path, line: u32) -> anyhow::Result<String> {
208    let text = std::fs::read_to_string(path)?;
209    let index = usize::try_from(line)?
210        .checked_sub(1)
211        .context("lines start at 1")?;
212    text.lines()
213        .nth(index)
214        .map(str::to_owned)
215        .with_context(|| format!("{} has no line {line}", path.display()))
216}
217
218fn write_if_changed(path: &Path, content: &str) -> std::io::Result<()> {
219    if std::fs::read_to_string(path).is_ok_and(|current| current == content) {
220        return Ok(());
221    }
222    std::fs::write(path, content)
223}

225/// Local date and time, e.g. "2026-11-17 18:20 -05:00".
226pub fn filed_at(report: &Report) -> String {
227    report
228        .created
229        .with_timezone(&chrono::Local)
230        .format("%Y-%m-%d %H:%M %:z")
231        .to_string()
232}

234fn location(report: &Report) -> String {
235    format!("{}/{}:{}", report.repo, report.file, report.line)

238fn report_page(report: &Report) -> anyhow::Result<String> {
239    // JSON strings are valid YAML scalars, so the front matter stays parseable.
240    let quote = |value: &str| serde_json::to_string(value);
241    let status = serde_json::to_value(report.status)?;
242    let mut page = format!(
243        "---\ntitle: {}\nnumber: {}\nstatus: {}\nrepo: {}\nfile: {}\nline: {}\ncreated: {}\nsource: {}\n---\n\n",
244        quote(&format!("#{} {}", report.number, location(report)))?,
245        report.number,
246        status.as_str().unwrap_or_default(),
247        quote(&report.repo)?,
248        quote(&report.file)?,
249        report.line,
250        report.created.to_rfc3339(),
251        quote(&report.source)?,
252    );
253    page += &format!("# #{} `{}`\n\n", report.number, location(report));
254    page += &format!(
255        "Filed {} from {} · {}\n\n",
256        filed_at(report),
257        report.source,
258        status.as_str().unwrap_or_default()
259    );
260    if let Some(note) = &report.note {
261        page += &format!("{}\n\n", note.trim());
262    }
263    let language = Path::new(&report.file)
264        .extension()
265        .map(|extension| extension.to_string_lossy().into_owned())
266        .unwrap_or_default();
267    let fence = if report.line_text.contains("```") {
268        "````"
269    } else {
270        "```"
271    };
272    page += &format!("{fence}{language}\n{}\n{fence}\n", report.line_text);
273    if let Some(selection) = &report.quote {
274        let quoted = selection
275            .lines()
276            .map(|line| format!("> {line}"))
277            .collect::<Vec<_>>()
278            .join("\n");
279        page += &format!("\nSelected:\n\n{quoted}\n");
280    }
281    Ok(page)
282}

284fn index_page(reports: &[Report]) -> String {
285    let mut page = String::from("---\ntitle: \"Radar\"\n---\n\n# Radar\n\nOpen reports.\n\n");
286    if reports.is_empty() {
287        return page + "None.\n";
288    }
289    page += "| # | Filed | Location | Note |\n| - | - | - | - |\n";
290    for report in reports {
291        let note = report
292            .note
293            .as_deref()
294            .and_then(|note| note.lines().next())
295            .unwrap_or("")
296            .replace('|', "\\|");
297        page += &format!(
298            "| [{}]({:04}.md) | {} | `{}` | {note} |\n",
299            report.number,
300            report.number,
301            filed_at(report),
302            location(report)
303        );
304    }
305    page
