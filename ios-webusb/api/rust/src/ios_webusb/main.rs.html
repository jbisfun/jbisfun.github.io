14fn paths() -> anyhow::Result<Paths> {
15    let data = dirs::data_dir()
16        .context("no data directory")?
17        .join("ios-webusb");
18    let runtime = dirs::runtime_dir()
19        .map(|dir| dir.join("ios-webusb"))
20        .unwrap_or_else(|| data.clone());
21    Ok(Paths {
22        socket: std::env::var_os("IOS_WEBUSB_SOCKET")
23            .map(PathBuf::from)
24            .unwrap_or_else(|| runtime.join("usbmuxd")),
25        lockdown: data.join("lockdown"),
26    })
27}

29fn main() -> anyhow::Result<()> {
30    tracing_subscriber::fmt()
31        .with_env_filter(
32            tracing_subscriber::EnvFilter::try_from_default_env()
33                .unwrap_or_else(|_| "info,mdns_sd=warn".into()),
34        )
35        .init();
36
37    let show = std::env::args().any(|arg| arg == "--show");
38    let paths = paths()?;
39
40    if std::os::unix::net::UnixStream::connect(&paths.socket).is_ok() {
41        bail!(
42            "another driver is already serving {}",
43            paths.socket.display()
44        );
45    }
46    if let Some(parent) = paths.socket.parent() {
47        std::fs::create_dir_all(parent)?;
48    }
49    if rusbmux::handler::set_lockdown_path(paths.lockdown.clone()).is_err() {
50        bail!("lockdown path already set");
51    }
52
53    let runtime = tokio::runtime::Runtime::new()?;
54
55    let (ui_tx, ui_rx) = mpsc::unbounded_channel();
56    let (device_tx, device_rx) = mpsc::unbounded_channel();
57    let state = State::new(paths, ui_tx, device_tx, runtime.handle().clone());
58
59    runtime.spawn({
60        let state = state.clone();
61        let socket = state.paths.socket.to_string_lossy().into_owned();
62        async move {
63            if let Err(error) = rusbmux::daemon::run(Some(socket)).await {
64                log::error!("USB driver stopped: {error}");
65            }
66            state.quit();
67        }
68    });
69    runtime.block_on(wait_for_socket(&state));
70
71    devices::spawn(state.clone(), device_rx);
72    runtime.spawn(tray::run(state.clone()));
73    log::info!("Serving usbmuxd on {}", state.paths.socket.display());
74
75    ui::run(state.clone(), ui_rx, show);
76
77    runtime.shutdown_timeout(Duration::from_millis(500));
78    std::fs::remove_file(&state.paths.socket).ok();
79    Ok(())
80}

82async fn wait_for_socket(state: &State) {
83    for _ in 0..50 {
84        if tokio::net::UnixStream::connect(&state.paths.socket)
85            .await
86            .is_ok()
87        {
88            return;
89        }
90        tokio::time::sleep(Duration::from_millis(100)).await;
91    }
92    log::warn!(
93        "usbmuxd socket {} did not come up",
94        state.paths.socket.display()
95    );
96}
