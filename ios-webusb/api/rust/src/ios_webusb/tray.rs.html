6struct Tray {
7    state: State,
8    snapshot: Snapshot,
9}
10
11pub async fn run(state: State) {
12    let tray = Tray {
13        snapshot: state.snapshot(),
14        state: state.clone(),
15    };
16    let handle = match tray.spawn().await {
17        Ok(handle) => handle,
18        Err(error) => {
19            log::error!("Could not create the tray icon: {error}");
20            return;
21        }
22    };
23    let mut revisions = state.subscribe();
24    while revisions.changed().await.is_ok() {
25        let snapshot = state.snapshot();
26        if handle
27            .update(|tray| tray.snapshot = snapshot)
28            .await
29            .is_none()
30        {
31            return;
32        }
33    }
34}

36fn item(label: impl Into<String>, activate: impl Fn(&mut Tray) + Send + 'static) -> MenuItem<Tray> {
37    StandardItem {
38        label: label.into(),
39        activate: Box::new(activate),
40        ..Default::default()
41    }
42    .into()
43}
44
45fn note(label: impl Into<String>) -> MenuItem<Tray> {
46    StandardItem {
47        label: label.into(),
48        enabled: false,
49        ..Default::default()
50    }
51    .into()
52}

54impl ksni::Tray for Tray {
55    fn id(&self) -> String {
56        "ios-webusb".into()
57    }
58
59    fn title(&self) -> String {
60        "iOS WebUSB".into()
61    }
62
63    fn icon_name(&self) -> String {
64        "phone-symbolic".into()
65    }
66
67    fn tool_tip(&self) -> ksni::ToolTip {
68        let devices = self.snapshot.devices.len();
69        ksni::ToolTip {
70            title: "iOS WebUSB".into(),
71            description: format!("{devices} device(s)"),
72            ..Default::default()
73        }
74    }
75
76    fn activate(&mut self, _x: i32, _y: i32) {
77        self.state.show_window();
78    }
79
80    fn menu(&self) -> Vec<MenuItem<Self>> {
81        let mut menu = Vec::new();
82        if self.snapshot.devices.is_empty() {
83            menu.push(note("No devices"));
84        }
85        for device in &self.snapshot.devices {
86            let transport = match device.transport {
87                Transport::Usb => "USB",
88                Transport::WiFi => "Wi-Fi",
89            };
90            let device_id = device.device_id;
91            let mut submenu = vec![note(device.udid.clone())];
92            if let Some(status) = &device.status {
93                submenu.push(note(status.clone()));
94            }
95            submenu.push(item(
96                if device.paired { "Pair again" } else { "Pair" },
97                move |tray: &mut Tray| tray.state.pair(device_id),
98            ));
99            if device.transport == Transport::Usb {
100                submenu.push(item("Release for WebUSB", move |tray: &mut Tray| {
101                    tray.state.release(device_id)
102                }));
103            }
104            menu.push(
105                SubMenu {
106                    label: format!(
107                        "{} ({transport}{})",
108                        device.name,
109                        if device.paired { "" } else { ", not paired" }
110                    ),
111                    submenu,
112                    ..Default::default()
113                }
114                .into(),
115            );
116        }
117
118        menu.push(MenuItem::Separator);
119        menu.push(item("Open", |tray: &mut Tray| tray.state.show_window()));
120        menu.push(item("Quit", |tray: &mut Tray| tray.state.quit()));
121        menu
122    }
123}
