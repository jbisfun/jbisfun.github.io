15/// Watches our own usbmuxd socket with idevice. It gets its own thread because
16/// idevice's listen stream isn't `Send`.
17pub fn spawn(state: State, commands: mpsc::UnboundedReceiver<DeviceCommand>) {
18    let spawned = std::thread::Builder::new()
19        .name("devices".into())
20        .spawn(move || {
21            let runtime = match tokio::runtime::Builder::new_current_thread()
22                .enable_all()
23                .build()
24            {
25                Ok(runtime) => runtime,
26                Err(error) => {
27                    log::error!("Could not start the device runtime: {error}");
28                    return;
29                }
30            };
31            LocalSet::new().block_on(&runtime, run(state, commands));
32        });
33    if let Err(error) = spawned {
34        log::error!("Could not spawn the device thread: {error}");
35    }
36}

43async fn run(state: State, mut commands: mpsc::UnboundedReceiver<DeviceCommand>) {
44    let address = UsbmuxdAddr::UnixSocket(state.paths.socket.to_string_lossy().into_owned());
45    let (events_tx, mut events) = mpsc::unbounded_channel();
46    tokio::task::spawn_local(watch(address.clone(), events_tx));
47
48    let mut devices: HashMap<u32, UsbmuxdDevice> = HashMap::new();
49    let mut tick = tokio::time::interval(Duration::from_secs(2));
50    loop {
51        tokio::select! {
52            Some(event) = events.recv() => match event {
53                Event::Connected(device, name) => {
54                    state.upsert_device(entry(&state, &device, name));
55                    devices.insert(device.device_id, device);
56                }
57                Event::Disconnected(device_id) => {
58                    devices.remove(&device_id);
59                    state.remove_device(device_id);
60                }
61            },
62            Some(command) = commands.recv() => match command {
63                DeviceCommand::Pair(device_id) => {
64                    let Some(device) = devices.get(&device_id).cloned() else { continue };
65                    state.set_device_status(
66                        device_id,
67                        Some("Pairing: unlock the device and tap Trust".into()),
68                    );
69                    let state = state.clone();
70                    let address = address.clone();
71                    tokio::task::spawn_local(async move {
72                        let status = match pair(&address, &device).await {
73                            Ok(()) => None,
74                            Err(error) => Some(format!("Pairing failed: {error}")),
75                        };
76                        state.set_device_status(device_id, status);
77                        state.refresh_pairing();
78                    });
79                }
80            },
81            _ = tick.tick() => state.refresh_pairing(),
82            else => return,
83        }
84    }
85}

87/// Reconnects to the socket until the receiver is dropped; a lost connection
88/// invalidates its device IDs, so they're reported as gone.
89async fn watch(address: UsbmuxdAddr, events: mpsc::UnboundedSender<Event>) {
90    let mut connected = Vec::new();
91    while !events.is_closed() {
92        if let Err(error) = listen(&address, &events, &mut connected).await {
93            log::debug!("usbmuxd listen interrupted: {error}");
94        }
95        for device_id in connected.drain(..) {
96            events.send(Event::Disconnected(device_id)).ok();
97        }
98        tokio::time::sleep(Duration::from_secs(2)).await;
99    }
100}

102async fn listen(
103    address: &UsbmuxdAddr,
104    events: &mpsc::UnboundedSender<Event>,
105    connected: &mut Vec<u32>,
106) -> Result<(), idevice::IdeviceError> {
107    let mut connection = address.connect(0).await?;
108    let mut stream = connection.listen().await?;
109    while let Some(event) = stream.next().await {
110        match event? {
111            UsbmuxdListenEvent::Connected(device) => {
112                let name = device_name(address, &device)
113                    .await
114                    .unwrap_or_else(|| device.udid.clone());
115                connected.push(device.device_id);
116                events.send(Event::Connected(device, name)).ok();
117            }
118            UsbmuxdListenEvent::Disconnected(device_id) => {
119                connected.retain(|id| *id != device_id);
120                events.send(Event::Disconnected(device_id)).ok();
121            }
122        }
123    }
124    Ok(())
125}

127async fn device_name(address: &UsbmuxdAddr, device: &UsbmuxdDevice) -> Option<String> {
128    let lookup = async {
129        let provider = device.to_provider(address.clone(), LABEL);
130        let mut lockdown = LockdownClient::connect(&provider).await?;
131        lockdown.get_value(Some("DeviceName"), None).await
132    };
133    match tokio::time::timeout(Duration::from_secs(3), lookup).await {
134        Ok(Ok(value)) => value
135            .as_string()
136            .filter(|name| !name.is_empty())
137            .map(str::to_owned),
138        result => {
139            log::warn!("Could not read the name of {}: {result:?}", device.udid);
140            None
141        }
142    }
143}

145async fn pair(address: &UsbmuxdAddr, device: &UsbmuxdDevice) -> Result<(), idevice::IdeviceError> {
146    let mut usbmuxd = address.connect(0).await?;
147    let mut lockdown = LockdownClient::connect(&device.to_provider(address.clone(), LABEL)).await?;
148    let host_id = uuid::Uuid::new_v4().to_string().to_uppercase();
149    let buid = usbmuxd.get_buid().await?;
150    let mut pairing_file = lockdown.pair(host_id, buid, None).await?;
151    pairing_file.udid = Some(device.udid.clone());
152    usbmuxd
153        .save_pair_record(&device.udid, pairing_file.serialize()?)
154        .await
155}

157fn entry(state: &State, device: &UsbmuxdDevice, name: String) -> DeviceEntry {
158    DeviceEntry {
159        device_id: device.device_id,
160        udid: device.udid.clone(),
161        name,
162        transport: match device.connection_type {
163            Connection::Network(_) => Transport::WiFi,
164            _ => Transport::Usb,
165        },
166        paired: state.is_paired(&device.udid),
167        status: None,
168    }
