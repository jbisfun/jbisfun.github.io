1//! The pullfile `hooks` section: shell commands run around commits and syncs.
2
3use anyhow::{Result, bail};
4use serde::{Deserialize, Serialize};
5use std::path::Path;
6use std::process::Command;
7
8/// Pullfile section key.
9pub const SECTION: &str = "hooks";
10
11#[derive(Serialize, Deserialize, Debug, Default, Clone)]
12#[serde(default)]
13pub struct Hooks {
14    pub precommit: Vec<String>,
15    pub postcommit: Vec<String>,
16    pub pull: Vec<String>,
17}
18
19/// Run each hook command through `sh -c` in the pullfile root.
20/// Env: the workspace env (`pull env`), PULL_ROOT, PULL_HOOK, PULL_REPOS
21/// (space-separated), plus `extra`.
22pub fn run(
23    kind: &str,
24    cmds: &[String],
25    root: &Path,
26    repos: &[String],
27    env: &[(String, String)],
28    extra: &[(&str, &str)],
29) -> Result<()> {
30    for cmd in cmds {
31        eprintln!("[{kind}] {cmd}");
32        let mut c = Command::new("sh");
33        c.arg("-c")
34            .arg(cmd)
35            .current_dir(root)
36            .envs(env.iter().map(|(k, v)| (k, v)))
37            .env("PULL_ROOT", root)
38            .env("PULL_HOOK", kind)
39            .env("PULL_REPOS", repos.join(" "));
40        for (k, v) in extra {
41            c.env(k, v);
42        }
43        let st = c.status()?;
44        if !st.success() {
45            bail!("{kind} hook failed ({st}): {cmd}");
46        }
47    }
48    Ok(())
49}
