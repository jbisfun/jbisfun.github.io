1//! `pull mcp`: an MCP server (stdio) over the workspace, for agents.
2//!
3//! Multi-repo context (workspace overview, status, diffs, logs, search, files), programs
4//! (one-shot commands with the workspace env) and stateful terminals (the launchd
5//! sessions: they persist between calls and across agents, and a human can attach to the
6//! same terminal with `pull session attach`).
7//!
8//! The session/job tools need the `sessions` feature (the launchd daemon). The workspace
9//! env is injected by the caller ([`EnvProvider`]), so this crate doesn't depend on how
10//! (or whether) the env feature is built.
11

41/// Keep tool output within what an agent can take in one go.
42pub(crate) fn truncate(mut s: String, max: usize) -> String {
43    if s.len() > max {
44        let mut cut = max;
45        while !s.is_char_boundary(cut) {
46            cut -= 1;
47        }
48        let total = s.len();
49        s.truncate(cut);
50        let _ = write!(s, "\n… truncated ({total} bytes total)");
51    }
52    s
53}

129impl PullMcp {
130    pub fn new(root: PathBuf, env: EnvProvider) -> Self {
131        #[allow(unused_mut)]
132        let mut tool_router = Self::tool_router();
133        #[cfg(feature = "sessions")]
134        {
135            tool_router += Self::session_router();
136        }
137        Self {
138            root,
139            env,
140            tool_router,
141        }
142    }
143
144    fn pullfile(&self) -> Result<Pullfile, String> {
145        Pullfile::load(&self.root).map_err(err)
146    }
147
148    /// (name, repo) for `only`, or every cloned repo.
149    fn repos(&self, only: Option<&str>) -> Result<Vec<(String, Repository)>, String> {
150        let pf = self.pullfile()?;
151        if let Some(name) = only
152            && !pf.repos.iter().any(|r| r.name == name)
153        {
154            return Err(format!("no repo named {name} in the pullfile"));
155        }
156        Ok(pf
157            .repos
158            .iter()
159            .filter(|r| only.is_none_or(|n| n == r.name))
160            .filter_map(|r| {
161                Repository::open(self.root.join(&r.name))
162                    .ok()
163                    .map(|repo| (r.name.clone(), repo))
164            })
165            .collect())
166    }
167
168    /// A path inside the workspace (no escaping it with `..`).
169    fn inside(&self, rel: Option<&str>) -> Result<PathBuf, String> {
170        let p = match rel {
171            Some(r) if !r.is_empty() => self.root.join(r),
172            _ => self.root.clone(),
173        };
174        let canon = p
175            .canonicalize()
176            .map_err(|e| format!("{}: {e}", p.display()))?;
177        let root = self.root.canonicalize().map_err(err)?;
178        if !canon.starts_with(&root) {
179            return Err(format!("{} is outside the workspace", p.display()));
180        }
181        Ok(canon)
182    }
183
184    fn rel(&self, p: &Path) -> String {
185        p.strip_prefix(&self.root)
186            .unwrap_or(p)
187            .display()
188            .to_string()
189    }
190
191    fn env(&self) -> Vec<(String, String)> {
192        (self.env)()
193    }
194}

198    #[tool(
199        description = "Overview of the pull workspace: root, every repo (upstream, branch, HEAD, pinned commit, ahead/behind origin, changed files), lint rules, env and jobs. Start here."
200    )]
201    fn workspace(&self) -> ToolResult {
202        let pf = self.pullfile()?;
203        let mut out = format!("root: {}\n", self.root.display());
204        if let Some(plist::Value::Dictionary(lint)) = pf.sections.get("lint") {
205            let rules: Vec<_> = lint
206                .iter()
207                .map(|(k, v)| format!("{k}={}", scalar(v)))
208                .collect();
209            let _ = writeln!(out, "lint: {}", rules.join(" "));
210        }
211        out.push_str("repos:\n");
212        for r in &pf.repos {
213            let _ = write!(out, "  {} ({}, branch {})", r.name, r.upstream, r.branch);
214            let Ok(repo) = Repository::open(self.root.join(&r.name)) else {
215                out.push_str(" not cloned\n");
216                continue;
217            };
218            let head = git::head_oid(&repo);
219            let branch = repo
220                .head()
221                .ok()
222                .and_then(|h| h.shorthand().ok().map(String::from))
223                .unwrap_or("(unborn)".into());
224            let _ = write!(
225                out,
226                ": on {branch} at {}",
227                head.map(git::short).unwrap_or("-".into())
228            );
229            match (&r.commit, head) {
230                (Some(p), Some(h)) if *p != h.to_string() => {
231                    let _ = write!(out, ", pinned {}", &p[..8.min(p.len())]);
232                }
233                (None, _) => out.push_str(", unpinned"),
234                _ => {}
235            }
236            if let (Some(h), Some(o)) = (head, git::remote_oid(&repo, &r.branch))
237                && h != o
238                && let Ok((a, b)) = repo.graph_ahead_behind(h, o)
239            {
240                let _ = write!(out, ", +{a}/-{b} origin");
241            }
242            let changed = git::status_lines(&repo).map(|l| l.len()).unwrap_or(0);
243            let _ = writeln!(out, ", {changed} changed");
244        }
245        let env = self.env();
246        if !env.is_empty() {
247            out.push_str("env:\n");
248            for (k, v) in env.iter().filter(|(k, _)| k != "PATH") {
249                let _ = writeln!(out, "  {k}={v}");
250            }
251        }
252        #[cfg(feature = "sessions")]
253        {
254            let jobs: Vec<pull_launchd::Job> = pf.section(pull_launchd::SECTION).map_err(err)?;
255            if !jobs.is_empty() {
256                out.push_str("jobs:\n");
257                for j in &jobs {
258                    let _ = writeln!(out, "  {}: {}", j.label, j.argv().join(" "));
259                }
260            }
261        }
262        Ok(out)
263    }

265    #[tool(description = "Changed files per repo (git status, XY format like `git status -s`).")]
266    fn status(&self, Parameters(a): Parameters<RepoArg>) -> ToolResult {
267        let mut out = String::new();
268        for (name, repo) in self.repos(a.repo.as_deref())? {
269            let lines = git::status_lines(&repo).map_err(err)?;
270            let _ = writeln!(
271                out,
272                "{name}: {}",
273                if lines.is_empty() { "clean" } else { "" }
274            );
275            for l in lines {
276                let _ = writeln!(out, "  {l}");
277            }
278        }
279        Ok(out)
280    }

282    #[tool(
283        description = "Unified diff of uncommitted changes, per repo (unstaged by default, or staged)."
284    )]
285    fn diff(&self, Parameters(a): Parameters<DiffArgs>) -> ToolResult {
286        let mut out = String::new();
287        for (name, repo) in self.repos(a.repo.as_deref())? {
288            let mut opts = DiffOptions::new();
289            if let Some(p) = &a.path {
290                opts.pathspec(p);
291            }
292            let diff = if a.staged {
293                let head = repo.head().ok().and_then(|h| h.peel_to_tree().ok());
294                repo.diff_tree_to_index(head.as_ref(), None, Some(&mut opts))
295            } else {
296                repo.diff_index_to_workdir(None, Some(&mut opts))
297            }
298            .map_err(err)?;
299            if diff.deltas().len() == 0 {
300                continue;
301            }
302            let _ = writeln!(out, "### {name}");
303            diff.print(DiffFormat::Patch, |_, _, line| {
304                match line.origin() {
305                    '+' | '-' | ' ' => out.push(line.origin()),
306                    _ => {}
307                }
308                out.push_str(&String::from_utf8_lossy(line.content()));
309                out.len() < MAX_OUTPUT * 2
310            })
311            .ok();
312        }
313        if out.is_empty() {
314            out = "no changes".into();
315        }
316        Ok(truncate(out, MAX_OUTPUT))
317    }

319    #[tool(description = "Recent commits per repo: short id, date, author, subject.")]
320    fn log(&self, Parameters(a): Parameters<LogArgs>) -> ToolResult {
321        let limit = a.limit.unwrap_or(10);
322        let mut out = String::new();
323        for (name, repo) in self.repos(a.repo.as_deref())? {
324            let _ = writeln!(out, "{name}:");
325            let mut walk = repo.revwalk().map_err(err)?;
326            if walk.push_head().is_err() {
327                continue;
328            }
329            walk.set_sorting(Sort::TIME).ok();
330            for oid in walk.take(limit).flatten() {
331                let Ok(c) = repo.find_commit(oid) else {
332                    continue;
333                };
334                let when = c.time().seconds();
335                let _ = writeln!(
336                    out,
337                    "  {} {} {} {}",
338                    git::short(oid),
339                    format_time(when),
340                    c.author().name().unwrap_or("?"),
341                    c.summary().ok().flatten().unwrap_or("")
342                );
343            }
344        }
345        Ok(out)
346    }

348    #[tool(
349        description = "Search file contents across the workspace's repos with a regex (ripgrep's engine; respects .gitignore). Returns repo/path:line: text."
350    )]
351    fn grep(&self, Parameters(a): Parameters<GrepArgs>) -> ToolResult {
352        use grep_regex::RegexMatcherBuilder;
353        use grep_searcher::{BinaryDetection, SearcherBuilder, sinks::UTF8};
354        let max = a.max_results.unwrap_or(200);
355        let matcher = RegexMatcherBuilder::new()
356            .case_smart(true)
357            .build(&a.pattern)
358            .map_err(err)?;
359        let mut searcher = SearcherBuilder::new()
360            .binary_detection(BinaryDetection::quit(0))
361            .line_number(true)
362            .build();
363        let mut out = String::new();
364        let mut n = 0;
365        'repos: for (name, _) in self.repos(a.repo.as_deref())? {
366            let dir = self.root.join(&name);
367            for entry in self.walker(&dir, a.glob.as_deref())? {
368                let Ok(entry) = entry else { continue };
369                if !entry.file_type().is_some_and(|t| t.is_file()) {
370                    continue;
371                }
372                let rel = self.rel(entry.path());
373                let mut done = false;
374                searcher
375                    .search_path(
376                        &matcher,
377                        entry.path(),
378                        UTF8(|line_no, line| {
379                            let mut line = line.trim_end().to_string();
380                            if line.len() > 300 {
381                                line = truncate(line, 300);
382                            }
383                            let _ = writeln!(out, "{rel}:{line_no}: {line}");
384                            n += 1;
385                            done = n >= max;
386                            Ok(!done)
387                        }),
388                    )
389                    .ok();
390                if done {
391                    let _ = writeln!(out, "… stopped at {max} results");
392                    break 'repos;
393                }
394            }
395        }
396        if out.is_empty() {
397            out = "no matches".into();
398        }
399        Ok(truncate(out, MAX_OUTPUT))
400    }

402    #[tool(description = "List files under a workspace directory (respects .gitignore).")]
403    fn list_files(&self, Parameters(a): Parameters<FilesArgs>) -> ToolResult {
404        let dir = self.inside(a.path.as_deref())?;
405        let max = a.max_results.unwrap_or(500);
406        let mut out = String::new();
407        let mut n = 0;
408        for entry in self.walker(&dir, a.glob.as_deref())?.flatten() {
409            if !entry.file_type().is_some_and(|t| t.is_file()) {
410                continue;
411            }
412            let _ = writeln!(out, "{}", self.rel(entry.path()));
413            n += 1;
414            if n >= max {
415                let _ = writeln!(out, "… stopped at {max} entries");
416                break;
417            }
418        }
419        Ok(out)
420    }

422    #[tool(description = "Read a file (path relative to the workspace root), with line numbers.")]
423    fn read_file(&self, Parameters(a): Parameters<ReadArgs>) -> ToolResult {
424        let path = self.inside(Some(&a.path))?;
425        let text = std::fs::read_to_string(&path).map_err(|e| format!("{}: {e}", a.path))?;
426        let start = a.start_line.unwrap_or(1).max(1);
427        let end = a.end_line.unwrap_or(start + 400);
428        let total = text.lines().count();
429        let mut out = String::new();
430        for (i, line) in text
431            .lines()
432            .enumerate()
433            .skip(start - 1)
434            .take(end + 1 - start)
435        {
436            let _ = writeln!(out, "{:>5}|{line}", i + 1);
437        }
438        if end < total {
439            let _ = writeln!(
440                out,
441                "… {total} lines total; continue with start_line={}",
442                end + 1
443            );
444        }
445        Ok(truncate(out, MAX_OUTPUT))
446    }

448    #[tool(
449        description = "Run a shell command to completion with the workspace env (PATH, toolchains) and return its exit code and output. For long-running or interactive programs use session_new instead."
450    )]
451    async fn exec(&self, Parameters(a): Parameters<ExecArgs>) -> ToolResult {
452        let cwd = self.inside(a.cwd.as_deref())?;
453        let env = self.env();
454        let timeout = Duration::from_millis(a.timeout_ms.unwrap_or(120_000));
455        let mut cmd = tokio::process::Command::new("sh");
456        cmd.arg("-c")
457            .arg(&a.command)
458            .current_dir(cwd)
459            .envs(env)
460            .stdin(std::process::Stdio::null())
461            .kill_on_drop(true);
462        match tokio::time::timeout(timeout, cmd.output()).await {
463            Err(_) => Err(format!(
464                "timed out after {} ms (killed)",
465                timeout.as_millis()
466            )),
467            Ok(Err(e)) => Err(err(e)),
468            Ok(Ok(o)) => {
469                let mut out = format!("exit {}\n", o.status.code().unwrap_or(-1));
470                out.push_str(&String::from_utf8_lossy(&o.stdout));
471                if !o.stderr.is_empty() {
472                    out.push_str("\n[stderr]\n");
473                    out.push_str(&String::from_utf8_lossy(&o.stderr));
474                }
475                Ok(truncate(out, MAX_OUTPUT))
476            }
477        }
478    }

481impl PullMcp {
482    fn walker(&self, dir: &Path, glob: Option<&str>) -> Result<ignore::Walk, String> {
483        let mut b = ignore::WalkBuilder::new(dir);
484        b.hidden(true).git_ignore(true).git_exclude(true);
485        if let Some(g) = glob {
486            let mut o = ignore::overrides::OverrideBuilder::new(dir);
487            o.add(g).map_err(err)?;
488            b.overrides(o.build().map_err(err)?);
489        }
490        Ok(b.build())
491    }
492}

494/// A plist scalar as text (for showing pullfile sections).
495fn scalar(v: &plist::Value) -> String {
496    match v {
497        plist::Value::Boolean(b) => b.to_string(),
498        plist::Value::Integer(i) => i.to_string(),
499        plist::Value::Real(r) => r.to_string(),
500        plist::Value::String(s) => s.clone(),
501        other => format!("{other:?}"),
502    }
503}

505fn format_time(secs: i64) -> String {
506    jiff::Timestamp::from_second(secs)
507        .map(|t| t.strftime("%Y-%m-%d").to_string())
508        .unwrap_or_default()

512impl ServerHandler for PullMcp {
513    fn get_info(&self) -> ServerConfig {
514        ServerConfig::new(ServerCapabilities::builder().enable_tools().build()).with_instructions(
515            "pull workspace: several git repos under one root, managed by `pull`. Use `workspace` \
516             for an overview, then status/diff/log/grep/list_files/read_file across repos. \
517             `exec` runs a command to completion with the workspace env; sessions are stateful \
518             terminals (PTYs) that persist between calls and can be attached to by a human. \
519             Commit with `pull commit -m MSG PATHS` via exec (subject lint applies).",
520        )
521    }

524/// Serve MCP on stdin/stdout until the client disconnects.
525pub fn serve(root: PathBuf, env: EnvProvider) -> anyhow::Result<()> {
526    let rt = tokio::runtime::Builder::new_multi_thread()
527        .enable_all()
528        .build()?;
529    rt.block_on(async move {
530        let service = PullMcp::new(root, env)
531            .serve(rmcp::transport::stdio())
532            .await?;
533        service.waiting().await?;
534        Ok(())
535    })
