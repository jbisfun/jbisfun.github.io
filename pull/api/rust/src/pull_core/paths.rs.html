1//! Paths in a workspace: routing user paths to repos, root-level ignore files.
2
3use crate::{PULLFILE, Pullfile};
4use anyhow::{Context, Result, bail};
5use std::collections::BTreeMap;
6use std::path::{Component, Path, PathBuf};
7
8/// Map user paths onto (repo name -> repo-relative pathspecs), CODEOWNERS-style:
9/// the first path component under the root picks the repo.
10pub fn route(
11    root: &Path,
12    pf: &Pullfile,
13    cwd: &Path,
14    paths: &[PathBuf],
15) -> Result<BTreeMap<String, Vec<String>>> {
16    let mut out: BTreeMap<String, Vec<String>> = BTreeMap::new();
17    for p in paths {
18        let abs = absolutize(&cwd.join(p))?;
19        let rel = abs
20            .strip_prefix(root)
21            .map_err(|_| anyhow::anyhow!("{} is outside the pullfile root", p.display()))?;
22        let mut comps = rel.components();
23        let Some(Component::Normal(first)) = comps.next() else {
24            // The root itself: everything in every repo.
25            for r in &pf.repos {
26                out.entry(r.name.clone()).or_default().push("*".into());
27            }
28            continue;
29        };
30        let name = first.to_string_lossy().to_string();
31        if !pf.repos.iter().any(|r| r.name == name) {
32            bail!("{} is not inside any repo in {PULLFILE}", p.display());
33        }
34        let inner = comps.as_path().to_string_lossy().to_string();
35        out.entry(name)
36            .or_default()
37            .push(if inner.is_empty() { "*".into() } else { inner });
38    }
39    Ok(out)
40}

42/// Canonicalize, tolerating a missing final component (deleted files).
43pub fn absolutize(p: &Path) -> Result<PathBuf> {
44    if let Ok(c) = p.canonicalize() {
45        return Ok(c);
46    }
47    let parent = p.parent().context("bad path")?.canonicalize()?;
48    Ok(parent.join(p.file_name().context("bad path")?))
49}

51/// Append `line` to the root's .gitignore unless it is already there.
52pub fn ensure_ignored(root: &Path, line: &str) -> Result<()> {
53    let p = root.join(".gitignore");
54    let cur = std::fs::read_to_string(&p).unwrap_or_default();
55    if cur.lines().any(|l| l.trim() == line) {
56        return Ok(());
57    }
58    let mut s = cur;
59    if !s.is_empty() && !s.ends_with('\n') {
60        s.push('\n');
61    }
62    s.push_str(line);
63    s.push('\n');
64    std::fs::write(p, s)?;
65    Ok(())
66}

68/// Directory name for a clone URL (`.../foo.git` -> `foo`).
69pub fn repo_name_from_url(url: &str) -> Result<String> {
70    let last = url
71        .trim_end_matches('/')
72        .rsplit(['/', ':'])
73        .next()
74        .unwrap_or("")
75        .trim_end_matches(".git");
76    if last.is_empty() {
77        bail!("can't derive a name from `{url}`; pass --name");
78    }
79    Ok(last.to_string())
80}
