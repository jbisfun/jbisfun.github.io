1//! The workspace environment: pullfile `env`, then local.toml `[env]` on top.

3use anyhow::{Context, Result};
4use serde::{Deserialize, Serialize};

6use std::ffi::OsString;
7use std::path::{Path, PathBuf};

9/// Section key, in the pullfile and in local.toml.
10pub const SECTION: &str = "env";
11
12/// Environment for the workspace. Values expand `~`, `$VAR` and `${VAR}` (including
13/// `$PULL_ROOT`); relative PATH entries are relative to the pullfile root.
14#[derive(Serialize, Deserialize, Debug, Default, Clone)]
15#[serde(default)]
16pub struct Env {
17    /// Prepended to PATH, first entry wins.
18    #[serde(skip_serializing_if = "Vec::is_empty")]
19    pub path: Vec<String>,
20    /// Exported variables (toolchain selection, etc.).
21    #[serde(skip_serializing_if = "BTreeMap::is_empty")]
22    pub vars: BTreeMap<String, String>,
23}
24
25impl Env {
26    pub fn is_empty(&self) -> bool {
27        self.path.is_empty() && self.vars.is_empty()
28    }
29}
30
31/// Resolved variables to export, in order. `layers` go from shared to most specific
32/// (pullfile, then local.toml). `PATH` is complete (new entries + current PATH).
33pub fn resolve(root: &Path, layers: &[&Env]) -> Result<Vec<(String, String)>> {
34    let mut vars: BTreeMap<String, String> = BTreeMap::new();
35    vars.insert("PULL_ROOT".into(), root.display().to_string());
36    for env in layers {
37        for (k, v) in &env.vars {
38            let value = expand(v, &vars);
39            vars.insert(k.clone(), value);
40        }
41    }
42
43    // Local entries first so a machine can shadow the shared toolchain.
44    let mut entries: Vec<PathBuf> = Vec::new();
45    for env in layers.iter().rev() {
46        for p in &env.path {
47            let p = PathBuf::from(expand(p, &vars));
48            let p = if p.is_relative() { root.join(p) } else { p };
49            if !entries.contains(&p) {
50                entries.push(p);
51            }
52        }
53    }
54    let mut out: Vec<(String, String)> = vars.into_iter().collect();
55    if !entries.is_empty() {
56        let current = std::env::var_os("PATH").unwrap_or_default();
57        let rest = std::env::split_paths(&current).filter(|p| !entries.contains(p));
58        let joined: OsString = std::env::join_paths(entries.iter().cloned().chain(rest))
59            .context("PATH entry contains a separator")?;
60        out.push(("PATH".into(), joined.to_string_lossy().into_owned()));
61    }
62    Ok(out)
63}

65fn expand(s: &str, vars: &BTreeMap<String, String>) -> String {
66    shellexpand::full_with_context_no_errors(
67        s,
68        || dirs::home_dir().map(|h| h.display().to_string()),
69        |name: &str| vars.get(name).cloned().or_else(|| std::env::var(name).ok()),
70    )
71    .into_owned()
72}

74/// POSIX shell `export` lines (bash/zsh/sh; what direnv's `eval` expects).
75pub fn exports(vars: &[(String, String)]) -> Result<String> {
76    let mut out = String::new();
77    for (k, v) in vars {
78        let q = shlex::try_quote(v).with_context(|| format!("can't quote value of {k}"))?;
79        out.push_str(&format!("export {k}={q}\n"));
80    }
81    Ok(out)
82}

89#[cfg(test)]
90mod tests {
91    use super::*;
92
93    #[test]
94    fn layers_expand_and_path() {
95        let mut shared = Env::default();
96        shared
97            .vars
98            .insert("TOOLS".into(), "$PULL_ROOT/tools".into());
99        shared.vars.insert("GOTOOLCHAIN".into(), "local".into());
100        shared.path = vec!["bin".into(), "$TOOLS".into(), "~/x".into()];
101        let mut local = Env::default();
102        local.vars.insert("GOTOOLCHAIN".into(), "go1.27.4".into());
103        local.path = vec!["/opt/mine".into()];
104        let root = Path::new("/w");
105        let v: BTreeMap<_, _> = resolve(root, &[&shared, &local])
106            .unwrap()
107            .into_iter()
108            .collect();
109        assert_eq!(v["PULL_ROOT"], "/w");
110        assert_eq!(v["TOOLS"], "/w/tools");
111        assert_eq!(v["GOTOOLCHAIN"], "go1.27.4");
112        let home = dirs::home_dir().unwrap();
113        let path = &v["PATH"];
114        let want = format!("/opt/mine:/w/bin:/w/tools:{}/x:", home.display());
115        assert!(path.starts_with(&want), "{path}");
116        // round-trip through a real shell
117        let tricky = "it's $HOME \"q\" `x` \\ end";
118        let script = exports(&[("A".into(), tricky.into())]).unwrap() + "printf %s \"$A\"";
119        let out = std::process::Command::new("sh")
120            .arg("-c")
121            .arg(script)
122            .output()
123            .unwrap();
124        assert_eq!(String::from_utf8(out.stdout).unwrap(), tricky);
125    }
