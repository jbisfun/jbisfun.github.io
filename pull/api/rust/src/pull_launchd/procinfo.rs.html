1//! What is running in a job's terminal, from /proc (Linux).
2
3use std::path::Path;
4
5/// (foreground program, its working directory) for the terminal session led by `pid`:
6/// the terminal's foreground process group (`tpgid` in /proc/PID/stat), like tmux's
7/// pane_current_command / pane_current_path.
8pub fn foreground(pid: u32) -> (Option<String>, Option<String>) {
9    let fg = tpgid(pid).filter(|p| *p > 0).unwrap_or(pid as i64);
10    let base = format!("/proc/{fg}");
11    let comm = std::fs::read_to_string(format!("{base}/comm"))
12        .ok()
13        .map(|s| s.trim().to_string());
14    let cwd = std::fs::read_link(format!("{base}/cwd"))
15        .ok()
16        .map(|p| p.display().to_string());
17    (comm, cwd)
18}
19
20fn tpgid(pid: u32) -> Option<i64> {
21    let stat =
22        std::fs::read_to_string(Path::new("/proc").join(pid.to_string()).join("stat")).ok()?;
23    // Fields after the parenthesised comm: state ppid pgrp session tty_nr tpgid ...
24    let rest = &stat[stat.rfind(')')? + 2..];
25    rest.split_whitespace().nth(5)?.parse().ok()
26}
