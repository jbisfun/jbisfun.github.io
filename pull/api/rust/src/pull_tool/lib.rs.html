1//! Workspace packages: the pullfile `tool` section is a Brewfile-shaped package list
2//! (`tap`, `brew`) that jbisfun/tool pours from Linuxbrew bottles into its shared store and
3//! links into the workspace profile `.pull/tool/`. The env adds the profile's `bin`/`sbin`
4//! to PATH.
5
6use anyhow::Result;
7use pull_core::Pullfile;
8use std::path::{Path, PathBuf};
9pub use tool::{Manifest, Packages, Store};
10
11/// Pullfile section key.
12pub const SECTION: &str = "tool";
13
14/// The workspace profile, relative to the root.
15pub const PROFILE: &str = ".pull/tool";
16
17pub fn profile(root: &Path) -> PathBuf {
18    root.join(PROFILE)
19}
20
21/// The package list from the pullfile.
22pub fn packages(pf: &Pullfile) -> Result<Packages> {
23    pf.section(SECTION)
24}
25
26/// PATH entries for the workspace env (none unless the pullfile has a `tool` section).
27pub fn path_entries(root: &Path, pf: &Pullfile) -> Vec<PathBuf> {
28    if pf.has_section(SECTION) {
29        tool::path_entries(&profile(root))
30    } else {
31        Vec::new()
32    }
33}
34
35/// Install the pullfile's packages and swap the workspace profile to exactly that set.
36pub fn sync(root: &Path, pf: &Pullfile) -> Result<Manifest> {
37    tool::sync(&Store::open()?, &packages(pf)?, &profile(root))
38}
39
40/// Edit the package list in the pullfile (saved only if it changed). Returns whether it did.
41pub fn edit(root: &Path, pf: &mut Pullfile, f: impl FnOnce(&mut Packages) -> bool) -> Result<bool> {
42    let mut pkgs = packages(pf)?;
43    if !f(&mut pkgs) && pf.has_section(SECTION) {
44        return Ok(false);
45    }
46    pf.set_section(SECTION, &pkgs)?;
47    pf.save(root)?;
48    Ok(true)
49}
50
51/// The workspace profile's manifest, if it has been synced.
52pub fn manifest(root: &Path) -> Result<Option<Manifest>> {
53    Manifest::load(&profile(root))
54}
55
56/// Delete store kegs no live profile (in any workspace) uses.
57pub fn gc() -> Result<Vec<String>> {
58    tool::gc(&Store::open()?)
59}
60
61/// Read a Brewfile into a package list.
62pub fn read_brewfile(path: &Path) -> Result<Packages> {
63    Packages::from_brewfile(path)
64}
