15pub fn open(root: &Path, name: &str) -> Result<Repository> {
16    Repository::open(root.join(name))
17        .with_context(|| format!("{name}: not cloned (run `pull sync`)"))
18}

20/// The pullfile root itself, if it is the top of a git repo.
21pub fn root_repo(root: &Path) -> Option<Repository> {
22    let r = Repository::open(root).ok()?;
23    (r.workdir()?.canonicalize().ok()? == root).then_some(r)
24}

26/// Keep a repo dir out of the root repo's `git status`, if the root is a git repo.
27/// Uses `.git/info/exclude`, not `.gitignore`: editors (Zed/ycode) honour `.gitignore`
28/// outside of git too and would stop scanning the repo, hiding it from the git panel.
29pub fn exclude_from_root_repo(root: &Path, name: &str) -> Result<()> {
30    let Some(repo) = root_repo(root) else {
31        return Ok(());
32    };
33    let info = repo.path().join("info");
34    std::fs::create_dir_all(&info)?;
35    let p = info.join("exclude");
36    let line = format!("/{name}/");
37    let cur = std::fs::read_to_string(&p).unwrap_or_default();
38    if cur.lines().any(|l| l.trim() == line) {
39        return Ok(());
40    }
41    let sep = if cur.is_empty() || cur.ends_with('\n') {
42        ""
43    } else {
44        "\n"
45    };
46    std::fs::write(&p, format!("{cur}{sep}{line}\n"))?;
47    Ok(())
48}

50/// Clone missing repos, fetch, and fast-forward to the pins (or, `to_upstream`, to the
51/// upstream branches and re-pin). Saves the pullfile if pins changed.
52pub fn sync(root: &Path, pf: &mut Pullfile, to_upstream: bool) -> Result<()> {
53    let mut failed = Vec::new();
54    let mut changed = false;
55    for r in pf.repos.iter_mut() {
56        let dir = root.join(&r.name);
57        let res: Result<()> = (|| {
58            let fresh = !dir.join(".git").exists();
59            let repo = if fresh {
60                println!("{}: cloning {}", r.name, r.upstream);
61                git::clone(&r.upstream, &dir, &r.branch)?
62            } else {
63                let repo = Repository::open(&dir)?;
64                git::fetch(&repo)?;
65                repo
66            };
67            // A fresh clone lands on upstream HEAD; put it on the pin instead.
68            if fresh
69                && !to_upstream
70                && let Some(pin) = r.commit.as_deref()
71            {
72                git::reset_hard(&repo, git2::Oid::from_str(pin)?)?;
73            }
74            let target = if to_upstream {
75                git::remote_oid(&repo, &r.branch)
76            } else {
77                r.commit.as_deref().map(git2::Oid::from_str).transpose()?
78            };
79            if let Some(t) = target
80                && git::fast_forward(&repo, t)?
81            {
82                println!("{}: -> {}", r.name, git::short(t));
83            }
84            if to_upstream {
85                let head = git::head_oid(&repo).map(|o| o.to_string());
86                if head.is_some() && head != r.commit {
87                    r.commit = head;
88                    changed = true;
89                }
90            }
91            Ok(())
92        })();
93        if let Err(e) = res {
94            eprintln!("{}: {e:#}", r.name);
95            failed.push(r.name.clone());
96        }
97    }
98    if changed {
99        pf.save(root)?;
100    }
101    if !failed.is_empty() {
102        bail!("failed: {}", failed.join(", "));
103    }
104    Ok(())
105}

107fn branch_name(repo: &Repository) -> Option<String> {
108    repo.head()
109        .ok()
110        .and_then(|h| h.shorthand().ok().map(String::from))
111}

113/// Human-readable status of every repo.
114pub fn print_status(root: &Path, pf: &Pullfile) -> Result<()> {
115    for r in &pf.repos {
116        let Ok(repo) = Repository::open(root.join(&r.name)) else {
117            println!("{}: not cloned", r.name);
118            continue;
119        };
120        let branch = branch_name(&repo).unwrap_or("(unborn)".into());
121        let head = git::head_oid(&repo);
122        let mut flags = Vec::new();
123        match (head, r.commit.as_deref()) {
124            (Some(h), Some(p)) if h.to_string() != p => {
125                flags.push(format!("pin {}", &p[..8.min(p.len())]))
126            }
127            (Some(_), None) => flags.push("unpinned".into()),
128            _ => {}
129        }
130        if let (Some(h), Some(o)) = (head, git::remote_oid(&repo, &r.branch))
131            && h != o
132        {
133            let (a, b) = repo.graph_ahead_behind(h, o)?;
134            flags.push(format!("+{a}/-{b} origin"));
135        }
136        let h = head.map(git::short).unwrap_or("-".into());
137        println!("{} [{branch} {h}] {}", r.name, flags.join(", "));
138        for l in git::status_lines(&repo)? {
139            println!("  {l}");
140        }
141    }
142    Ok(())
143}

146pub struct RepoStatus {
147    pub name: String,
148    pub path: String,
149    pub cloned: bool,
150    pub branch: Option<String>,
151    pub head: Option<String>,
152    pub pinned: Option<String>,
153    pub ahead: usize,
154    pub behind: usize,
155    pub changes: Vec<Change>,
156}

158/// Machine-readable status of every repo (`pull status --json`).
159pub fn repo_statuses(root: &Path, pf: &Pullfile) -> Result<Vec<RepoStatus>> {
160    let mut out = Vec::new();
161    for r in &pf.repos {
162        let mut st = RepoStatus {
163            name: r.name.clone(),
164            path: root.join(&r.name).display().to_string(),
165            cloned: false,
166            branch: None,
167            head: None,
168            pinned: r.commit.clone(),
169            ahead: 0,
170            behind: 0,
171            changes: Vec::new(),
172        };
173        if let Ok(repo) = Repository::open(root.join(&r.name)) {
174            st.cloned = true;
175            st.branch = branch_name(&repo);
176            let head = git::head_oid(&repo);
177            st.head = head.map(|h| h.to_string());
178            if let (Some(h), Some(o)) = (head, git::remote_oid(&repo, &r.branch))
179                && h != o
180            {
181                (st.ahead, st.behind) = repo.graph_ahead_behind(h, o)?;
182            }
183            st.changes = git::changes(&repo)?;
184        }
185        out.push(st);
186    }
187    Ok(out)
188}

190/// Stage routed pathspecs (and, `all`, every tracked modification) in each repo.
191pub fn stage(
192    root: &Path,
193    pf: &Pullfile,
194    routed: &BTreeMap<String, Vec<String>>,
195    all: bool,
196) -> Result<()> {
197    for r in &pf.repos {
198        let Ok(repo) = Repository::open(root.join(&r.name)) else {
199            continue;
200        };
201        if all {
202            git::stage_tracked(&repo)?;
203        }
204        if let Some(specs) = routed.get(&r.name) {
205            git::stage(&repo, specs)?;
206        }
207    }
208    Ok(())
209}

211/// Repos with something staged.
212pub fn staged_repos(root: &Path, pf: &Pullfile) -> Result<Vec<String>> {
213    let mut out = Vec::new();
214    for r in &pf.repos {
215        if let Ok(repo) = Repository::open(root.join(&r.name))
216            && git::has_staged(&repo)?
217        {
218            out.push(r.name.clone());
219        }
220    }
221    Ok(out)
222}

224/// Commit `targets` with one message and identity, pin the new commits, and (if the root
225/// is a git repo) commit the updated pullfile there too.
226pub fn commit(
227    root: &Path,
228    pf: &mut Pullfile,
229    targets: &[String],
230    id: &Identity,
231    msg: &str,
232) -> Result<()> {
233    for name in targets {
234        let repo = open(root, name)?;
235        let oid = git::commit(&repo, id, msg).with_context(|| name.clone())?;
236        println!("{name}: {}", git::short(oid));
237        if let Some(r) = pf.repo_mut(name) {
238            r.commit = Some(oid.to_string());
239        }
240    }
241    pf.save(root)?;
242    if let Some(repo) = root_repo(root) {
243        git::stage(&repo, &[PULLFILE.into(), ".gitignore".into()])?;
244        if git::has_staged(&repo)? {
245            let oid = git::commit(&repo, id, msg).context("(root)")?;
246            println!("(root): {}", git::short(oid));
247        }
248    }
249    Ok(())
250}

252/// Push every repo whose branch is ahead of origin, then the root repo (if any).
253pub fn push(root: &Path, pf: &Pullfile) -> Result<()> {
254    for r in &pf.repos {
255        let repo = open(root, &r.name)?;
256        let head = git::head_oid(&repo);
257        if head.is_none() || head == git::remote_oid(&repo, &r.branch) {
258            continue;
259        }
260        println!("{}: pushing {}", r.name, r.branch);
261        git::push(&repo, &r.branch).with_context(|| r.name.clone())?;
262    }
263    // The root repo (if any) carries the pullfile itself.
264    if let Some(repo) = root_repo(root)
265        && repo.find_remote("origin").is_ok()
266        && let Some(b) = branch_name(&repo)
267    {
268        println!("(root): pushing {b}");
269        git::push(&repo, &b).context("(root)")?;
270    }
271    Ok(())
