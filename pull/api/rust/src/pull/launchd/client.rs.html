13/// Connect to the workspace daemon, starting it (detached, in its own session) if needed.
14pub fn connect(root: &Path) -> Result<UnixStream> {
15    let sock = socket_path(root);
16    if let Ok(s) = UnixStream::connect(&sock) {
17        return Ok(s);
18    }
19    std::fs::create_dir_all(state_dir(root))?;
20    let log = std::fs::OpenOptions::new()
21        .create(true)
22        .append(true)
23        .open(state_dir(root).join("launchd.log"))?;
24    let mut cmd = Command::new(std::env::current_exe()?);
25    cmd.arg("launchd")
26        .current_dir(root)
27        .stdin(Stdio::null())
28        .stdout(log.try_clone()?)
29        .stderr(log);
30    // SAFETY: setsid is async-signal-safe; detaches the daemon from our terminal.
31    unsafe {
32        cmd.pre_exec(|| {
33            nix::unistd::setsid().map_err(std::io::Error::from)?;
34            Ok(())
35        });
36    }
37    cmd.spawn().context("starting pull launchd")?;
38    let deadline = Instant::now() + Duration::from_secs(5);
39    while Instant::now() < deadline {
40        if let Ok(s) = UnixStream::connect(&sock) {
41            return Ok(s);
42        }
43        std::thread::sleep(Duration::from_millis(50));
44    }
45    bail!(
46        "launchd did not start; see {}",
47        state_dir(root).join("launchd.log").display()
48    )
49}

51pub fn request(root: &Path, req: &Request) -> Result<Response> {
52    let mut s = connect(root)?;
53    writeln!(s, "{}", serde_json::to_string(req)?)?;
54    let mut line = String::new();
55    BufReader::new(s).read_line(&mut line)?;
56    let resp: Response = serde_json::from_str(&line).context("bad reply from launchd")?;
57    if let Response::Error(e) = resp {
58        bail!("{e}");
59    }
60    Ok(resp)
61}

63/// Is a daemon running for this workspace (without starting one)?
64pub fn running(root: &Path) -> bool {
65    UnixStream::connect(socket_path(root)).is_ok()
66}

68fn frame(kind: u8, payload: &[u8]) -> Vec<u8> {
69    let mut v = Vec::with_capacity(3 + payload.len());
70    v.push(kind);
71    v.extend_from_slice(&(payload.len() as u16).to_be_bytes());
72    v.extend_from_slice(payload);
73    v
74}

76/// Attach the current terminal to a job. Detach with ctrl-\.
77pub fn attach(root: &Path, label: &str) -> Result<()> {
78    use crossterm::terminal;
79    let (cols, rows) = terminal::size().unwrap_or((120, 40));
80    let mut s = connect(root)?;
81    writeln!(
82        s,
83        "{}",
84        serde_json::to_string(&Request::Attach {
85            label: label.into(),
86            rows,
87            cols
88        })?
89    )?;
90    // Read exactly one reply line (unbuffered: raw output follows it).
91    let mut line = Vec::new();
92    let mut byte = [0u8; 1];
93    while s.read(&mut byte)? == 1 && byte[0] != b'\n' {
94        line.push(byte[0]);
95    }
96    if let Response::Error(e) = serde_json::from_slice(&line)? {
97        bail!("{e}");
98    }
99
100    terminal::enable_raw_mode()?;
101    eprint!("\x1b]0;pull: {label}\x07");
102    let mut from_daemon = s.try_clone()?;
103    let label_out = label.to_string();
104    let output = std::thread::spawn(move || {
105        let mut stdout = std::io::stdout();
106        let mut buf = [0u8; 8192];
107        while let Ok(n) = from_daemon.read(&mut buf) {
108            if n == 0 || stdout.write_all(&buf[..n]).is_err() {
109                break;
110            }
111            stdout.flush().ok();
112        }
113        // The session ended or was removed: the main thread is blocked reading stdin, so
114        // restore the terminal and exit from here.
115        terminal::disable_raw_mode().ok();
116        eprintln!("\r\n[session {label_out} ended]");
117        std::process::exit(0);
118    });
119    let mut to_daemon = s.try_clone()?;
120    std::thread::spawn(move || {
121        let mut last = (cols, rows);
122        loop {
123            std::thread::sleep(Duration::from_millis(250));
124            let Ok(size) = terminal::size() else { break };
125            if size != last {
126                last = size;
127                let mut p = size.1.to_be_bytes().to_vec();
128                p.extend_from_slice(&size.0.to_be_bytes());
129                if to_daemon.write_all(&frame(FRAME_RESIZE, &p)).is_err() {
130                    break;
131                }
132            }
133        }
134    });
135    let mut stdin = std::io::stdin();
136    let mut buf = [0u8; 4096];
137    loop {
138        let n = match stdin.read(&mut buf) {
139            Ok(0) | Err(_) => break,
140            Ok(n) => n,
141        };
142        // ctrl-\ detaches.
143        if let Some(i) = buf[..n].iter().position(|b| *b == 0x1c) {
144            if i > 0 {
145                s.write_all(&frame(FRAME_DATA, &buf[..i])).ok();
146            }
147            break;
148        }
149        if s.write_all(&frame(FRAME_DATA, &buf[..n])).is_err() {
150            break;
151        }
152    }
153    drop(output);
154    s.shutdown(std::net::Shutdown::Both).ok();
155    terminal::disable_raw_mode()?;
156    eprintln!("\r\n[detached from {label}]");
157    Ok(())
158}
