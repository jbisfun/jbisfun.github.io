1//! Workspace worktrees: `pull worktree add NAME` makes `{root}/.worktrees/NAME/`, a full
2//! pull workspace whose repos are git worktrees of the main ones on branch NAME.
3//! Run any pull command inside it (it has its own pullfile.plist with the new branch).
4
5use crate::config::{LOCAL, PULLFILE, Pullfile};
6use crate::git;
7use anyhow::{Context, Result, bail};
8use git2::{BranchType, Repository, WorktreeAddOptions, WorktreePruneOptions};
9use std::path::{Path, PathBuf};
10
11pub const DIR: &str = ".worktrees";
12
13pub fn path(root: &Path, name: &str) -> PathBuf {
14    root.join(DIR).join(name)
15}
16
17fn check_name(name: &str) -> Result<()> {
18    if name.is_empty() || name.contains(['/', '\\']) || name.starts_with('.') {
19        bail!("invalid worktree name `{name}`");
20    }
21    Ok(())
22}
23
24pub fn add(root: &Path, pf: &Pullfile, name: &str, branch: Option<&str>) -> Result<PathBuf> {
25    check_name(name)?;
26    let branch = branch.unwrap_or(name);
27    let dest = path(root, name);
28    if dest.exists() {
29        bail!("{} already exists", dest.display());
30    }
31    std::fs::create_dir_all(&dest)?;
32
33    let mut wpf = pf.clone();
34    for r in wpf.repos.iter_mut() {
35        let repo = Repository::open(root.join(&r.name))
36            .with_context(|| format!("{}: not cloned (run `pull sync`)", r.name))?;
37        // Branch from the main checkout's HEAD unless it already exists.
38        let reference = match repo.find_branch(branch, BranchType::Local) {
39            Ok(b) => b.into_reference(),
40            Err(_) => {
41                let head = repo
42                    .head()
43                    .and_then(|h| h.peel_to_commit())
44                    .with_context(|| format!("{}: no commits to branch from", r.name))?;
45                repo.branch(branch, &head, false)?.into_reference()
46            }
47        };
48        let mut opts = WorktreeAddOptions::new();
49        opts.reference(Some(&reference));
50        repo.worktree(name, &dest.join(&r.name), Some(&opts))
51            .with_context(|| format!("{}: adding worktree", r.name))?;
52        println!("{}: {} on {branch}", r.name, dest.join(&r.name).display());
53        r.branch = branch.to_string();
54    }
55    wpf.save(&dest)?;
56    for f in [LOCAL, ".gitignore", ".envrc"] {
57        if root.join(f).exists() {
58            std::fs::copy(root.join(f), dest.join(f))?;
59        }
60    }
61    Ok(dest)
62}
63
64pub fn list(root: &Path) -> Result<Vec<(String, PathBuf)>> {
65    let dir = root.join(DIR);
66    let mut out = Vec::new();
67    for e in std::fs::read_dir(&dir).into_iter().flatten().flatten() {
68        if e.path().join(PULLFILE).is_file() {
69            out.push((e.file_name().to_string_lossy().into_owned(), e.path()));
70        }
71    }
72    out.sort();
73    Ok(out)
74}
75
76pub fn remove(root: &Path, pf: &Pullfile, name: &str, force: bool) -> Result<()> {
77    check_name(name)?;
78    let dest = path(root, name);
79    // Refuse before touching anything if any worktree has uncommitted work.
80    if !force {
81        for r in &pf.repos {
82            if let Ok(wt) = Repository::open(dest.join(&r.name))
83                && git::is_dirty(&wt)?
84            {
85                bail!("{}: uncommitted changes in worktree (use --force)", r.name);
86            }
87        }
88    }
89    for r in &pf.repos {
90        let Ok(repo) = Repository::open(root.join(&r.name)) else {
91            continue;
92        };
93        let Ok(wt) = repo.find_worktree(name) else {
94            continue;
95        };
96        let wt_path = wt.path().to_path_buf();
97        if wt_path.exists() {
98            std::fs::remove_dir_all(&wt_path)?;
99        }
100        wt.prune(Some(
101            WorktreePruneOptions::new()
102                .valid(true)
103                .locked(force)
104                .working_tree(true),
105        ))
106        .with_context(|| format!("{}: pruning worktree", r.name))?;
107        println!("{}: removed worktree {name} (branch kept)", r.name);
108    }
109    if dest.exists() {
110        std::fs::remove_dir_all(&dest)?;
111    }
112    Ok(())
113}
