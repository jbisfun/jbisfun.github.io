3/// Returns a list of violations (empty = ok). `msg` should already be prettified.
4pub fn check(rules: &Lint, msg: &str) -> Vec<String> {
5    let mut out = Vec::new();
6    let subject = msg.lines().next().unwrap_or("").trim_end();
7
8    if subject.trim().is_empty() {
9        out.push("empty commit message".into());
10        return out;
11    }
12    if rules.no_description && msg.lines().skip(1).any(|l| !l.trim().is_empty()) {
13        out.push("no_description: message must be a single subject line".into());
14    }
15    if let Some(max) = rules.max_subject_length {
16        let n = subject.chars().count() as u64;
17        if n > max {
18            out.push(format!(
19                "max_subject_length: subject is {n} chars (max {max})"
20            ));
21        }
22    }
23    if rules.no_trailing_period && subject.ends_with('.') {
24        out.push("no_trailing_period: subject ends with '.'".into());
25    }
26    if rules.conventional
27        && let Err(e) = git_conventional::Commit::parse(msg)
28    {
29        out.push(format!("conventional: {e}"));
30    }
31    out
32}
