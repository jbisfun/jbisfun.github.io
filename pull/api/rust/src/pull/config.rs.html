148impl Local {
149    /// `{root}/local.toml`, falling back to `$XDG_CONFIG_HOME/pull/local.toml`.
150    pub fn path(root: &Path) -> Option<PathBuf> {
151        let here = root.join(LOCAL);
152        if here.exists() {
153            return Some(here);
154        }
155        let global = dirs::config_dir()?.join("pull").join(LOCAL);
156        global.exists().then_some(global)
157    }
158
159    pub fn load(root: &Path) -> Result<Self> {
160        let Some(p) = Self::path(root) else {
161            bail!(
162                "no {LOCAL} found in {} or ~/.config/pull (run `pull init`)",
163                root.display()
164            );
165        };
166        let s = std::fs::read_to_string(&p).with_context(|| format!("reading {}", p.display()))?;
167        toml::from_str(&s).with_context(|| format!("parsing {}", p.display()))
168    }
169}

173pub struct Identity {
174    pub name: String,
175    pub email: String,
176    pub sign: bool,
177    pub signing_key: Option<String>,
178    pub gpg_program: String,

181pub fn resolve_identity(pf: &Pullfile, local: &Local) -> Result<Identity> {
182    let org = match &pf.identity.org {
183        Some(o) => Some(local.orgs.get(o).with_context(|| {
184            format!("pullfile wants org `{o}` but local.toml has no [orgs.{o}]")
185        })?),
186        None => None,
187    };
188    let name = org
189        .and_then(|o| o.name.clone())
190        .unwrap_or_else(|| local.name.clone());
191    if name.trim().is_empty() {
192        bail!("local.toml: `name` is empty");
193    }
194    let email = match pf.identity.email {
195        EmailPolicy::None => String::new(),
196        EmailPolicy::Local => org
197            .and_then(|o| o.email.clone())
198            .or_else(|| local.email.clone())
199            .unwrap_or_default(),
200    };
201    let sign = org.and_then(|o| o.sign).unwrap_or(local.sign) || pf.identity.require_signing;
202    let signing_key = org
203        .and_then(|o| o.signing_key.clone())
204        .or_else(|| local.signing_key.clone());
205    Ok(Identity {
206        name,
207        email,
208        sign,
209        signing_key,
210        gpg_program: local.gpg_program.clone().unwrap_or_else(|| "gpg".into()),
211    })
212}

214/// Walk up from `start` to the directory containing pullfile.plist.
215pub fn find_root(start: &Path) -> Result<PathBuf> {
216    let mut d = start.canonicalize()?;
217    loop {
218        if d.join(PULLFILE).is_file() {
219            return Ok(d);
220        }
221        if !d.pop() {
222            bail!(
223                "no {PULLFILE} found in {} or any parent (run `pull init`)",
224                start.display()
225            );
226        }
227    }
