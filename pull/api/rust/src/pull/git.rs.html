17pub fn clone(url: &str, dest: &Path, branch: &str) -> Result<Repository> {
18    let repo = auth()
19        .clone_repo(url, dest)
20        .with_context(|| format!("cloning {url}"))?;
21    checkout_branch(&repo, branch)?;
22    Ok(repo)
23}

25/// Make `branch` the checked-out branch, creating it from origin/<branch> if needed.
26pub fn checkout_branch(repo: &Repository, branch: &str) -> Result<()> {
27    if repo
28        .head()
29        .ok()
30        .and_then(|h| h.shorthand().ok().map(String::from))
31        .as_deref()
32        == Some(branch)
33    {
34        return Ok(());
35    }
36    if repo.find_branch(branch, BranchType::Local).is_err() {
37        let Ok(remote) = repo.find_branch(&format!("origin/{branch}"), BranchType::Remote) else {
38            // Empty / new upstream: point HEAD at the unborn branch.
39            repo.set_head(&format!("refs/heads/{branch}"))?;
40            return Ok(());
41        };
42        let c = remote.get().peel_to_commit()?;
43        let mut b = repo.branch(branch, &c, false)?;
44        b.set_upstream(Some(&format!("origin/{branch}")))?;
45    }
46    let refname = format!("refs/heads/{branch}");
47    let obj = repo.revparse_single(&refname)?;
48    repo.checkout_tree(&obj, None)?;
49    repo.set_head(&refname)?;
50    Ok(())
51}

53pub fn fetch(repo: &Repository) -> Result<()> {
54    let mut remote = repo.find_remote("origin")?;
55    auth()
56        .fetch(
57            repo,
58            &mut remote,
59            &["+refs/heads/*:refs/remotes/origin/*"],
60            None,
61        )
62        .context("fetch origin")?;
63    Ok(())
64}
65
66pub fn push(repo: &Repository, branch: &str) -> Result<()> {
67    let mut remote = repo.find_remote("origin")?;
68    let spec = format!("refs/heads/{branch}:refs/heads/{branch}");
69    auth()
70        .push(repo, &mut remote, &[&spec])
71        .context("push origin")?;
72    // Mirror what `git push` does to the remote-tracking ref.
73    if let Some(oid) = head_oid(repo) {
74        repo.reference(&format!("refs/remotes/origin/{branch}"), oid, true, "push")?;
75    }
76    Ok(())
77}

88/// Tracked changes in index or worktree (untracked files ignored).
89pub fn is_dirty(repo: &Repository) -> Result<bool> {
90    let mut o = StatusOptions::new();
91    o.include_untracked(false).include_ignored(false);
92    Ok(!repo.statuses(Some(&mut o))?.is_empty())
93}

95/// One changed path: index (staged) and worktree status letters, as in `git status -s`
96/// (`A M D R` / `?` for untracked / space for unchanged).
97#[derive(serde::Serialize, Debug, Clone)]
98pub struct Change {
99    pub path: String,
100    pub staged: char,
101    pub unstaged: char,
102}
103
104pub fn changes(repo: &Repository) -> Result<Vec<Change>> {
105    let mut o = StatusOptions::new();
106    o.include_untracked(true).recurse_untracked_dirs(true);
107    let mut out = Vec::new();
108    for e in repo.statuses(Some(&mut o))?.iter() {
109        let s = e.status();
110        if s.is_ignored() {
111            continue;
112        }
113        let staged = if s.is_index_new() {
114            'A'
115        } else if s.is_index_modified() || s.is_index_typechange() {
116            'M'
117        } else if s.is_index_deleted() {
118            'D'
119        } else if s.is_index_renamed() {
120            'R'
121        } else {
122            ' '
123        };
124        let unstaged = if s.is_wt_new() {
125            '?'
126        } else if s.is_wt_modified() || s.is_wt_typechange() {
127            'M'
128        } else if s.is_wt_deleted() {
129            'D'
130        } else if s.is_wt_renamed() {
131            'R'
132        } else {
133            ' '
134        };
135        let staged = if unstaged == '?' { '?' } else { staged };
136        out.push(Change {
137            path: e.path().unwrap_or("?").to_string(),
138            staged,
139            unstaged,
140        });
141    }
142    Ok(out)
143}

145pub fn status_lines(repo: &Repository) -> Result<Vec<String>> {
146    let mut o = StatusOptions::new();
147    o.include_untracked(true).recurse_untracked_dirs(false);
148    let mut out = Vec::new();
149    for e in repo.statuses(Some(&mut o))?.iter() {
150        let s = e.status();
151        let idx = if s.is_index_new() {
152            'A'
153        } else if s.is_index_modified() {
154            'M'
155        } else if s.is_index_deleted() {
156            'D'
157        } else if s.is_index_renamed() {
158            'R'
159        } else {
160            ' '
161        };
162        let wt = if s.is_wt_new() {
163            '?'
164        } else if s.is_wt_modified() {
165            'M'
166        } else if s.is_wt_deleted() {
167            'D'
168        } else {
169            ' '
170        };
171        let idx = if wt == '?' { '?' } else { idx };
172        out.push(format!("{idx}{wt} {}", e.path().unwrap_or("?")));
173    }
174    Ok(out)
175}

177/// Move the current branch to `target` if that is a fast-forward and the tree is clean.
178/// Returns true if HEAD moved.
179pub fn fast_forward(repo: &Repository, target: Oid) -> Result<bool> {
180    let Some(head) = head_oid(repo) else {
181        let obj = repo.find_object(target, Some(ObjectType::Commit))?;
182        repo.reset(&obj, ResetType::Hard, None)?;
183        return Ok(true);
184    };
185    if head == target || repo.graph_descendant_of(head, target)? {
186        return Ok(false);
187    }
188    if !repo.graph_descendant_of(target, head)? {
189        bail!(
190            "diverged from {} (HEAD {}); resolve manually",
191            short(target),
192            short(head)
193        );
194    }
195    if is_dirty(repo)? {
196        bail!(
197            "has uncommitted changes; not fast-forwarding to {}",
198            short(target)
199        );
200    }
201    let obj = repo.find_object(target, Some(ObjectType::Commit))?;
202    repo.reset(&obj, ResetType::Hard, None)?;
203    Ok(true)
204}

206pub fn reset_hard(repo: &Repository, target: Oid) -> Result<()> {
207    let obj = repo.find_object(target, Some(ObjectType::Commit))?;
208    repo.reset(&obj, ResetType::Hard, None)?;
209    Ok(())

212/// Stage paths (relative to the repo) like `git add`, including deletions.
213pub fn stage(repo: &Repository, paths: &[String]) -> Result<()> {
214    let mut idx = repo.index()?;
215    idx.add_all(paths, IndexAddOption::DEFAULT, None)?;
216    idx.update_all(paths, None)?;
217    idx.write()?;
218    Ok(())
219}
220
221/// Remove paths from the index back to HEAD (`git reset -- paths`).
222pub fn unstage(repo: &Repository, paths: &[String]) -> Result<()> {
223    match repo.head().ok().and_then(|h| h.peel_to_commit().ok()) {
224        Some(head) => repo.reset_default(Some(head.as_object()), paths)?,
225        None => {
226            let mut idx = repo.index()?;
227            idx.remove_all(paths, None)?;
228            idx.write()?;
229        }
230    }
231    Ok(())
232}

234/// `git commit -a` staging: tracked files only.
235pub fn stage_tracked(repo: &Repository) -> Result<()> {
236    let mut idx = repo.index()?;
237    idx.update_all(["*"], None)?;
238    idx.write()?;
239    Ok(())

242pub fn has_staged(repo: &Repository) -> Result<bool> {
243    let head_tree = match repo.head() {
244        Ok(h) => Some(h.peel_to_tree()?),
245        Err(_) => None,
246    };
247    let idx = repo.index()?;
248    let diff = repo.diff_tree_to_index(head_tree.as_ref(), Some(&idx), None)?;
249    Ok(diff.deltas().len() > 0)
250}

252/// Commit the index as `id`, optionally GPG-signed. Updates HEAD's branch.
253pub fn commit(repo: &Repository, id: &Identity, msg: &str) -> Result<Oid> {
254    let mut idx = repo.index()?;
255    let tree = repo.find_tree(idx.write_tree()?)?;
256    // libgit2 rejects empty emails, but git (and the object format) allow `Name <>`.
257    // Use a placeholder and blank it in the header before hashing/signing.
258    const NO_EMAIL: &str = "no-email.pull.invalid";
259    let email = if id.email.is_empty() {
260        NO_EMAIL
261    } else {
262        id.email.as_str()
263    };
264    let sig = Signature::now(&id.name, email)?;
265    let parent = match repo.head() {
266        Ok(h) => Some(h.peel_to_commit()?),
267        Err(_) => None,
268    };
269    let parents: Vec<_> = parent.iter().collect();
270    let buf = repo.commit_create_buffer(&sig, &sig, msg, &tree, &parents)?;
271    let raw = buf.as_str().context("commit buffer not utf-8")?;
272    let content = if id.email.is_empty() {
273        let (header, body) = raw.split_once("\n\n").unwrap_or((raw, ""));
274        let header: Vec<String> = header
275            .lines()
276            .map(|l| {
277                if l.starts_with("author ") || l.starts_with("committer ") {
278                    l.replace(&format!("<{NO_EMAIL}>"), "<>")
279                } else {
280                    l.to_string()
281                }
282            })
283            .collect();
284        format!("{}\n\n{body}", header.join("\n"))
285    } else {
286        raw.to_string()
287    };
288    let content = content.as_str();
289
290    let oid = if id.sign {
291        let signature = gpg_sign(id, content)?;
292        repo.commit_signed(content, &signature, None)?
293    } else {
294        repo.odb()?.write(ObjectType::Commit, content.as_bytes())?
295    };
296
297    let subject = msg.lines().next().unwrap_or("");
298    let reflog = format!("commit: {subject}");
299    let head = repo.find_reference("HEAD")?;
300    match head.symbolic_target()? {
301        Some(branch_ref) => {
302            repo.reference(branch_ref, oid, true, &reflog)?;
303        }
304        None => repo.set_head_detached(oid)?,
305    }
306    Ok(oid)
307}

309/// Same protocol git uses for gpg.program: detached, armored signature over the commit buffer.
310fn gpg_sign(id: &Identity, content: &str) -> Result<String> {
311    let mut c = Command::new(&id.gpg_program);
312    c.args(["--status-fd=2", "-bsa"]);
313    if let Some(k) = &id.signing_key {
314        c.args(["-u", k]);
315    }
316    let mut child = c
317        .stdin(Stdio::piped())
318        .stdout(Stdio::piped())
319        .stderr(Stdio::piped())
320        .spawn()
321        .with_context(|| format!("running {}", id.gpg_program))?;
322    child.stdin.take().unwrap().write_all(content.as_bytes())?;
323    let out = child.wait_with_output()?;
324    let status = String::from_utf8_lossy(&out.stderr);
325    if !out.status.success() || !status.contains("[GNUPG:] SIG_CREATED ") {
326        bail!("gpg failed to sign:\n{status}");
327    }
328    Ok(String::from_utf8(out.stdout)?)
329}
