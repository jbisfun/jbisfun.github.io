1//! Glue between the core and the optional features: each helper does the feature's work
2//! when it is compiled in and a neutral fallback otherwise, so commands never need to
3//! know which features exist.
4
5use anyhow::Result;
6use pull_core::Pullfile;
7use std::path::Path;
8
9/// Enabled cargo features (see build.rs).
10pub fn features() -> Vec<&'static str> {
11    env!("PULL_FEATURES").split_whitespace().collect()
12}
13
14/// Pullfile sections and the feature that acts on them.
15const SECTIONS: &[(&str, &str)] = &[
16    ("env", "env"),
17    ("hooks", "hooks"),
18    ("lint", "lint"),
19    ("ignore", "ignore"),
20    ("jobs", "launchd"),
21    ("tool", "tool"),
22];
23
24/// Say so when the pullfile configures something this build can't do (the section is
25/// still kept when the pullfile is rewritten).
26pub fn warn_unsupported(pf: &Pullfile) {
27    let have = features();
28    for (key, feature) in SECTIONS {
29        if pf.has_section(key) && !have.contains(feature) {
30            eprintln!("pull: ignoring pullfile `{key}` (built without the `{feature}` feature)");
31        }
32    }
33}

35/// The workspace environment: pullfile `env` + local.toml `[env]` with the `env` feature,
36/// otherwise just PULL_ROOT. With `tool`, the workspace profile's bin dirs are on PATH
37/// (after the `env` paths, so those can shadow packages).
38pub fn env(root: &Path, pf: &Pullfile) -> Result<Vec<(String, String)>> {
39    let tool_paths = tool_paths(root, pf);
40    #[cfg(feature = "env")]
41    {
42        // Lowest-priority layer: resolve puts later layers' paths first.
43        let tool = pull_env::Env {
44            path: tool_paths.iter().map(|p| p.display().to_string()).collect(),
45            ..Default::default()
46        };
47        let shared: pull_env::Env = pf.section(pull_env::SECTION)?;
48        let local: Option<pull_env::Env> = pull_core::Local::load(root)
49            .ok()
50            .map(|l| l.section(pull_env::SECTION))
51            .transpose()?;
52        let layers: Vec<&pull_env::Env> = [&tool, &shared].into_iter().chain(local.as_ref()).collect();
53        pull_env::resolve(root, &layers)
54    }
55    #[cfg(not(feature = "env"))]
56    {
57        let mut out = vec![("PULL_ROOT".to_string(), root.display().to_string())];
58        if !tool_paths.is_empty() {
59            let current = std::env::var_os("PATH").unwrap_or_default();
60            let joined = std::env::join_paths(
61                tool_paths.into_iter().chain(std::env::split_paths(&current)),
62            )?;
63            out.push(("PATH".into(), joined.to_string_lossy().into_owned()));
64        }
65        Ok(out)
66    }
67}

69/// PATH entries of the workspace package profile (none without the `tool` feature).
70fn tool_paths(root: &Path, pf: &Pullfile) -> Vec<std::path::PathBuf> {
71    #[cfg(feature = "tool")]
72    {
73        pull_tool::path_entries(root, pf)
74    }
75    #[cfg(not(feature = "tool"))]
76    {
77        let _ = (root, pf);
78        Vec::new()
79    }
80}

82/// [`env`] re-read from disk on every call, for long-running servers (daemon, MCP).
83#[allow(dead_code)]
84pub fn env_provider(root: &Path) -> impl Fn() -> Vec<(String, String)> + Send + Sync + 'static {
85    let root = root.to_path_buf();
86    move || {
87        Pullfile::load(&root)
88            .and_then(|pf| env(&root, &pf))
89            .unwrap_or_default()
90    }
91}

93/// Which hook list to run.
94#[derive(Clone, Copy)]
95pub enum Hook {
96    Precommit,
97    Postcommit,
98    Pull,
99}
100
101/// Run a pullfile hook (no-op without the `hooks` feature).
102pub fn run_hook(
103    hook: Hook,
104    root: &Path,
105    pf: &Pullfile,
106    repos: &[String],
107    extra: &[(&str, &str)],
108) -> Result<()> {
109    #[cfg(feature = "hooks")]
110    {
111        let hooks: pull_hooks::Hooks = pf.section(pull_hooks::SECTION)?;
112        let (kind, cmds) = match hook {
113            Hook::Precommit => ("precommit", &hooks.precommit),
114            Hook::Postcommit => ("postcommit", &hooks.postcommit),
115            Hook::Pull => ("pull", &hooks.pull),
116        };
117        if cmds.is_empty() {
118            return Ok(());
119        }
120        pull_hooks::run(kind, cmds, root, repos, &env(root, pf)?, extra)
121    }
122    #[cfg(not(feature = "hooks"))]
123    {
124        let _ = (hook, root, pf, repos, extra);
125        Ok(())
126    }
127}

129/// Lint violations for a commit message (none without the `lint` feature).
130pub fn lint(pf: &Pullfile, msg: &str) -> Result<Vec<String>> {
131    #[cfg(feature = "lint")]
132    {
133        let rules: pull_lint::Lint = pf.section(pull_lint::SECTION)?;
134        Ok(pull_lint::check(&rules, msg))
135    }
136    #[cfg(not(feature = "lint"))]
137    {
138        let _ = (pf, msg);
139        Ok(Vec::new())
140    }
141}

143/// Keep the pullfile `ignore` patterns in every repo's exclude file.
144pub fn apply_ignores(root: &Path, pf: &Pullfile) {
145    #[cfg(feature = "ignore")]
146    match pf.section::<Vec<String>>(pull_ignore::SECTION) {
147        Ok(patterns) => pull_ignore::apply(root, &pf.repo_names(), &patterns),
148        Err(e) => eprintln!("pull: {e:#}"),
149    }
150    #[cfg(not(feature = "ignore"))]
151    let _ = (root, pf);
152}
