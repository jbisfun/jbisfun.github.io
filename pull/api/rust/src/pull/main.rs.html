1//! `pull`: the CLI. The core (repos, sync, status, commit, push) lives in pull-core and
2//! pull-git; every other feature is an optional crate behind a cargo feature, wired in
3//! through `workspace` (hooks, lint, env, ignore) or its own subcommands.

18fn main() {
19    if let Err(e) = run(Cli::parse()) {
20        eprintln!("pull: {e:#}");
21        std::process::exit(1);
22    }
23}

25fn run(cli: Cli) -> Result<()> {
26    let cwd = std::env::current_dir()?;
27    if let Cmd::Init { org } = &cli.cmd {
28        return init(&cwd, org.clone());
29    }
30    #[cfg(feature = "mcp")]
31    if let Cmd::Mcp { root: Some(r) } = &cli.cmd {
32        return mcp(pull_core::find_root(r)?);
33    }
34    let root = pull_core::find_root(&cwd)?;
35    let mut pf = Pullfile::load(&root)?;
36    workspace::warn_unsupported(&pf);
37    workspace::apply_ignores(&root, &pf);
38
39    match cli.cmd {
40        Cmd::Init { .. } => unreachable!(),
41        Cmd::Repo(RepoCmd::Add {
42            upstream,
43            name,
44            branch,
45        }) => {
46            let name = match name {
47                Some(n) => n,
48                None => paths::repo_name_from_url(&upstream)?,
49            };
50            if pf.repos.iter().any(|r| r.name == name) {
51                bail!("repo `{name}` already in {PULLFILE}");
52            }
53            pf.repos.push(Repo {
54                name: name.clone(),
55                upstream,
56                branch,
57                commit: None,
58            });
59            pf.save(&root)?;
60            pull_git::exclude_from_root_repo(&root, &name)?;
61            println!("added {name}; run `pull sync` to clone");
62        }
63        Cmd::Repo(RepoCmd::Rm { name }) => {
64            let n = pf.repos.len();
65            pf.repos.retain(|r| r.name != name);
66            if pf.repos.len() == n {
67                bail!("no repo `{name}`");
68            }
69            pf.save(&root)?;
70        }
71        Cmd::Repo(RepoCmd::List) => {
72            for r in &pf.repos {
73                let pin = r
74                    .commit
75                    .as_deref()
76                    .map(|c| &c[..8.min(c.len())])
77                    .unwrap_or("-");
78                println!("{:<20} {:<10} {pin}  {}", r.name, r.branch, r.upstream);
79            }
80        }
81        Cmd::Sync => sync(&root, &mut pf, false)?,
82        Cmd::Update => sync(&root, &mut pf, true)?,
83        Cmd::Status { json: false } => pull_git::print_status(&root, &pf)?,
84        Cmd::Status { json: true } => status_json(&root, &pf)?,
85        Cmd::Unstage { paths } => {
86            for (name, rel) in paths::route(&root, &pf, &cwd, &paths)? {
87                git::unstage(&pull_git::open(&root, &name)?, &rel)?;
88            }
89        }
90        Cmd::Add { paths } => {
91            for (name, rel) in paths::route(&root, &pf, &cwd, &paths)? {
92                git::stage(&pull_git::open(&root, &name)?, &rel)?;
93            }
94        }
95        Cmd::Commit {
96            message,
97            all,
98            no_verify,
99            paths,
100        } => commit(&root, &mut pf, &cwd, &message, all, no_verify, &paths)?,
101        Cmd::Push => pull_git::push(&root, &pf)?,
102        #[cfg(feature = "lint")]
103        Cmd::Lint { message } => {
104            let msg = git2::message_prettify(&message, None)?;
105            let v = workspace::lint(&pf, &msg)?;
106            if v.is_empty() {
107                println!("ok");
108            } else {
109                for e in &v {
110                    eprintln!("lint: {e}");
111                }
112                bail!("{} lint violation(s)", v.len());
113            }
114        }
115        Cmd::Whoami => {
116            let id = pull_core::resolve_identity(&pf, &Local::load(&root)?)?;
117            println!("{} <{}>", id.name, id.email);
118            match (id.sign, &id.signing_key) {
119                (true, Some(k)) => println!("signing: gpg key {k}"),
120                (true, None) => println!("signing: gpg default key"),
121                (false, _) => println!("signing: off"),
122            }
123            if let Some(o) = &pf.identity.org {
124                println!("org: {o}");
125            }
126        }
127        #[cfg(feature = "mcp")]
128        Cmd::Mcp { .. } => mcp(root)?,
129        #[cfg(feature = "mcp")]
130        Cmd::McpServers => mcp_servers_json(&root)?,
131        #[cfg(feature = "launchd")]
132        Cmd::Launchd => launchd::daemon(&root, &pf)?,
133        #[cfg(feature = "launchd")]
134        Cmd::Launchctl(cmd) => launchd::launchctl(&root, &pf, cmd)?,
135        #[cfg(feature = "launchd")]
136        Cmd::Session(cmd) => launchd::session(&root, cmd)?,
137        #[cfg(feature = "worktree")]
138        Cmd::Worktree(cmd) => worktree(&root, &pf, cmd)?,
139        #[cfg(feature = "tool")]
140        Cmd::Tool(cmd) => tool(&root, &mut pf, &cwd, cmd)?,
141        #[cfg(feature = "env")]
142        Cmd::Env { envrc } => {
143            if envrc {
144                let p = root.join(".envrc");
145                std::fs::write(&p, pull_env::ENVRC)?;
146                println!("wrote {} (run `direnv allow` there)", p.display());
147            } else {
148                print!("{}", pull_env::exports(&workspace::env(&root, &pf)?)?);
149            }
150        }
151        #[cfg(feature = "env")]
152        Cmd::Shell { cmd } => {
153            use anyhow::Context as _;
154            let shell = std::env::var("SHELL").unwrap_or_else(|_| "/bin/sh".into());
155            let (prog, args) = match cmd.split_first() {
156                Some((p, a)) => (p.clone(), a.to_vec()),
157                None => (shell, Vec::new()),
158            };
159            let st = std::process::Command::new(&prog)
160                .args(&args)
161                .envs(workspace::env(&root, &pf)?)
162                .status()
163                .with_context(|| format!("running {prog}"))?;
164            std::process::exit(st.code().unwrap_or(1));
165        }
166        Cmd::Each { cmd } => {
167            let vars = workspace::env(&root, &pf)?;
168            for r in &pf.repos {
169                let dir = root.join(&r.name);
170                if !dir.exists() {
171                    continue;
172                }
173                println!("== {}", r.name);
174                let st = std::process::Command::new(&cmd[0])
175                    .args(&cmd[1..])
176                    .envs(vars.iter().map(|(k, v)| (k, v)))
177                    .current_dir(&dir)
178                    .status()?;
179                if !st.success() {
180                    bail!("{}: {st}", r.name);
181                }
182            }
183        }
184    }
185    Ok(())
186}

188fn init(cwd: &Path, org: Option<String>) -> Result<()> {
189    let pfp = cwd.join(PULLFILE);
190    if pfp.exists() {
191        println!("{PULLFILE} already exists");
192    } else {
193        let mut pf = Pullfile::default();
194        pf.identity.org = org.clone();
195        pf.save(cwd)?;
196        println!("wrote {}", pfp.display());
197    }
198    let lp = cwd.join(LOCAL);
199    if !lp.exists() && Local::path(cwd).is_none() {
200        let mut local = Local {
201            name: std::env::var("USER").unwrap_or_else(|_| "anon".into()),
202            ..Default::default()
203        };
204        if let Some(o) = org {
205            local.orgs.insert(o, Default::default());
206        }
207        std::fs::write(&lp, toml::to_string_pretty(&local)?)?;
208        println!("wrote {}", lp.display());
209    }
210    paths::ensure_ignored(cwd, &format!("/{LOCAL}"))
211}

213fn sync(root: &Path, pf: &mut Pullfile, to_upstream: bool) -> Result<()> {
214    pull_git::sync(root, pf, to_upstream)?;
215    workspace::run_hook(Hook::Pull, root, pf, &pf.repo_names(), &[])
216}

218fn commit(
219    root: &Path,
220    pf: &mut Pullfile,
221    cwd: &Path,
222    message: &str,
223    all: bool,
224    no_verify: bool,
225    paths: &[PathBuf],
226) -> Result<()> {
227    let msg = git2::message_prettify(message, None)?;
228    let violations = workspace::lint(pf, &msg)?;
229    if !violations.is_empty() {
230        for v in &violations {
231            eprintln!("lint: {v}");
232        }
233        bail!("commit message rejected");
234    }
235    let id = pull_core::resolve_identity(pf, &Local::load(root)?)?;
236    if pf.identity.require_signing && !id.sign {
237        bail!("pullfile requires signed commits");
238    }
239
240    let routed = paths::route(root, pf, cwd, paths)?;
241    pull_git::stage(root, pf, &routed, all)?;
242    let mut targets = pull_git::staged_repos(root, pf)?;
243    let extra = [("PULL_MESSAGE", msg.as_str())];
244    if !no_verify {
245        workspace::run_hook(Hook::Precommit, root, pf, &targets, &extra)?;
246        // Like git, whatever the hook staged is part of the commit.
247        targets = pull_git::staged_repos(root, pf)?;
248    }
249    if targets.is_empty() {
250        bail!("nothing staged in any repo");
251    }
252    pull_git::commit(root, pf, &targets, &id, &msg)?;
253    if !no_verify {
254        workspace::run_hook(Hook::Postcommit, root, pf, &targets, &extra)?;
255    }
256    Ok(())
257}

259fn status_json(root: &Path, pf: &Pullfile) -> Result<()> {
260    #[allow(unused_mut)]
261    let mut doc = serde_json::json!({
262        "root": root.display().to_string(),
263        "features": workspace::features(),
264        "repos": pull_git::repo_statuses(root, pf)?,
265    });
266    #[cfg(feature = "lint")]
267    {
268        let lint: pull_lint::Lint = pf.section(pull_lint::SECTION)?;
269        doc["lint"] = serde_json::json!({
270            "max_subject_length": lint.max_subject_length,
271            "no_description": lint.no_description,
272        });
273    }
274    println!("{}", serde_json::to_string_pretty(&doc)?);
275    Ok(())
276}

284/// A pullfile `mcp_servers` entry: `{command, args}`, kept raw (no env/root here — that's
285/// filled in per workspace by [`mcp_servers_json`]).
286#[cfg(feature = "mcp")]
287#[derive(serde::Deserialize, Default)]
288#[serde(default)]
289struct McpServerConfig {
290    /// Defaults to the section key.
291    command: Option<String>,
292    args: Vec<String>,
293}
294
295/// `pull` itself, plus every pullfile `mcp_servers` entry, as `{name: {command, args,
296/// env}}` ready for an editor/agent's MCP settings: `command` resolved on PATH (`command
297/// -v`, falling back to the bare name) and `PULL_ROOT` always set.
298#[cfg(feature = "mcp")]
299fn mcp_servers_json(root: &Path) -> Result<()> {
300    let pf = Pullfile::load(root)?;
301    let mut servers = serde_json::Map::new();
302    servers.insert(
303        "pull".to_string(),
304        serde_json::json!({
305            "command": resolve_command("pull"),
306            "args": ["mcp", "--root", root.display().to_string()],
307        }),
308    );
309    let configured: std::collections::BTreeMap<String, McpServerConfig> =
310        pf.section("mcp_servers")?;
311    for (name, cfg) in configured {
312        let command = resolve_command(cfg.command.as_deref().unwrap_or(&name));
313        servers.insert(
314            name,
315            serde_json::json!({
316                "command": command,
317                "args": cfg.args,
318                "env": { "PULL_ROOT": root.display().to_string() },
319            }),
320        );
321    }
322    println!("{}", serde_json::to_string_pretty(&servers)?);
323    Ok(())
324}

326/// `name`'s absolute path on `$PATH` (`command -v`), or `name` itself if that fails.
327#[cfg(feature = "mcp")]
328fn resolve_command(name: &str) -> String {
329    std::process::Command::new("sh")
330        .arg("-c")
331        .arg(format!("command -v -- {}", shell_quote(name)))
332        .output()
333        .ok()
334        .filter(|o| o.status.success())
335        .and_then(|o| String::from_utf8(o.stdout).ok())
336        .map(|s| s.trim().to_string())
337        .filter(|s| !s.is_empty())
338        .unwrap_or_else(|| name.to_string())
339}

341#[cfg(feature = "mcp")]
342fn shell_quote(s: &str) -> String {
343    format!("'{}'", s.replace('\'', "'\\''"))
344}

346#[cfg(feature = "worktree")]
347fn worktree(root: &Path, pf: &Pullfile, cmd: cli::WorktreeCmd) -> Result<()> {
348    use cli::WorktreeCmd;
349    match cmd {
350        WorktreeCmd::Add { name, branch } => {
351            paths::ensure_ignored(root, &format!("/{}/", pull_worktree::DIR))?;
352            let dest = pull_worktree::add(root, pf, &name, branch.as_deref())?;
353            println!("workspace: {}", dest.display());
354        }
355        WorktreeCmd::List => {
356            for (name, path) in pull_worktree::list(root)? {
357                let wpf = Pullfile::load(&path)?;
358                let mut branches: Vec<_> = wpf.repos.iter().map(|r| r.branch.as_str()).collect();
359                branches.dedup();
360                println!("{name:<20} {:<12} {}", branches.join(","), path.display());
361            }
362        }
363        WorktreeCmd::Rm { name, force } => pull_worktree::remove(root, pf, &name, force)?,
364    }
365    Ok(())
366}

368#[cfg(feature = "tool")]
369fn tool(root: &Path, pf: &mut Pullfile, cwd: &Path, cmd: cli::ToolCmd) -> Result<()> {
370    use cli::ToolCmd;
371    let print = |m: &pull_tool::Manifest| {
372        for (name, version) in &m.closure {
373            let mark = if m.linked.contains(name) { "*" } else { " " };
374            println!("{mark} {name} {version}");
375        }
376    };
377    match cmd {
378        ToolCmd::Sync => {}
379        ToolCmd::Add { names } => {
380            pull_tool::edit(root, pf, |p| p.add(&names))?;
381        }
382        ToolCmd::Rm { names } => {
383            pull_tool::edit(root, pf, |p| p.remove(&names))?;
384        }
385        ToolCmd::Import { file } => {
386            let other = pull_tool::read_brewfile(&cwd.join(file))?;
387            pull_tool::edit(root, pf, |p| {
388                let before = p.clone();
389                p.merge(other);
390                *p != before
391            })?;
392        }
393        ToolCmd::List => {
394            match pull_tool::manifest(root)? {
395                Some(m) => print(&m),
396                None => println!("not synced yet (run `pull tool sync`)"),
397            }
398            return Ok(());
399        }
400        ToolCmd::Gc => {
401            for k in pull_tool::gc()? {
402                println!("removed {k}");
403            }
404            return Ok(());
405        }
406    }
407    print(&pull_tool::sync(root, pf)?);
408    Ok(())
