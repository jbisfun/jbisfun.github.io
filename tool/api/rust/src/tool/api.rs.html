1//! Formula metadata from the Homebrew JSON API (https://formulae.brew.sh/docs/api/), seen
2//! from Linux: per-platform `variations` replace the macOS dependency lists and
3//! `uses_from_macos` entries become real dependencies, as Homebrew does on Linux.
4
5use anyhow::{Context, Result, bail};
6use serde::Deserialize;
7use std::collections::BTreeMap;
8use std::path::Path;
9
10const API: &str = "https://formulae.brew.sh/api/formula";

12#[derive(Deserialize, Debug, Clone)]
13pub struct Formula {
14    pub name: String,
15    pub tap: String,
16    pub versions: Versions,
17    #[serde(default)]
18    pub revision: u32,
19    #[serde(default)]
20    pub dependencies: Vec<String>,
21    #[serde(default)]
22    pub uses_from_macos: Vec<serde_json::Value>,
23    #[serde(default)]
24    pub variations: BTreeMap<String, Variation>,
25    pub bottle: Bottles,
26    #[serde(default)]
27    pub keg_only: bool,
28    #[serde(default)]
29    pub post_install_defined: bool,
30    #[serde(default)]
31    pub disabled: bool,
32}

58pub struct BottleFile {

65/// Homebrew bottle tag for this machine.
66pub fn bottle_tag() -> &'static str {
67    match std::env::consts::ARCH {
68        "aarch64" => "arm64_linux",
69        _ => "x86_64_linux",
70    }
71}
72
73impl Formula {
74    /// Keg directory name: version, plus `_N` for formula revisions.
75    pub fn pkg_version(&self) -> String {
76        match self.revision {
77            0 => self.versions.stable.clone(),
78            r => format!("{}_{r}", self.versions.stable),
79        }
80    }
81
82    /// Runtime dependencies on this platform.
83    pub fn runtime_deps(&self) -> Vec<String> {
84        let var = self.variations.get(bottle_tag());
85        let deps = var
86            .and_then(|v| v.dependencies.clone())
87            .unwrap_or_else(|| self.dependencies.clone());
88        let ufm = var
89            .and_then(|v| v.uses_from_macos.clone())
90            .unwrap_or_else(|| self.uses_from_macos.clone());
91        let mut out = deps;
92        for u in ufm {
93            // "zlib" or {"python": "build"} / {"m4": ["build", "test"]}: only runtime ones.
94            match u {
95                serde_json::Value::String(s) => out.push(s),
96                serde_json::Value::Object(m) => {
97                    for (name, kind) in m {
98                        let k = kind.to_string();
99                        if !k.contains("build") && !k.contains("test") {
100                            out.push(name);
101                        }
102                    }
103                }
104                _ => {}
105            }
106        }
107        out.sort();
108        out.dedup();
109        out
110    }
111
112    /// The bottle for this machine (or an arch-independent `all` bottle).
113    pub fn bottle(&self) -> Result<&BottleFile> {
114        let spec = self.bottle.stable.as_ref();
115        spec.and_then(|s| s.files.get(bottle_tag()).or_else(|| s.files.get("all")))
116            .with_context(|| format!("{} has no {} bottle", self.name, bottle_tag()))
117    }
118}

120/// Formula by name, alias (`python` -> `python@3.14`) or old name.
121pub async fn fetch(client: &reqwest::Client, cache: &Path, name: &str) -> Result<Formula> {
122    if name.contains('/') {
123        bail!("{name}: only homebrew/core formulae are supported for now");
124    }
125    let url = format!("{API}/{name}.json");
126    let resp = client.get(&url).send().await.with_context(|| url.clone())?;
127    if resp.status() == reqwest::StatusCode::NOT_FOUND {
128        // The per-formula endpoint only knows canonical names; aliases are in the index.
129        let Some(canonical) = resolve_alias(client, cache, name).await? else {
130            bail!("no formula named `{name}`");
131        };
132        return Box::pin(fetch(client, cache, &canonical)).await;
133    }
134    let f: Formula = resp
135        .error_for_status()?
136        .json()
137        .await
138        .with_context(|| format!("parsing {url}"))?;
139    if f.disabled {
140        bail!("{name} is disabled in Homebrew");
141    }
142    Ok(f)
143}

145#[derive(Deserialize)]
146struct IndexEntry {
147    name: String,
148    #[serde(default)]
149    aliases: Vec<String>,
150    #[serde(default)]
151    oldnames: Vec<String>,
152}
153
154/// Canonical name for an alias/old name, from the full formula index (cached for a day).
155async fn resolve_alias(client: &reqwest::Client, cache: &Path, name: &str) -> Result<Option<String>> {
156    let path = cache.join("formula.json");
157    let fresh = std::fs::metadata(&path)
158        .and_then(|m| m.modified())
159        .is_ok_and(|t| t.elapsed().is_ok_and(|age| age.as_secs() < 24 * 3600));
160    if !fresh {
161        let url = format!("{API}.json");
162        let bytes = client
163            .get(&url)
164            .send()
165            .await
166            .with_context(|| url.clone())?
167            .error_for_status()?
168            .bytes()
169            .await?;
170        std::fs::create_dir_all(cache)?;
171        std::fs::write(&path, &bytes)?;
172    }
173    let index: Vec<IndexEntry> = serde_json::from_slice(&std::fs::read(&path)?)
174        .with_context(|| path.display().to_string())?;
175    Ok(index
176        .into_iter()
177        .find(|e| e.aliases.iter().chain(&e.oldnames).any(|a| a == name))
178        .map(|e| e.name))
179}
