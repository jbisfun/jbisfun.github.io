1//! GrandSlam constants and the anisette header abstraction used by the developer APIs.
2//!
3//! Logging in (SRP, 2FA, app tokens) and producing anisette data is left to the
4//! caller: implement [`AnisetteDataProvider`] over whatever anisette source you use.
5use crate::Result;
6use futures::future::BoxFuture;
7use reqwest::header::HeaderMap;
8
9pub const XCODE_APP: &str = "com.apple.gs.xcode.auth";
10pub const XCODE_VERSION: &str = "26.5 (17F42)";
11
12pub trait AnisetteDataProvider: Send + Sync {
13    fn headers(&self) -> BoxFuture<'_, Result<HeaderMap>>;
14}
15
16/// Useful when the caller already owns a fresh anisette snapshot.
17impl AnisetteDataProvider for HeaderMap {
18    fn headers(&self) -> BoxFuture<'_, Result<HeaderMap>> {
19        Box::pin(async { Ok(self.clone()) })
20    }
21}
