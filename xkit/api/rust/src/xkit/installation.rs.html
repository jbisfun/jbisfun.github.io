25pub enum SearchMode {
26    Usb,
27    Network,
28    #[default]
29    All,
30}
31impl SearchMode {
32    fn allows(self, connection: &ConnectionType) -> bool {
33        matches!(
34            (self, connection),
35            (Self::All, _)
36                | (Self::Usb, ConnectionType::Usb)
37                | (Self::Network, ConnectionType::Network(_))
38        )
39    }
40}

52pub struct Connection {
53    pub udid: String,
54    pub provider: Arc<dyn IdeviceProvider>,
55    pub client: Mutex<LockdownClient>,
56}
57type ConnectionSlot = Arc<Mutex<Weak<Connection>>>;
58type ConnectionPool = Mutex<HashMap<(String, ConnectionPreferences, u64), ConnectionSlot>>;
59impl Connection {
60    pub async fn connection(
61        udid: &str,
62        preferences: ConnectionPreferences,
63        progress: &(dyn Fn(f64) + Send + Sync),
64    ) -> Result<Arc<Self>> {
65        Self::connection_with_messages(udid, preferences, progress, &|_| {}).await
66    }
67    pub async fn connection_with_messages(
68        udid: &str,
69        preferences: ConnectionPreferences,
70        progress: &(dyn Fn(f64) + Send + Sync),
71        message: &(dyn Fn(Option<ConnectionMessage>) + Send + Sync),
72    ) -> Result<Arc<Self>> {
73        static POOL: OnceLock<ConnectionPool> = OnceLock::new();
74        progress(0.0);
75        let (address, revision) = loop {
76            let before = crate::usbmuxd_address::usbmuxd_revision();
77            let address = crate::usbmuxd_address();
78            let after = crate::usbmuxd_address::usbmuxd_revision();
79            if before == after {
80                break (address, after);
81            }
82        };
83        let key = (udid.to_owned(), preferences, revision);
84        let slot = {
85            let mut pool = POOL.get_or_init(Mutex::default).lock().await;
86            pool.retain(|_, slot| {
87                Arc::strong_count(slot) > 1
88                    || slot.try_lock().is_ok_and(|entry| entry.strong_count() != 0)
89            });
90            pool.entry(key).or_default().clone()
91        };
92        // Only connection attempts for the same descriptor serialize. Pairing
93        // one device cannot block discovery or connection of another device.
94        let mut cached = slot.lock().await;
95        if let Some(connection) = cached.upgrade() {
96            progress(1.0);
97            return Ok(connection);
98        }
99        let mut mux = address.connect(0).await?;
100        let device = mux
101            .get_devices()
102            .await?
103            .into_iter()
104            .find(|d| d.udid == udid && preferences.lookup_mode.allows(&d.connection_type))
105            .ok_or_else(|| invalid(format!("Device '{udid}' was not found")))?;
106        progress(0.5);
107        let provider = Arc::new(device.to_provider(
108            address,
109            "xtool",
110        ));
111        match provider.get_pairing_file().await {
112            Ok(_) => (),
113            Err(idevice::IdeviceError::UnexpectedResponse(reason))
114                if reason == "missing PairRecordData in pair record response" =>
115            {
116                message(Some(ConnectionMessage::PairDevice));
117                Self::pair(provider.as_ref(), &mut mux, udid).await?;
118                message(None);
119            }
120            Err(error) => return Err(error.into()),
121        }
122        let connection = match Self::with_provider(udid, provider.clone()).await {
123            Err(crate::Error::Device(idevice::IdeviceError::InvalidHostID)) => {
124                message(Some(ConnectionMessage::PairDevice));
125                Self::pair(provider.as_ref(), &mut mux, udid).await?;
126                message(None);
127                Self::with_provider(udid, provider).await?
128            }
129            result => result?,
130        };
131        *cached = Arc::downgrade(&connection);
132        progress(1.0);
133        Ok(connection)
134    }
135    async fn pair(
136        provider: &dyn IdeviceProvider,
137        mux: &mut UsbmuxdConnection,
138        udid: &str,
139    ) -> Result<()> {
140        let mut client = LockdownClient::connect(provider).await?;
141        let mut record = client
142            .pair(
143                uuid::Uuid::new_v4().to_string().to_uppercase(),
144                mux.get_buid().await?,
145                None,
146            )
147            .await?;
148        record.udid = Some(udid.to_owned());
149        mux.save_pair_record(udid, record.serialize()?).await?;
150        Ok(())
151    }
152    /// Custom lookup/tunnel handlers supply a provider and own its lifetime.
153    pub async fn with_provider(
154        udid: impl Into<String>,
155        provider: Arc<dyn IdeviceProvider>,
156    ) -> Result<Arc<Self>> {
157        let mut client = LockdownClient::connect(provider.as_ref()).await?;
158        client
159            .start_session(&provider.get_pairing_file().await?)
160            .await?;
161        Ok(Arc::new(Self {
162            udid: udid.into(),
163            provider,
164            client: Mutex::new(client),
165        }))
166    }
167    pub async fn start_client<T: IdeviceService>(&self) -> Result<T> {
168        Ok(T::connect(self.provider.as_ref()).await?)
169    }
170    pub async fn device_name(&self) -> Result<String> {
171        self.client
172            .lock()
173            .await
174            .get_value(Some("DeviceName"), None)
175            .await?
176            .as_string()
177            .map(str::to_owned)
178            .ok_or_else(|| invalid("Device name is unavailable"))
179    }
180    /// Enables wireless connections and returns a separate host pairing record.
181    /// The system usbmuxd record is left intact.
182    pub async fn configure_wireless(&self) -> Result<Vec<u8>> {
183        let mut client = self.client.lock().await;
184        client
185            .set_value(
186                "WirelessBuddyID",
187                self.udid.clone().into(),
188                Some("com.apple.mobile.wireless_lockdown"),
189            )
190            .await?;
191        client
192            .set_value(
193                "EnableWifiConnections",
194                true.into(),
195                Some("com.apple.mobile.wireless_lockdown"),
196            )
197            .await?;
198        let old = self.provider.get_pairing_file().await?;
199        let mut buid = uuid::Uuid::parse_str(&old.system_buid)
200            .map_err(|e| invalid(e.to_string()))?
201            .into_bytes();
202        buid[0] = !buid[0];
203        let new_buid = uuid::Uuid::from_bytes(buid).to_string().to_uppercase();
204        let mut record = client
205            .pair(
206                uuid::Uuid::new_v4().to_string().to_uppercase(),
207                new_buid,
208                None,
209            )
210            .await?;
211        record.udid = Some(self.udid.clone());
212        Ok(record.serialize()?)
213    }
214}

216#[derive(Clone, Debug)]
217pub struct ClientDevice {
218    pub device: UsbmuxdDevice,
219    pub device_name: String,
220}
221impl ClientDevice {
222    pub fn udid(&self) -> &str {
223        &self.device.udid
224    }
225    pub async fn all(mode: SearchMode) -> Result<Vec<Self>> {
226        let address = crate::usbmuxd_address();
227        let mut mux = address.connect(0).await?;
228        let mut devices = Vec::new();
229        for device in mux.get_devices().await? {
230            if !mode.allows(&device.connection_type) {
231                continue;
232            }
233            let provider = device.to_provider(
234                address.clone(),
235                "xtool",
236            );
237            let mut client = LockdownClient::connect(&provider).await?;
238            let device_name = client
239                .get_value(Some("DeviceName"), None)
240                .await?
241                .as_string()
242                .map(str::to_owned)
243                .ok_or_else(|| invalid("Device name is unavailable"))?;
244            devices.push(Self {
245                device,
246                device_name,
247            });
248        }
249        devices.sort_by_key(|d| {
250            (
251                d.device.udid.clone(),
252                matches!(d.device.connection_type, ConnectionType::Usb),
253            )
254        });
255        Ok(devices)
256    }
257    /// Emits an initial snapshot, then changed device lists. Dropping the stream
258    /// cancels discovery; no detached discovery task is retained.
259    pub fn search(mode: SearchMode) -> BoxStream<'static, Result<Vec<Self>>> {
260        futures::stream::try_unfold(
261            None::<Vec<(String, u32, String)>>,
262            move |mut previous| async move {
263                loop {
264                    if previous.is_some() {
265                        tokio::time::sleep(Duration::from_secs(1)).await;
266                    }
267                    let devices = Self::all(mode).await?;
268                    let snapshot = devices
269                        .iter()
270                        .map(|d| {
271                            (
272                                d.device.udid.clone(),
273                                d.device.device_id,
274                                d.device_name.clone(),
275                            )
276                        })
277                        .collect::<Vec<_>>();
278                    if previous.as_ref() != Some(&snapshot) {
279                        return Ok(Some((devices, Some(snapshot))));
280                    }
281                    previous = Some(snapshot);
282                }
283            },
284        )
285        .boxed()
286    }
287}

289#[derive(Clone, Debug, PartialEq)]
290pub enum InstallationStage {
291    Connecting(f64),
292    Uploading(f64),
293    Installing {
294        details: String,
295        progress: Option<f64>,
296    },
297}
298impl InstallationStage {
299    pub fn display_name(&self) -> &str {
300        match self {
301            Self::Connecting(_) => "Connecting",
302            Self::Uploading(_) => "Installing",
303            Self::Installing { .. } => "Verifying",
304        }
305    }
306    pub fn display_progress(&self) -> f64 {
307        match self {
308            Self::Connecting(v) | Self::Uploading(v) => *v,
309            Self::Installing { progress, .. } => progress.unwrap_or(0.0),
310        }
311    }
312}

317pub struct UploadedIPA {
318    pub location: String,
319    client: Option<Arc<Mutex<AfcClient>>>,
320}
321impl UploadedIPA {
322    pub async fn delete(&mut self) -> Result<()> {
323        if let Some(client) = self.client.as_ref() {
324            client.lock().await.remove(&self.location).await?;
325            self.client = None;
326        }
327        Ok(())
328    }
329}
330impl Drop for UploadedIPA {
331    fn drop(&mut self) {
332        if let (Some(client), Ok(runtime)) =
333            (self.client.take(), tokio::runtime::Handle::try_current())
334        {
335            let location = self.location.clone();
336            runtime.spawn(async move {
337                let _ = client.lock().await.remove(location).await;
338            });
339        }
340    }
341}
342impl IPAUploader {
343    pub async fn new(connection: &Connection) -> Result<Self> {
344        Ok(Self {
345            client: Arc::new(Mutex::new(connection.start_client().await?)),
346        })
347    }
348    pub async fn upload(
349        &self,
350        app: &Path,
351        progress: &(dyn Fn(f64) + Send + Sync),
352    ) -> Result<UploadedIPA> {
353        let mut source = tokio::fs::File::open(app).await?;
354        let size = source.metadata().await?.len();
355        let mut client = self.client.lock().await;
356        if let Err(error) = client.get_file_info("/PublicStaging").await {
357            if matches!(
358                error,
359                idevice::IdeviceError::Afc(idevice::afc::errors::AfcError::ObjectNotFound)
360            ) {
361                client.mk_dir("/PublicStaging").await?;
362            } else {
363                return Err(error.into());
364            }
365        }
366        let uploaded = UploadedIPA {
367            location: format!("/PublicStaging/XTool-{}.ipa", uuid::Uuid::new_v4()),
368            client: Some(self.client.clone()),
369        };
370        let mut target = client
371            .open(&uploaded.location, AfcFopenMode::WrOnly)
372            .await?;
373        let mut buffer = vec![0; 1 << 20];
374        let mut written = 0u64;
375        progress(0.0);
376        // Close `target` on every exit: idevice panics (debug) on an unclosed AFC handle.
377        let copied: std::result::Result<(), std::io::Error> = async {
378            loop {
379                let count = source.read(&mut buffer).await?;
380                if count == 0 {
381                    break;
382                }
383                target.write_all(&buffer[..count]).await?;
384                written += count as u64;
385                if size != 0 {
386                    progress((written as f64 / size as f64).min(1.0));
387                }
388            }
389            Ok(())
390        }
391        .await;
392        if let Err(error) = copied {
393            let _ = target.close().await;
394            return Err(error.into());
395        }
396        target.close().await?;
397        progress(1.0);
398        Ok(uploaded)
399    }
400}

434pub struct AppInstaller {
435    pub ipa: PathBuf,
436    pub udid: String,
437    pub connection_preferences: ConnectionPreferences,
438}
439impl AppInstaller {
440    pub async fn install(
441        &self,
442        progress: &(dyn Fn(InstallationStage) + Send + Sync),
443    ) -> Result<()> {
444        let connection = Connection::connection(&self.udid, self.connection_preferences, &|v| {
445            progress(InstallationStage::Connecting(v * 4.0 / 6.0))
446        })
447        .await?;
448        self.install_with_connection(&connection, progress).await
449    }
450    pub async fn install_with_connection(
451        &self,
452        connection: &Connection,
453        progress: &(dyn Fn(InstallationStage) + Send + Sync),
454    ) -> Result<()> {
455        let uploader = IPAUploader::new(connection).await?;
456        progress(InstallationStage::Connecting(5.0 / 6.0));
457        // Start installation_proxy before uploading to avoid its startup timeout.
458        let mut installer = IPAInstaller::new(connection).await?;
459        progress(InstallationStage::Connecting(1.0));
460        let mut uploaded = uploader
461            .upload(&self.ipa, &|v| progress(InstallationStage::Uploading(v)))
462            .await?;
463        let result = installer.install(&uploaded, progress).await;
464        let cleanup = uploaded.delete().await;
465        result?;
466        cleanup
467    }
468}

470pub struct ProfileManager {
471    client: MisagentClient,
472    modern: bool,
473}
474impl ProfileManager {
475    pub async fn new(connection: &Connection) -> Result<Self> {
476        let version = connection
477            .client
478            .lock()
479            .await
480            .get_value(Some("ProductVersion"), None)
481            .await?;
482        let components = version
483            .as_string()
484            .unwrap_or("")
485            .split('.')
486            .filter_map(|v| v.parse::<u32>().ok())
487            .collect::<Vec<_>>();
488        let modern = components.first().is_some_and(|major| {
489            *major > 9 || (*major == 9 && components.get(1).is_some_and(|minor| *minor >= 3))
490        });
491        Ok(Self {
492            client: connection.start_client().await?,
493            modern,
494        })
495    }
496    pub async fn all_profiles(&mut self) -> Result<Vec<Mobileprovision>> {
497        let data = if self.modern {
498            self.client.copy_all().await?
499        } else {
500            let request = plist::Dictionary::from_iter([
501                ("MessageType", plist::Value::from("Copy")),
502                ("ProfileType", plist::Value::from("Provisioning")),
503            ]);
504            let mut bytes = Vec::new();
505            plist::to_writer_xml(&mut bytes, &request)?;
506            let length: u32 = bytes
507                .len()
508                .try_into()
509                .map_err(|_| invalid("Profile request is too large"))?;
510            let mut frame = length.to_be_bytes().to_vec();
511            frame.extend_from_slice(&bytes);
512            self.client.idevice.send_raw(&frame).await?;
513            let length = self.client.idevice.read_raw(4).await?;
514            let length = u32::from_be_bytes(
515                length
516                    .try_into()
517                    .map_err(|_| invalid("Invalid profile response length"))?,
518            );
519            if length > 64 * 1024 * 1024 {
520                return Err(invalid("Profile response is too large"));
521            }
522            let response: plist::Dictionary =
523                plist::from_bytes(&self.client.idevice.read_raw(length as usize).await?)?;
524            if response
525                .get("Status")
526                .and_then(plist::Value::as_signed_integer)
527                .is_some_and(|status| status != 0)
528            {
529                return Err(invalid("Could not enumerate legacy provisioning profiles"));
530            }
531            response
532                .get("Payload")
533                .and_then(plist::Value::as_array)
534                .ok_or_else(|| invalid("Missing profile list"))?
535                .iter()
536                .map(|value| {
537                    value
538                        .as_data()
539                        .map(<[u8]>::to_vec)
540                        .ok_or_else(|| invalid("Invalid profile data"))
541                })
542                .collect::<Result<Vec<_>>>()?
543        };
544        data.into_iter().map(Mobileprovision::from_bytes).collect()
545    }
546    pub async fn install(&mut self, profiles: &[Mobileprovision]) -> Result<()> {
547        fn group(profiles: &[Mobileprovision]) -> BTreeMap<String, Vec<&Mobileprovision>> {
548            let mut grouped: BTreeMap<String, Vec<&Mobileprovision>> = BTreeMap::new();
549            for profile in profiles {
550                if let Some(id) = profile
551                    .entitlements()
552                    .get("application-identifier")
553                    .and_then(plist::Value::as_string)
554                {
555                    grouped.entry(id.to_owned()).or_default().push(profile);
556                }
557            }
558            grouped
559        }
560        let installed = self.all_profiles().await?;
561        let installed = group(&installed);
562        for (id, profiles) in group(profiles) {
563            if profiles.len() != 1 {
564                continue;
565            }
566            if let Some(old) = installed.get(&id) {
567                for profile in old {
568                    self.client.remove(&profile.digest().uuid).await?;
569                }
570            }
571            self.client.install(profiles[0].data().to_vec()).await?;
572        }
573        Ok(())
574    }
575}

577pub struct HeartbeatHandler(tokio::task::JoinHandle<()>);
578impl HeartbeatHandler {
579    pub async fn new(connection: &Connection) -> Result<Self> {
580        let mut client: idevice::heartbeat::HeartbeatClient = connection.start_client().await?;
581        let provider = connection.provider.clone();
582        Ok(Self(tokio::spawn(async move {
583            let mut first = true;
584            loop {
585                let beat = async {
586                    let interval = client.get_marco(if first { 30 } else { 5 }).await?;
587                    client.send_polo().await?;
588                    first = false;
589                    tokio::time::sleep(Duration::from_secs(interval)).await;
590                    Ok::<_, idevice::IdeviceError>(())
591                }
592                .await;
593                if beat.is_err() {
594                    loop {
595                        tokio::time::sleep(Duration::from_secs(1)).await;
596                        if let Ok(reconnected) =
597                            idevice::heartbeat::HeartbeatClient::connect(provider.as_ref()).await
598                        {
599                            client = reconnected;
600                            first = true;
601                            break;
602                        }
603                    }
604                }
605            }
606        })))
607    }
608    pub fn stop(&self) {
609        self.0.abort();
610    }
611}
612impl Drop for HeartbeatHandler {
613    fn drop(&mut self) {
614        self.stop();
615    }

618pub struct Debugger {
619    client: idevice::debug_proxy::DebugProxyClient<Box<dyn idevice::ReadWrite>>,
620}
621impl Debugger {
622    pub async fn new(connection: &Connection) -> Result<Self> {
623        let mut lockdown = connection.client.lock().await;
624        let (port, ssl) = lockdown
625            .start_service("com.apple.debugserver.DVTSecureSocketProxy")
626            .await?;
627        let mut socket = connection.provider.connect(port).await?;
628        if ssl {
629            socket
630                .start_session(&connection.provider.get_pairing_file().await?, false)
631                .await?;
632        }
633        let mut client = idevice::debug_proxy::DebugProxyClient::new(
634            socket
635                .get_socket()
636                .ok_or_else(|| invalid("Debugger socket is unavailable"))?,
637        );
638        client
639            .send_command(idevice::debug_proxy::DebugserverCommand::new(
640                "QSetMaxPacketSize:".into(),
641                vec!["1024".into()],
642            ))
643            .await?;
644        Ok(Self { client })
645    }
646    pub async fn attach(&mut self, process: &str) -> Result<()> {
647        loop {
648            let response = self
649                .client
650                .send_command(idevice::debug_proxy::DebugserverCommand::new(
651                    "vAttachWait;".into(),
652                    vec![process.into()],
653                ))
654                .await?;
655            if let Some(response) = response.filter(|s| !s.is_empty()) {
656                if response.starts_with('E') {
657                    return Err(invalid(format!("Debugger attach failed: {response}")));
658                }
659                break;
660            }
661            tokio::task::yield_now().await;
662        }
663        self.client
664            .send_command(idevice::debug_proxy::DebugserverCommand::new(
665                "D".into(),
666                vec![],
667            ))
668            .await?;
669        Ok(())
670    }
