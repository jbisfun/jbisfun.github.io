52pub(crate) fn bundle_metadata(app: &Path) -> Result<(String, String)> {
53    let info: plist::Dictionary = plist::from_file(app.join("Info.plist"))?;
54    let string = |key| {
55        info.get(key)
56            .and_then(plist::Value::as_string)
57            .filter(|v| !v.is_empty())
58            .map(str::to_owned)
59            .ok_or_else(|| invalid(format!("Missing {key} in {}", app.display())))
60    };
61    let executable = string("CFBundleExecutable")?;
62    if Path::new(&executable).components().count() != 1
63        || !matches!(
64            Path::new(&executable).components().next(),
65            Some(Component::Normal(_))
66        )
67    {
68        return Err(invalid("Bundle executable must be a file name"));
69    }
70    if !app.join(&executable).is_file() {
71        return Err(invalid(format!("Missing executable in {}", app.display())));
72    }
73    Ok((string("CFBundleIdentifier")?, executable))
74}

76/// Rejects an expired certificate or one that doesn't match the private key
77/// before signing, so the device doesn't fail the install later (0xe8008018).
78fn validate_signing_identity(
79    cert: &x509_certificate::CapturedX509Certificate,
80    key: &dyn PrivateKey,
81) -> Result<()> {
82    let now = chrono::Utc::now();
83    if now < cert.validity_not_before() || now >= cert.validity_not_after() {
84        return Err(invalid(
85            "Signing certificate is expired or not yet valid. Choose a valid certificate and re-sign (0xe8008018).",
86        ));
87    }
88    if key.public_key_data().as_ref() != cert.public_key_data().as_ref() {
89        return Err(invalid(
90            "Signing certificate does not match the private key. Import the matching key and re-sign (0xe8008018).",
91        ));
92    }
93    Ok(())
94}

101impl Signer {
102    pub fn name() -> &'static str {
103        "apple-codesign"
104    }
105    pub fn analyze(executable: &Path) -> Result<Entitlements> {
106        let data = fs::read(executable)?;
107        let file = MachFile::parse(&data)?;
108        if let Some(signature) = file.nth_macho(0)?.code_signature()?
109            && let Ok(Some(slot)) = signature.entitlements()
110            && let plist::Value::Dictionary(dict) =
111                plist::Value::from_reader_xml(slot.to_string().as_bytes())?
112        {
113            return Ok(Entitlements(dict));
114        }
115        Ok(Entitlements(Default::default()))
116    }
117    pub fn sign(
118        app: &Path,
119        identity: SigningIdentity<'_>,
120        entitlement_mapping: &BTreeMap<PathBuf, Entitlements>,
121        progress: &(dyn Fn(Option<f64>) + Send + Sync),
122    ) -> Result<()> {
123        progress(None);
124        let key;
125        let mut settings = SigningSettings::default();
126        if let SigningIdentity::Real(info) = identity {
127            let pem =
128                pem::parse(info.private_key.data.as_ref()).map_err(|e| invalid(e.to_string()))?;
129            key = match pem.tag() {
130                "PRIVATE KEY" => InMemoryPrivateKey::from_pkcs8_der(pem.contents())?,
131                "RSA PRIVATE KEY" => InMemoryPrivateKey::from_pkcs1_der(pem.contents())?,
132                _ => return Err(invalid("Expected a PEM private key")),
133            };
134            validate_signing_identity(info.certificate.raw(), &key)?;
135            settings.set_signing_key(&key, info.certificate.raw().clone());
136            settings.chain_apple_certificates();
137            settings.set_team_id_from_signing_certificate();
138        }
139        // Use the bundle library's exact scope strings. On Windows, nested
140        // bundle scopes can contain both native separators and `/` delimiters.
141        let nested = apple_bundles::DirectoryBundle::new_from_path(app)
142            .and_then(|bundle| bundle.nested_bundles(true))
143            .map_err(|error| invalid(error.to_string()))?;
144        for (path, entitlements) in entitlement_mapping {
145            let relative = path
146                .strip_prefix(app)
147                .map_err(|_| invalid("Entitlement mapping is outside the app"))?;
148            let scope = if relative.as_os_str().is_empty() {
149                SettingsScope::Main
150            } else {
151                let (scope, _) = nested
152                    .iter()
153                    .find(|(scope, _)| Path::new(scope) == relative)
154                    .ok_or_else(|| {
155                        invalid(format!("No nested signing bundle at {}", path.display()))
156                    })?;
157                SettingsScope::Path(scope.clone())
158            };
159            let xml =
160                String::from_utf8(entitlements.to_xml()?).map_err(|e| invalid(e.to_string()))?;
161            settings.set_entitlements_xml(scope, xml)?;
162        }
163        UnifiedSigner::new(settings).sign_path_in_place(app)?;
164        progress(Some(1.0));
165        Ok(())
166    }
167}

169pub struct AutoSigner {
170    pub context: SigningContext,
171}
172impl AutoSigner {
173    pub async fn sign(
174        &self,
175        app: &Path,
176        confirm: &ConfirmRevocation,
177        status: &(dyn Fn(&str) + Send + Sync),
178        progress: &(dyn Fn(Option<f64>) + Send + Sync),
179        did_provision: &(dyn Fn() -> Result<()> + Send + Sync),
180    ) -> Result<String> {
181        status("Provisioning");
182        let response = DeveloperServicesProvisioningOperation {
183            context: &self.context,
184        }
185        .perform(app, confirm, &|value| progress(Some(value)))
186        .await?;
187        let main_id = response
188            .provisioning
189            .get(app)
190            .ok_or_else(|| invalid("Missing main app provisioning"))?
191            .new_bundle_id
192            .clone();
193        did_provision()?;
194        let mut entitlements = BTreeMap::new();
195        for (path, info) in response.provisioning {
196            let info_path = path.join("Info.plist");
197            let mut plist: plist::Dictionary = plist::from_file(&info_path)?;
198            plist.insert("CFBundleIdentifier".into(), info.new_bundle_id.into());
199            plist::to_file_xml(info_path, &plist)?;
200            fs::write(
201                path.join("embedded.mobileprovision"),
202                info.mobileprovision.data(),
203            )?;
204            entitlements.insert(path, info.entitlements);
205        }
206        status("Signing");
207        Signer::sign(
208            app,
209            SigningIdentity::Real(&response.signing_info),
210            &entitlements,
211            progress,
212        )?;
213        Ok(main_id)
214    }
