8pub struct ZIPCompressor;
9impl ZIPCompressor {
10    pub fn decompress(file: &Path, directory: &Path, progress: &dyn Fn(Option<f64>)) -> Result<()> {
11        progress(None);
12        fs::create_dir_all(directory)?;
13        // zip validates enclosed paths and symlink destinations during extraction.
14        ZipArchive::new(fs::File::open(file)?)?.extract(directory)?;
15        progress(Some(1.0));
16        Ok(())
17    }
18    /// Writes a ZIP containing the supplied directory itself (normally Payload).
19    pub fn compress(
20        directory: &Path,
21        destination: &Path,
22        progress: &dyn Fn(Option<f64>),
23    ) -> Result<()> {
24        let directory = directory.canonicalize()?;
25        let parent = directory
26            .parent()
27            .ok_or_else(|| invalid("Cannot archive filesystem root"))?;
28        let destination_parent = destination
29            .parent()
30            .filter(|p| !p.as_os_str().is_empty())
31            .unwrap_or_else(|| Path::new("."));
32        let destination_absolute = destination_parent.canonicalize()?.join(
33            destination
34                .file_name()
35                .ok_or_else(|| invalid("Invalid archive path"))?,
36        );
37        if destination_absolute.starts_with(&directory) {
38            return Err(invalid(
39                "Archive destination must be outside the source directory",
40            ));
41        }
42        let mut files = Vec::new();
43        collect(&directory, &mut files)?;
44        let mut writer = ZipWriter::new(fs::File::create(destination)?);
45        let total = files.len().max(1) as f64;
46        progress(Some(0.0));
47        for (index, path) in files.iter().enumerate() {
48            let name = path
49                .strip_prefix(parent)
50                .map_err(|e| invalid(e.to_string()))?
51                .to_string_lossy()
52                .replace('\\', "/");
53            let metadata = fs::symlink_metadata(path)?;
54            #[cfg(unix)]
55            let mode = {
56                use std::os::unix::fs::PermissionsExt;
57                metadata.permissions().mode()
58            };
59            #[cfg(not(unix))]
60            let mode = if metadata.is_dir() { 0o755 } else { 0o644 };
61            let options = SimpleFileOptions::default().unix_permissions(mode);
62            if metadata.file_type().is_symlink() {
63                writer.add_symlink(
64                    name,
65                    fs::read_link(path)?.to_string_lossy().replace('\\', "/"),
66                    options,
67                )?;
68            } else if metadata.is_dir() {
69                writer.add_directory(format!("{name}/"), options)?;
70            } else {
71                writer.start_file(name, options)?;
72                io::copy(&mut fs::File::open(path)?, &mut writer)?;
73            }
74            progress(Some((index + 1) as f64 / total));
75        }
76        writer.finish()?;
77        progress(Some(1.0));
78        Ok(())
79    }
80}

82fn collect(path: &Path, result: &mut Vec<PathBuf>) -> Result<()> {
83    result.push(path.to_path_buf());
84    if fs::symlink_metadata(path)?.is_dir() {
85        let mut entries = fs::read_dir(path)?.collect::<std::result::Result<Vec<_>, _>>()?;
86        entries.sort_by_key(|e| e.file_name());
87        for entry in entries {
88            collect(&entry.path(), result)?;
89        }
90    }
91    Ok(())
92}

94pub fn copy_directory(source: &Path, destination: &Path) -> Result<()> {
95    fs::create_dir_all(destination)?;
96    for entry in fs::read_dir(source)? {
97        let entry = entry?;
98        let path = entry.path();
99        let target = destination.join(entry.file_name());
100        let kind = entry.file_type()?;
101        if kind.is_symlink() {
102            let link = fs::read_link(&path)?;
103            #[cfg(unix)]
104            std::os::unix::fs::symlink(link, target)?;
105            #[cfg(windows)]
106            {
107                if path.is_dir() {
108                    std::os::windows::fs::symlink_dir(link, target)?;
109                } else {
110                    std::os::windows::fs::symlink_file(link, target)?;
111                }
112            }
113            #[cfg(not(any(unix, windows)))]
114            return Err(invalid("Symlinks are unsupported on this platform"));
115        } else if kind.is_dir() {
116            copy_directory(&path, &target)?;
117        } else {
118            fs::copy(path, target)?;
119        }
120    }
121    Ok(())
122}
