1//! Injectable HTTP transport. Requests never log authentication headers or bodies.
2use crate::{Error, Result};
3use futures::future::BoxFuture;
4use reqwest::{Client, Method, header::HeaderMap};
5
6#[derive(Clone, Debug)]
7pub struct HTTPRequest {
8    pub method: Method,
9    pub url: String,
10    pub headers: HeaderMap,
11    pub body: Vec<u8>,
12}
13
14impl HTTPRequest {
15    pub fn new(method: Method, url: impl Into<String>) -> Self {
16        Self {
17            method,
18            url: url.into(),
19            headers: HeaderMap::new(),
20            body: Vec::new(),
21        }
22    }
23}
24
25#[derive(Clone, Debug)]
26pub struct HTTPResponse {
27    pub status: u16,
28    pub headers: HeaderMap,
29    pub body: Vec<u8>,
30}
31
32impl HTTPResponse {
33    pub fn require_success(self, request: &HTTPRequest) -> Result<Self> {
34        if (200..300).contains(&self.status) {
35            return Ok(self);
36        }
37        Err(Error::Http {
38            method: request.method.to_string(),
39            url: request.url.clone(),
40            status: self.status,
41            body: String::from_utf8_lossy(&self.body).trim().to_owned(),
42        })
43    }
44}
45
46pub trait HTTPClientProtocol: Send + Sync {
47    fn send(&self, request: HTTPRequest) -> BoxFuture<'_, Result<HTTPResponse>>;
48}
49
50#[derive(Clone)]
51pub struct HTTPClient(pub Client);
52
53impl HTTPClient {
54    pub fn new() -> Result<Self> {
55        const APPLE_ROOT: &[u8] = include_bytes!("apple_root.der");
56        Ok(Self(
57            Client::builder()
58                .add_root_certificate(reqwest::Certificate::from_der(APPLE_ROOT)?)
59                .http1_title_case_headers()
60                .build()?,
61        ))
62    }
63}
64
65impl HTTPClientProtocol for HTTPClient {
66    fn send(&self, request: HTTPRequest) -> BoxFuture<'_, Result<HTTPResponse>> {
67        Box::pin(async move {
68            let response = self
69                .0
70                .request(request.method, &request.url)
71                .headers(request.headers)
72                .body(request.body)
73                .send()
74                .await?;
75            Ok(HTTPResponse {
76                status: response.status().as_u16(),
77                headers: response.headers().clone(),
78                body: response.bytes().await?.to_vec(),
79            })
80        })
81    }
