7/// Validated, DER encoded X.509 certificate, serialized as plist data.
8#[derive(Clone, Debug)]
9pub struct Certificate(CapturedX509Certificate);
10
11impl Certificate {
12    pub fn from_der(data: impl AsRef<[u8]>) -> Result<Self> {
13        Ok(Self(CapturedX509Certificate::from_der(
14            data.as_ref().to_vec(),
15        )?))
16    }
17    pub fn from_path(path: impl AsRef<Path>) -> Result<Self> {
18        Self::from_der(fs::read(path)?)
19    }
20    pub fn data(&self) -> &[u8] {
21        self.0.constructed_data()
22    }
23    pub fn raw(&self) -> &CapturedX509Certificate {
24        &self.0
25    }
26    pub fn developer_identity(&self) -> Result<String> {
27        self.0
28            .subject_common_name()
29            .ok_or_else(|| invalid("Certificate has no common name"))
30    }
31    pub fn team_id(&self) -> Result<String> {
32        self.0
33            .subject_name()
34            .iter_organizational_unit()
35            .next()
36            .and_then(|v| v.to_string().ok())
37            .ok_or_else(|| invalid("Certificate has no team ID"))
38    }
39    pub fn serial_number(&self) -> String {
40        // Keep leading zero nibbles, but drop sign-padding bytes.
41        let serial = self.0.serial_number_asn1().clone();
42        let bytes = serial.as_slice();
43        let start = bytes.iter().position(|b| *b != 0).unwrap_or(bytes.len());
44        hex::encode_upper(&bytes[start..])
45    }
46    pub fn has_serial_number(&self, serial: &str) -> bool {
47        self.serial_number()
48            .trim_start_matches('0')
49            .eq_ignore_ascii_case(serial.trim_start_matches('0'))
50    }
51    pub fn was_issued_before(&self, other: &Self) -> bool {
52        self.0.validity_not_before() < other.0.validity_not_before()
53    }
54}

56impl Serialize for Certificate {
57    fn serialize<S: serde::Serializer>(
58        &self,
59        serializer: S,
60    ) -> std::result::Result<S::Ok, S::Error> {
61        serializer.serialize_bytes(self.data())
62    }
63}
64impl<'de> Deserialize<'de> for Certificate {
65    fn deserialize<D: serde::Deserializer<'de>>(
66        deserializer: D,
67    ) -> std::result::Result<Self, D::Error> {
68        let data = plist::Data::deserialize(deserializer)?;
69        Self::from_der(data.as_ref()).map_err(serde::de::Error::custom)
70    }
71}

89pub struct Keypair(rsa::RsaPrivateKey);
90impl Keypair {
91    pub fn new() -> Result<Self> {
92        Ok(Self(
93            rsa::RsaPrivateKey::new(&mut rand::rngs::OsRng, 2048)
94                .map_err(|e| invalid(e.to_string()))?,
95        ))
96    }
97    pub fn private_key(&self) -> Result<PrivateKey> {
98        use rsa::pkcs8::EncodePrivateKey;
99        Ok(PrivateKey {
100            data: self
101                .0
102                .to_pkcs8_pem(Default::default())
103                .map_err(|e| invalid(e.to_string()))?
104                .as_bytes()
105                .to_vec()
106                .into(),
107        })
108    }
109    pub fn generate_csr(&self) -> Result<CSR> {
110        use rsa::pkcs8::EncodePrivateKey;
111        let der = self.0.to_pkcs8_der().map_err(|e| invalid(e.to_string()))?;
112        let mut params = rcgen::CertificateParams::new(Vec::new());
113        params.alg = &rcgen::PKCS_RSA_SHA256;
114        params.key_pair =
115            Some(rcgen::KeyPair::from_der(der.as_bytes()).map_err(|e| invalid(e.to_string()))?);
116        params.distinguished_name = rcgen::DistinguishedName::new();
117        params
118            .distinguished_name
119            .push(rcgen::DnType::CountryName, "US");
120        params
121            .distinguished_name
122            .push(rcgen::DnType::CommonName, "xtool");
123        let pem_string = rcgen::Certificate::from_params(params)
124            .and_then(|c| c.serialize_request_pem())
125            .map_err(|e| invalid(e.to_string()))?;
126        Ok(CSR { pem_string })
127    }
128}

157#[derive(Clone, Debug)]
158pub struct Mobileprovision {
159    data: Vec<u8>,
160    digest: MobileprovisionDigest,
161}
162
163impl Mobileprovision {
164    pub fn from_bytes(data: Vec<u8>) -> Result<Self> {
165        // Apple's CMS payload contains an XML plist. Also accept a standalone
166        // plist (including binary plists) for tooling and offline fixtures.
167        let value = match Value::from_reader(std::io::Cursor::new(&data)) {
168            Ok(value) => value,
169            Err(_) => {
170                let start = data
171                    .windows(6)
172                    .position(|v| v == b"<plist")
173                    .ok_or_else(|| invalid("Invalid provisioning profile"))?;
174                let end = data
175                    .windows(8)
176                    .rposition(|v| v == b"</plist>")
177                    .map(|v| v + 8)
178                    .filter(|v| *v > start)
179                    .ok_or_else(|| invalid("Invalid provisioning profile"))?;
180                Value::from_reader_xml(&data[start..end])?
181            }
182        };
183        let digest = plist::from_value(&value)?;
184        Ok(Self { data, digest })
185    }
186    pub fn from_path(path: impl AsRef<Path>) -> Result<Self> {
187        Self::from_bytes(fs::read(path)?)
188    }
189    pub fn data(&self) -> &[u8] {
190        &self.data
191    }
192    pub fn digest(&self) -> &MobileprovisionDigest {
193        &self.digest
194    }
195    pub fn entitlements(&self) -> &Dictionary {
196        &self.digest.entitlements.0
197    }
