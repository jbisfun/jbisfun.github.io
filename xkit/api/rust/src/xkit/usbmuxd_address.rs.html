1use std::sync::atomic::{AtomicU64, Ordering};
2use std::sync::{OnceLock, RwLock};
3
4use idevice::usbmuxd::UsbmuxdAddr;
5
6static OVERRIDE: OnceLock<RwLock<Option<UsbmuxdAddr>>> = OnceLock::new();
7static REVISION: AtomicU64 = AtomicU64::new(0);
8
9pub(crate) fn usbmuxd_revision() -> u64 {
10    REVISION.load(Ordering::Acquire)
11}
12
13pub fn parse_usbmuxd_address(address: &str) -> Option<UsbmuxdAddr> {
14    #[cfg(unix)]
15    if !address.contains(':') {
16        return Some(UsbmuxdAddr::UnixSocket(address.to_owned()));
17    }
18    address.parse().ok().map(UsbmuxdAddr::TcpSocket)
19}
20
21/// The USB service used by device discovery and subsequent operations.
22pub fn usbmuxd_address() -> UsbmuxdAddr {
23    OVERRIDE
24        .get_or_init(RwLock::default)
25        .read()
26        .expect("USB service address lock poisoned")
27        .clone()
28        .unwrap_or_else(|| UsbmuxdAddr::from_env_var().unwrap_or_default())
29}
30
31/// Changes the USB service for future connections. `None` uses the environment
32/// override or the platform default.
33pub fn set_usbmuxd_address(address: Option<UsbmuxdAddr>) {
34    *OVERRIDE
35        .get_or_init(RwLock::default)
36        .write()
37        .expect("USB service address lock poisoned") = address;
38    REVISION.fetch_add(1, Ordering::Release);
39}
