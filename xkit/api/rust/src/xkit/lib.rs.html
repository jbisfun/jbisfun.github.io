35#[derive(Debug, thiserror::Error)]
36pub enum Error {
37    #[error("{0}")]
38    InvalidData(String),
39    #[error("The operation was cancelled")]
40    Cancelled,
41    #[error("An installation is already in progress")]
42    AlreadyInstalling,
43    #[error("Developer API error {code}: {message}")]
44    DeveloperApi { code: String, message: String },
45    #[error("{method} {url} failed: HTTP {status}: {body}")]
46    Http {
47        method: String,
48        url: String,
49        status: u16,
50        body: String,
51    },
52    #[error("This operation requires Xcode authentication")]
53    RequiresXcode,
54    #[error(transparent)]
55    Io(#[from] std::io::Error),
56    #[error(transparent)]
57    Plist(#[from] plist::Error),
58    #[error(transparent)]
59    Json(#[from] serde_json::Error),
60    #[error(transparent)]
61    Reqwest(#[from] reqwest::Error),
62    /// Errors from caller-supplied providers (anisette, storage, ...).
63    #[error(transparent)]
64    External(#[from] Box<dyn std::error::Error + Send + Sync>),
65    #[error(transparent)]
66    Device(#[from] idevice::IdeviceError),
67    #[error(transparent)]
68    Zip(#[from] zip::result::ZipError),
69    #[error(transparent)]
70    Certificate(#[from] x509_certificate::X509CertificateError),
71    #[error(transparent)]
72    Codesign(#[from] Box<apple_codesign::AppleCodesignError>),
73}
74
75impl From<apple_codesign::AppleCodesignError> for Error {
76    fn from(error: apple_codesign::AppleCodesignError) -> Self {
77        Self::Codesign(Box::new(error))
78    }
79}
80
81pub type Result<T> = std::result::Result<T, Error>;
82
83pub(crate) fn invalid(message: impl Into<String>) -> Error {
84    Error::InvalidData(message.into())
