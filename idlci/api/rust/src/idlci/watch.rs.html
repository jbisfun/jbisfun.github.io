1//! `serve --watch`: poll the repos of every project (and of its targets' dirs) and queue
2//! `build` then `test` when a HEAD moves.
3
4use std::collections::HashMap;
5use std::path::PathBuf;
6use std::sync::Arc;
7use std::time::Duration;
8
9use crate::db::{Command, NewJob, RequestedBy};
10use crate::projects::head_oid;
11use crate::web::App;
12
13pub async fn run(app: Arc<App>, every: Duration) {
14    let mut seen: HashMap<String, Vec<(PathBuf, git2::Oid)>> = HashMap::new();
15    loop {
16        for p in app.projects().iter() {
17            let mut dirs = vec![p.dir.clone()];
18            if let Ok(targets) = app.targets(p).await {
19                dirs.extend(targets.iter().map(|t| p.dir.join(&t.dir)));
20            }
21            let mut heads: Vec<_> = dirs.iter().filter_map(|d| head_oid(d)).collect();
22            heads.sort();
23            heads.dedup();
24            let changed = seen.insert(p.path.clone(), heads.clone()).is_some_and(|old| old != heads);
25            if !changed {
26                continue;
27            }
28            for command in [Command::Build, Command::Test] {
29                let new = NewJob {
30                    project: p.path.clone(),
31                    command,
32                    requested_by: RequestedBy::Watch,
33                    ..Default::default()
34                };
35                match app.enqueue(new).await {
36                    Ok(e) if !e.deduplicated => eprintln!("idlci: watch: {} moved, queued #{}", p.path, e.job.id),
37                    Ok(_) => {}
38                    Err(e) => eprintln!("idlci: watch: {}: {e:#}", p.path),
39                }
40            }
41        }
42        tokio::time::sleep(every).await;
43    }
44}
