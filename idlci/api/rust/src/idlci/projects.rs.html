10#[derive(Clone, Debug, Serialize)]
11pub struct Project {
12    /// Path relative to the workspace root ("." for the root itself).
13    pub path: String,
14    #[serde(skip)]
15    pub dir: PathBuf,
16}

18const SKIP: &[&str] = &["target", "node_modules", "build", "DerivedData", "Pods"];
19
20/// Every directory holding an `idlproj.yml`, up to 5 levels below the root.
21pub fn discover(root: &Path) -> Vec<Project> {
22    let walk = walkdir::WalkDir::new(root).max_depth(5).sort_by_file_name();
23    let mut out: Vec<Project> = walk
24        .into_iter()
25        .filter_entry(|e| {
26            let name = e.file_name().to_string_lossy();
27            e.depth() == 0 || !(e.file_type().is_dir() && (name.starts_with('.') || SKIP.contains(&&*name)))
28        })
29        .filter_map(Result::ok)
30        .filter(|e| e.file_type().is_file() && e.file_name() == "idlproj.yml")
31        .filter_map(|e| {
32            let dir = e.path().parent()?.to_path_buf();
33            let rel = dir.strip_prefix(root).ok()?.to_string_lossy().into_owned();
34            Some(Project { path: if rel.is_empty() { ".".into() } else { rel }, dir })
35        })
36        .collect();
37    out.sort_by(|a, b| a.path.cmp(&b.path));
38    out
39}

41/// Find a project by relative path, pull repo name, or unique last path segment.
42pub fn resolve<'a>(root: &Path, projects: &'a [Project], arg: &str) -> Result<&'a Project> {
43    let arg = arg.trim_end_matches('/');
44    if let Some(p) = projects.iter().find(|p| p.path == arg) {
45        return Ok(p);
46    }
47    // Pull repo name: the project at the repo root, or the repo's only project.
48    let in_repo: Vec<_> = projects.iter().filter(|p| repo_name(root, &p.dir).as_deref() == Some(arg)).collect();
49    if let Some(p) = in_repo.iter().find(|p| p.dir == root.join(arg)).or(in_repo.first().filter(|_| in_repo.len() == 1)) {
50        return Ok(p);
51    }
52    let by_name: Vec<_> = projects.iter().filter(|p| p.path.rsplit('/').next() == Some(arg)).collect();
53    match by_name[..] {
54        [p] => Ok(p),
55        _ => bail!(
56            "unknown project {arg:?}; projects: {}",
57            projects.iter().map(|p| p.path.as_str()).collect::<Vec<_>>().join(", ")
58        ),
59    }
60}

62#[derive(Clone, Debug, Serialize, serde::Deserialize)]
63pub struct Target {
64    pub name: String,
65    pub kind: String,
66    pub builder: String,
67    pub bundle_id: Option<String>,
68    pub platform: String,
69    pub dir: String,
70}
71
72const KINDS: &[&str] = &["app", "framework", "dylib", "appex", "static", "test"];
73
74/// Parse `idl list`: `NAME KIND BUILDER [BUNDLE_ID] PLATFORM VERSION DIR` per line.
75pub fn parse_list(text: &str) -> Vec<Target> {
76    text.lines()
77        .filter_map(|line| {
78            let t: Vec<&str> = line.split_whitespace().collect();
79            let n = t.len();
80            if n < 6 || !KINDS.contains(&t[1]) {
81                return None;
82            }
83            Some(Target {
84                name: t[0].into(),
85                kind: t[1].into(),
86                builder: t[2].into(),
87                bundle_id: (n > 6).then(|| t[3..n - 3].join(" ")),
88                platform: format!("{} {}", t[n - 3], t[n - 2]),
89                dir: t[n - 1].into(),
90            })
91        })
92        .collect()
93}

95pub async fn targets(idl: &Path, dir: &Path) -> Result<Vec<Target>> {
96    let out = tokio::process::Command::new(idl)
97        .arg("list")
98        .current_dir(dir)
99        .stdin(std::process::Stdio::null())
100        .output()
101        .await
102        .with_context(|| format!("running {} list", idl.display()))?;
103    if !out.status.success() {
104        bail!("idl list failed: {}", String::from_utf8_lossy(&out.stderr).trim());
105    }
106    Ok(parse_list(&String::from_utf8_lossy(&out.stdout)))
107}

109/// The `idl` binary: $IDLCI_IDL, else `idl` on PATH, else ~/.local/bin/idl.
110pub fn idl_path() -> PathBuf {
111    if let Some(p) = std::env::var_os("IDLCI_IDL") {
112        return p.into();
113    }
114    let on_path = std::env::var_os("PATH")
115        .into_iter()
116        .flat_map(|p| std::env::split_paths(&p).collect::<Vec<_>>())
117        .map(|d| d.join("idl"))
118        .find(|p| p.is_file());
119    on_path.unwrap_or_else(|| {
120        let local = std::env::home_dir().unwrap_or_default().join(".local/bin/idl");
121        if local.is_file() { local } else { "idl".into() }
122    })
123}

125fn repo_name(root: &Path, dir: &Path) -> Option<String> {
126    let repo = git2::Repository::discover(dir).ok()?;
127    let rel = repo.workdir()?.strip_prefix(root).ok()?.to_string_lossy().trim_end_matches('/').to_string();
128    (!rel.is_empty()).then_some(rel)
129}

131/// Repo name (relative to the root), branch, short HEAD and dirty flag.
132pub fn git_info(root: &Path, dir: &Path) -> GitInfo {
133    let Ok(repo) = git2::Repository::discover(dir) else {
134        return GitInfo::default();
135    };
136    let head = repo.head().ok();
137    let mut opts = git2::StatusOptions::new();
138    opts.include_untracked(true).include_ignored(false).exclude_submodules(true);
139    GitInfo {
140        repo: repo_name(root, dir),
141        branch: head.as_ref().and_then(|h| h.shorthand().ok().map(str::to_string)),
142        head: head
143            .as_ref()
144            .and_then(|h| h.peel_to_commit().ok())
145            .and_then(|c| c.as_object().short_id().ok())
146            .and_then(|b| b.as_str().ok().map(str::to_string)),
147        dirty: repo.statuses(Some(&mut opts)).map(|s| !s.is_empty()).unwrap_or(false),
148    }
149}

151/// HEAD oid of the repo containing `dir` (for the watcher).
152pub fn head_oid(dir: &Path) -> Option<(PathBuf, git2::Oid)> {
153    let repo = git2::Repository::discover(dir).ok()?;
154    let oid = repo.head().ok()?.target()?;
155    Some((repo.workdir()?.to_path_buf(), oid))
156}

163    fn parses_idl_list() {
164        let text = "FossifyCalculator        app       direct  org.fossify.math.ios                 iOS 17.0   .\n\
165                    CalculatorKit            static    direct                                       iOS 17.0   .\n\
166                    FossifyCommons           static    direct                                       iOS 17.0   ../commons-ios\n\
167                    warning: something\n";
168        let t = parse_list(text);
169        assert_eq!(t.len(), 3);
170        assert_eq!(t[0].bundle_id.as_deref(), Some("org.fossify.math.ios"));
171        assert_eq!(t[1].bundle_id, None);
172        assert_eq!(t[1].platform, "iOS 17.0");
173        assert_eq!(t[2].dir, "../commons-ios");
174    }

176    #[test]
177    fn discovers_and_resolves() {
178        let tmp = tempfile::tempdir().unwrap();
179        let root = tmp.path();
180        for d in ["group/a-ios", "b", "b/target/x", ".hidden/c"] {
181            std::fs::create_dir_all(root.join(d)).unwrap();
182            std::fs::write(root.join(d).join("idlproj.yml"), "").unwrap();
183        }
184        let p = discover(root);
185        assert_eq!(p.iter().map(|p| p.path.as_str()).collect::<Vec<_>>(), ["b", "group/a-ios"]);
186        assert_eq!(resolve(root, &p, "a-ios").unwrap().path, "group/a-ios");
187        assert_eq!(resolve(root, &p, "group/a-ios/").unwrap().path, "group/a-ios");
188        assert!(resolve(root, &p, "nope").is_err());
189    }
