20#[derive(Parser)]
21#[command(version, about)]
22struct Cli {
23    /// Workspace root (default: $PULL_ROOT, else the nearest parent with pullfile.plist).
24    #[arg(long, global = true, env = "PULL_ROOT")]
25    root: Option<PathBuf>,
26    /// Server address (serve: listen address, default 127.0.0.1:8737; clients: default from
27    /// .pull/idlci/serve.addr).
28    #[arg(long, global = true, env = "IDLCI_ADDR")]
29    addr: Option<String>,
30    #[command(subcommand)]
31    cmd: Cmd,
32}

34#[derive(Subcommand)]
35enum Cmd {
36    /// Web UI + JSON API + job workers.
37    Serve {
38        /// Concurrent jobs (builds are CPU-heavy).
39        #[arg(long, default_value_t = 1)]
40        jobs: usize,
41        /// Queue build + test when a project's git HEAD changes.
42        #[arg(long)]
43        watch: bool,
44        /// Seconds between HEAD polls with --watch.
45        #[arg(long, default_value_t = 15)]
46        watch_interval: u64,
47    },
48    /// MCP server on stdio (starts `idlci serve` if it isn't running).
49    Mcp,
50    /// Queue a job.
51    Queue {
52        project: String,
53        command: Command,
54        targets: Vec<String>,
55        #[arg(short, long, value_enum, default_value_t = Configuration::None)]
56        configuration: Configuration,
57        #[arg(long)]
58        strict: bool,
59        /// Follow the log until the job finishes; exit nonzero unless it passed.
60        #[arg(short, long)]
61        wait: bool,
62    },
63    /// List jobs.
64    Jobs {
65        #[arg(long, value_enum)]
66        status: Option<Status>,
67        #[arg(long)]
68        project: Option<String>,
69        #[arg(long, default_value_t = 20)]
70        limit: usize,
71    },
72    /// Print a job's log.
73    Log {
74        id: i64,
75        /// Follow until the job finishes.
76        #[arg(short, long)]
77        follow: bool,
78    },
79    /// Cancel a queued or running job.
80    Cancel { id: i64 },
81    /// Queue a job again.
82    Retry { id: i64 },
83}

85fn find_root(arg: Option<PathBuf>) -> Result<PathBuf> {
86    let root = match arg {
87        Some(r) => r,
88        None => {
89            let cwd = std::env::current_dir()?;
90            cwd.ancestors()
91                .find(|d| d.join("pullfile.plist").is_file())
92                .map(Path::to_path_buf)
93                .context("no workspace root: pass --root, set PULL_ROOT, or run inside a pull workspace")?
94        }
95    };
96    root.canonicalize().with_context(|| format!("workspace root {}", root.display()))
97}

99fn main() -> Result<()> {
100    let cli = Cli::parse();
101    let root = find_root(cli.root)?;
102    let rt = tokio::runtime::Builder::new_multi_thread().enable_all().build()?;
103    rt.block_on(async move {
104        let client = || Client::connect(&root, cli.addr.as_deref(), true);
105        match cli.cmd {
106            Cmd::Serve { jobs, watch, watch_interval } => {
107                let addr = cli.addr.clone().unwrap_or_else(|| client::DEFAULT_ADDR.into());
108                web::serve(root.clone(), &addr, jobs, watch.then_some(watch_interval)).await
109            }
110            Cmd::Mcp => mcp::serve(root.clone(), cli.addr.clone()).await,
111            Cmd::Queue { project, command, targets, configuration, strict, wait } => {
112                let c = client().await?;
113                let new = NewJob { project, command, targets, configuration, strict, requested_by: RequestedBy::Cli, retry_of: None };
114                let e: Enqueued = c.post("/api/jobs", &new).await?;
115                println!("{}{}", e.job.summary(), if e.deduplicated { " (already queued)" } else { "" });
116                if wait {
117                    let job = follow(&c, e.job.id).await?;
118                    if job.status != Status::Passed {
119                        bail!("job #{} {}", job.id, name(&job.status));
120                    }
121                }
122                Ok(())
123            }
124            Cmd::Jobs { status, project, limit } => {
125                let mut q = vec![("limit", limit.to_string())];
126                q.extend(status.map(|s| ("status", name(&s))));
127                q.extend(project.map(|p| ("project", p)));
128                let jobs: Vec<Job> = client().await?.get_query("/api/jobs", &q).await?;
129                jobs.iter().for_each(|j| println!("{}", j.summary()));
130                Ok(())
131            }
132            Cmd::Log { id, follow: true } => follow(&client().await?, id).await.map(drop),
133            Cmd::Log { id, follow: false } => {
134                let l: web::LogChunk = client().await?.get(&format!("/api/jobs/{id}/log?tail=1000000")).await?;
135                println!("{}", l.text);
136                Ok(())
137            }
138            Cmd::Cancel { id } => {
139                let job: Job = client().await?.post(&format!("/api/jobs/{id}/cancel"), &()).await?;
140                println!("{}", job.summary());
141                Ok(())
142            }
143            Cmd::Retry { id } => {
144                let e: Enqueued = client().await?.post(&format!("/api/jobs/{id}/retry?by=cli"), &()).await?;
145                println!("{}", e.job.summary());
146                Ok(())
147            }
148        }
149    })
150}

152/// Stream a job's log to stdout until it finishes.
153async fn follow(c: &Client, id: i64) -> Result<Job> {
154    use std::io::Write;
155    let mut offset = 0;
156    loop {
157        let l: web::LogChunk = c.get(&format!("/api/jobs/{id}/log?offset={offset}")).await?;
158        print!("{}", l.text);
159        std::io::stdout().flush()?;
160        offset = l.next_offset;
161        if l.status.is_final() && offset >= l.size {
162            return c.get(&format!("/api/jobs/{id}")).await;
163        }
164        if l.text.is_empty() {
165            tokio::time::sleep(Duration::from_secs(1)).await;
166        }
167    }
168}
