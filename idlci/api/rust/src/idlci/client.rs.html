1//! HTTP client for a running `idlci serve` (MCP server and CLI), starting one if needed.
2
3use std::os::unix::process::CommandExt;
4use std::path::{Path, PathBuf};
5use std::time::Duration;
6
7use anyhow::{Context, Result, bail};
8use serde::{Serialize, de::DeserializeOwned};
9
10use crate::db::data_dir;
11
12pub const DEFAULT_ADDR: &str = "127.0.0.1:8737";
13
14pub struct Client {
15    http: reqwest::Client,
16    base: String,
17}
18
19impl Client {
20    /// Connect to the server for `root` (address: `addr`, else `.pull/idlci/serve.addr`, else
21    /// the default); with `autostart`, spawn a detached `idlci serve` when none answers.
22    pub async fn connect(root: &Path, addr: Option<&str>, autostart: bool) -> Result<Self> {
23        let http = reqwest::Client::builder().connect_timeout(Duration::from_secs(2)).build()?;
24        let addr_file = data_dir(root).join("serve.addr");
25        let current = || {
26            addr.map(str::to_string)
27                .or_else(|| std::fs::read_to_string(&addr_file).ok().map(|s| s.trim().to_string()))
28                .filter(|s| !s.is_empty())
29                .unwrap_or_else(|| DEFAULT_ADDR.into())
30        };
31        if let Some(c) = Self::probe(&http, &current(), root).await? {
32            return Ok(c);
33        }
34        if !autostart {
35            bail!("idlci serve is not running for {} (start it with `idlci serve`)", root.display());
36        }
37        let log = spawn_server(root, addr)?;
38        for _ in 0..80 {
39            tokio::time::sleep(Duration::from_millis(250)).await;
40            if let Some(c) = Self::probe(&http, &current(), root).await? {
41                return Ok(c);
42            }
43        }
44        bail!("started idlci serve but it did not answer; see {}", log.display())
45    }
46
47    /// `Some` if a server for this root answers at `addr`; error if it serves another root.
48    async fn probe(http: &reqwest::Client, addr: &str, root: &Path) -> Result<Option<Self>> {
49        let base = format!("http://{addr}");
50        let Ok(resp) = http.get(format!("{base}/api/health")).timeout(Duration::from_secs(3)).send().await else {
51            return Ok(None);
52        };
53        let health: serde_json::Value = resp.json().await.unwrap_or_default();
54        match health["root"].as_str() {
55            Some(r) if Path::new(r) == root => Ok(Some(Self { http: http.clone(), base })),
56            Some(r) => bail!("{addr} is an idlci server for {r}, not {}; pass --addr", root.display()),
57            None => bail!("{addr} answers but isn't idlci; pass --addr"),
58        }
59    }
60
61    pub async fn get<T: DeserializeOwned>(&self, path: &str) -> Result<T> {
62        self.get_query(path, &[]).await
63    }
64
65    pub async fn get_query<T: DeserializeOwned>(&self, path: &str, query: &[(&str, String)]) -> Result<T> {
66        decode(self.http.get(format!("{}{path}", self.base)).query(query).send().await?).await
67    }
68
69    pub async fn post<T: DeserializeOwned>(&self, path: &str, body: &impl Serialize) -> Result<T> {
70        decode(self.http.post(format!("{}{path}", self.base)).json(body).send().await?).await
71    }
72}
73
74async fn decode<T: DeserializeOwned>(resp: reqwest::Response) -> Result<T> {
75    let status = resp.status();
76    let v: serde_json::Value = resp.json().await.context("decoding response")?;
77    if !status.is_success() {
78        bail!("{}", v["error"].as_str().unwrap_or(status.as_str()));
79    }
80    Ok(serde_json::from_value(v)?)
81}
82
83/// `idlci serve --root ROOT` in its own session, logging to `.pull/idlci/serve.log`.
84fn spawn_server(root: &Path, addr: Option<&str>) -> Result<PathBuf> {
85    let dir = data_dir(root);
86    std::fs::create_dir_all(&dir)?;
87    let log_path = dir.join("serve.log");
88    let log = std::fs::OpenOptions::new().create(true).append(true).open(&log_path)?;
89    let mut cmd = std::process::Command::new(std::env::current_exe()?);
90    cmd.arg("serve").arg("--root").arg(root);
91    if let Some(a) = addr {
92        cmd.arg("--addr").arg(a);
93    }
94    cmd.stdin(std::process::Stdio::null()).stdout(log.try_clone()?).stderr(log);
95    // SAFETY: setsid is async-signal-safe; it detaches the server from our session.
96    unsafe {
97        cmd.pre_exec(|| nix::unistd::setsid().map(drop).map_err(std::io::Error::from));
98    }
99    cmd.spawn().context("starting idlci serve")?;
100    Ok(log_path)
101}
