32    /// Arguments for `idl`, run from the project directory.
33    pub fn idl_args(self, cfg: Configuration, strict: bool, targets: &[String]) -> Vec<String> {
34        let mut args: Vec<String> = match self {
35            Command::FmtCheck => vec!["fmt".into(), "--check".into()],
36            _ => vec![name(&self)],
37        };
38        let cfg_flag = matches!(self, Command::Build | Command::Test | Command::Clean);
39        if cfg_flag && cfg != Configuration::None {
40            args.extend(["-c".into(), name(&cfg)]);
41        }
42        if strict && matches!(self, Command::Build | Command::Check) {
43            args.push("--strict".into());
44        }
45        args.extend(targets.iter().cloned());
46        args
47    }
48}

85/// The serde name of a unit enum variant ("fmt-check", "queued", ...).
86pub fn name<T: Serialize>(v: &T) -> String {
87    match serde_json::to_value(v) {
88        Ok(serde_json::Value::String(s)) => s,
89        _ => String::new(),
90    }

118    pub finished: Option<i64>,

126impl Job {
127    /// One line for the CLI and MCP tools.
128    pub fn summary(&self) -> String {
129        let mut s = format!("#{} {} {} {}", self.id, name(&self.status), name(&self.command), self.project);
130        if !self.targets.is_empty() {
131            s += &format!(" [{}]", self.targets.join(" "));
132        }
133        if self.configuration != Configuration::None {
134            s += &format!(" -c {}", name(&self.configuration));
135        }
136        if self.strict {
137            s += " --strict";
138        }
139        if let Some(code) = self.exit_code {
140            s += &format!(" exit={code}");
141        }
142        if let (Some(a), Some(b)) = (self.started, self.finished) {
143            s += &format!(" {}s", b - a);
144        }
145        if let Some(head) = &self.git.head {
146            let dirty = if self.git.dirty { "+dirty" } else { "" };
147            s += &format!(" @{}{head}{dirty}", self.git.branch.as_deref().map(|b| format!("{b}:")).unwrap_or_default());
148        }
149        s + &format!(" by {}", name(&self.requested_by))
150    }
151}

191const COLS: &str = "id, project, command, targets, configuration, strict, status, exit_code, created, \
192                    started, finished, repo, branch, head, dirty, requested_by, retry_of, log";
193
194fn row(r: &Row) -> rusqlite::Result<Job> {
195    let enum_col = |i: usize| -> rusqlite::Result<String> { r.get(i) };
196    let bad = |i, e: anyhow::Error| rusqlite::Error::FromSqlConversionFailure(i, rusqlite::types::Type::Text, e.into());
197    Ok(Job {
198        id: r.get(0)?,
199        project: r.get(1)?,
200        command: parse(&enum_col(2)?).map_err(|e| bad(2, e))?,
201        targets: serde_json::from_str(&enum_col(3)?).unwrap_or_default(),
202        configuration: parse(&enum_col(4)?).map_err(|e| bad(4, e))?,
203        strict: r.get(5)?,
204        status: parse(&enum_col(6)?).map_err(|e| bad(6, e))?,
205        exit_code: r.get(7)?,
206        created: r.get(8)?,
207        started: r.get(9)?,
208        finished: r.get(10)?,
209        git: GitInfo { repo: r.get(11)?, branch: r.get(12)?, head: r.get(13)?, dirty: r.get(14)? },
210        requested_by: parse(&enum_col(15)?).map_err(|e| bad(15, e))?,
211        retry_of: r.get(16)?,
212        log: PathBuf::from(r.get::<_, String>(17)?),
213    })
214}

221impl Store {
222    /// Open (creating) the store under `$ROOT/.pull/idlci`.
223    pub fn open(root: &Path) -> Result<Self> {
224        let dir = data_dir(root);
225        std::fs::create_dir_all(dir.join("logs"))?;
226        let conn = Connection::open(dir.join("idlci.db"))?;
227        conn.busy_timeout(std::time::Duration::from_secs(5))?;
228        conn.execute_batch(
229            "PRAGMA journal_mode=WAL;
230             CREATE TABLE IF NOT EXISTS jobs (
231               id INTEGER PRIMARY KEY AUTOINCREMENT,
232               project TEXT NOT NULL, command TEXT NOT NULL, targets TEXT NOT NULL,
233               configuration TEXT NOT NULL, strict INTEGER NOT NULL, status TEXT NOT NULL,
234               exit_code INTEGER, created INTEGER NOT NULL, started INTEGER, finished INTEGER,
235               repo TEXT, branch TEXT, head TEXT, dirty INTEGER NOT NULL DEFAULT 0,
236               requested_by TEXT NOT NULL, retry_of INTEGER, log TEXT NOT NULL DEFAULT '',
237               pid INTEGER);
238             CREATE INDEX IF NOT EXISTS jobs_project ON jobs(project, command);
239             CREATE INDEX IF NOT EXISTS jobs_status ON jobs(status);",
240        )?;
241        Ok(Self { conn: Mutex::new(conn), dir })
242    }
243
244    fn conn(&self) -> std::sync::MutexGuard<'_, Connection> {
245        self.conn.lock().unwrap_or_else(|e| e.into_inner())
246    }
247
248    /// Queue a job, or return the identical job that is already queued (`false`).
249    pub fn insert(&self, new: &NewJob, git: &GitInfo) -> Result<(Job, bool)> {
250        let conn = self.conn();
251        let targets = serde_json::to_string(&new.targets)?;
252        let (command, cfg) = (name(&new.command), name(&new.configuration));
253        let existing = conn
254            .query_row(
255                &format!(
256                    "SELECT {COLS} FROM jobs WHERE status = 'queued' AND project = ?1 AND command = ?2 \
257                     AND targets = ?3 AND configuration = ?4 AND strict = ?5 ORDER BY id LIMIT 1"
258                ),
259                params![new.project, command, targets, cfg, new.strict],
260                row,
261            )
262            .optional()?;
263        if let Some(job) = existing {
264            return Ok((job, false));
265        }
266        conn.execute(
267            "INSERT INTO jobs (project, command, targets, configuration, strict, status, created, \
268             repo, branch, head, dirty, requested_by, retry_of) \
269             VALUES (?1, ?2, ?3, ?4, ?5, 'queued', ?6, ?7, ?8, ?9, ?10, ?11, ?12)",
270            params![
271                new.project,
272                command,
273                targets,
274                cfg,
275                new.strict,
276                now(),
277                git.repo,
278                git.branch,
279                git.head,
280                git.dirty,
281                name(&new.requested_by),
282                new.retry_of
283            ],
284        )?;
285        let id = conn.last_insert_rowid();
286        let log = self.dir.join("logs").join(format!("{id}.log"));
287        conn.execute("UPDATE jobs SET log = ?1 WHERE id = ?2", params![log.to_string_lossy(), id])?;
288        let job = conn.query_row(&format!("SELECT {COLS} FROM jobs WHERE id = ?1"), [id], row)?;
289        Ok((job, true))
290    }

292    pub fn get(&self, id: i64) -> Result<Option<Job>> {
293        Ok(self
294            .conn()
295            .query_row(&format!("SELECT {COLS} FROM jobs WHERE id = ?1"), [id], row)
296            .optional()?)
297    }

299    pub fn list(&self, f: &JobFilter) -> Result<Vec<Job>> {
300        let conn = self.conn();
301        let mut stmt = conn.prepare(&format!(
302            "SELECT {COLS} FROM jobs WHERE (?1 IS NULL OR status = ?1) AND (?2 IS NULL OR project = ?2) \
303             ORDER BY id DESC LIMIT ?3"
304        ))?;
305        let status = f.status.map(|s| name(&s));
306        let limit = f.limit.unwrap_or(100) as i64;
307        let jobs = stmt.query_map(params![status, f.project, limit], row)?;
308        Ok(jobs.collect::<rusqlite::Result<_>>()?)
309    }

311    /// Queued and running jobs, oldest first.
312    pub fn active(&self) -> Result<Vec<Job>> {
313        let conn = self.conn();
314        let mut stmt = conn.prepare(&format!(
315            "SELECT {COLS} FROM jobs WHERE status IN ('queued', 'running') ORDER BY id"
316        ))?;
317        let jobs = stmt.query_map([], row)?;
318        Ok(jobs.collect::<rusqlite::Result<_>>()?)
319    }

321    /// Mark the oldest queued job running and return it.
322    pub fn claim_next(&self) -> Result<Option<Job>> {
323        Ok(self
324            .conn()
325            .query_row(
326                &format!(
327                    "UPDATE jobs SET status = 'running', started = ?1 WHERE id = \
328                     (SELECT id FROM jobs WHERE status = 'queued' ORDER BY id LIMIT 1) RETURNING {COLS}"
329                ),
330                [now()],
331                row,
332            )
333            .optional()?)
334    }

336    pub fn set_pid(&self, id: i64, pid: Option<i32>) -> Result<()> {
337        self.conn().execute("UPDATE jobs SET pid = ?1 WHERE id = ?2", params![pid, id])?;
338        Ok(())
339    }

341    pub fn finish(&self, id: i64, status: Status, exit_code: Option<i32>) -> Result<()> {
342        self.conn().execute(
343            "UPDATE jobs SET status = ?1, exit_code = ?2, finished = ?3, pid = NULL WHERE id = ?4",
344            params![name(&status), exit_code, now(), id],
345        )?;
346        Ok(())
347    }

349    /// Cancel a job that hasn't started; false if it isn't queued.
350    pub fn cancel_queued(&self, id: i64) -> Result<bool> {
351        let n = self.conn().execute(
352            "UPDATE jobs SET status = 'cancelled', finished = ?1 WHERE id = ?2 AND status = 'queued'",
353            params![now(), id],
354        )?;
355        Ok(n > 0)
356    }

358    /// Jobs left `running` by a previous server: mark them failed, return (job, pid).
359    pub fn recover(&self) -> Result<Vec<(Job, Option<i32>)>> {
360        let conn = self.conn();
361        let mut stmt = conn.prepare(&format!("SELECT {COLS}, pid FROM jobs WHERE status = 'running'"))?;
362        let jobs = stmt.query_map([], |r| Ok((row(r)?, r.get(18)?)))?.collect::<rusqlite::Result<_>>()?;
363        conn.execute(
364            "UPDATE jobs SET status = 'failed', finished = ?1, pid = NULL WHERE status = 'running'",
365            [now()],
366        )?;
367        Ok(jobs)
368    }

370    /// Newest finished jobs per project (for the dashboard), newest first.
371    pub fn finished_per_project(&self, per_project: usize) -> Result<Vec<Job>> {
372        let conn = self.conn();
373        let mut stmt = conn.prepare(&format!(
374            "SELECT {COLS} FROM (SELECT *, ROW_NUMBER() OVER (PARTITION BY project ORDER BY id DESC) AS n \
375             FROM jobs WHERE status IN ('passed', 'failed', 'cancelled')) WHERE n <= ?1 ORDER BY id DESC"
376        ))?;
377        let jobs = stmt.query_map([per_project as i64], row)?;
378        Ok(jobs.collect::<rusqlite::Result<_>>()?)
379    }

381    /// Latest job per (project, command).
382    pub fn latest_per_command(&self) -> Result<Vec<Job>> {
383        let conn = self.conn();
384        let mut stmt = conn.prepare(&format!(
385            "SELECT {COLS} FROM jobs WHERE id IN (SELECT MAX(id) FROM jobs GROUP BY project, command)"
386        ))?;
387        let jobs = stmt.query_map([], row)?;
388        Ok(jobs.collect::<rusqlite::Result<_>>()?)
389    }

391    /// Duration of the last passed run of the same project + command (progress estimate).
392    pub fn last_duration(&self, project: &str, command: Command) -> Result<Option<i64>> {
393        Ok(self
394            .conn()
395            .query_row(
396                "SELECT finished - started FROM jobs WHERE project = ?1 AND command = ?2 \
397                 AND status = 'passed' ORDER BY id DESC LIMIT 1",
398                params![project, name(&command)],
399                |r| r.get(0),
400            )
401            .optional()?)
402    }

413    #[test]
414    fn queue_dedupe_claim_finish() {
415        let tmp = tempfile::tempdir().unwrap();
416        let store = Store::open(tmp.path()).unwrap();
417        let new = NewJob {
418            project: "a/b".into(),
419            command: Command::Build,
420            targets: vec!["App".into()],
421            ..Default::default()
422        };
423        let (j1, created) = store.insert(&new, &GitInfo::default()).unwrap();
424        assert!(created);
425        assert!(j1.log.ends_with(format!("logs/{}.log", j1.id)));
426        let (j2, created) = store.insert(&new, &GitInfo::default()).unwrap();
427        assert!(!created);
428        assert_eq!(j1.id, j2.id);
429        let claimed = store.claim_next().unwrap().unwrap();
430        assert_eq!(claimed.status, Status::Running);
431        assert!(store.claim_next().unwrap().is_none());
432        // A running job doesn't block a new identical one.
433        assert!(store.insert(&new, &GitInfo::default()).unwrap().1);
434        store.finish(j1.id, Status::Passed, Some(0)).unwrap();
435        let latest = store.latest_per_command().unwrap();
436        assert_eq!(latest.len(), 1);
437        assert_eq!(store.finished_per_project(5).unwrap().len(), 1);
438        assert!(store.last_duration("a/b", Command::Build).unwrap().is_some());
439        let queued = store.list(&JobFilter { status: Some(Status::Queued), ..Default::default() }).unwrap();
440        assert_eq!(queued.len(), 1);
441        assert!(store.cancel_queued(queued[0].id).unwrap());
442        assert!(store.recover().unwrap().is_empty());
443        store.insert(&new, &GitInfo::default()).unwrap();
444        let running = store.claim_next().unwrap().unwrap();
445        store.set_pid(running.id, Some(4242)).unwrap();
446        let recovered = store.recover().unwrap();
447        assert_eq!(recovered.len(), 1);
448        assert_eq!(recovered[0].1, Some(4242));
449        assert_eq!(store.get(running.id).unwrap().unwrap().status, Status::Failed);
450    }
